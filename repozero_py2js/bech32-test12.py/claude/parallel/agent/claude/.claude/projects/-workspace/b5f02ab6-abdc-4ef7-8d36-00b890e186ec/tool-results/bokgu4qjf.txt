     1	// The BCH checksum over GF(32) that bech32 strings carry.
     2	//
     3	// Every intermediate stays inside 30 bits -- `chk` is masked to 25 bits before
     4	// being shifted left by 5, and the generator constants are all below 2**30 --
     5	// so JavaScript's 32-bit bitwise operators are exact here and agree with
     6	// Python's arbitrary-precision integers bit for bit.
     7	
     8	import { CHECKSUM_LENGTH, GENERATOR } from './charset.mjs';
     9	import { codePoints } from '../python/text.mjs';
    10	
    11	export function bech32Polymod(values) {
    12	  let chk = 1;
    13	  for (const value of values) {
    14	    const top = chk >> 25;
    15	    chk = ((chk & 0x1ffffff) << 5) ^ value;
    16	    for (let i = 0; i < GENERATOR.length; i += 1) {
    17	      if ((top >> i) & 1) {
    18	        chk ^= GENERATOR[i];
    19	      }
    20	    }
    21	  }
    22	  return chk;
    23	}
    24	
    25	/** High bits of the HRP, a zero separator, then the low bits. */
    26	export function bech32HrpExpand(hrp) {
    27	  const points = codePoints(hrp);
    28	  return [...points.map((point) => point >> 5), 0, ...points.map((point) => point & 31)];
    29	}
    30	
    31	export function bech32VerifyChecksum(hrp, data) {
    32	  return bech32Polymod([...bech32HrpExpand(hrp), ...data]) === 1;
    33	}
    34	
    35	export function bech32CreateChecksum(hrp, data) {
    36	  const values = [...bech32HrpExpand(hrp), ...data, ...new Array(CHECKSUM_LENGTH).fill(0)];
    37	  const polymod = bech32Polymod(values) ^ 1;
    38	  const checksum = [];
    39	  for (let i = 0; i < CHECKSUM_LENGTH; i += 1) {
    40	    checksum.push((polymod >> (5 * (5 - i))) & 31);
    41	  }
    42	  return checksum;
    43	}

    19	/**
    20	 * Validate a bech32 string and split it into `[hrp, data]`, or `[null, null]`.
    21	 *
    22	 * `data` excludes the six checksum symbols. The HRP is returned lower-cased,
    23	 * so an all-upper-case address never matches an upper-case HRP argument.
    24	 */
    25	export function bech32Decode(bech) {
    26	  if (!isPrintableAsciiRange(bech) || isMixedCase(bech)) {
    27	    return failure();
    28	  }
    29	
    30	  const lowered = bech.toLowerCase();
    31	  const pos = lowered.lastIndexOf('1');
    32	  // `pos` is both the separator index and the HRP length.
    33	  if (pos < 1 || pos + 1 + CHECKSUM_LENGTH > lowered.length || pos > MAX_HRP_LENGTH) {
    34	    return failure();
    35	  }
    36	
    37	  const tail = lowered.slice(pos + 1);
    38	  for (const char of tail) {
    39	    if (!isCharsetChar(char)) {
    40	      return failure();
    41	    }
    42	  }
    43	
    44	  const hrp = lowered.slice(0, pos);
    45	  const data = [];
    46	  for (const char of tail) {
    47	    data.push(charsetIndex(char));
    48	  }
    49	  if (!bech32VerifyChecksum(hrp, data)) {
    50	    return failure();
    51	  }
    52	  return [hrp, data.slice(0, -CHECKSUM_LENGTH)];
    53	}

     1	// General power-of-2 base conversion, as `bech32.convertbits`.
     2	
     3	/**
     4	 * Regroup `data` from `frombits`-wide symbols into `tobits`-wide ones.
     5	 *
     6	 * Returns `null` (Python's `None`) when a symbol does not fit in `frombits` --
     7	 * which is what makes `convertbits(witprog, 5, 8)` fail for any witness program
     8	 * byte >= 32 -- and, when `pad` is false, when the trailing partial group is
     9	 * either too wide or not all zeroes.
    10	 */
    11	export function convertbits(data, frombits, tobits, pad = true) {
    12	  let acc = 0;
    13	  let bits = 0;
    14	  const ret = [];
    15	  const maxv = (1 << tobits) - 1;
    16	  const maxAcc = (1 << (frombits + tobits - 1)) - 1;
    17	
    18	  for (const value of data) {
    19	    if (value < 0 || value >> frombits) {
    20	      return null;
    21	    }
    22	    acc = ((acc << frombits) | value) & maxAcc;
    23	    bits += frombits;
    24	    while (bits >= tobits) {
    25	      bits -= tobits;
    26	      ret.push((acc >> bits) & maxv);
    27	    }
    28	  }
    29	
    30	  if (pad) {
    31	    if (bits) {
    32	      ret.push((acc << (tobits - bits)) & maxv);
    33	    }
    34	  } else if (bits >= frombits || ((acc << (tobits - bits)) & maxv)) {
    35	    return null;
    36	  }
    37	  return ret;
    38	}

     2	//
     3	// This mirrors the BIP-173-only behaviour of the reference module: a witness
     4	// version of 1..16 is accepted with an ordinary bech32 checksum, and bech32m
     5	// (BIP-350) strings are rejected outright.
     6	
     7	import { bech32Decode, bech32Encode } from './codec.mjs';
     8	import { convertbits } from './convert.mjs';
     9	
    10	export const MIN_PROGRAM_LENGTH = 2;
    11	export const MAX_PROGRAM_LENGTH = 40;
    12	export const MAX_WITNESS_VERSION = 16;
    13	
    14	function failure() {
    15	  return [null, null];
    16	}
    17	
    18	/**
    19	 * Decode a segwit address into `[witnessVersion, witnessProgram]`, where the
    20	 * program is a plain array of byte values, or `[null, null]` on any failure.
    21	 *
    22	 * The program-length check deliberately precedes the `data[0]` read: for a
    23	 * checksum-only string the payload is empty, the conversion yields `[]`, and the
    24	 * length test bails out before the witness version would be indexed.
    25	 */
    26	export function decode(hrp, addr) {
    27	  const [hrpGot, data] = bech32Decode(addr);
    28	  if (hrpGot !== hrp) {
    29	    return failure();
    30	  }
    31	
    32	  const decoded = convertbits(data.slice(1), 5, 8, false);
    33	  if (
    34	    decoded === null ||
    35	    decoded.length < MIN_PROGRAM_LENGTH ||
    36	    decoded.length > MAX_PROGRAM_LENGTH
    37	  ) {
    38	    return failure();
    39	  }
    40	  if (data[0] > MAX_WITNESS_VERSION) {
    41	    return failure();
    42	  }
    43	  if (data[0] === 0 && decoded.length !== 20 && decoded.length !== 32) {
    44	    return failure();
    45	  }
    46	  return [data[0], decoded];
    47	}
    48	
    49	/** Encode a segwit address, or `null` when the result would not round-trip. */
    50	export function encode(hrp, witver, witprog) {
    51	  const addr = bech32Encode(hrp, [witver, ...convertbits(witprog, 8, 5)]);
    52	  const [version, program] = decode(hrp, addr);
    53	  if (version === null && program === null) {
    54	    return null;
    55	  }
    56	  return addr;
    57	}

     1	// A faithful port of the slice of Python's `argparse.ArgumentParser` that this
     2	// CLI exercises: two required value-taking long options plus the automatic
     3	// `-h/--help`, and no positionals.
     4	//
     5	// The token-classification and consumption algorithm follows CPython's
     6	// `_parse_known_args` closely, because its quirks are observable:
     7	//   * a required-argument error is raised before unrecognised arguments are
     8	//     reported, so a missing `--a` wins over trailing junk;
     9	//   * `-5` and `-1.5` are values (they look like negative numbers) while `-x`
    10	//     and `-5x` are option strings, so `--b -x` fails with "expected one
    11	//     argument" rather than storing "-x";
    12	//   * `--h`, `--he` and `--hel` all resolve to `--help` by unique-prefix match;
    13	//   * `--` is a separator that is itself left over as an unrecognised argument,
    14	//     since no positional action is declared to absorb it.
    15	
    16	import { basename } from 'node:path';
    17	
    18	import { HELP, STORE, Action, destFromOptionStrings, makeHelpAction } from './actions.mjs';
    19	import { ArgumentError, ParserExit } from './errors.mjs';
    20	import { formatHelp, formatUsage } from './formatter.mjs';
    21	import { pyRepr } from '../python/repr.mjs';
    22	
    23	const PREFIX_CHARS = '-';
    24	
    25	/** argparse's own "this token is a negative number, not an option" test. */
    26	const NEGATIVE_NUMBER = /^-\d+$|^-\d*\.\d+$/;
    27	
    28	/** argparse's default `prog`: `os.path.basename(sys.argv[0])`. */
    29	export function defaultProg() {
    30	  const script = process.argv[1];
    31	  return script === undefined ? '' : basename(script);
    32	}
    33	
    34	export class ArgumentParser {
    35	  constructor({ prog = defaultProg(), addHelp = true } = {}) {
    36	    this.prog = prog;
    37	    this.actions = [];
    38	    this.optionStringActions = new Map();
    39	    if (addHelp) {
    40	      this.addAction(makeHelpAction());
    41	    }
    42	  }
    43	
    44	  addAction(action) {
    45	    this.actions.push(action);
    46	    for (const optionString of action.optionStrings) {
    47	      this.optionStringActions.set(optionString, action);
    48	    }
    49	    return action;
    50	  }
    51	
    52	  addArgument(flags, options = {}) {
    53	    const optionStrings = Array.isArray(flags) ? flags : [flags];
    54	    return this.addAction(
    55	      new Action({
    56	        optionStrings,
    57	        dest: options.dest ?? destFromOptionStrings(optionStrings),
    58	        kind: STORE,
    59	        nargs: null,
    60	        required: options.required === true,
    61	        metavar: options.metavar ?? null,
    62	        help: options.help ?? '',
    63	        type: options.type ?? null,
    64	      }),
    65	    );
    66	  }
    67	
    68	  formatUsage() {
    69	    return formatUsage(this.prog, this.actions);
    70	  }
    71	
    72	  formatHelp() {
    73	    return formatHelp(this.prog, this.actions);
    74	  }
    75	
    76	  /** Print usage plus a message on stderr and terminate with status 2. */
    77	  error(message) {
    78	    throw new ParserExit(2, {
    79	      stderr: `${this.formatUsage()}${this.prog}: error: ${message}\n`,
    80	    });
    81	  }
    82	
    83	  parseArgs(argv) {
    84	    const [namespace, extras] = this.parseKnownArgs(argv);
    85	    if (extras.length > 0) {
    86	      this.error(`unrecognized arguments: ${extras.join(' ')}`);
    87	    }
    88	    return namespace;
    89	  }
    90	
    91	  parseKnownArgs(argv) {
    92	    try {
    93	      return this.parseKnownArgsUnchecked(argv);
    94	    } catch (error) {
    95	      if (error instanceof ArgumentError) {
    96	        this.error(error.message);
    97	      }
    98	      throw error;
    99	    }
   100	  }
   101	
   102	  parseKnownArgsUnchecked(argv) {
   103	    const namespace = {};
   104	    for (const action of this.actions) {
   105	      if (action.kind === STORE) {
   106	        namespace[action.dest] = null;
   107	      }
   108	    }
   109	
   110	    // Classify every token up front: 'O' for something that looks like an
   111	    // option, 'A' for a value, '-' for the `--` separator. Option matching then
   112	    // works against this pattern rather than against the raw strings, which is
   113	    // what makes `--b -- value` fail instead of storing "--".
   114	    const pattern = [];
   115	    const optionTuples = new Map();
   116	    let index = 0;
   117	    while (index < argv.length) {
   118	      if (argv[index] === '--') {
   119	        pattern.push('-');
   120	        index += 1;
   121	        while (index < argv.length) {
   122	          pattern.push('A');
   123	          index += 1;
   124	        }
   125	        break;
   126	      }
   127	      const tuple = this.parseOptional(argv[index]);
   128	      if (tuple === null) {
   129	        pattern.push('A');
   130	      } else {
   131	        optionTuples.set(index, tuple);
   132	        pattern.push('O');
   133	      }
   134	      index += 1;
   135	    }
   136	
   137	    const extras = [];
   138	    const seen = new Set();
   139	    const context = { argv, pattern, optionTuples, extras, seen, namespace };
   140	
   141	    const optionIndices = [...optionTuples.keys()].sort((left, right) => left - right);
   142	    const maxOptionIndex = optionIndices.length === 0 ? -1 : optionIndices[optionIndices.length - 1];
   143	    let start = 0;
   144	    while (start <= maxOptionIndex) {
   145	      const next = optionIndices.find((candidate) => candidate >= start);
   146	      if (next === undefined) {
   147	        break;
   148	      }
   149	      if (start !== next) {
   150	        // No positional action is declared, so nothing before the next option
   151	        // can be consumed -- it all becomes unrecognised.
   152	        extras.push(...argv.slice(start, next));
   153	        start = next;
   154	      }
   155	      start = this.consumeOptional(start, context);
   156	    }
   157	    extras.push(...argv.slice(start));
   158	
   159	    const missing = this.actions
   160	      .filter((action) => action.required && !seen.has(action))
   161	      .map((action) => action.optionStrings.join('/'));
   162	    if (missing.length > 0) {
   163	      this.error(`the following arguments are required: ${missing.join(', ')}`);
   164	    }
   165	
   166	    return [namespace, extras];
   167	  }
   168	
   169	  consumeOptional(startIndex, context) {
   170	    const { argv, pattern, optionTuples, extras, seen, namespace } = context;
   171	    let { action, optionString, explicitArg } = optionTuples.get(startIndex);
   172	    const actionTuples = [];
   173	    let stop = startIndex + 1;
   174	
   175	    for (;;) {
   176	      if (action === null) {
   177	        // An option string no action claims: report the whole token verbatim.
   178	        extras.push(argv[startIndex]);
   179	        return startIndex + 1;
   180	      }
   181	
   182	      if (explicitArg !== null) {
   183	        const argCount = this.matchArgument(action, 'A');
   184	        if (argCount === 0 && !PREFIX_CHARS.includes(optionString[1]) && explicitArg !== '') {
   185	          // A single-dash flag with a tail: `-hz` means `-h` then `-z`.
   186	          actionTuples.push([action, []]);
   187	          const prefix = optionString[0];
   188	          optionString = prefix + explicitArg[0];
   189	          const nextAction = this.optionStringActions.get(optionString);
   190	          if (nextAction === undefined) {
   191	            extras.push(prefix + explicitArg);
   192	            stop = startIndex + 1;
   193	            break;
   194	          }
   195	          action = nextAction;
   196	          explicitArg = explicitArg.slice(1);
   197	          if (explicitArg === '') {
   198	            explicitArg = null;
   199	          } else if (explicitArg[0] === '=') {
   200	            explicitArg = explicitArg.slice(1);
   201	          }
   202	        } else if (argCount === 1) {
   203	          stop = startIndex + 1;
   204	          actionTuples.push([action, [explicitArg]]);
   205	          break;
   206	        } else {
   207	          throw new ArgumentError(action, `ignored explicit argument ${pyRepr(explicitArg)}`);
   208	        }
   209	      } else {
   210	        const valueStart = startIndex + 1;
   211	        const argCount = this.matchArgument(action, pattern.slice(valueStart).join(''));
   212	        stop = valueStart + argCount;
   213	        actionTuples.push([action, argv.slice(valueStart, stop)]);
   214	        break;
   215	      }
   216	    }
   217	
   218	    for (const [chosen, values] of actionTuples) {
   219	      seen.add(chosen);
   220	      this.takeAction(chosen, values, namespace);
   221	    }
   222	    return stop;
   223	  }
   224	
   225	  /** How many following tokens this action consumes, or a parse error. */
   226	  matchArgument(action, patternText) {
   227	    if (action.nargs === 0) {
   228	      return 0;
   229	    }
   230	    if (patternText.startsWith('A')) {
   231	      return 1;
   232	    }
   233	    throw new ArgumentError(action, 'expected one argument');
   234	  }
   235	
   236	  takeAction(action, values, namespace) {
   237	    if (action.kind === HELP) {
   238	      throw new ParserExit(0, { stdout: this.formatHelp() });
   239	    }
   240	    const [value] = values;
   241	    namespace[action.dest] = action.type === null ? value : action.type(value);
   242	  }
   243	
   244	  hasNegativeNumberOptionals() {
   245	    for (const optionString of this.optionStringActions.keys()) {
   246	      if (NEGATIVE_NUMBER.test(optionString)) {
   247	        return true;
   248	      }
   249	    }
   250	    return false;
   251	  }
   252	
   253	  /**
   254	   * Decide whether a token is an option. Returns `null` when it should be
   255	   * treated as a value, otherwise `{action, optionString, explicitArg}` where a
   256	   * `null` action means "looks like an option, but nothing matches".
   257	   */
   258	  parseOptional(argString) {
   259	    if (argString === '') {
   260	      return null;
   261	    }
   262	    if (!PREFIX_CHARS.includes(argString[0])) {
   263	      return null;
   264	    }
   265	    const exact = this.optionStringActions.get(argString);
   266	    if (exact !== undefined) {
   267	      return { action: exact, optionString: argString, explicitArg: null };
   268	    }
   269	    if (argString.length === 1) {
   270	      // A lone '-' is a value.
   271	      return null;

    13	function formatUsagePart(action) {
    14	  const option = action.optionStrings[0];

    16	  return action.required ? part : `[${part}]`;

    20	  const parts = actions.map(formatUsagePart);
    21	  const tail = parts.length === 0 ? '' : ` ${parts.join(' ')}`;
    22	  return `usage: ${prog}${tail}\n`;
    23	}

    26	  const invocations = actions.map(formatInvocation);
    27	  const actionMaxLength = Math.max(0, ...invocations.map((text) => text.length + INDENT));
    28	  const helpPosition = Math.min(actionMaxLength + 2, MAX_HELP_POSITION);
    29	  const actionWidth = helpPosition - INDENT - 2;
    30	
    31	  const pad = ' '.repeat(INDENT);
    32	  const lines = [formatUsage(prog, actions), '\n', 'options:\n'];
    33	
    34	  actions.forEach((action, index) => {
    35	    const invocation = invocations[index];
    36	    if (!action.help) {
    37	      // No help text: the invocation stands alone, with no trailing padding.
    38	      lines.push(`${pad}${invocation}\n`);
    39	    } else if (invocation.length <= actionWidth) {
    40	      lines.push(`${pad}${invocation.padEnd(actionWidth)}  ${action.help}\n`);
    41	    } else {
    42	      // Too wide to share a line, so the help text starts on the next one.
    43	      lines.push(`${pad}${invocation}\n`);
    44	      lines.push(`${' '.repeat(helpPosition)}${action.help}\n`);
    45	    }
    46	  });
