import { buildHelpText, buildUsageLine, CliParseError, getProgramName, parseArguments } from "./lib/cli/parser.mjs";
import { convertbits, decode } from "./lib/bech32/index.mjs";
import { formatPythonValue } from "./lib/format/python.mjs";

function toFileUrl(filePath) {
  const normalized = filePath.startsWith("/")
    ? filePath
    : `${process.cwd().replace(/\\/g, "/")}/${filePath}`;

  return `file://${encodeURI(normalized.replace(/\\/g, "/"))}`;
}

export function main(argv = process.argv) {
  let parsed;

  try {
    parsed = parseArguments(argv);
  } catch (error) {
    if (!(error instanceof CliParseError)) {
      throw error;
    }

    const prog = getProgramName(argv);
    process.stderr.write(buildUsageLine(prog));
    process.stderr.write(`${prog}: error: ${error.message}\n`);
    return 2;
  }

  if (parsed.kind === "help") {
    process.stdout.write(buildHelpText(parsed.prog));
    return 0;
  }

  const decoded = decode(parsed.a, parsed.b);
  if (decoded[0] !== null && decoded[1] !== null) {
    const [witnessVersion, witnessProgram] = decoded;
    const converted = convertbits(witnessProgram, 5, 8);
    process.stdout.write(`${formatPythonValue(witnessVersion)}\n`);
    process.stdout.write(`${formatPythonValue(converted)}\n`);
  }

  return 0;
}

if (process.argv[1] !== undefined && import.meta.url === toFileUrl(process.argv[1])) {
  const exitCode = main(process.argv);
  if (exitCode !== 0) {
    process.exit(exitCode);
  }
}
