     1	/**
     2	 * construct's FormatField - a fixed-width integer serialised through Python's
     3	 * struct module. Only the little-endian unsigned formats this program uses
     4	 * are provided.
     5	 */
     6	
     7	import { FormatFieldError, StructError } from './exceptions.mjs';
     8	
     9	class FormatField {
    10	  /**
    11	   * @param {string} fmt the struct format string, e.g. "<B"
    12	   * @param {string} code the format character used in struct's own message
    13	   * @param {number} byteCount width of the encoded value
    14	   */
    15	  constructor(fmt, code, byteCount) {
    16	    this.fmt = fmt;
    17	    this.code = code;
    18	    this.byteCount = byteCount;
    19	    this.max = (1n << BigInt(byteCount * 8)) - 1n;
    20	  }
    21	
    22	  /**
    23	   * @param {bigint} value
    24	   * @returns {number[]} little-endian bytes
    25	   */
    26	  build(value) {
    27	    if (value < 0n || value > this.max) {
    28	      // struct raises first; construct catches it and re-raises with context,
    29	      // which is what produces the chained traceback.
    30	      const inner = new StructError(
    31	        `'${this.code}' format requires 0 <= number <= ${this.max}`,
    32	      );
    33	      const outer = new FormatFieldError(
    34	        `struct '${this.fmt}' error during building, given value ${value}`,
    35	      );
    36	      outer.during = inner;
    37	      throw outer;
    38	    }
    39	
    40	    const bytes = [];
    41	    let remaining = value;
    42	    for (let i = 0; i < this.byteCount; i += 1) {
    43	      bytes.push(Number(remaining & 0xffn));
    44	      remaining >>= 8n;
    45	    }
    46	    return bytes;
    47	  }
    48	}

     1	/**
     2	 * UTF-8 codec matching CPython, including the `surrogateescape` error handler
     3	 * that the interpreter applies when decoding argv.
     4	 *
     5	 * Bytes are represented as plain arrays of integers 0-255.
     6	 */
     7	
     8	import { PyUnicodeEncodeError } from './exceptions.mjs';
     9	import { codePoints } from './unicode.mjs';
    10	import { pyStrRepr } from './repr.mjs';
    11	
    12	const SURROGATE_LOW = 0xd800;
    13	const SURROGATE_HIGH = 0xdfff;
    14	
    15	function isContinuation(byte) {
    16	  return byte !== undefined && byte >= 0x80 && byte <= 0xbf;
    17	}
    18	
    19	/**
    20	 * Length and second-byte range of a well-formed UTF-8 sequence starting with
    21	 * `lead`, or null if `lead` can never start one. The narrowed second-byte
    22	 * ranges are what reject overlong forms, encoded surrogates, and codepoints
    23	 * above U+10FFFF.
    24	 */
    25	function sequenceShape(lead) {
    26	  if (lead <= 0x7f) return { length: 1, secondMin: 0, secondMax: 0 };
    27	  if (lead >= 0xc2 && lead <= 0xdf) return { length: 2, secondMin: 0x80, secondMax: 0xbf };
    28	  if (lead === 0xe0) return { length: 3, secondMin: 0xa0, secondMax: 0xbf };
    29	  if (lead >= 0xe1 && lead <= 0xec) return { length: 3, secondMin: 0x80, secondMax: 0xbf };
    30	  if (lead === 0xed) return { length: 3, secondMin: 0x80, secondMax: 0x9f };
    31	  if (lead >= 0xee && lead <= 0xef) return { length: 3, secondMin: 0x80, secondMax: 0xbf };
    32	  if (lead === 0xf0) return { length: 4, secondMin: 0x90, secondMax: 0xbf };
    33	  if (lead >= 0xf1 && lead <= 0xf3) return { length: 4, secondMin: 0x80, secondMax: 0xbf };
    34	  if (lead === 0xf4) return { length: 4, secondMin: 0x80, secondMax: 0x8f };
    35	  return null;
    36	}
    37	
    38	/**
    39	 * Decode bytes as UTF-8 the way Python decodes argv: every byte that cannot
    40	 * begin or continue a valid sequence becomes the lone surrogate U+DC00+byte,
    41	 * so the original bytes survive round-tripping.
    42	 */
    43	export function fsDecode(bytes) {
    44	  let out = '';
    45	  let i = 0;
    46	  while (i < bytes.length) {
    47	    const lead = bytes[i];
    48	    const shape = sequenceShape(lead);
    49	
    50	    if (shape === null) {
    51	      out += String.fromCharCode(0xdc00 + lead);
    52	      i += 1;
    53	      continue;
    54	    }
    55	    if (shape.length === 1) {
    56	      out += String.fromCharCode(lead);
    57	      i += 1;
    58	      continue;
    59	    }
    60	
    61	    const second = bytes[i + 1];
    62	    const secondOk = second !== undefined
    63	      && second >= shape.secondMin
    64	      && second <= shape.secondMax;
    65	    let valid = secondOk;
    66	    for (let k = 2; valid && k < shape.length; k += 1) {
    67	      valid = isContinuation(bytes[i + k]);
    68	    }
    69	    if (!valid) {
    70	      out += String.fromCharCode(0xdc00 + lead);
    71	      i += 1;
    72	      continue;
    73	    }
    74	
    75	    let cp = lead & (0x7f >> shape.length);
    76	    for (let k = 1; k < shape.length; k += 1) cp = (cp << 6) | (bytes[i + k] & 0x3f);
    77	    out += String.fromCodePoint(cp);
    78	    i += shape.length;
    79	  }
    80	  return out;
    81	}
    82	
    83	/**
    84	 * `str.encode()` with the default strict handler. A surrogate left behind by
    85	 * fsDecode cannot be encoded, and Python reports the run of offending
    86	 * characters by position.
    87	 */
    88	export function encodeUtf8(text) {
    89	  const points = [...codePoints(text)];
    90	  const bytes = [];
    91	
    92	  for (let index = 0; index < points.length; index += 1) {
    93	    const cp = points[index];
    94	
    95	    if (cp >= SURROGATE_LOW && cp <= SURROGATE_HIGH) {
    96	      let end = index + 1;
    97	      while (end < points.length
    98	        && points[end] >= SURROGATE_LOW
    99	        && points[end] <= SURROGATE_HIGH) end += 1;
   100	
   101	      const position = end - index === 1
   102	        ? `character ${pyStrRepr(String.fromCharCode(cp))} in position ${index}`
   103	        : `characters in position ${index}-${end - 1}`;
   104	      throw new PyUnicodeEncodeError(
   105	        `'utf-8' codec can't encode ${position}: surrogates not allowed`,
   106	      );
   107	    }
   108	
   109	    if (cp < 0x80) {
   110	      bytes.push(cp);
   111	    } else if (cp < 0x800) {
   112	      bytes.push(0xc0 | (cp >> 6), 0x80 | (cp & 0x3f));
   113	    } else if (cp < 0x10000) {
   114	      bytes.push(0xe0 | (cp >> 12), 0x80 | ((cp >> 6) & 0x3f), 0x80 | (cp & 0x3f));
   115	    } else {
   116	      bytes.push(
   117	        0xf0 | (cp >> 18),
   118	        0x80 | ((cp >> 12) & 0x3f),
   119	        0x80 | ((cp >> 6) & 0x3f),
   120	        0x80 | (cp & 0x3f),
   121	      );
   122	    }
   123	  }
   124	  return bytes;
   125	}

     1	/**
     2	 * Python-compatible Unicode character predicates.
     3	 *
     4	 * CPython's `int(str)` and `str.__repr__` consult the Unicode database directly,
     5	 * so a faithful port needs the same classifications rather than JavaScript's
     6	 * (subtly different) built-in equivalents.
     7	 */
     8	
     9	// Codepoints for which CPython's Py_UNICODE_ISSPACE() is true. This is *not*
    10	// the same set as JavaScript's /\s/u: Python includes the C0 separators
    11	// U+001C..U+001F and U+0085 (NEL) but excludes U+FEFF (ZWNBSP).
    12	const PY_UNICODE_WHITESPACE = new Set([
    13	  0x09, 0x0a, 0x0b, 0x0c, 0x0d, // \t \n \v \f \r
    14	  0x1c, 0x1d, 0x1e, 0x1f, // FS GS RS US
    15	  0x20, // SPACE
    16	  0x85, // NEL
    17	  0xa0, // NBSP
    18	  0x1680, // OGHAM SPACE MARK
    19	  0x2000, 0x2001, 0x2002, 0x2003, 0x2004, 0x2005,
    20	  0x2006, 0x2007, 0x2008, 0x2009, 0x200a,
    21	  0x2028, // LINE SEPARATOR
    22	  0x2029, // PARAGRAPH SEPARATOR
    23	  0x202f, // NARROW NO-BREAK SPACE
    24	  0x205f, // MEDIUM MATHEMATICAL SPACE
    25	  0x3000, // IDEOGRAPHIC SPACE
    26	]);
    27	
    28	// The narrower ASCII-only set recognised by the C macro Py_ISSPACE().
    29	const ASCII_WHITESPACE = new Set([0x09, 0x0a, 0x0b, 0x0c, 0x0d, 0x20]);
    30	
    31	const DECIMAL_DIGIT = /\p{Nd}/u;
    32	
    33	// str.isprintable() is false for these categories; U+0020 is the one exception.
    34	const NON_PRINTABLE = /[\p{Cc}\p{Cf}\p{Cs}\p{Co}\p{Cn}\p{Zl}\p{Zp}\p{Zs}]/u;
    35	
    36	/** Equivalent of CPython's Py_UNICODE_ISSPACE(). */
    37	export function isPyWhitespace(codePoint) {
    38	  return PY_UNICODE_WHITESPACE.has(codePoint);
    39	}
    40	
    41	/**
    42	 * Whitespace as int(str) sees it - a hybrid of the two sets above, and a
    43	 * genuine CPython quirk rather than an approximation.
    44	 *
    45	 * int() first runs the text through _PyUnicode_TransformDecimalAndSpaceToASCII,
    46	 * which rewrites every *non-ASCII* Py_UNICODE_ISSPACE character to a plain
    47	 * space, and then parses the result with PyLong_FromString, which strips using
    48	 * the ASCII-only Py_ISSPACE. So U+0085 and U+00A0 pad an integer literal fine,
    49	 * while the ASCII separators U+001C..U+001F do not - int('\x1c12') raises even
    50	 * though '\x1c'.isspace() is True.
    51	 */
    52	export function isIntLiteralWhitespace(codePoint) {
    53	  return codePoint < 0x80
    54	    ? ASCII_WHITESPACE.has(codePoint)
    55	    : PY_UNICODE_WHITESPACE.has(codePoint);
    56	}
    57	
    58	function isDecimalDigit(codePoint) {
    59	  // Lone surrogates can never be Nd and String.fromCodePoint would produce an
    60	  // unpaired unit, so short-circuit them.
    61	  if (codePoint >= 0xd800 && codePoint <= 0xdfff) return false;
    62	  return DECIMAL_DIGIT.test(String.fromCodePoint(codePoint));
    63	}
    64	
    65	// Walking back more than this from a digit means the Unicode database grew a
    66	// run far larger than anything that has ever existed; bail out rather than spin.
    67	const MAX_RUN_SCAN = 4096;
    68	
    69	/**
    70	 * Value 0-9 of a Unicode decimal digit, or -1 if the codepoint is not one.
    71	 * Mirrors CPython's Py_UNICODE_TODECIMAL(), which is what int(str) uses.
    72	 *
    73	 * Unicode guarantees that decimal digits appear as blocks of exactly ten
    74	 * consecutive codepoints ordered 0..9. Several blocks can sit back-to-back
    75	 * (U+1D7CE..U+1D7FF is one unbroken run of fifty mathematical digits), so the
    76	 * digit's value is its offset from the start of the maximal run, modulo ten.
    77	 */
    78	export function decimalDigitValue(codePoint) {
    79	  if (codePoint >= 0x30 && codePoint <= 0x39) return codePoint - 0x30; // ASCII fast path
    80	  if (!isDecimalDigit(codePoint)) return -1;
    81	
    82	  let runStart = codePoint;
    83	  for (let steps = 0; steps < MAX_RUN_SCAN; steps += 1) {
    84	    if (runStart === 0 || !isDecimalDigit(runStart - 1)) break;
    85	    runStart -= 1;
    86	  }
    87	  return (codePoint - runStart) % 10;
    88	}
    89	
    90	/** Equivalent of Python's str.isprintable() for a single codepoint. */
    91	export function isPyPrintable(codePoint) {
    92	  if (codePoint === 0x20) return true;
    93	  if (codePoint >= 0xd800 && codePoint <= 0xdfff) return false; // Cs
    94	  return !NON_PRINTABLE.test(String.fromCodePoint(codePoint));
    95	}
    96	
    97	/** Iterate a JS string as codepoints, yielding lone surrogates as-is. */
    98	export function* codePoints(text) {
    99	  for (let i = 0; i < text.length; i += 1) {
   100	    const unit = text.charCodeAt(i);
   101	    if (unit >= 0xd800 && unit <= 0xdbff && i + 1 < text.length) {
   102	      const low = text.charCodeAt(i + 1);
   103	      if (low >= 0xdc00 && low <= 0xdfff) {
   104	        yield (unit - 0xd800) * 0x400 + (low - 0xdc00) + 0x10000;
   105	        i += 1;
   106	        continue;
   107	      }
   108	    }
   109	    yield unit;
   110	  }
   111	}

     1	/**
     2	 * CPython's int(str) conversion.
     3	 *
     4	 * argparse's `type=int` calls this, so the accepted grammar has to match
     5	 * exactly - including Unicode whitespace padding, underscore digit grouping,
     6	 * and non-ASCII decimal digits such as U+0665 ARABIC-INDIC DIGIT FIVE.
     7	 */
     8	
     9	import { PyValueError } from './exceptions.mjs';
    10	import { codePoints, decimalDigitValue, isIntLiteralWhitespace } from './unicode.mjs';
    11	import { pyStrRepr } from './repr.mjs';
    12	
    13	const UNDERSCORE = 0x5f;
    14	const PLUS = 0x2b;
    15	const MINUS = 0x2d;
    16	
    17	/**
    18	 * Parse a Python base-10 integer literal.
    19	 *
    20	 * Grammar, after stripping leading and trailing whitespace:
    21	 *   ['+' | '-'] digit (['_'] digit)*
    22	 * Note there is no whitespace allowed after the sign, and an underscore is
    23	 * only legal between two digits.
    24	 *
    25	 * @returns {bigint} the value, with arbitrary precision like Python's int
    26	 * @throws {PyValueError} if the text is not a valid literal
    27	 */
    28	export function pyInt(text) {
    29	  const invalid = () => new PyValueError(
    30	    `invalid literal for int() with base 10: ${pyStrRepr(text)}`,
    31	  );
    32	
    33	  const points = [...codePoints(text)];
    34	  let start = 0;
    35	  let end = points.length;
    36	  while (start < end && isIntLiteralWhitespace(points[start])) start += 1;
    37	  while (end > start && isIntLiteralWhitespace(points[end - 1])) end -= 1;
    38	
    39	  let negative = false;
    40	  if (start < end && (points[start] === PLUS || points[start] === MINUS)) {
    41	    negative = points[start] === MINUS;
    42	    start += 1;
    43	  }
    44	  if (start >= end) throw invalid();
    45	
    46	  let value = 0n;
    47	  let digitCount = 0;
    48	  let previousWasUnderscore = false;
    49	
    50	  for (let i = start; i < end; i += 1) {
    51	    const cp = points[i];
    52	
    53	    if (cp === UNDERSCORE) {
    54	      // Legal only between digits: never leading, doubled, or trailing.
    55	      if (digitCount === 0 || previousWasUnderscore) throw invalid();
    56	      previousWasUnderscore = true;
    57	      continue;
    58	    }
    59	
    60	    const digit = decimalDigitValue(cp);
    61	    if (digit < 0) throw invalid();
    62	    value = value * 10n + BigInt(digit);
    63	    digitCount += 1;
    64	    previousWasUnderscore = false;
    65	  }
    66	
    67	  if (digitCount === 0 || previousWasUnderscore) throw invalid();
    68	  return negative ? -value : value;
    69	}
    70	
    71	/** Python's `%` on integers: the result carries the sign of the divisor. */
    72	export function pyMod(value, modulus) {
    73	  const remainder = value % modulus;
    74	  if (remainder !== 0n && (remainder < 0n) !== (modulus < 0n)) {
    75	    return remainder + modulus;
    76	  }
    77	  return remainder;
    78	}

     9	/**
    10	 * Tag frames from the frozen script onto an exception raised beneath it, so
    11	 * the traceback reads from module scope down into the library.
    12	 *
    13	 * @param {number} line source line of the failing statement inside main()
    14	 * @param {() => T} body
    15	 * @returns {T}
    16	 * @template T
    17	 */
    18	export function atSourceLine(line, body) {
    19	  try {
    20	    return body();
    21	  } catch (err) {
    22	    if (err instanceof PyException) {
    23	      throw err.withOuterFrames([
    24	        { file: SCRIPT_FILE, line: MODULE_CALL_LINE, func: '<module>' },
    25	        { file: SCRIPT_FILE, line, func: 'main' },
    26	      ]);
    27	    }
    28	    throw err;
    29	  }
    30	}

    32	function renderOne(exception) {
    33	  let out = 'Traceback (most recent call last):\n';
    34	  for (const frame of exception.frames) {
    35	    out += `  File "${frame.file}", line ${frame.line}, in ${frame.func}\n`;
    36	  }
    37	  return `${out}${exception.pyType}: ${exception.pyMessage}\n`;
    38	}

    40	/**
    41	 * Full stderr text for an unhandled exception, including any implicitly
    42	 * chained predecessor.
    43	 */
    44	export function renderTraceback(exception) {
    45	  let out = '';
    46	  if (exception.during) {
    47	    out += renderOne(exception.during);
    48	    out += '\nDuring handling of the above exception, another exception occurred:\n\n';
    49	  }
    50	  out += renderOne(exception);
    51	  // The bootloader tags its banner with the process id.
    52	  out += `[PYI-${process.pid}:ERROR] Failed to execute script '${SCRIPT_NAME}' due to unhandled exception!\n`;
    53	  return out;

     1	/**
     2	 * Recovering argv exactly as CPython sees it.
     3	 *
     4	 * Node decodes process.argv as UTF-8 with lossy replacement, so a byte
     5	 * sequence like 0xFF arrives as U+FFFD and becomes indistinguishable from a
     6	 * genuine replacement character. CPython instead applies the `surrogateescape`
     7	 * handler, preserving the original bytes as lone surrogates - which is exactly
     8	 * why the reference build can go on to raise UnicodeEncodeError.
     9	 *
    10	 * On Linux the untouched bytes are still available from /proc/self/cmdline, so
    11	 * read them there and decode them ourselves.
    12	 */
    13	
    14	import { readFileSync } from 'node:fs';
    15	
    16	import { fsDecode } from '../python/codec.mjs';
    17	
    18	const CMDLINE = '/proc/self/cmdline';
    19	
    20	/** Split the NUL-delimited cmdline into per-argument byte arrays. */
    21	function readRawArguments() {
    22	  // 'latin1' maps each byte to the codepoint of the same value, giving us the
    23	  // raw bytes without going near typed arrays.
    24	  const raw = readFileSync(CMDLINE, 'latin1');
    25	  const text = raw.endsWith('\0') ? raw.slice(0, -1) : raw;
    26	  if (text === '') return [];
    27	  return text.split('\0').map((entry) => {
    28	    const bytes = [];
    29	    for (let i = 0; i < entry.length; i += 1) bytes.push(entry.charCodeAt(i));
    30	    return bytes;
    31	  });
    32	}
    33	
    34	/**
    35	 * The user-supplied arguments, decoded the way Python decodes them.
    36	 * Falls back to Node's own view if /proc is unavailable.
    37	 */
    38	export function getPythonArgv() {
    39	  const userCount = process.argv.length - 2;
    40	  if (userCount <= 0) return [];
    41	
    42	  try {
    43	    const entries = readRawArguments();
    44	    // The user's arguments are always the trailing entries: anything Node adds
    45	    // (the interpreter path, exec flags, the script path) comes first.
    46	    if (entries.length >= userCount) {
    47	      return entries.slice(entries.length - userCount).map(fsDecode);
    48	    }
    49	  } catch {
    50	    // Not Linux, or /proc not mounted - fall through.
    51	  }
    52	  return process.argv.slice(2);
    53	}

     1	/**
     2	 * Minimal model of the Python exception objects this program can raise.
     3	 *
     4	 * Each exception carries what the interpreter needs to render a traceback:
     5	 * the fully-qualified type name shown on the final line, the message body
     6	 * (which may span several lines), the stack frames, and an optional
     7	 * "during handling of the above exception" predecessor.

    10	export class PyException extends Error {
    11	  /**
    12	   * @param {string} pyType   name printed before the colon, e.g. "struct.error"
    13	   * @param {string} pyMessage text printed after the colon, may be multi-line
    14	   * @param {{file: string, line: number, func: string}[]} frames
    15	   */
    16	  constructor(pyType, pyMessage, frames = []) {
    17	    super(`${pyType}: ${pyMessage}`);
    18	    this.name = pyType;
    19	    this.pyType = pyType;
    20	    this.pyMessage = pyMessage;
    21	    this.frames = frames;
    22	    /** @type {PyException|null} predecessor for implicit exception chaining */
    23	    this.during = null;
    24	  }

    26	  /** Prepend caller frames so the traceback reads outermost-first. */
    27	  withOuterFrames(outer) {

    30	  }
    31	}

=================== python/repr.mjs
     1	/**
     2	 * Python 3 repr() for str and bytes.
     3	 *
     4	 * These drive both the program's stdout (print of a bytes object) and
     5	 * argparse's "invalid int value: %r" diagnostics, so they must be exact.
     6	 */
     7	
     8	import { codePoints, isPyPrintable } from './unicode.mjs';
     9	
    10	const BACKSLASH = 0x5c;
    11	const SINGLE_QUOTE = 0x27;
    12	const DOUBLE_QUOTE = 0x22;
    13	
    14	function hex(value, width) {
    15	  return value.toString(16).padStart(width, '0');
    16	}
    17	
    18	/**
    19	 * Python picks the quote character that needs the least escaping: a single
    20	 * quote normally, switching to a double quote only when the value contains a
    21	 * single quote but no double quote.
    22	 */
    23	function chooseQuote(hasSingle, hasDouble) {
    24	  return hasSingle && !hasDouble ? DOUBLE_QUOTE : SINGLE_QUOTE;
    25	}
    26	
    27	/** Python 3 `repr(some_str)`. */
    28	export function pyStrRepr(text) {
    29	  const points = [...codePoints(text)];
    30	  const quote = chooseQuote(
    31	    points.includes(SINGLE_QUOTE),
    32	    points.includes(DOUBLE_QUOTE),
    33	  );
    34	
    35	  let out = String.fromCharCode(quote);
    36	  for (const cp of points) {
    37	    if (cp === quote || cp === BACKSLASH) out += `\\${String.fromCharCode(cp)}`;
    38	    else if (cp === 0x09) out += '\\t';
    39	    else if (cp === 0x0a) out += '\\n';
    40	    else if (cp === 0x0d) out += '\\r';
    41	    else if (cp < 0x20 || cp === 0x7f) out += `\\x${hex(cp, 2)}`;
    42	    else if (cp < 0x7f) out += String.fromCharCode(cp);
    43	    else if (isPyPrintable(cp)) out += String.fromCodePoint(cp);
    44	    else if (cp < 0x100) out += `\\x${hex(cp, 2)}`;
    45	    else if (cp < 0x10000) out += `\\u${hex(cp, 4)}`;
    46	    else out += `\\U${hex(cp, 8)}`;
    47	  }
    48	  return out + String.fromCharCode(quote);
    49	}
    50	
    51	/**
    52	 * Python 3 `repr(some_bytes)`, given the bytes as an array of 0-255 integers.
    53	 * Only \t, \n, \r and \\ get short escapes; every other non-ASCII-printable
    54	 * byte becomes a two-digit lowercase \xHH.
    55	 */
    56	export function pyBytesRepr(bytes) {
    57	  const quote = chooseQuote(
    58	    bytes.includes(SINGLE_QUOTE),
    59	    bytes.includes(DOUBLE_QUOTE),
    60	  );
    61	
    62	  let out = `b${String.fromCharCode(quote)}`;
    63	  for (const byte of bytes) {
    64	    if (byte === quote || byte === BACKSLASH) out += `\\${String.fromCharCode(byte)}`;
    65	    else if (byte === 0x09) out += '\\t';
    66	    else if (byte === 0x0a) out += '\\n';
    67	    else if (byte === 0x0d) out += '\\r';
    68	    else if (byte < 0x20 || byte >= 0x7f) out += `\\x${hex(byte, 2)}`;
    69	    else out += String.fromCharCode(byte);
    70	  }
    71	  return out + String.fromCharCode(quote);
    72	}
