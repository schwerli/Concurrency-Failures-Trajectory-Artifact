import {
  bech32_create_checksum,
  bech32_verify_checksum,
} from './lib/bech32/index.mjs';
import { parseTest10Arguments, ParserExit } from './lib/cli/argparse.mjs';
import { parseCommaSeparatedIntegers } from './lib/data/integers.mjs';
import {
  formatPythonBoolean,
  formatPythonList,
} from './lib/format/python.mjs';

export function main(argv = process.argv) {
  const args = parseTest10Arguments(argv);
  const data = parseCommaSeparatedIntegers(args.b);
  const checksum = bech32_create_checksum(args.a, data);
  const combined = data.concat(checksum);
  const isValid = bech32_verify_checksum(args.a, combined);

  process.stdout.write(`${formatPythonList(checksum)}\n`);
  process.stdout.write(`${formatPythonBoolean(isValid)}\n`);
}

export function runCli(argv = process.argv) {
  try {
    main(argv);
  } catch (error) {
    if (error instanceof ParserExit) {
      if (error.stdout) {
        process.stdout.write(error.stdout);
      }

      if (error.stderr) {
        process.stderr.write(error.stderr);
      }

      process.exit(error.code);
    }

    throw error;
  }
}

function fileUrlToPath(moduleUrl) {
  return decodeURIComponent(new URL(moduleUrl).pathname);
}

function normalizePath(pathLike) {
  const rawPath = String(pathLike);
  const absolutePath = rawPath.startsWith('/') ? rawPath : `${process.cwd()}/${rawPath}`;
  const segments = absolutePath.split('/');
  const normalized = [];

  for (const segment of segments) {
    if (segment === '' || segment === '.') {
      continue;
    }

    if (segment === '..') {
      normalized.pop();
      continue;
    }

    normalized.push(segment);
  }

  return `/${normalized.join('/')}`;
}

if (process.argv[1] && normalizePath(fileUrlToPath(import.meta.url)) === normalizePath(process.argv[1])) {
  runCli(process.argv);
}
