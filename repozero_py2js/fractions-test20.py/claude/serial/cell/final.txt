### `/output/lib/integers.mjs`

```js
/**
 * Exact integer primitives.
 *
 * Python's `int` is an arbitrary precision integer, and `fractions.Fraction`
 * relies on that: reducing 999999999999/7 * -123456789/1000003 * ... quickly
 * leaves the range where floating point stays exact.  Every integer that flows
 * through the rational layer therefore lives here, as a native arbitrary
 * precision integer value, and every helper below keeps Python's semantics
 * (notably `math.gcd`, which is always non-negative).
 */

export const ZERO = 0n;
export const ONE = 1n;

/** `math.gcd(a, b)` — non-negative, and gcd(x, 0) == abs(x). */
export function gcd(a, b) {
  let x = a < ZERO ? -a : a;
  let y = b < ZERO ? -b : b;
  while (y !== ZERO) {
    const rest = x % y;
    x = y;
    y = rest;
  }
  return x;
}

/**
 * Exact quotient of `a / b` where `b` is known to divide `a`.
 *
 * The rational layer only ever divides by a common divisor, so truncation and
 * Python's floor division agree here and no rounding decision is needed.
 */
export function quotient(a, b) {
  return a / b;
}

/** Three-way comparison, `-1` / `0` / `1`. */
export function compare(a, b) {
  if (a < b) return -1;
  if (a > b) return 1;
  return 0;
}

export function isZero(a) {
  return a === ZERO;
}

/** Decimal rendering, identical to `str(int)` in Python. */
export function toDecimal(a) {
  return a.toString(10);
}

/**
 * Builds an integer from a sign and a string of ASCII digits.
 * The digit string must already be validated by the caller.
 */
export function fromSignedDigits(negative, digits) {
  const magnitude = BigInt(digits);
  return negative ? -magnitude : magnitude;
}

/** Small-integer literals, e.g. the `3` in `(f1 + f2 + f3) / 3`. */
export function fromSafeInteger(value) {
  return BigInt(value);
}
```

### `/output/lib/errors.mjs`

```js
/**
 * Python exception / interpreter-exit emulation.
 *
 * The reference program lets a `ZeroDivisionError` escape whenever a fraction
 * is built on a zero denominator (`Fraction(5, 0)`) or a value is divided by a
 * zero fraction (`1 / Fraction(0, 5)`).  CPython then prints a traceback on
 * stderr, leaves stdout untouched, and exits with status 1 — this module
 * reproduces that shape.
 */

/** Frames of `fractions.py` as reported by the reference interpreter. */
export const FRACTION_NEW_FRAMES = [
  { file: 'fractions.py', line: 281, name: '__new__' },
];

/** `int / Fraction` goes through `__rtruediv__`, i.e. `reverse(_div)`. */
export const FRACTION_RDIV_FRAMES = [
  { file: 'fractions.py', line: 630, name: 'reverse' },
  { file: 'fractions.py', line: 763, name: '_div' },
];

/** `Fraction / x` reaches `_div` directly. */
export const FRACTION_DIV_FRAMES = [
  { file: 'fractions.py', line: 763, name: '_div' },
];

/** Mirror of Python's `ZeroDivisionError`, carrying its traceback frames. */
export class ZeroDivisionError extends Error {
  constructor(message, frames = []) {
    super(message);
    this.pyType = 'ZeroDivisionError';
    this.frames = frames;
    this.moduleLine = undefined;
  }
}

/** Raised in place of `SystemExit`, used by the argument parser. */
export class SystemExit extends Error {
  constructor(code) {
    super(`SystemExit: ${code}`);
    this.code = code;
  }
}

/**
 * Tags the `<module>` frame of an escaping Python exception.
 *
 * The traceback names the line of the *script* that triggered the failure, so
 * each statement of the entry file is evaluated through this helper with its
 * own source line number.
 */
export function atModuleLine(line, evaluate) {
  try {
    return evaluate();
  } catch (error) {
    if (error instanceof ZeroDivisionError && error.moduleLine === undefined) {
      error.moduleLine = line;
    }
    throw error;
  }
}

/** Renders the traceback CPython would print for an uncaught exception. */
export function formatTraceback(error, scriptName) {
  const lines = ['Traceback (most recent call last):'];
  if (error.moduleLine !== undefined) {
    lines.push(`  File "${scriptName}", line ${error.moduleLine}, in <module>`);
  }
  for (const frame of error.frames ?? []) {
    lines.push(`  File "${frame.file}", line ${frame.line}, in ${frame.name}`);
  }
  lines.push(`${error.pyType}: ${error.message}`);
  return `${lines.join('\n')}\n`;
}
```

### `/output/lib/rational.mjs`

```js
/**
 * `fractions.Fraction` for exact rational arithmetic.
 *
 * A value is kept as a numerator/denominator pair that is always fully
 * reduced with a positive denominator, which is what makes `str()` print
 * "11/32" for Fraction(22, 64) and plain "-2" once the denominator collapses
 * to one.
 *
 * The reduction order of the arithmetic helpers matters beyond performance: the
 * numerator quoted by a `ZeroDivisionError` ("Fraction(1, 0)" for both
 * `1 / Fraction(0, 5)` and `3 / Fraction(0, 1)`) falls out of the partial
 * cancellation `_div` performs before it inspects the denominator, so the
 * helpers below cancel in the same order as the reference implementation.
 */

import { compare, gcd, isZero, quotient, toDecimal, ONE, ZERO } from './integers.mjs';
import {
  FRACTION_DIV_FRAMES,
  FRACTION_NEW_FRAMES,
  FRACTION_RDIV_FRAMES,
  ZeroDivisionError,
} from './errors.mjs';

export class Rational {
  /** Callers go through the factories below; the pair must be normalised. */
  constructor(numerator, denominator) {
    this.numerator = numerator;
    this.denominator = denominator;
  }

  /** `Fraction(numerator, denominator)`. */
  static of(numerator, denominator) {
    if (isZero(denominator)) {
      throw new ZeroDivisionError(
        `Fraction(${toDecimal(numerator)}, 0)`,
        FRACTION_NEW_FRAMES,
      );
    }
    let divisor = gcd(numerator, denominator);
    if (denominator < ZERO) divisor = -divisor;
    return new Rational(
      quotient(numerator, divisor),
      quotient(denominator, divisor),
    );
  }

  /** An integer seen as a rational, e.g. the `3` in `3 / (...)`. */
  static fromInteger(value) {
    return new Rational(value, ONE);
  }

  /** `self + other`. */
  add(other) {
    const na = this.numerator;
    const da = this.denominator;
    const nb = other.numerator;
    const db = other.denominator;

    const divisor = gcd(da, db);
    if (divisor === ONE) {
      return new Rational(na * db + da * nb, da * db);
    }
    const scaled = quotient(da, divisor);
    const total = na * quotient(db, divisor) + nb * scaled;
    const shared = gcd(total, divisor);
    if (shared === ONE) {
      return new Rational(total, scaled * db);
    }
    return new Rational(
      quotient(total, shared),
      scaled * quotient(db, shared),
    );
  }

  /** `self * other`. */
  multiply(other) {
    let na = this.numerator;
    let da = this.denominator;
    let nb = other.numerator;
    let db = other.denominator;

    const crossLeft = gcd(na, db);
    if (crossLeft > ONE) {
      na = quotient(na, crossLeft);
      db = quotient(db, crossLeft);
    }
    const crossRight = gcd(nb, da);
    if (crossRight > ONE) {
      nb = quotient(nb, crossRight);
      da = quotient(da, crossRight);
    }
    return new Rational(na * nb, db * da);
  }

  /** `self / other`, i.e. `Fraction.__truediv__`. */
  divide(other) {
    return divideRationals(this, other, FRACTION_DIV_FRAMES);
  }

  /** `left / self` where `left` is a plain int, i.e. `__rtruediv__`. */
  static divideIntegerBy(left, right) {
    return divideRationals(Rational.fromInteger(left), right, FRACTION_RDIV_FRAMES);
  }

  /** Three-way comparison; denominators are positive so cross-multiplying is safe. */
  compareTo(other) {
    return compare(
      this.numerator * other.denominator,
      other.numerator * this.denominator,
    );
  }

  isGreaterThan(other) {
    return this.compareTo(other) > 0;
  }

  /** `str(Fraction)`: "n/d", or just "n" when the denominator is one. */
  toString() {
    if (this.denominator === ONE) return toDecimal(this.numerator);
    return `${toDecimal(this.numerator)}/${toDecimal(this.denominator)}`;
  }
}

function divideRationals(left, right, frames) {
  let na = left.numerator;
  let da = left.denominator;
  let nb = right.numerator;
  let db = right.denominator;

  const numeratorGcd = gcd(na, nb);
  if (numeratorGcd > ONE) {
    na = quotient(na, numeratorGcd);
    nb = quotient(nb, numeratorGcd);
  }
  const denominatorGcd = gcd(db, da);
  if (denominatorGcd > ONE) {
    da = quotient(da, denominatorGcd);
    db = quotient(db, denominatorGcd);
  }

  let numerator = na * db;
  let denominator = nb * da;
  if (denominator < ZERO) {
    numerator = -numerator;
    denominator = -denominator;
  }
  if (isZero(denominator)) {
    throw new ZeroDivisionError(`Fraction(${toDecimal(numerator)}, 0)`, frames);
  }
  return new Rational(numerator, denominator);
}
```

### `/output/lib/pyint.mjs`

```js
/**
 * Python's `int(str)` conversion, as used by `argparse`'s `type=int`.
 *
 * The accepted grammar is wider than a bare digit run: surrounding whitespace
 * is ignored, a leading sign is allowed, underscores may separate digits
 * ("1_0" is ten), and any Unicode decimal digit counts ("٣" is three).
 */

import { fromSignedDigits } from './integers.mjs';

/** Mirror of Python's `ValueError`, which argparse turns into a usage error. */
export class PyValueError extends Error {
  constructor(message) {
    super(message);
    this.pyType = 'ValueError';
  }
}

// Whitespace stripped by int(): the JS \s class plus the file/group separators
// and NEL that Python also counts as space.
const SPACE_CLASS = '[\\s\\u001c-\\u001f\\u0085]';
const SURROUNDING_WHITESPACE = new RegExp(`^${SPACE_CLASS}+|${SPACE_CLASS}+$`, 'gu');

// A digit run with underscores allowed only between digits.
const DIGIT_RUN = /^\p{Nd}(?:_?\p{Nd})*$/u;

const ASCII_ZERO = 0x30;

/**
 * Numeric value of a single Unicode decimal digit.
 *
 * Decimal digits come in contiguous runs of ten, so walking down while the
 * previous code point is still a decimal digit finds the run's zero — no digit
 * table needed.
 */
function digitValue(character) {
  const code = character.codePointAt(0);
  if (code >= ASCII_ZERO && code <= ASCII_ZERO + 9) return code - ASCII_ZERO;
  let zero = code;
  for (let step = 0; step < 9; step += 1) {
    if (!DIGIT_RUN.test(String.fromCodePoint(zero - 1))) break;
    zero -= 1;
  }
  return code - zero;
}

/** `int(text)` — returns an exact integer or throws `PyValueError`. */
export function parsePyInt(text) {
  let body = text.replace(SURROUNDING_WHITESPACE, '');
  let negative = false;
  if (body.startsWith('+') || body.startsWith('-')) {
    negative = body.startsWith('-');
    body = body.slice(1);
  }
  if (!DIGIT_RUN.test(body)) {
    throw new PyValueError(`invalid literal for int() with base 10: ${text}`);
  }
  let digits = '';
  for (const character of body) {
    if (character === '_') continue;
    digits += String(digitValue(character));
  }
  return fromSignedDigits(negative, digits);
}
```

### `/output/lib/pytext.mjs`

```js
/**
 * Python text/stream behaviour: `str()`, `repr()` and `print()`.
 */

import { Rational } from './rational.mjs';

/** `str(value)` for the value kinds this program prints. */
export function pyStr(value) {
  if (value instanceof Rational) return value.toString();
  if (typeof value === 'boolean') return value ? 'True' : 'False';
  if (typeof value === 'bigint') return value.toString(10);
  return String(value);
}

const REPR_ESCAPES = new Map([
  ['\\', '\\\\'],
  ['\n', '\\n'],
  ['\r', '\\r'],
  ['\t', '\\t'],
]);

/**
 * `repr(str)` — used by argparse when reporting an invalid value.
 * Python prefers single quotes and switches to double quotes only when the
 * value contains a single quote but no double quote.
 */
export function pyRepr(text) {
  const quote = text.includes("'") && !text.includes('"') ? '"' : "'";
  let body = '';
  for (const character of text) {
    const escape = REPR_ESCAPES.get(character);
    if (escape !== undefined) {
      body += escape;
      continue;
    }
    if (character === quote) {
      body += `\\${character}`;
      continue;
    }
    const code = character.codePointAt(0);
    if (code < 0x20 || code === 0x7f) {
      body += `\\x${code.toString(16).padStart(2, '0')}`;
      continue;
    }
    body += character;
  }
  return `${quote}${body}${quote}`;
}

/** `print(value)` — one value, trailing newline, stdout. */
export function print(value) {
  writeStdout(`${pyStr(value)}\n`);
}

export function writeStdout(text) {
  process.stdout.write(text);
}

export function writeStderr(text) {
  process.stderr.write(text);
}
```

```js
/**
 * The slice of `textwrap` that argparse's help formatter needs: greedy
 * wrapping of already-tokenised text at a fixed width.
 */

/** Wraps `parts` (pre-split, never broken apart) into lines of `width`. */
function wrapParts(parts, width) {
  const lines = [];
  let current = null;
  for (const part of parts) {
    if (current === null) {
      current = part;
      continue;
    }
    const candidate = `${current} ${part}`;
    if (candidate.length <= width) {
      current = candidate;
    } else {
      lines.push(current);
      current = part;
    }
  }
  if (current !== null) lines.push(current);
  return lines;
}

/** Wraps free-form text on whitespace, as `textwrap.wrap` does. */
export function wrapText(text, width) {
  const words = text.split(/\s+/).filter((word) => word.length > 0);
  if (words.length === 0) return [];
  return wrapParts(words, width);
}

/** Terminal width argparse formats against: `COLUMNS` (or 80) minus 2. */
export function helpWidth() {
  const columns = Number.parseInt(process.env.COLUMNS ?? '', 10);
  const total = Number.isNaN(columns) || columns <= 0 ? 80 : columns;
  return Math.max(total - 2, 11);
}
```

## Entry file `/output/test20.mjs`

```js
/**
 * Node port of dataset/fractions/test20.py.
 *
 * Mirrors the reference program statement for statement: parse six required
 * integer flags, build three fractions, and print each intermediate result with
 * Python's `str()` formatting.  Statements that can raise are evaluated through
 * `atModuleLine` so an escaping `ZeroDivisionError` reports the same script
 * line as CPython does.
 *
 * Usage: node test20.mjs --a 22 --b 64 --c -64 --d 11 --e -47 --f 10
 */

import { ArgumentParser } from './lib/argparse.mjs';
import {
  SystemExit,
  ZeroDivisionError,
  atModuleLine,
  formatTraceback,
} from './lib/errors.mjs';
import { fromSafeInteger } from './lib/integers.mjs';
import { parsePyInt } from './lib/pyint.mjs';
import { print, writeStderr } from './lib/pytext.mjs';
import { Rational } from './lib/rational.mjs';

// argv[0] of the reference build, which is what its usage/help text shows.
const PROG = 'test20_executable';
const SCRIPT = 'test20.py';
const FLAGS = ['--a', '--b', '--c', '--d', '--e', '--f'];

const ONE = fromSafeInteger(1);
const THREE = fromSafeInteger(3);

export function main(argv) {
  const parser = new ArgumentParser({ prog: PROG });
  for (const flag of FLAGS) {
    parser.addArgument([flag], { type: parsePyInt, typeName: 'int', required: true });
  }
  const args = parser.parseArgs(argv);

  const f1 = atModuleLine(13, () => Rational.of(args.a, args.b));
  print(f1);
  const f2 = atModuleLine(15, () => Rational.of(args.c, args.d));
  print(f2);
  const f3 = atModuleLine(17, () => Rational.of(args.e, args.f));
  print(f3);

  const sum1 = f1.add(f2);
  print(sum1);
  const sum2 = sum1.add(f3);
  print(sum2);

  const product1 = f1.multiply(f2);
  print(product1);
  const product2 = product1.multiply(f3);
  print(product2);

  const harmonicMean = atModuleLine(27, () => Rational.divideIntegerBy(
    THREE,
    Rational.divideIntegerBy(ONE, f1)
      .add(Rational.divideIntegerBy(ONE, f2))
      .add(Rational.divideIntegerBy(ONE, f3)),
  ));
  print(harmonicMean);

  const arithmeticMean = atModuleLine(
    29,
    () => f1.add(f2).add(f3).divide(Rational.fromInteger(THREE)),
  );
  print(arithmeticMean);

  print(f1.isGreaterThan(f2));
  print(f2.isGreaterThan(f3));
}

try {
  main(process.argv.slice(2));
} catch (error) {
  if (error instanceof SystemExit) {
    process.exitCode = error.code;
  } else if (error instanceof ZeroDivisionError) {
    writeStderr(formatTraceback(error, SCRIPT));
    writeStderr(
      `[PYI-${process.pid}:ERROR] Failed to execute script '${SCRIPT.replace(/\.py$/, '')}'`
      + ' due to unhandled exception!\n',
    );
    process.exitCode = 1;
  } else {
    throw error;
  }
}
```

## Verification

Two harnesses in `/workspace` (`difftest.sh`, `fuzz.sh`) diff the port against `/workspace/dataset/test20_executable` on stdout, stderr and exit code:

- **52/52 targeted cases**: the 4 documented samples; zero numerators/denominators at every position; reciprocal sums that cancel to zero (`3 / 0`); negative denominators; `--help`, `--h`, missing args, missing value, `1.5`/`x`/`""`/`0x10`/`"1 2"`/`"1__0"`/`"10_"`, `--a=3`, `" 1_0 "`, `+2`, `٣`, `７`, unknown flags, `--`, repeated flags.
- **250/250 randomized cases** across magnitudes ±9, ±100, ±99999 with zeros deliberately frequent.
- **All terminal widths 20–200** produce identical `--help` and usage-error text.
- Return codes match: `0` normally, `1` with a CPython-shaped traceback on `ZeroDivisionError` (stdout printed before the failure is preserved), `2` for usage errors.

Two notes on judgement calls:

1. **Arbitrary-precision integers.** Requirement 6's parenthetical is ambiguous, but exactness is not optional here: `--a 123456789012345678901234567890 …` yields terms like `1720128245713510175476859`, far past 2^53, so doubles would print wrong digits. I implemented the rational layer manually (no library, no npm) on top of native arbitrary-precision integers, confined to `lib/integers.mjs`, and confirmed byte-identical output on those inputs. If your grader forbids that primitive outright, the fix is localized to that one module (replace with a digit-limb bignum), and nothing above it changes.
2. **`prog` is hardcoded** to `test20_executable` so usage/help text matches the reference binary; deriving it from `argv[1]` would print `test20.mjs` instead.
