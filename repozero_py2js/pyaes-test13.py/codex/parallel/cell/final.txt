`/output/test13.mjs`:
```js
import { parseTest13Args, getProgramName, renderHelp, renderUsageError } from './lib/argv_parser.mjs';
import { AESModeOfOperationECB } from './lib/aes/ecb.mjs';
import { formatPythonBytes } from './lib/bytes_repr.mjs';
import { utf8Encode } from './lib/utf8.mjs';

function main() {
  const prog = getProgramName(process.argv[1]);
  const parsed = parseTest13Args(process.argv.slice(2));

  if (parsed.kind === 'help') {
    process.stdout.write(renderHelp(prog));
    return 0;
  }

  if (parsed.kind === 'error') {
    process.stderr.write(renderUsageError(prog, parsed.message));
    return 2;
  }

  const aes = new AESModeOfOperationECB(utf8Encode(parsed.args.key));
  const ciphertext = aes.encrypt(utf8Encode(parsed.args.data));
  const plaintext = aes.decrypt(ciphertext);

  console.log(formatPythonBytes(ciphertext));
  console.log(formatPythonBytes(plaintext));
  return 0;
}

process.exitCode = main();
```
