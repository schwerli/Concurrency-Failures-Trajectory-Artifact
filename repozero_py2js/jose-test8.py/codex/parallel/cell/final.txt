**Entry File**
[test8.mjs](/output/test8.mjs:1)
```js
import {
  CliHelpRequested,
  CliUsageError,
  encodeJwt,
  formatHelp,
  formatUsage,
  getProgramName,
  getUnverifiedClaims,
  getUnverifiedHeader,
  parseTest8Arguments,
  toPythonRepr,
} from './lib/index.mjs';

const programName = getProgramName(process.argv[1]);

async function main() {
  const args = parseTest8Arguments(process.argv.slice(2));
  const secret = args.secret;
  const payload = { data: args.payload };

  const token = await encodeJwt(payload, secret);
  console.log(token);
  console.log(toPythonRepr(getUnverifiedClaims(token)));
  console.log(toPythonRepr(getUnverifiedHeader(token)));
}

try {
  await main();
} catch (error) {
  if (error instanceof CliHelpRequested) {
    process.stdout.write(formatHelp(programName));
    process.exit(0);
  }

  if (error instanceof CliUsageError) {
    process.stderr.write(
      `${formatUsage(programName)}\n${programName}: error: ${error.message}\n`,
    );
    process.exit(2);
  }

  throw error;
}
```
