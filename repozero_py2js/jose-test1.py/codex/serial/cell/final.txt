**Entry File**
[test1.mjs](/output/test1.mjs)

```js
import { parseCommandLine } from './lib/cli/parser.mjs';
import { writeStderrLine, writeStdoutLine } from './lib/io/write.mjs';
import { encodeToken } from './lib/jwt/encode-token.mjs';

const SECRET = 'my_secret_key';

export function main(argv = process.argv) {
  const parseResult = parseCommandLine(argv);

  if (parseResult.kind === 'help') {
    writeStdoutLine(parseResult.text);
    return parseResult.exitCode;
  }

  if (parseResult.kind === 'error') {
    writeStderrLine(parseResult.text);
    return parseResult.exitCode;
  }

  const token = encodeToken({ data: parseResult.payload }, SECRET);
  writeStdoutLine(token);
  return 0;
}

process.exitCode = main();
```
