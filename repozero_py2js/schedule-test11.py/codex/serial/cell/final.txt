```js
import { parseArguments } from './lib/cli/parse-args.mjs';
import { printLine } from './lib/runtime/print.mjs';
import { repeatCount } from './lib/runtime/repeat.mjs';
import * as schedule from './lib/schedule/index.mjs';

export function main(argv = process.argv.slice(2), programName = 'test11_executable', io = { stdout: process.stdout, stderr: process.stderr }) {
  const parsed = parseArguments(argv, programName, io);

  if (parsed.kind === 'exit') {
    process.exitCode = parsed.code;
    return;
  }

  const args = parsed.values;

  let jobsCreated = 0n;

  repeatCount(args.a, () => {
    schedule.every(1).seconds.tag('group1').do(() => null);
    jobsCreated += 1n;
  });
  printLine(jobsCreated);

  repeatCount(args.b, () => {
    schedule.every(1).minutes.tag('group2').do(() => null);
    jobsCreated += 1n;
  });
  printLine(jobsCreated);

  const group1Jobs = schedule.getJobs('group1');
  printLine(group1Jobs.length);

  const group2Jobs = schedule.getJobs('group2');
  printLine(group2Jobs.length);

  const allJobs = schedule.getJobs();
  printLine(allJobs.length);

  schedule.clear('group1');
  printLine(schedule.getJobs().length);

  schedule.clear('group2');
  printLine(schedule.getJobs().length);

  repeatCount(args.a + args.b, () => {
    schedule.every(1).seconds.do(() => null);
  });
  printLine(schedule.getJobs().length);
}

main();
```
