import { KeyAndValueDuplicationError, ValueDuplicationError, formatBidictError } from './lib/bidictErrors.mjs';
import { CliSignal } from './lib/cliSignal.mjs';
import { runTest16 } from './lib/runTest16.mjs';

function getProgramName(scriptPath) {
  if (typeof scriptPath !== 'string' || scriptPath.length === 0) {
    return 'test16.mjs';
  }

  const segments = scriptPath.split(/[\\/]/);
  return segments[segments.length - 1] || 'test16.mjs';
}

function flushStdoutBuffer(stdoutBuffer) {
  if (stdoutBuffer.length > 0) {
    process.stdout.write(stdoutBuffer.join(''));
  }
}

function writeSignal(signal) {
  const target = signal.stream === 'stdout' ? process.stdout : process.stderr;
  target.write(signal.text);
}

function main() {
  const stdoutBuffer = [];
  const programName = getProgramName(process.argv[1]);

  try {
    runTest16(process.argv.slice(2), programName, stdoutBuffer);
    flushStdoutBuffer(stdoutBuffer);
    return 0;
  } catch (error) {
    if (error instanceof CliSignal) {
      writeSignal(error);
      return error.code;
    }

    if (error instanceof ValueDuplicationError || error instanceof KeyAndValueDuplicationError) {
      process.stderr.write(`${formatBidictError(error)}\n`);
      flushStdoutBuffer(stdoutBuffer);
      return 1;
    }

    throw error;
  }
}

const exitCode = main();
if (exitCode !== 0) {
  process.exitCode = exitCode;
}
