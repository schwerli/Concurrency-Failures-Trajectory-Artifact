The implementation is pure ESM, uses only local `.mjs` imports, manually parses `process.argv`, and reproduces the Python executable’s successful output format as `b'...'`. I verified representative CLI/error cases plus a wider random sample against `/workspace/dataset/test3_executable`.

**Entry**
```js
import { encodeCanonicalJson } from './lib/canonical_json.mjs';
import { parseCommandLine, getProgramName } from './lib/cli.mjs';
import { formatPythonBytesLiteral } from './lib/python_bytes.mjs';

const programName = getProgramName(process.argv);
const parseResult = parseCommandLine(process.argv.slice(2), programName);

if (parseResult.kind === 'help') {
  process.stdout.write(parseResult.text);
  process.exit(0);
}

if (parseResult.kind === 'error') {
  process.stderr.write(parseResult.text);
  process.exit(parseResult.exitCode);
}

try {
  const result = encodeCanonicalJson(parseResult.a);
  process.stdout.write(`${formatPythonBytesLiteral(result)}\n`);
} catch (error) {
  if (error instanceof Error) {
    process.stderr.write(`${error.name}: ${error.message}\n`);
  } else {
    process.stderr.write(`${String(error)}\n`);
  }

  process.exit(1);
}
```
