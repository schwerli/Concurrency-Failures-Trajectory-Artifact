import { getProgramName, parseCommandLine } from './lib/cli.mjs';
import { parsePythonLiteral } from './lib/python_literal.mjs';
import { dumpYaml } from './lib/yaml_dump.mjs';

export function main(argv = process.argv) {
  const programName = getProgramName(argv[1]);
  const parsed = parseCommandLine(argv.slice(2), programName);

  if (parsed.kind === 'help') {
    process.stdout.write(parsed.text);
    return 0;
  }

  if (parsed.kind === 'error') {
    process.stderr.write(parsed.text);
    return parsed.exitCode;
  }

  const value = parsePythonLiteral(parsed.args.a);
  const result = dumpYaml(value);
  process.stdout.write(`${result}\n`);
  return 0;
}

const exitCode = (() => {
  try {
    return main(process.argv);
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    process.stderr.write(`${message}\n`);
    return 1;
  }
})();

if (exitCode !== 0) {
  process.exit(exitCode);
}
