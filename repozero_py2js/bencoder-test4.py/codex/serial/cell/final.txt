```js
import { CliExit, decode, ensureAscii, formatPythonValue, parseArgs } from './lib/index.mjs';
import { formatPythonTraceback } from './lib/shared/named-error.mjs';

function main(argv = process.argv) {
  const { a } = parseArgs(argv);
  const result = decode(ensureAscii(a));
  console.log(formatPythonValue(result));
}

try {
  main(process.argv);
} catch (error) {
  if (error instanceof CliExit) {
    if (error.stdout) {
      process.stdout.write(error.stdout);
    }

    if (error.stderr) {
      process.stderr.write(error.stderr);
    }

    process.exitCode = error.exitCode;
  } else if (error && error.isPythonStyleError) {
    if (error.stdout) {
      process.stdout.write(error.stdout);
    }

    process.stderr.write(`${formatPythonTraceback(error)}\n`);
    process.exitCode = 1;
  } else {
    throw error;
  }
}
```
