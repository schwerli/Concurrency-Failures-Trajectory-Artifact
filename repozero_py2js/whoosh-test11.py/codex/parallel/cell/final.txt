import { parseCommandLine } from './lib/cli/parse_args.mjs';
import { mkdtemp } from './lib/temp/tempfile.mjs';
import { Schema, TEXT } from './lib/whoosh/fields.mjs';
import { createIn } from './lib/whoosh/index.mjs';
import { Every } from './lib/whoosh/query.mjs';

const args = parseCommandLine(process.argv);

const schema = Schema({
  title: TEXT({ stored: true }),
  content: TEXT,
});

const tempDir = mkdtemp();
const index = createIn(tempDir, schema);
const writer = index.writer();

writer.addDocument({ title: args.a, content: args.b });
writer.addDocument({ title: args.b, content: args.c });
writer.addDocument({ title: args.c, content: args.a });
writer.commit();

const searcher = index.searcher();
const everyQuery = new Every();
const results = searcher.search(everyQuery);

console.log(results.length);
