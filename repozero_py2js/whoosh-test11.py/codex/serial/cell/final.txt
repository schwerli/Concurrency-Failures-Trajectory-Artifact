```js
import { parseTest11Args } from './lib/cli/argparse.mjs';
import { mkdtemp } from './lib/tempfile.mjs';
import { Schema, TEXT } from './lib/whoosh/fields.mjs';
import { create_in } from './lib/whoosh/index.mjs';
import { Every } from './lib/whoosh/query.mjs';

const args = parseTest11Args();

const schema = new Schema({
  title: TEXT({ stored: true }),
  content: TEXT(),
});
const temp_dir = mkdtemp();
const ix = create_in(temp_dir, schema);
const writer = ix.writer();

writer.add_document({ title: args.a, content: args.b });
writer.add_document({ title: args.b, content: args.c });
writer.add_document({ title: args.c, content: args.a });
writer.commit();

const searcher = ix.searcher();
const every_query = new Every();
const results = searcher.search(every_query);

console.log(results.length);
```
