## Entry file — `/output/test11.mjs`

```js
import { ArgumentParser } from './lib/support/argparse.mjs';
import { mkdtemp } from './lib/support/tempfile.mjs';
import { Schema, TEXT, ID } from './lib/whoosh/fields.mjs';
import { createIn } from './lib/whoosh/index.mjs';
import { QueryParser } from './lib/whoosh/qparser.mjs';
import { Every } from './lib/whoosh/query.mjs';

const parser = new ArgumentParser();
parser.addArgument('--a', { required: true });
parser.addArgument('--b', { required: true });
parser.addArgument('--c', { required: true });
const args = parser.parseArgs();

const schema = new Schema({ title: TEXT({ stored: true }), content: TEXT }); // Create schema
const tempDir = mkdtemp();                                  // Create temp dir
const ix = createIn(tempDir, schema);                       // Create index
const writer = ix.writer();                                 // Create writer
writer.add_document({ title: args.a, content: args.b });    // Add first document
writer.add_document({ title: args.b, content: args.c });    // Add second document
writer.add_document({ title: args.c, content: args.a });    // Add third document
writer.commit();                                            // Commit
const searcher = ix.searcher();                             // Create searcher
const everyQuery = new Every();                             // Create query to match all documents
const results = searcher.search(everyQuery);                // Search all documents
console.log(results.length());                              // Get total document count
```
