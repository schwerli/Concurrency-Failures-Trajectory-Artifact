import { CliAbort, DeepDiff, extract, parseCommandLine, pythonStr } from './lib/index.mjs';

function printPython(value) {
  console.log(pythonStr(value));
}

function main(argv) {
  const args = parseCommandLine(argv);

  const obj = {
    users: [
      { name: args.a, age: args.b },
      { name: `${args.a}2`, age: args.b + 1n },
    ],
  };

  const r1 = DeepDiff(obj, obj);
  printPython(r1);

  const r2 = extract(obj, "root['users'][0]['name']");
  printPython(r2);

  const r3 = extract(obj, "root['users'][1]['age']");
  printPython(r3);

  const r4 = DeepDiff(obj.users[0], obj.users[1]);
  printPython(r4);

  const r5 = extract(obj, "root['users'][0]");
  printPython(r5);

  const r6 = DeepDiff(obj.users, []);
  printPython(r6);

  const r7 = DeepDiff(obj, { users: [] });
  printPython(r7);

  const r8 = extract(obj, "root['users']");
  printPython(r8);
}

try {
  main(process.argv);
} catch (error) {
  if (error instanceof CliAbort) {
    const stream = error.outputChannel === 'stdout' ? process.stdout : process.stderr;
    stream.write(error.message);
    process.exitCode = error.exitCode;
  } else {
    throw error;
  }
}
