===== bech32/charset.mjs =====
     1	/**
     2	 * The Bech32 character set and Python's string-indexing semantics over it.
     3	 *
     4	 * `bech32_encode` looks up `CHARSET[d]` for every value it encodes, including
     5	 * the caller-supplied witness version, so out-of-range versions surface as the
     6	 * same `IndexError` CPython raises.
     7	 */
     8	
     9	import { PythonIndexError } from '../py/errors.mjs';
    10	
    11	export const CHARSET = 'qpzry9x8gf2tvdw0s3jn54khce6mua7l';
    12	
    13	const LENGTH = BigInt(CHARSET.length);
    14	// CPython converts an index through `Py_ssize_t` before bounds-checking it.
    15	const SSIZE_MAX = (1n << 63n) - 1n;
    16	const SSIZE_MIN = -(1n << 63n);
    17	
    18	/** `CHARSET[index]` with CPython's negative indexing and error messages. */
    19	export function charsetAt(index) {
    20	  if (index > SSIZE_MAX || index < SSIZE_MIN) {
    21	    throw new PythonIndexError("cannot fit 'int' into an index-sized integer");
    22	  }
    23	  const resolved = index < 0n ? index + LENGTH : index;
    24	  if (resolved < 0n || resolved >= LENGTH) {
    25	    throw new PythonIndexError('string index out of range');
    26	  }
    27	  return CHARSET[Number(resolved)];
    28	}
    29	
    30	/** `CHARSET.find(character)`: the value of a data character, or -1. */
    31	export function charsetIndexOf(character) {
    32	  return CHARSET.indexOf(character);
    33	}
===== bech32/checksum.mjs =====
     1	/**
     2	 * The BCH checksum machinery behind Bech32.
     3	 *
     4	 * Every value is a BigInt so the arithmetic matches Python's unbounded integers:
     5	 * a witness version outside 0..31 still flows through `bech32_polymod`, where
     6	 * two's-complement XOR over negative values must agree bit for bit with CPython
     7	 * before the result is rejected downstream.
     8	 */
     9	
    10	const GENERATOR = [0x3b6a57b2n, 0x26508e6dn, 0x1ea119fan, 0x3d4233ddn, 0x2a1462b3n];
    11	
    12	const LOW_25_BITS = 0x1ffffffn;
    13	const FIVE_BITS = 31n;
    14	
    15	/** `bech32_polymod`. */
    16	export function bech32Polymod(values) {
    17	  let checksum = 1n;
    18	  for (const value of values) {
    19	    const top = checksum >> 25n;
    20	    checksum = ((checksum & LOW_25_BITS) << 5n) ^ value;
    21	    for (let bit = 0; bit < 5; bit += 1) {
    22	      if ((top >> BigInt(bit)) & 1n) checksum ^= GENERATOR[bit];
    23	    }
    24	  }
    25	  return checksum;
    26	}
    27	
    28	/** `bech32_hrp_expand`: high bits of the HRP, a zero, then the low bits. */
    29	export function bech32HrpExpand(hrp) {
    30	  const high = [];
    31	  const low = [];
    32	  for (const character of hrp) {
    33	    const codePoint = BigInt(character.codePointAt(0));
    34	    high.push(codePoint >> 5n);
    35	    low.push(codePoint & FIVE_BITS);
    36	  }
    37	  return [...high, 0n, ...low];
    38	}
    39	
    40	/** `bech32_verify_checksum`. */
    41	export function bech32VerifyChecksum(hrp, data) {
    42	  return bech32Polymod([...bech32HrpExpand(hrp), ...data]) === 1n;
    43	}
    44	
    45	/** `bech32_create_checksum`: the six trailing data values. */
    46	export function bech32CreateChecksum(hrp, data) {
    47	  const values = [...bech32HrpExpand(hrp), ...data, 0n, 0n, 0n, 0n, 0n, 0n];
    48	  const polymod = bech32Polymod(values) ^ 1n;
    49	  const checksum = [];
    50	  for (let position = 0; position < 6; position += 1) {
    51	    checksum.push((polymod >> (5n * BigInt(5 - position))) & FIVE_BITS);
    52	  }
    53	  return checksum;
    54	}
===== bech32/codec.mjs =====
     1	/**
     2	 * Bech32 string codec: the raw `hrp` + separator + data-with-checksum format,
     3	 * independent of any segwit interpretation.
     4	 */
     5	
     6	import { CHARSET, charsetAt, charsetIndexOf } from './charset.mjs';
     7	import { bech32CreateChecksum, bech32VerifyChecksum } from './checksum.mjs';
     8	
     9	const MIN_CODE_POINT = 33;
    10	const MAX_CODE_POINT = 126;
    11	const SEPARATOR = '1';
    12	const CHECKSUM_LENGTH = 6;
    13	
    14	/** `bech32_encode(hrp, data)`; may raise `IndexError` for out-of-range values. */
    15	export function bech32Encode(hrp, data) {
    16	  const combined = [...data, ...bech32CreateChecksum(hrp, data)];
    17	  let encoded = '';
    18	  for (const value of combined) encoded += charsetAt(value);
    19	  return hrp + SEPARATOR + encoded;
    20	}
    21	
    22	/**
    23	 * `bech32_decode(bech)` -> `[hrp, data]`, or `[null, null]` when the string is
    24	 * not a well-formed Bech32 string with a valid checksum.
    25	 */
    26	export function bech32Decode(bech) {
    27	  for (const character of bech) {
    28	    const codePoint = character.codePointAt(0);
    29	    if (codePoint < MIN_CODE_POINT || codePoint > MAX_CODE_POINT) return [null, null];
    30	  }
    31	  // Mixed case is not allowed; the range check above guarantees pure ASCII here.
    32	  if (bech.toLowerCase() !== bech && bech.toUpperCase() !== bech) return [null, null];
    33	
    34	  const normalized = bech.toLowerCase();
    35	  const position = normalized.lastIndexOf(SEPARATOR);
    36	  if (position < 1 || position + 1 + CHECKSUM_LENGTH > normalized.length) {
    37	    return [null, null];
    38	  }
    39	
    40	  const payload = normalized.slice(position + 1);
    41	  for (const character of payload) {
    42	    if (!CHARSET.includes(character)) return [null, null];
    43	  }
    44	
    45	  const hrp = normalized.slice(0, position);
    46	  const data = [...payload].map((character) => BigInt(charsetIndexOf(character)));
    47	  if (!bech32VerifyChecksum(hrp, data)) return [null, null];
    48	
    49	  return [hrp, data.slice(0, -CHECKSUM_LENGTH)];
    50	}
===== bech32/convertbits.mjs =====
     1	/**
     2	 * `convertbits`: general power-of-two base conversion over a list of integers.
     3	 *
     4	 * Values stay BigInt so that an out-of-range witness program byte is rejected by
     5	 * the same `value >> frombits` test Python applies, whatever its magnitude.
     6	 * A rejected input yields `null`, mirroring the Python return value.
     7	 */
     8	
     9	export function convertBits(data, fromBits, toBits, pad = true) {
    10	  const from = BigInt(fromBits);
    11	  const to = BigInt(toBits);
    12	  const maxValue = (1n << to) - 1n;
    13	  const maxAccumulator = (1n << (from + to - 1n)) - 1n;
    14	
    15	  let accumulator = 0n;
    16	  let bits = 0n;
    17	  const result = [];
    18	
    19	  for (const value of data) {
    20	    if (value < 0n || (value >> from) !== 0n) return null;
    21	    accumulator = ((accumulator << from) | value) & maxAccumulator;
    22	    bits += from;
    23	    while (bits >= to) {
    24	      bits -= to;
    25	      result.push((accumulator >> bits) & maxValue);
    26	    }
    27	  }
    28	
    29	  if (pad) {
    30	    if (bits !== 0n) result.push((accumulator << (to - bits)) & maxValue);
    31	  } else if (bits >= from || ((accumulator << (to - bits)) & maxValue) !== 0n) {
    32	    return null;
    33	  }
    34	
    35	  return result;
    36	}

===== bech32/segwit.mjs =====
     1	/**
     2	 * Segwit address layer: `encode(hrp, witver, witprog)` and `decode(hrp, addr)`.
     3	 *
     4	 * `encode` validates by round-tripping through `decode`, exactly as the Python
     5	 * package does, and returns `null` (Python's `None`) whenever that round trip
     6	 * rejects the address it just built.  Note that this build checksums every
     7	 * witness version with plain Bech32 -- there is no Bech32m variant.
     8	 */
     9	
    10	import { bech32Decode, bech32Encode } from './codec.mjs';
    11	import { convertBits } from './convertbits.mjs';
    12	
    13	const MAX_WITNESS_VERSION = 16n;
    14	const MIN_PROGRAM_LENGTH = 2;
    15	const MAX_PROGRAM_LENGTH = 40;
    16	const VERSION_ZERO_LENGTHS = [20, 32];
    17	
    18	/** `decode(hrp, addr)` -> `[witver, witprog]` or `[null, null]`. */
    19	export function decode(hrp, addr) {
    20	  const [decodedHrp, data] = bech32Decode(addr);
    21	  if (decodedHrp !== hrp) return [null, null];
    22	
    23	  const program = convertBits(data.slice(1), 5, 8, false);
    24	  if (
    25	    program === null ||
    26	    program.length < MIN_PROGRAM_LENGTH ||
    27	    program.length > MAX_PROGRAM_LENGTH
    28	  ) {
    29	    return [null, null];
    30	  }
    31	  if (data[0] > MAX_WITNESS_VERSION) return [null, null];
    32	  if (data[0] === 0n && !VERSION_ZERO_LENGTHS.includes(program.length)) {
    33	    return [null, null];
    34	  }
    35	  return [data[0], program];
    36	}
    37	
    38	/** `encode(hrp, witver, witprog)` -> the address string, or `null`. */
    39	export function encode(hrp, witver, witprog) {
    40	  const fiveBitProgram = convertBits(witprog, 8, 5);
    41	  if (fiveBitProgram === null) return null;
    42	
    43	  const address = bech32Encode(hrp, [witver, ...fiveBitProgram]);
    44	  const [decodedVersion, decodedProgram] = decode(hrp, address);
    45	  if (decodedVersion === null && decodedProgram === null) return null;
    46	  return address;
    47	}
===== py/argparse.mjs =====
     1	/**
     2	 * The slice of `argparse.ArgumentParser` this script depends on.
     3	 *
     4	 * Reproduced from observed behaviour of the reference build (CPython 3.11+
     5	 * argparse): usage/help layout, the `--x=value` form, long-option abbreviation,
     6	 * the negative-number heuristic that lets `--b -1` pass a value through, `--`
     7	 * ending option parsing, and the three error messages (`expected one argument`,
     8	 * `invalid <type> value`, `the following arguments are required`) which all exit
     9	 * with status 2 after printing usage.
    10	 */
    11	
    12	import { writeStdout, writeStderr } from './stdio.mjs';
    13	import { pythonRepr } from './repr.mjs';
    14	import { PythonError } from './errors.mjs';
    15	
    16	const PREFIX = '-';
    17	const HELP_INVOCATION = '-h, --help';
    18	const HELP_TEXT = 'show this help message and exit';
    19	const INDENT = 2;
    20	const MAX_HELP_POSITION = 24;
    21	
    22	/**
    23	 * argparse's `_negative_number_matcher`.  Python's `\d` covers every Unicode
    24	 * decimal digit, hence `\p{Nd}` rather than `[0-9]`.
    25	 */
    26	const NEGATIVE_NUMBER = new RegExp(
    27	  '^-\\p{Nd}+$|^-\\p{Nd}*\\.\\p{Nd}+$',
    28	  'u',
    29	);
    30	
    31	class Action {
    32	  constructor(optionStrings, { dest, metavar, type, typeName, required, isHelp }) {
    33	    this.optionStrings = optionStrings;
    34	    this.dest = dest;
    35	    this.metavar = metavar;
    36	    this.type = type;
    37	    this.typeName = typeName;
    38	    this.required = required === true;
    39	    this.isHelp = isHelp === true;
    40	  }
    41	
    42	  /** `_get_action_name`: how the action is named inside error messages. */
    43	  get name() {
    44	    return this.optionStrings.join('/');
    45	  }
    46	
    47	  /** `_format_action_invocation`: `-h, --help` / `--a A`. */
    48	  get invocation() {
    49	    if (this.isHelp) return this.optionStrings.join(', ');
    50	    return this.optionStrings.map((option) => `${option} ${this.metavar}`).join(', ');
    51	  }
    52	
    53	  /** `_format_actions_usage`: the usage line names only the first option string. */
    54	  get usagePart() {
    55	    const option = this.optionStrings[0];
    56	    return this.isHelp ? option : `${option} ${this.metavar}`;
    57	  }
    58	}
    59	
    60	export class ArgumentParser {
    61	  constructor({ prog }) {
    62	    this.prog = prog;
    63	    this.actions = [];
    64	    this.optionMap = new Map();
    65	    this.#register(
    66	      new Action(['-h', '--help'], { dest: 'help', isHelp: true, required: false }),
    67	    );
    68	  }
    69	
    70	  #register(action) {
    71	    this.actions.push(action);
    72	    for (const option of action.optionStrings) this.optionMap.set(option, action);
    73	    return action;
    74	  }
    75	
    76	  /**
    77	   * `add_argument("--x", type=..., required=...)`.  `type` is a converter that
    78	   * throws for invalid input; `typeName` is the `__name__` argparse reports.
    79	   */
    80	  add_argument(optionString, { type, typeName, required } = {}) {
    81	    const dest = optionString.replace(/^-+/, '').replace(/-/g, '_');
    82	    return this.#register(
    83	      new Action([optionString], {
    84	        dest,
    85	        metavar: dest.toUpperCase(),
    86	        type,
    87	        typeName: typeName ?? 'str',
    88	        required,
    89	      }),
    90	    );
    91	  }
    92	
    93	  format_usage() {
    94	    const parts = this.actions.map((action) =>
    95	      action.required ? action.usagePart : `[${action.usagePart}]`,
    96	    );
    97	    return `usage: ${[this.prog, ...parts].join(' ')}\n`;
    98	  }
    99	
   100	  format_help() {
   101	    // HelpFormatter aligns help text at `min(longest + 2, 24)` columns, where
   102	    // `longest` already includes the two-space action indent.
   103	    const longest = Math.max(
   104	      ...this.actions.map((action) => action.invocation.length + INDENT),
   105	    );
   106	    const helpPosition = Math.min(longest + 2, MAX_HELP_POSITION);
   107	    const invocationWidth = helpPosition - INDENT - 2;
   108	
   109	    const lines = [this.format_usage().trimEnd(), '', 'options:'];
   110	    for (const action of this.actions) {
   111	      const invocation = action.invocation;
   112	      if (!action.isHelp) {
   113	        lines.push(`${' '.repeat(INDENT)}${invocation}`);
   114	        continue;
   115	      }
   116	      lines.push(
   117	        `${' '.repeat(INDENT)}${invocation.padEnd(invocationWidth)}  ${HELP_TEXT}`,
   118	      );
   119	    }
   120	    return `${lines.join('\n')}\n`;
   121	  }
   122	
   123	  /** Prints usage plus the message on stderr and exits with status 2. */
   124	  error(message) {
   125	    writeStderr(this.format_usage());
   126	    writeStderr(`${this.prog}: error: ${message}\n`);
   127	    process.exit(2);
   128	  }
   129	
   130	  print_help() {
   131	    writeStdout(this.format_help());
   132	  }
   133	
   134	  parse_args(argv) {
   135	    const [namespace, extras] = this.parse_known_args(argv);
   136	    if (extras.length > 0) {
   137	      this.error(`unrecognized arguments: ${extras.join(' ')}`);
   138	    }
   139	    return namespace;
   140	  }
   141	
   142	  parse_known_args(argv) {
   143	    const namespace = Object.create(null);
   144	    const seen = new Set();
   145	    const extras = [];
   146	    let index = 0;
   147	    let optionsEnded = false;
   148	
   149	    while (index < argv.length) {
   150	      const argument = argv[index];
   151	
   152	      if (optionsEnded) {
   153	        extras.push(argument);
   154	        index += 1;
   155	        continue;
   156	      }
   157	      // A bare `--` stops option parsing; argparse leaves it among the leftovers.
   158	      if (argument === '--') {
   159	        extras.push(argument);
   160	        optionsEnded = true;
   161	        index += 1;
   162	        continue;
   163	      }
   164	
   165	      const candidate = this.#parseOptional(argument);
   166	      if (candidate === null || candidate.action === null) {
   167	        // Either a positional-looking token or an unknown option: both survive
   168	        // as leftovers for `unrecognized arguments`.
   169	        extras.push(argument);
   170	        index += 1;
   171	        continue;
   172	      }
   173	
   174	      const { action, optionString, explicitArgument } = candidate;
   175	      if (action.isHelp) {
   176	        this.print_help();
   177	        process.exit(0);
   178	      }
   179	
   180	      let rawValue;
   181	      if (explicitArgument !== null) {
   182	        rawValue = explicitArgument;
   183	        index += 1;
   184	      } else {
   185	        const next = index + 1 < argv.length ? argv[index + 1] : null;
   186	        if (next === null || next === '--' || this.#parseOptional(next) !== null) {
   187	          this.error(`argument ${action.name}: expected one argument`);
   188	        }
   189	        rawValue = next;
   190	        index += 2;
   191	      }
   192	
   193	      namespace[action.dest] = this.#convert(action, rawValue);
   194	      seen.add(action.dest);
   195	    }
   196	
   197	    const missing = this.actions
   198	      .filter((action) => action.required && !seen.has(action.dest))
   199	      .map((action) => action.name);
   200	    if (missing.length > 0) {
   201	      this.error(`the following arguments are required: ${missing.join(', ')}`);
   202	    }
   203	
   204	    return [namespace, extras];
   205	  }
   206	
   207	  #convert(action, rawValue) {
   208	    if (action.type === undefined) return rawValue;
   209	    try {
   210	      return action.type(rawValue);
   211	    } catch (error) {
   212	      if (error instanceof PythonError || error instanceof TypeError) {
   213	        this.error(
   214	          `argument ${action.name}: invalid ${action.typeName} value: ${pythonRepr(rawValue)}`,
   215	        );
   216	      }
   217	      throw error;
   218	    }
   219	  }
   220	
   221	  /**
   222	   * `_parse_optional`.  Returns `null` when the token should be treated as a
   223	   * value, `{action: null, ...}` when it looks like an unknown option, and a
   224	   * resolved action otherwise.
   225	   */
   226	  #parseOptional(argument) {
   227	    if (argument.length === 0) return null;
   228	    if (argument[0] !== PREFIX) return null;
   229	
   230	    const exact = this.optionMap.get(argument);
   231	    if (exact !== undefined) {
   232	      return { action: exact, optionString: argument, explicitArgument: null };
   233	    }
   234	    // A lone `-` was meant to be positional.
   235	    if (argument.length === 1) return null;
   236	
   237	    const equals = argument.indexOf('=');
   238	    if (equals >= 0) {
   239	      const optionString = argument.slice(0, equals);
   240	      const action = this.optionMap.get(optionString);
   241	      if (action !== undefined) {
   242	        return { action, optionString, explicitArgument: argument.slice(equals + 1) };
   243	      }
   244	    }
   245	
   246	    const matches = this.#optionTuples(argument);
   247	    if (matches.length > 1) {
   248	      const options = matches.map((match) => match.optionString).join(', ');
   249	      this.error(`ambiguous option: ${argument} could match ${options}`);
   250	    }
   251	    if (matches.length === 1) return matches[0];
   252	
   253	    // Negative numbers are values, since no option string looks like one.
   254	    if (NEGATIVE_NUMBER.test(argument)) return null;
   255	    // Anything containing a space cannot be an option.
   256	    if (argument.includes(' ')) return null;
   257	
   258	    return { action: null, optionString: argument, explicitArgument: null };
   259	  }
   260	
   261	  /** `_get_option_tuples`: abbreviation matching. */
   262	  #optionTuples(argument) {
   263	    const matches = [];
   264	    if (argument.startsWith('--')) {
   265	      const equals = argument.indexOf('=');
   266	      const prefix = equals >= 0 ? argument.slice(0, equals) : argument;
   267	      const explicitArgument = equals >= 0 ? argument.slice(equals + 1) : null;
   268	      for (const [optionString, action] of this.optionMap) {
   269	        if (optionString.startsWith(prefix)) {
   270	          matches.push({ action, optionString, explicitArgument });
   271	        }
   272	      }
   273	      return matches;
   274	    }
   275	
   276	    const shortPrefix = argument.slice(0, 2);
   277	    const shortExplicit = argument.slice(2);
   278	    for (const [optionString, action] of this.optionMap) {
   279	      if (optionString === shortPrefix) {
   280	        matches.push({ action, optionString, explicitArgument: shortExplicit });
   281	      } else if (optionString.startsWith(argument)) {
   282	        matches.push({ action, optionString, explicitArgument: null });
   283	      }
   284	    }
   285	    return matches;
   286	  }
   287	}

===== py/int.mjs =====
     1	/**
     2	 * CPython's `int(str)` conversion, base 10.
     3	 *
     4	 * Values are returned as BigInt because Python integers are unbounded: the
     5	 * witness version reaches `CHARSET[witver]` unchanged, and the difference
     6	 * between "string index out of range" and "cannot fit 'int' into an index-sized
     7	 * integer" depends on the true magnitude, not on a clamped double.
     8	 *
     9	 * Accepted grammar (after stripping Python whitespace from both ends):
    10	 *
    11	 *     [+-] digit (('_')? digit)*
    12	 *
    13	 * where `digit` is any Unicode decimal digit, and underscores may neither lead,
    14	 * trail, nor repeat.
    15	 */
    16	
    17	import { decimalDigitValue, isPythonSpace } from './unicode.mjs';
    18	import { PythonValueError } from './errors.mjs';
    19	import { pythonRepr } from './repr.mjs';
    20	
    21	const PLUS = 0x2b;
    22	const MINUS = 0x2d;
    23	const UNDERSCORE = 0x5f;
    24	
    25	/** Parses like `int()` but reports failure as `null` instead of raising. */
    26	export function parsePythonInt(text) {
    27	  const codePoints = Array.from(text, (character) => character.codePointAt(0));
    28	
    29	  let start = 0;
    30	  let end = codePoints.length;
    31	  while (start < end && isPythonSpace(codePoints[start])) start += 1;
    32	  while (end > start && isPythonSpace(codePoints[end - 1])) end -= 1;
    33	
    34	  let negative = false;
    35	  if (start < end && (codePoints[start] === PLUS || codePoints[start] === MINUS)) {
    36	    negative = codePoints[start] === MINUS;
    37	    start += 1;
    38	  }
    39	
    40	  let digits = '';
    41	  // Doubles as "an underscore here would be illegal": true before the first
    42	  // digit and immediately after an underscore.
    43	  let awaitingDigit = true;
    44	  for (let index = start; index < end; index += 1) {
    45	    const codePoint = codePoints[index];
    46	    if (codePoint === UNDERSCORE) {
    47	      if (awaitingDigit) return null;
    48	      awaitingDigit = true;
    49	      continue;
    50	    }
    51	    const value = decimalDigitValue(codePoint);
    52	    if (value < 0) return null;
    53	    digits += String(value);
    54	    awaitingDigit = false;
    55	  }
    56	  // Catches an empty body as well as a trailing underscore.
    57	  if (awaitingDigit) return null;
    58	
    59	  const magnitude = BigInt(digits);
    60	  return negative ? -magnitude : magnitude;
    61	}
    62	
    63	/** `int(text)`; raises `ValueError` with CPython's message on bad input. */
    64	export function pythonInt(text) {
    65	  const value = parsePythonInt(text);
    66	  if (value === null) {
    67	    throw new PythonValueError(
    68	      `invalid literal for int() with base 10: ${pythonRepr(text)}`,
    69	    );
    70	  }
    71	  return value;
    72	}
===== py/repr.mjs =====
     1	/**
     2	 * `repr()` of a Python `str`.
     3	 *
     4	 * Both `int()`'s ValueError text and argparse's "invalid int value" text embed
     5	 * the offending argument through `%r`, so the escaping rules have to match
     6	 * CPython exactly: single quotes unless that would need escaping, short escapes
     7	 * for backslash / quote / newline / carriage return / tab, and `\xXX`, `\uXXXX`
     8	 * or `\UXXXXXXXX` for anything `str.isprintable()` rejects.
     9	 */
    10	
    11	import { isPythonPrintable } from './unicode.mjs';
    12	
    13	const SHORT_ESCAPES = new Map([
    14	  ['\\', '\\\\'],
    15	  ['\n', '\\n'],
    16	  ['\r', '\\r'],
    17	  ['\t', '\\t'],
    18	]);
    19	
    20	function hexEscape(prefix, codePoint, width) {
    21	  return `\\${prefix}${codePoint.toString(16).padStart(width, '0')}`;
    22	}
    23	
    24	export function pythonRepr(text) {
    25	  const quote = text.includes("'") && !text.includes('"') ? '"' : "'";
    26	  let body = '';
    27	  for (const character of text) {
    28	    const short = SHORT_ESCAPES.get(character);
    29	    if (short !== undefined) {
    30	      body += short;
    31	      continue;
    32	    }
    33	    if (character === quote) {
    34	      body += `\\${quote}`;
    35	      continue;
    36	    }
    37	    const codePoint = character.codePointAt(0);
    38	    if (isPythonPrintable(codePoint)) body += character;
    39	    else if (codePoint < 0x100) body += hexEscape('x', codePoint, 2);
    40	    else if (codePoint < 0x10000) body += hexEscape('u', codePoint, 4);
    41	    else body += hexEscape('U', codePoint, 8);
    42	  }
    43	  return quote + body + quote;
    44	}
===== py/stdio.mjs =====
     1	/**
     2	 * Synchronous stdio writers.
     3	 *
     4	 * `process.stdout.write` is asynchronous when stdout is a pipe, so anything
     5	 * written just before `process.exit()` can be lost.  Every diagnostic path here
     6	 * ends in an exit, so all of them go through `writeSync`.
     7	 */
     8	
     9	import { writeSync } from 'node:fs';
    10	
    11	const STDOUT_FD = 1;
    12	const STDERR_FD = 2;
    13	
    14	function write(fileDescriptor, text) {
    15	  const buffer = Buffer.from(text, 'utf8');
    16	  let written = 0;
    17	  while (written < buffer.length) {
    18	    try {
    19	      written += writeSync(fileDescriptor, buffer, written);
    20	    } catch (error) {
    21	      if (error.code === 'EAGAIN') continue;
    22	      if (error.code === 'EPIPE') return;
    23	      throw error;
    24	    }
    25	  }
    26	}
    27	
    28	export function writeStdout(text) {
    29	  write(STDOUT_FD, text);
    30	}
    31	
    32	export function writeStderr(text) {
    33	  write(STDERR_FD, text);
    34	}

     3	 *
     4	 * The reference build is a PyInstaller bundle, so an unhandled exception prints
     5	 * a CPython traceback whose frames point at the original `test4.py` source
     6	 * lines, followed by PyInstaller's own one-line summary.  Frames inside the

    14	const SCRIPT_FILE = 'test4.py';

    53	function frameLines(frame) {
    54	  const lines = [`  File "${frame.file}", line ${frame.line}, in ${frame.name}`];
    55	  if (frame.source === undefined) return lines;
    56	  lines.push(`    ${frame.source}`);
    57	  if (frame.anchorStart !== undefined) {
    58	    const end = frame.anchorEnd ?? frame.source.length;
    59	    lines.push(
    60	      `    ${' '.repeat(frame.anchorStart)}${'^'.repeat(end - frame.anchorStart)}`,
    61	    );
    62	  }
    63	  return lines;
    64	}

    68	 * does, including PyInstaller's trailing failure notice.
    69	 */
    70	export function reportUnhandledException(error, frames) {
    71	  const lines = ['Traceback (most recent call last):'];
    72	  for (const frame of frames) lines.push(...frameLines(frame));
    73	  lines.push(`${error.typeName}: ${error.pythonMessage}`);
    74	  lines.push(
    75	    `[PYI-${process.pid}:ERROR] Failed to execute script '${SCRIPT_MODULE}' due to unhandled exception!`,
    76	  );
    77	  writeStderr(`${lines.join('\n')}\n`);
    78	}
===== py/unicode.mjs =====
     1	/**
     2	 * Unicode predicates that mirror the CPython helpers used by `int()` and `repr()`.
     3	 *
     4	 * CPython does not restrict `int("...")` to ASCII: any character whose Unicode
     5	 * general category is Nd carries a decimal value, and `str.strip()` (applied by
     6	 * `int()` before parsing) removes every character for which `str.isspace()` is
     7	 * true.  These helpers reproduce those two predicates plus `str.isprintable()`,
     8	 * which `repr()` needs in order to decide what to escape.
     9	 */
    10	
    11	/**
    12	 * One entry per contiguous run of ten decimal digits in the Unicode database
    13	 * (the code point of each family's zero).  Unicode guarantees a decimal digit
    14	 * family occupies exactly ten consecutive code points, so the value of a digit
    15	 * is `codePoint - base`.
    16	 */
    17	const DECIMAL_ZERO_BASES = [
    18	  0x30, 0x660, 0x6f0, 0x7c0, 0x966, 0x9e6, 0xa66, 0xae6, 0xb66, 0xbe6,
    19	  0xc66, 0xce6, 0xd66, 0xde6, 0xe50, 0xed0, 0xf20, 0x1040, 0x1090, 0x17e0,
    20	  0x1810, 0x1946, 0x19d0, 0x1a80, 0x1a90, 0x1b50, 0x1bb0, 0x1c40, 0x1c50,
    21	  0xa620, 0xa8d0, 0xa900, 0xa9d0, 0xa9f0, 0xaa50, 0xabf0, 0xff10, 0x104a0,
    22	  0x10d30, 0x11066, 0x110f0, 0x11136, 0x111d0, 0x112f0, 0x11450, 0x114d0,
    23	  0x11650, 0x116c0, 0x11730, 0x118e0, 0x11950, 0x11c50, 0x11d50, 0x11da0,
    24	  0x11f50, 0x16a60, 0x16ac0, 0x16b50, 0x1d7ce, 0x1d7d8, 0x1d7e2, 0x1d7ec,
    25	  0x1d7f6, 0x1e140, 0x1e2f0, 0x1e4f0, 0x1e950, 0x1fbf0,
    26	];
    27	
    28	/** Code points `Py_UNICODE_ISSPACE` accepts that are outside the Z* categories. */
    29	const EXTRA_SPACE_CODE_POINTS = new Set([
    30	  0x09, 0x0a, 0x0b, 0x0c, 0x0d, 0x1c, 0x1d, 0x1e, 0x1f, 0x20, 0x85,
    31	]);
    32	
    33	/** Zs plus LINE SEPARATOR (U+2028) and PARAGRAPH SEPARATOR (U+2029). */
    34	const SPACE_SEPARATOR = new RegExp('^[\\p{Zs}\\u2028\\u2029]$', 'u');
    35	
    36	/** Categories for which `str.isprintable()` is false (ASCII space is the one exception). */
    37	const NON_PRINTABLE = buildNonPrintableMatcher();
    38	
    39	function buildNonPrintableMatcher() {
    40	  // `\p{Cn}` (unassigned) is not accepted by every engine; degrade gracefully.
    41	  try {
    42	    return new RegExp('^[\\p{Cc}\\p{Cf}\\p{Cs}\\p{Co}\\p{Cn}\\p{Zl}\\p{Zp}\\p{Zs}]$', 'u');
    43	  } catch {
    44	    return new RegExp('^[\\p{Cc}\\p{Cf}\\p{Cs}\\p{Co}\\p{Zl}\\p{Zp}\\p{Zs}]$', 'u');
    45	  }
    46	}
    47	
    48	/**
    49	 * Decimal value of a code point, or -1 when it is not a decimal digit.
    50	 * Equivalent to CPython's `Py_UNICODE_TODECIMAL`.
    51	 */
    52	export function decimalDigitValue(codePoint) {
    53	  let low = 0;
    54	  let high = DECIMAL_ZERO_BASES.length - 1;
    55	  while (low <= high) {
    56	    const mid = (low + high) >> 1;
    57	    const base = DECIMAL_ZERO_BASES[mid];
    58	    if (codePoint < base) high = mid - 1;
    59	    else if (codePoint >= base + 10) low = mid + 1;
    60	    else return codePoint - base;
    61	  }
    62	  return -1;
    63	}
    64	
    65	/** `str.isspace()` for a single code point. */
    66	export function isPythonSpace(codePoint) {
    67	  return (
    68	    EXTRA_SPACE_CODE_POINTS.has(codePoint) ||
    69	    SPACE_SEPARATOR.test(String.fromCodePoint(codePoint))
    70	  );
    71	}
    72	
    73	/** `str.isprintable()` for a single code point. */
    74	export function isPythonPrintable(codePoint) {
    75	  if (codePoint === 0x20) return true;
    76	  return !NON_PRINTABLE.test(String.fromCodePoint(codePoint));
    77	}
