import { bech32Decode, parseArgs, pythonTupleRepr } from './lib/index.mjs';

export function main(argv = process.argv) {
  const parsed = parseArgs(argv);

  if (parsed.kind === 'help' || parsed.kind === 'error') {
    const stream = parsed.stream === 'stderr' ? process.stderr : process.stdout;
    stream.write(parsed.output);
    return parsed.code;
  }

  const result = bech32Decode(parsed.args.a);
  process.stdout.write(`${pythonTupleRepr(result)}\n`);
  return 0;
}

const exitCode = main();

if (exitCode !== 0) {
  process.exitCode = exitCode;
}
