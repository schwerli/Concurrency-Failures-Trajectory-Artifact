    25	TAR_COMPRESSIONS = {
    26	    "": "",
    27	    "none": "",
    28	    "tar": "",
    29	    "bz2": "bz2",
    30	    "bzip2": "bz2",
    31	    "gz": "gz",
    32	    "gzip": "gz",
    33	    "lzma": "xz",
    34	    "tbz": "bz2",
    35	    "tgz": "gz",
    36	    "txz": "xz",

    41	def _get_tar_compression(compression: str | None) -> str:
    42	    """
    43	    Normalize the given compression name to a tarfile mode suffix.
    44	    """
    45	    key = (compression or "").strip().lower().lstrip(".")
    46	    try:
    47	        return TAR_COMPRESSIONS[key]
    48	    except KeyError:
    49	        raise ValueError(
    50	            f"Invalid tar compression: {compression!r}, "
    51	            f"it must be one of {sorted(TAR_COMPRESSIONS)}."
    52	        ) from None
    53	
    54	
    55	def _iter_content_paths(
    56	    content_path: PathIn, basedir: str = ""
    57	) -> Iterable[tuple[str, str]]:
    58	    """
    59	    Walk the given content path yielding (filepath, archive-name) pairs.
    60	
    61	    Directories are walked recursively and their name becomes part of the
    62	    archive-relative path, while files are added at the current basedir level.
    63	    """
    64	    path = _get_path(content_path)
    65	    assert_exists(path)
    66	    if is_file(path):
    67	        yield (path, join_path(basedir, get_filename(path)))
    68	    elif is_dir(path):
    69	        for item_name in sorted(os.listdir(path)):
    70	            item_path = join_path(path, item_name)
    71	            item_basedir = (
    72	                join_path(basedir, item_name) if is_dir(item_path) else basedir
    73	            )
    74	            yield from _iter_content_paths(item_path, item_basedir)

    77	def create_tar_file(
    78	    path: PathIn,
    79	    content_paths: list[PathIn],
    80	    overwrite: bool = True,
    81	    compression: str = "",
    82	) -> None:
    83	    """
    84	    Create tar file at path compressing directories/files listed in content_paths.
    85	    If overwrite is allowed and dest tar already exists, it will be overwritten.
    86	    The compression can be one of: "", "gzip"/"gz", "bzip2"/"bz2", "xz"/"lzma".
    87	    """
    88	    path = _get_path(path)
    89	    assert_not_dir(path)
    90	    if not overwrite:
    91	        assert_not_exists(path)
    92	    make_dirs_for_file(path)
    93	    compression_suffix = _get_tar_compression(compression)
    94	    mode = f"w:{compression_suffix}" if compression_suffix else "w"
    95	    with tarfile.open(path, mode=mode) as file:  # type: ignore[call-overload]
    96	        for content_path in content_paths:
    97	            for filepath, filename in _iter_content_paths(content_path):
    98	                file.add(filepath, filename)

   122	def extract_tar_file(
   123	    path: PathIn,
   124	    dest: PathIn,
   125	    content_paths: Iterable[tarfile.TarInfo] | None = None,
   126	    autodelete: bool = False,
   127	    filter: (  # noqa: A002
   128	        Callable[[tarfile.TarInfo, str], tarfile.TarInfo | None]
   129	        | Literal["fully_trusted", "tar", "data"]
   130	        | None
   131	    ) = None,
   132	) -> None:
   133	    """
   134	    Extract tar file at path to dest path.
   135	    If autodelete, the archive will be deleted after extraction.
   136	    If content_paths list is defined,
   137	    only listed items will be extracted, otherwise all.
   138	    """
   139	    path = _get_path(path)
   140	    dest = _get_path(dest)
   141	    assert_file(path)
   142	    assert_not_file(dest)
   143	    make_dirs(dest)
   144	    # "r:*" transparently handles uncompressed and gz/bz2/xz compressed archives.
   145	    with tarfile.open(path, "r:*") as file:
   146	        if sys.version_info < (3, 12):
   147	            file.extractall(dest, members=content_paths)
   148	        else:
   149	            # https://docs.python.org/3/library/tarfile.html#tarfile-extraction-filter
   150	            file.extractall(
   151	                dest,
   152	                members=content_paths,
   153	                numeric_owner=False,
   154	                filter=(filter or "data"),
   155	            )
   156	    if autodelete:
   157	        remove_file(path)

    17	def get_path(path: PathIn | None) -> str | None:
    18	    """
    19	    Normalize a path argument to a plain normalized string path.
    20	
    21	    Every public function funnels its path arguments through this helper, so
    22	    that ``str`` and ``pathlib.Path`` inputs are handled uniformly and paths are

    26	    ``None`` is returned unchanged to support optional path arguments.
    27	    """
    28	    if path is None:
    29	        return None
    30	    return os.path.normpath(str(path))

     3	SIZE_UNITS = ["bytes", "KB", "MB", "GB", "TB", "PB", "EB", "ZB", "YB"]
     4	
     5	
     6	def convert_size_bytes_to_string(size: int) -> str:
     7	    """
     8	    Convert the given size bytes to string using the right unit suffix.
     9	
    10	    Sizes below 1 KB are rendered without decimals ("512 bytes"), as are exact
    11	    KB values ("1 KB"), while bigger units always use 2 decimals ("1.50 MB").
    12	    Sizes exceeding the biggest known unit keep using it ("1024.00 YB").
    13	    """
    14	    size_num = float(size)
    15	    units = SIZE_UNITS
    16	    factor = 0
    17	    factor_limit = len(units) - 1
    18	    while (size_num >= 1024) and (factor < factor_limit):
    19	        size_num /= 1024
    20	        factor += 1
    21	    size_units = units[factor]
    22	    size_str = f"{size_num:.2f}" if (factor > 1) else f"{size_num:.0f}"
    23	    return f"{size_str} {size_units}"
    24	
    25	
    26	def convert_size_string_to_bytes(size: str) -> float | int:
    27	    """
    28	    Convert the given size string to bytes.
    29	
    30	    The counterpart of :func:`convert_size_bytes_to_string`, it accepts any
    31	    "<amount> <unit>" string whose unit is listed in ``SIZE_UNITS``.
    32	    """
    33	    units = [item.lower() for item in SIZE_UNITS]
    34	    parts = size.strip().replace("  ", " ").split(" ")
    35	    amount = float(parts[0])
    36	    unit = parts[1]
    37	    factor = units.index(unit.lower())
    38	    if not factor:
    39	        return amount
    40	    return int((1024**factor) * amount)

   139	def get_file_hash(path: PathIn | IO[bytes], func: str = "md5") -> str:
   140	    """
   141	    Get the hash of the file at the given path (or of the given
   142	    binary file-like object) using the specified algorithm
   143	    function (md5 by default).
   144	    """
   145	    hash_ = hashlib.new(func)
   146	    if isinstance(path, (str, pathlib.Path)):
   147	        filepath = _get_path(path)
   148	        assert_file(filepath)
   149	        with open(filepath, "rb") as file:
   150	            for chunk in iter(lambda: file.read(_HASH_CHUNK_SIZE), b""):
   151	                hash_.update(chunk)
   152	    else:
   153	        fileobj = path
   154	        # the stream is rewound to hash its whole content, then the
   155	        # original position is restored to leave the caller unaffected.
   156	        position = None
   157	        if fileobj.seekable():
   158	            position = fileobj.tell()
   159	            fileobj.seek(0)
   160	        for chunk in iter(lambda: fileobj.read(_HASH_CHUNK_SIZE), b""):
   161	            hash_.update(chunk)
   162	        if position is not None:
   163	            fileobj.seek(position)
   164	    hash_hex = hash_.hexdigest()
   165	    return hash_hex

    73	def _read_file_lines_in_range(
    74	    path: PathIn,
    75	    line_start: int = 0,
    76	    line_end: int = -1,
    77	    encoding: str = "utf-8",
    78	) -> Generator[str, None, None]:
    79	    path = _get_path(path)
    80	    line_start_negative = line_start < 0
    81	    line_end_negative = line_end < 0
    82	    if line_start_negative or line_end_negative:
    83	        # pre-calculate lines count only if using negative line indexes
    84	        lines_count = read_file_lines_count(path)
    85	        # normalize negative indexes
    86	        if line_start_negative:
    87	            line_start = max(0, line_start + lines_count)
    88	        if line_end_negative:
    89	            line_end = min(line_end + lines_count, lines_count - 1)
    90	    with open(path, "rb") as file:
    91	        file.seek(0)
    92	        line_index = 0
    93	        for line in file:
    94	            if line_index >= line_start and line_index <= line_end:
    95	                yield line.decode(encoding)
    96	            line_index += 1

   155	def _write_file_atomic(
   156	    path: PathIn,
   157	    content: str,
   158	    append: bool = False,
   159	    encoding: str = "utf-8",
   160	) -> None:
   161	    path = _get_path(path)
   162	    if append and exists(path):
   163	        content = read_file(path, encoding=encoding) + content
   164	    dirpath, _ = split_filepath(path)
   165	    temp_path = None
   166	    try:
   167	        # the content is fully written and flushed to a temporary file living in
   168	        # the same directory (hence on the same filesystem), then moved over the
   169	        # destination: readers either see the old or the new content, never both.
   170	        with tempfile.NamedTemporaryFile(
   171	            mode="w",
   172	            dir=dirpath or os.curdir,
   173	            delete=False,
   174	            encoding=encoding,
   175	        ) as file:
   176	            file.write(content)
   177	            file.flush()
   178	            os.fsync(file.fileno())
   179	            temp_path = file.name
   180	        # file is now closed, safe to replace on Windows
   181	        permissions = get_permissions(path) if exists(path) else None
   182	        os.replace(temp_path, path)
   183	        # temporary files are created private (0600), so the permissions of the
   184	        # replaced file are restored, or the default ones are applied.
   185	        if permissions is not None:
   186	            set_permissions(path, permissions)
   187	        else:
   188	            os.chmod(path, _get_default_file_permissions())
   189	    finally:
   190	        # manually delete the temporary file if it still exists,
   191	        # eg. if an error occurred before/during the replace.
   192	        if temp_path and exists(temp_path):
   193	            remove_file(temp_path)

   187	def download_file(
   188	    url: str,
   189	    dirpath: PathIn | None = None,
   190	    filename: str | None = None,
   191	    chunk_size: int = 8192,
   192	    **kwargs: Any,
   193	) -> str:
   194	    """
   195	    Download a file from url to dirpath.
   196	    If dirpath is not provided, the file will be downloaded to a temp directory.
   197	    If filename is provided, the file will be named using filename.
   198	    It is possible to pass extra request options
   199	    (eg. for authentication) using **kwargs.
   200	    """
   201	    requests = require_requests()
   202	    # the response is streamed and written in chunks, so that
   203	    # arbitrarily large files don't have to fit in memory.
   204	    kwargs["stream"] = True
   205	    with requests.get(url, **kwargs) as response:
   206	        response.raise_for_status()
   207	
   208	        # build filename
   209	        if not filename:
   210	            # detect filename from headers
   211	            content_disposition = response.headers.get("content-disposition", "") or ""
   212	            filename_pattern = r'filename="(.*)"'
   213	            filename_match = re.search(filename_pattern, content_disposition)
   214	            if filename_match:
   215	                # sanitize Content-Disposition filename to prevent path traversal
   216	                filename = filename_match.group(1).replace("\\", "/")
   217	                filename = os.path.basename(filename)
   218	            # or detect filename from url
   219	            if not filename:
   220	                filename = get_filename(url)
   221	            # or fallback to a unique name
   222	            if not filename:
   223	                filename = f"download-{uuid.uuid4()}"
   224	
   225	        # build filepath
   226	        dirpath = dirpath or tempfile.gettempdir()
   227	        dirpath = _get_path(dirpath)
   228	        filepath = join_path(dirpath, filename)
   229	        make_dirs_for_file(filepath)
   230	
   231	        # write file to disk
   232	        with open(filepath, "wb") as file:
   233	            for chunk in response.iter_content(chunk_size=chunk_size):
   234	                if chunk:
   235	                    file.write(chunk)
   236	    return filepath

   443	    """
   444	    Replace directory at the specified path with the directory located at src.
   445	    If autodelete, the src directory will be removed at the end of the operation.
   446	    Optimized for large files.
   447	    """
   448	    path = _get_path(path)
   449	    src = _get_path(src)

   177	    :returns: The filepath with the applied changes.
   178	    :rtype: str
   179	    """
   180	
   181	    def _get_value(
   182	        new_value: str | Callable[[str], str] | None,
   183	        old_value: str,
   184	    ) -> str:
   185	        if new_value is None:
   186	            return old_value
   187	        if callable(new_value):
   188	            return new_value(old_value)
   189	        if isinstance(new_value, str):
   190	            return new_value
   191	        return old_value
   192	
   193	    error_message = (
   194	        "Invalid arguments: at least one of "
   195	        "'dirpath', 'basename' or 'extension' is required."
   196	    )
   197	    if all([dirpath is None, basename is None, extension is None]):
   198	        raise ValueError(error_message)
   199	    old_dirpath, old_filename = split_filepath(path)
   200	    old_basename, old_extension = split_filename(old_filename)
   201	    new_dirpath = _get_value(dirpath, old_dirpath)
   202	    new_basename = _get_value(basename, old_basename)
   203	    new_extension = _get_value(extension, old_extension)
   204	    if not any([new_dirpath, new_basename, new_extension]):
   205	        raise ValueError(error_message)
   206	    new_filename = join_filename(new_basename, new_extension)
   207	    new_filepath = join_filepath(new_dirpath, new_filename)
   208	    return new_filepath

    10	def get_permissions(path: PathIn) -> int:
    11	    """
    12	    Get the file/directory permissions.
    13	
    14	    The returned value is the octal representation read as a decimal int, eg.
    15	    a file with mode ``0o644`` returns ``644``, so that it round-trips with
    16	    :func:`set_permissions`.
    17	    """
    18	    path = _get_path(path)
    19	    assert_exists(path)
    20	    st_mode = os.stat(path).st_mode
    21	    permissions = int(str(oct(st_mode & 0o777))[2:])
    22	    return permissions
    23	
    24	
    25	def set_permissions(path: PathIn, value: int) -> None:
    26	    """
    27	    Set the file/directory permissions.
    28	
    29	    The value is interpreted the same way it is returned by
    30	    :func:`get_permissions`, ie. ``set_permissions(path, 644)`` applies the
    31	    ``0o644`` mode. Values that cannot be read as octal digits (eg. the result
    32	    of an ``0o755`` literal, which is ``493``) are used as raw modes instead.
    33	    """
    34	    path = _get_path(path)
    35	    assert_exists(path)
    36	    try:
    37	        permissions = int(str(value), 8) & 0o777
    38	    except ValueError:
    39	        permissions = int(value) & 0o777
    40	    os.chmod(path, permissions)
