    17	    """Infers a tool schema from a function's signature and docstring."""
    18	    sig = inspect.signature(func)

    23	    description = doc_parts[0].strip()
    24

    26	    examples = []
    27	    return_description = ""
    28	    for part in doc_parts:
    29	        if part.startswith("Examples:"):
    30	            examples = [line.strip() for line in part.split("\n")[1:] if line.strip()]
    31	        elif part.startswith("Returns:"):
    32	            return_description = part.split("\n")[1].strip()
    33

    35	        sig.return_annotation.__name__
    36	        if sig.return_annotation != inspect.Parameter.empty
    37	        else "any"
    38	    )
    39	
    40	    print(f"return_description: {return_description} ({return_type})")
    41	    # Build args schema

    44	        param_doc = ""
    45	        for part in doc_parts:
    46	            if part.strip().startswith("Args:"):
    47	                for line in part.split("\n")[1:]:

    72	    """Formats tool schemas into a user-friendly description string."""
    73	    descriptions = []
    74	    for schema in schemas:
    75	        desc = [f"{schema['name']}: {schema['description']}"]
    76	
    77	        desc.append("\nArguments:")
    78	        for arg_name, arg_info in schema["args"].items():
    79	            default = (

    81	            )

   130	        import io
   131	        import signal

   134	
   135	        try:

   141	        if code_str.startswith("```python"):

   403	    def call_tool(self, tool_json: str, max_chars: int = 1024, **kwargs) -> str:
   404	        """Call a tool based on JSON command."""
   405	        try:
   406	            command = json.loads(tool_json)
   407	            if not isinstance(command, dict):
   408	                return 'Error: Tool command must be a JSON object, e.g. \'{"name": "tool_name", "args": {"arg1": "value1", "arg2": "value2"}}\''
   409	
   410	            tool_name = command.get("name")
   411	            if not tool_name:
   412	                return 'Error: Tool command must specify \'name\', e.g. \'{"name": "tool_name", "args": {"arg1": "value1", "arg2": "value2"}}\''
   413	
   414	            if tool_name not in self.tools:
   415	                return (
   416	                    f"Error: Unknown tool '{tool_name}. "
   417	                    + 'Please format your tool call as \'{"name": "tool_name", "args": {"arg1": "value1", "arg2": "value2"}}\''
   418	                )
   419	
   420	            tool_func = self.tools[tool_name]
   421	            tool_args = command.get("args", {})
   422	            if isinstance(tool_args, str):
   423	                tool_schema = next(
   424	                    (
   425	                        schema["args"]
   426	                        for schema in self.tool_schemas
   427	                        if schema["name"] == tool_name
   428	                    ),
   429	                    None,
   430	                )
   431	                return f"Error: Arguments for {tool_name} must be a JSON object with schema {tool_schema}, not a string."
   432	
   433	            # Call the tool function with arguments
   434	            result = tool_func(**tool_args)
   435	            if max_chars > 0 and len(str(result)) > max_chars:
   436	                result = str(result)[:max_chars] + "..."
   437	            return str(result)
   438	        except Exception as e:
   439	            return (
   440	                f"Error: {str(e)}. "
   441	                + 'Please format your tool call as \'{{"name": "tool_name", "args": {{"arg1": "value1", "arg2": "value2"}}}}\''
   442	            )
   443	
   444	    def env_response(
   445	        self, messages: Messages, state: State, **kwargs
   446	    ) -> Tuple[Messages, State]:
   447	        try:
   448	            parsed = self.parser.parse(messages[-1]["content"])  # type: ignore
   449	            # Check if we got a valid tool field (not just None from failed parsing)
   450	            if hasattr(parsed, "tool") and parsed.tool is not None:
   451	                result = self.call_tool(parsed.tool)
   452	                if len(result.strip()) > 0:
   453	                    return [
   454	                        {
   455	                            "role": "user",
   456	                            "content": self.env_parser.format(result=result),
   457	                        }
   458	                    ], state
   459	                else:
   460	                    return [
   461	                        {
   462	                            "role": "user",
   463	                            "content": "Error: Tool execution returned empty output.",
   464	                        }
   465	                    ], state
   466	        except Exception:
   467	            pass
   468	        return [
   469	            {
   470	                "role": "user",
   471	                "content": "Error: Tool command not found or invalid XML format. Please ensure correct formatting.",
   472	            }
   473	        ], state

    92	def load_environment(**kwargs) -> vf.Environment:
    93	    dataset: Dataset = load_dataset("agentlans/wikipedia-paragraphs", split="train")  # type: ignore
    94	    dataset = dataset.map(lambda x: {"sentences": get_sentences(x["text"])})
    95	    dataset = dataset.filter(lambda x: len(x["sentences"]) == 5)
    96	    dataset = dataset.map(get_sentence_questions)
    97	    parser = vf.Parser()
    98	
    99	    def compare_answers_reward_func(parser, completion, info, **kwargs) -> float:
   100	        assert isinstance(completion, list)
   101	        model_answers = [
   102	            m["content"] for m in parser.get_assistant_messages(completion)
   103	        ]
   104	        answers = info["answers"]
   105	        reward = 0
   106	        if len(model_answers) != len(answers):
   107	            return 0
   108	
   109	        def similarity(a, b):
   110	            return SequenceMatcher(None, a, b).ratio()
   111	
   112	        for model_answer, answer in zip(model_answers[: len(answers)], answers):
   113	            reward += similarity(model_answer, answer) / len(answers)
   114	        return reward
   115	
   116	    rubric = vf.Rubric(parser=parser, funcs=[compare_answers_reward_func])
   117	
   118	    vf_env = SentenceRepeaterEnv(
   119	        dataset=dataset, parser=parser, rubric=rubric, **kwargs
   120	    )
   121	
   122	    return vf_env
