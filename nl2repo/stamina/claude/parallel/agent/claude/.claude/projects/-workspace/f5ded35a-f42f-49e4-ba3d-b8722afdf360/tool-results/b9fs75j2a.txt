    35	    # The global configuration object is private API, but it's handy enough to
    36	    # be reachable as `stamina.CONFIG` without importing `stamina._config`.
    37	    if name == "CONFIG":
    38	        from ._config import CONFIG
    39	
    40	        return CONFIG
    41	
    42	    if name != "__version__":
    43	        msg = f"module {__name__} has no attribute {name}"
    44	        raise AttributeError(msg)
    45	
    46	    from importlib.metadata import version
    47	
    48	    return version("stamina")

    17	    """
    18	    Test mode specification.
    19	
    20	    Strictly private.
    21	    """

    23	    __slots__ = ("attempts", "cap")

    25	    attempts: int

    32	    def get_attempts(self, non_testing_attempts: int | None) -> int:
    33	        """
    34	        Get the number of attempts to use.
    35	
    36	        Args:
    37	            non_testing_attempts: The number of attempts specified by the user.

    39	        Returns:
    40	            The number of attempts to use.

    48	class _Config:
    49	    """
    50	    Global stamina configuration.
    51	
    52	    Strictly private.
    53	    """
    54	
    55	    __slots__ = (
    56	        "_get_on_retry",
    57	        "_is_active",
    58	        "_on_retry",
    59	        "_testing",
    60	        "lock",
    61	    )
    62	
    63	    lock: Lock
    64	    _is_active: bool
    65	    _testing: _Testing | None
    66	    _on_retry: (
    67	        tuple[RetryHook, ...] | tuple[RetryHook | RetryHookFactory, ...] | None
    68	    )
    69	    _get_on_retry: Callable[[], tuple[RetryHook, ...]]
    70	
    71	    def __init__(self, lock: Lock) -> None:
    72	        self.lock = lock
    73	        self._is_active = True
    74	        self._testing = None
    75	
    76	        # Prepare delayed initialization.
    77	        self._on_retry = None
    78	        self._get_on_retry = self._init_on_first_retry
    79	
    80	    @property
    81	    def is_active(self) -> bool:
    82	        return self._is_active
    83	
    84	    @is_active.setter
    85	    def is_active(self, value: bool) -> None:
    86	        with self.lock:
    87	            self._is_active = value
    88	
    89	    @property
    90	    def testing(self) -> _Testing | None:
    91	        return self._testing
    92	
    93	    @testing.setter
    94	    def testing(self, value: _Testing | None) -> None:
    95	        with self.lock:
    96	            self._testing = value
    97	
    98	    @property
    99	    def on_retry(self) -> tuple[RetryHook, ...]:
   100	        return self._get_on_retry()
   101	
   102	    @on_retry.setter
   103	    def on_retry(
   104	        self, value: tuple[RetryHook | RetryHookFactory, ...] | None
   105	    ) -> None:
   106	        with self.lock:
   107	            self._get_on_retry = self._init_on_first_retry
   108	            self._on_retry = value
   109	
   110	    def _init_on_first_retry(self) -> tuple[RetryHook, ...]:
   111	        """
   112	        Perform delayed initialization of on_retry hooks.
   113	        """
   114	        with self.lock:
   115	            # Ensure hooks didn't init while waiting for the lock.
   116	            if self._get_on_retry == self._init_on_first_retry:
   117	                if self._on_retry is None:
   118	                    self._on_retry = get_default_hooks()
   119	
   120	                self._on_retry = init_hooks(self._on_retry)
   121	
   122	                self._get_on_retry = lambda: self._on_retry  # type: ignore[assignment, return-value]
   123	
   124	        return self._on_retry  # type: ignore[return-value]
   125

   174	def set_testing(
   175	    testing: bool, *, attempts: int = 1, cap: bool = False
   176	) -> _RestoreTestingCM:
   177	    """
   178	    Activate or deactivate test mode.
   179	
   180	    In testing mode, backoffs are disabled, and attempts are set to *attempts*.
   181	
   182	    If *cap* is True, the number of attempts is not set but capped at
   183	    *attempts*. This means that if *attempts* is greater than the number of
   184	    attempts specified by the user, the user's value is used.
   185	
   186	    Is idempotent and can be called repeatedly with the same values.
   187	
   188	    .. versionadded:: 24.3.0
   189	    .. versionadded:: 25.1.0 *cap*
   190	    .. versionadded:: 25.1.0 Can be used as a context manager.
   191	    """
   192	    old = CONFIG.testing
   193	    CONFIG.testing = _Testing(attempts, cap) if testing else None
   194	
   195	    return _RestoreTestingCM(old)

   105	class Attempt:
   106	    """
   107	    A context manager that can be used to retry code blocks.
   108	
   109	    Instances are yielded by the :func:`stamina.retry_context` iterator.
   110	
   111	    .. versionadded:: 23.2.0
   112	    """
   113	
   114	    __slots__ = ("_next_wait_fn", "_t_attempt")
   115	
   116	    _t_attempt: _t.AttemptManager
   117	
   118	    def __init__(
   119	        self,
   120	        attempt: _t.AttemptManager,
   121	        next_wait_fn: Callable[[int], float] | None,
   122	    ):
   123	        self._t_attempt = attempt
   124	        self._next_wait_fn = next_wait_fn
   125	
   126	    def __repr__(self) -> str:
   127	        return f"<Attempt num={self.num}, next_wait={float(self.next_wait)}>"
   128	
   129	    @property
   130	    def num(self) -> int:
   131	        """
   132	        The number of the current attempt.
   133	        """
   134	        return self._t_attempt.retry_state.attempt_number  # type: ignore[no-any-return]
   135	
   136	    @property
   137	    def next_wait(self) -> float:
   138	        """
   139	        The number of seconds of backoff before the *next* attempt if *this*
   140	        attempt fails.
   141	
   142	
   143	        .. warning::
   144	            This value does **not** include a possible random jitter and is
   145	            therefore just a *lower bound* of the actual value.
   146	
   147	        .. versionadded:: 24.3.0
   148	        """
   149	        return (
   150	            self._next_wait_fn(self._t_attempt.retry_state.attempt_number + 1)
   151	            if self._next_wait_fn
   152	            else 0.0
   153	        )
   154	
   155	    def __enter__(self) -> None:
   156	        return self._t_attempt.__enter__()  # type: ignore[no-any-return]
   157	
   158	    def __exit__(
   159	        self,
   160	        exc_type: type[BaseException] | None,
   161	        exc_value: BaseException | None,
   162	        traceback: TracebackType | None,
   163	    ) -> bool | None:
   164	        return self._t_attempt.__exit__(  # type: ignore[no-any-return]
   165	            exc_type, exc_value, traceback
   166	        )

   178	class BaseRetryingCaller:
   179	    """
   180	    Simple base class that transforms retry parameters into a dictionary that
   181	    can be `**`-passed into `retry_context`.
   182	    """
   183	
   184	    __slots__ = ("_context_kws",)
   185	
   186	    _context_kws: RetryKWs
   187	
   188	    def __init__(
   189	        self,
   190	        attempts: int | None = 10,
   191	        timeout: float | dt.timedelta | None = 45.0,
   192	        wait_initial: float | dt.timedelta = 0.1,
   193	        wait_max: float | dt.timedelta = 5.0,
   194	        wait_jitter: float | dt.timedelta = 1.0,
   195	        wait_exp_base: float = 2.0,
   196	    ):
   197	        self._context_kws = {
   198	            "attempts": attempts,
   199	            "timeout": timeout,
   200	            "wait_initial": wait_initial,
   201	            "wait_max": wait_max,
   202	            "wait_jitter": wait_jitter,
   203	            "wait_exp_base": wait_exp_base,
   204	        }
   205	
   206	    def __repr__(self) -> str:
   207	        kws = ", ".join(
   208	            f"{k}={self._context_kws[k]!r}"  # type: ignore[literal-required]
   209	            for k in sorted(self._context_kws)
   210	            if k != "on"
   211	        )
   212	        return f"<{self.__class__.__name__}({kws})>"

   215	class RetryingCaller(BaseRetryingCaller):
   216	    """
   217	    Call your callables with retries.
   218	
   219	    Arguments have the same meaning as for :func:`stamina.retry`.
   220	
   221	    Tip:
   222	        Instances of ``RetryingCaller`` may be reused because they internally
   223	        create a new :func:`retry_context` iterator on each call.
   224	
   225	    .. versionadded:: 24.2.0
   226	    """
   227	
   228	    def __call__(
   229	        self,
   230	        on: ExcOrPredicate,
   231	        callable_: Callable[P, T],
   232	        /,
   233	        *args: P.args,
   234	        **kwargs: P.kwargs,
   235	    ) -> T:
   236	        r"""
   237	        Call ``callable_(*args, **kw)`` with retries if *on* is raised.
   238	
   239	        Args:
   240	            on: Exception(s) to retry on.
   241	
   242	            callable\_: Callable to call.
   243	
   244	            args: Positional arguments to pass to *callable_*.
   245	
   246	            kw: Keyword arguments to pass to *callable_*.
   247	        """
   248	        for attempt in retry_context(on, **self._context_kws):
   249	            with attempt:
   250	                return callable_(*args, **kwargs)
   251	
   252	        raise SystemError("unreachable")  # noqa: EM101
   253	
   254	    def on(self, on: ExcOrPredicate, /) -> BoundRetryingCaller:
   255	        """
   256	        Create a new instance of :class:`BoundRetryingCaller` with the same
   257	        parameters, but bound to a specific exception type.
   258	
   259	        .. versionadded:: 24.2.0
   260	        """
   261	        # This should be a `functools.partial`, but unfortunately it's
   262	        # impossible to provide a nicely typed API with it, so we use a
   263	        # separate class.
   264	        return BoundRetryingCaller(self, on)

   267	class BoundRetryingCaller:
   268	    """
   269	    Same as :class:`RetryingCaller`, but pre-bound to a specific exception
   270	    type.
   271	
   272	    Caution:
   273	        Returned by :meth:`RetryingCaller.on` -- do not instantiate directly.
   274	
   275	    .. versionadded:: 24.2.0
   276	    """
   277	
   278	    __slots__ = ("_caller", "_on")
   279	
   280	    _caller: RetryingCaller
   281	    _on: ExcOrPredicate
   282	
   283	    def __init__(
   284	        self,
   285	        caller: RetryingCaller,
   286	        on: ExcOrPredicate,
   287	    ):
   288	        self._caller = caller
   289	        self._on = on
   290	
   291	    def __repr__(self) -> str:
   292	        return (
   293	            f"<BoundRetryingCaller({guess_name(self._on)}, {self._caller!r})>"
   294	        )

   296	    def __call__(
   297	        self, callable_: Callable[P, T], /, *args: P.args, **kwargs: P.kwargs
   298	    ) -> T:
   299	        """
   300	        Same as :func:`RetryingCaller.__call__`, except retry on the exception

   302	        """
   303	        return self._caller(self._on, callable_, *args, **kwargs)

   306	class AsyncRetryingCaller(BaseRetryingCaller):
   307	    """
   308	    Same as :class:`RetryingCaller`, but for async callables.
   309	
   310	    .. versionadded:: 24.2.0
   311	    """
   312	
   313	    async def __call__(
   314	        self,
   315	        on: ExcOrPredicate,
   316	        callable_: Callable[P, Awaitable[T]],
   317	        /,
   318	        *args: P.args,
   319	        **kwargs: P.kwargs,
   320	    ) -> T:
   321	        """
   322	        Same as :meth:`RetryingCaller.__call__`, but *callable_* is awaited.
   323	        """
   324	        async for attempt in retry_context(on, **self._context_kws):
   325	            with attempt:
   326	                return await callable_(*args, **kwargs)
   327	
   328	        raise SystemError("unreachable")  # noqa: EM101
   329	
   330	    def on(self, on: ExcOrPredicate, /) -> BoundAsyncRetryingCaller:
   331	        """
   332	        Create a new instance of :class:`BoundAsyncRetryingCaller` with the
   333	        same parameters, but bound to a specific exception type.
   334	
   335	        .. versionadded:: 24.2.0
   336	        """
   337	        return BoundAsyncRetryingCaller(self, on)

   434	        timeout: float | dt.timedelta | None,
   435	        wait_initial: float | dt.timedelta,
   436	        wait_max: float | dt.timedelta,
   437	        wait_jitter: float | dt.timedelta,
   438	        wait_exp_base: float,
   439	        name: str,
   440	        args: tuple[object, ...],
   441	        kw: dict[str, object],
   442	    ) -> _RetryContextIterator:
   443	        if (
   444	            isinstance(on, type) and issubclass(on, BaseException)
   445	        ) or isinstance(on, tuple):
   446	            _retry = _t.retry_if_exception_type(on)
   447	        else:
   448	            _retry = _t.retry_if_exception(on)
   449	
   450	        if isinstance(wait_initial, dt.timedelta):
   451	            wait_initial = wait_initial.total_seconds()
   452	
   453	        if isinstance(wait_max, dt.timedelta):
   454	            wait_max = wait_max.total_seconds()
   455	
   456	        if isinstance(wait_jitter, dt.timedelta):
   457	            wait_jitter = wait_jitter.total_seconds()
   458	
   459	        inst = cls(
   460	            _name=name,
   461	            _args=args,
   462	            _kw=kw,
   463	            _attempts=attempts,
   464	            _wait_jitter=wait_jitter,
   465	            _wait_initial=wait_initial,
   466	            _wait_max=wait_max,
   467	            _wait_exp_base=wait_exp_base,
   468	            _t_kw={
   469	                "retry": _retry,
   470	                "stop": _make_stop(
   471	                    attempts=attempts,
   472	                    timeout=(
   473	                        timeout.total_seconds()
   474	                        if isinstance(timeout, dt.timedelta)
   475	                        else timeout
   476	                    ),
   477	                ),
   478	                "reraise": True,
   479	            },
   480	            _t_a_retrying=_LAZY_NO_ASYNC_RETRY,
   481	            _cms_to_exit=[],
   482	        )
   483	
   484	        inst._t_kw["wait"] = inst._jittered_backoff_for_rcs
   485	
   486	        return inst
   487	
   488	    def with_name(
   489	        self, name: str, args: tuple[object, ...], kw: dict[str, object]
   490	    ) -> _RetryContextIterator:
   491	        """
   492	        Recreate ourselves with a new name and arguments.
   493	        """
   494	        return replace(self, _name=name, _args=args, _kw=kw)
   495	
   496	    def _apply_maybe_test_mode_to_tenacity_kw(
   497	        self, testing: _Testing | None
   498	    ) -> dict[str, object]:
   499	        if testing is None:
   500	            return self._t_kw
   501	
   502	        t_kw = self._t_kw.copy()
   503	
   504	        t_kw["stop"] = _t.stop_after_attempt(
   505	            testing.get_attempts(self._attempts)
   506	        )
   507	
   508	        return t_kw
   509	
   510	    def _exit_cms(self, _: _t.RetryCallState | None) -> None:
   511	        for cm in reversed(self._cms_to_exit):
   512	            cm.__exit__(None, None, None)
   513	
   514	    def __iter__(self) -> Iterator[Attempt]:
   515	        if not CONFIG.is_active:
   516	            for r in _t.Retrying(reraise=True, stop=_STOP_NO_RETRY):
   517	                yield Attempt(r, None)
   518	
   519	            return
   520	
   521	        before_sleep = _make_before_sleep(
   522	            self._name, CONFIG, self._args, self._kw, self._cms_to_exit
   523	        )
   524	
   525	        for r in _t.Retrying(
   526	            before=self._exit_cms,
   527	            before_sleep=before_sleep,
   528	            **self._apply_maybe_test_mode_to_tenacity_kw(CONFIG.testing),
   529	        ):
   530	            yield Attempt(r, self._backoff_for_attempt_number)
   531	
   532	    def __aiter__(self) -> AsyncIterator[Attempt]:
   533	        if CONFIG.is_active:
   534	            self._t_a_retrying = _t.AsyncRetrying(
   535	                sleep=_smart_sleep,
   536	                before=self._exit_cms,
   537	                before_sleep=_make_before_sleep(
   538	                    self._name, CONFIG, self._args, self._kw, self._cms_to_exit
   539	                ),
   540	                **self._apply_maybe_test_mode_to_tenacity_kw(CONFIG.testing),
   541	            )
   542	
   543	        self._t_a_retrying = self._t_a_retrying.__aiter__()
   544	
   545	        return self
   546	
   547	    async def __anext__(self) -> Attempt:
   548	        return Attempt(
   549	            await self._t_a_retrying.__anext__(),
   550	            self._backoff_for_attempt_number,
   551	        )
   552	
   553	    def _backoff_for_attempt_number(self, num: int) -> float:
   554	        """
   555	        Compute a jitter-less lower bound for backoff number *num*.
   556	
   557	        *num* is 1-based.
   558	        """
   559	        return _compute_backoff(
   560	            num, self._wait_max, self._wait_initial, self._wait_exp_base, 0
   561	        )
   562	
   563	    def _jittered_backoff_for_rcs(self, rcs: _t.RetryCallState) -> float:
   564	        """
   565	        Compute the backoff for *rcs*.
   566	        """
   567	        return _compute_backoff(
   568	            rcs.attempt_number,
   569	            self._wait_max,
   570	            self._wait_initial,
   571	            self._wait_exp_base,
   572	            self._wait_jitter,
   573	        )
   574	
   575	
   576	def _compute_backoff(
   577	    num: int,
   578	    max_backoff: float,
   579	    initial: float,
   580	    exp_base: float,
   581	    max_jitter: float,
   582	) -> float:
   583	    """
   584	    If not in testing mode, compute the backoff for attempt *num* with the
   585	    given parameters and clamp it to *max_backoff*.
   586	    """
   587	    if CONFIG.testing is not None:
   588	        return 0.0
   589	
   590	    jitter = random.uniform(0, max_jitter) if max_jitter else 0  # noqa: S311
   591	
   592	    return min(max_backoff, initial * (exp_base ** (num - 1)) + jitter)
   593	
   594	
   595	def _make_before_sleep(
   596	    name: str,
   597	    config: _Config,
   598	    args: tuple[object, ...],
   599	    kw: dict[str, object],
   600	    hook_cms: list[contextlib.AbstractContextManager[None]],
   601	) -> Callable[[_t.RetryCallState], None]:
   602	    """
   603	    Create a `before_sleep` callback function that runs our `RetryHook`s with
   604	    the necessary arguments.
   605	
   606	    If a hook returns a context manager, it's entered before retries start and
   607	    exited after they finish by keeping track of the context managers in
   608	    *hook_cms*.
   609	    """
   610	
   611	    last_idle_for = 0.0
   612	
   613	    def before_sleep(rcs: _t.RetryCallState) -> None:
   614	        nonlocal last_idle_for
   615	
   616	        wait_for = rcs.idle_for - last_idle_for
   617	
   618	        details = RetryDetails(
   619	            name=name,
   620	            retry_num=rcs.attempt_number,
   621	            wait_for=wait_for,
   622	            waited_so_far=rcs.idle_for - wait_for,
   623	            caused_by=rcs.outcome.exception(),
   624	            args=args,
   625	            kwargs=kw,
   626	        )
   627	
   628	        for hook in config.on_retry:
   629	            maybe_cm = hook(details)
   630	
   631	            if isinstance(maybe_cm, AbstractContextManager):
   632	                maybe_cm.__enter__()
   633	                hook_cms.append(maybe_cm)
   634	
   635	        last_idle_for = rcs.idle_for
   636	
   637	    return before_sleep
   638	
   639	
   640	def _make_stop(*, attempts: int | None, timeout: float | None) -> _t.stop_base:
   641	    """
   642	    Combine *attempts* and *timeout* into one stop condition.
   643	    """
   644	    stops = []
   645	
   646	    if attempts:
   647	        stops.append(_t.stop_after_attempt(attempts))
   648	
   649	    if timeout:
   650	        stops.append(_t.stop_after_delay(timeout))
   651	
   652	    if len(stops) > 1:
   653	        return _t.stop_any(*stops)
   654	
   655	    if not stops:
   656	        return _t.stop_never
   657	
   658	    return stops[0]

   661	def retry(
   662	    *,
   663	    on: ExcOrPredicate,
   664	    attempts: int | None = 10,
   665	    timeout: float | dt.timedelta | None = 45.0,
   666	    wait_initial: float | dt.timedelta = 0.1,
   667	    wait_max: float | dt.timedelta = 5.0,
   668	    wait_jitter: float | dt.timedelta = 1.0,
   669	    wait_exp_base: float = 2.0,
   670	) -> Callable[[Callable[P, T]], Callable[P, T]]:
   671	    r"""
   672	    Retry if one of configured exceptions are raised.
   673	
   674	    The backoff delays between retries grow exponentially plus a random jitter.
   675	
   676	    The backoff for retry attempt number *attempt* is computed as:
   677	
   678	    .. keep in-sync with docs/motivation.md
   679	    .. math::
   680	
   681	       min(wait\_max, wait\_initial * wait\_exp\_base^{attempt - 1} + random(0, wait\_jitter))
   682	
   683	    Since :math:`x^0` is always 1, the first backoff is within the interval
   684	    :math:`[wait\_initial,wait\_initial+wait\_jitter]`. Thus, with default
   685	    values between 0.1 and 1.1 seconds.
   686	
   687	    If all retries fail, the *last* exception is let through.
   688	
   689	    All float-based time parameters are in seconds.
   690	
   691	    .. warning::
   692	       It is possible to get unbounded retries by passing `None` for *attempts*
   693	       (= infinite attempts) and *timeout* (= no timeout).
   694	
   695	    Args:
   696	        on:
   697	            An Exception or a tuple of Exceptions on which the decorated
   698	            callable will be retried.
   699	
   700	            You can also pass a *predicate* in the form of a callable that
   701	            takes an exception and returns a bool which decides whether the
   702	            exception should be retried -- True meaning yes.
   703	
   704	            This allows more fine-grained control over when to retry. For
   705	            example, to only retry on HTTP errors in the 500s range that
   706	            indicate server errors, but not those in the 400s which indicate a
   707	            client error.
   708	
   709	            There is no default -- you *must* pass this explicitly.
   710	
   711	        attempts:
   712	            Maximum total number of attempts. Can be combined with *timeout*.
   713	            `None` means *no* limit to attempts.
   714	
   715	        timeout:
   716	            Maximum total time for all retries. Can be combined with
   717	            *attempts*. `None` means *no* timeout.
   718	
   719	        wait_initial: Minimum backoff before the *first* retry.
   720	
   721	        wait_max: Maximum backoff time between retries at any time.
   722	
   723	        wait_jitter:
   724	            Maximum *jitter* that is added to retry back-off delays (the actual
   725	            jitter added is a random number between 0 and *wait_jitter*)
   726	
   727	        wait_exp_base: The exponential base used to compute the retry backoff.
   728	
   729	    .. versionchanged:: 23.1.0
   730	       All time-related parameters can now be specified as a
   731	       :class:`datetime.timedelta`.
   732	
   733	    .. versionadded:: 23.3.0 `Trio <https://trio.readthedocs.io/>`_ support.
   734	
   735	    .. versionadded:: 24.3.0 *on* can be a callable now.
   736	    """
   737	    retry_ctx = _RetryContextIterator.from_params(
   738	        on=on,
   739	        attempts=attempts,
   740	        timeout=timeout,
   741	        wait_initial=wait_initial,
   742	        wait_max=wait_max,
   743	        wait_jitter=wait_jitter,
   744	        wait_exp_base=wait_exp_base,
   745	        name="<unknown>",
   746	        args=(),
   747	        kw={},
   748	    )
   749	
   750	    def retry_decorator(wrapped: Callable[P, T]) -> Callable[P, T]:
   751	        name = guess_name(wrapped)
   752	
   753	        if not iscoroutinefunction(wrapped):
   754	
   755	            @wraps(wrapped)
   756	            def sync_inner(*args: P.args, **kw: P.kwargs) -> T:  # type: ignore[return]
   757	                for attempt in retry_ctx.with_name(  # noqa: RET503
   758	                    name, args, kw
   759	                ):
   760	                    with attempt:
   761	                        return wrapped(*args, **kw)
   762	
   763	            return sync_inner
   764	
   765	        @wraps(wrapped)
   766	        async def async_inner(*args: P.args, **kw: P.kwargs) -> T:  # type: ignore[return]
   767	            async for attempt in retry_ctx.with_name(  # noqa: RET503
   768	                name, args, kw
   769	            ):
   770	                with attempt:
   771	                    return await wrapped(*args, **kw)  # type: ignore[no-any-return]
   772	
   773	        return async_inner  # type: ignore[return-value]
   774	
   775	    return retry_decorator
