    57	def build_camel_case_aliases(PyQuery):
    58	    """add camelcase aliases to PyQuery"""
    59	    for alias in _camel_case_aliases:
    60	        parts = list(alias.split('_'))
    61	        name = parts[0] + ''.join([p.title() for p in parts[1:]])
    62	        func = getattr(PyQuery, alias)
    63	        f = types.FunctionType(func.__code__, func.__globals__,
    64	                               name, func.__defaults__)
    65	        f.__doc__ = (
    66	            'Alias for :func:`~pyquery.pyquery.PyQuery.%s`') % func.__name__
    67	        setattr(PyQuery, name, f.__get__(None, PyQuery))

    70	####################################################
    71	# libxml2 HTML parsing backward compatibility layer #
    72	####################################################
    73	
    74	# libxml2 2.14 replaced its historical HTML parser by an HTML5 conformant
    75	# one. Three of its behaviours changed the shape of the tree pyquery has
    76	# always exposed (and documented), so they are restored here to keep the
    77	# same document giving the same nodes whatever libxml2 is in use:
    78	#
    79	# * character data at the top level of a document used to open an implied
    80	#   ``<p>`` element,
    81	# * only the HTML 4 named entities were resolved, the HTML 5 only ones were
    82	#   kept as plain text,
    83	# * the content of a ``<textarea>`` was parsed as markup instead of raw text.
    84	
    85	# named entities known to the historical libxml2 parser
    86	_legacy_entities = frozenset(name2codepoint) | {'apos'}
    87	
    88	# entities only libxml2 >= 2.14 resolves
    89	_html5_entities = frozenset(
    90	    name.rstrip(';') for name in html5) - _legacy_entities
    91	
    92	_entity_regex = re.compile(r'&([a-zA-Z][a-zA-Z0-9]*);')
    93	
    94	# elements whose content is never parsed as markup nor unescaped
    95	_raw_text_regex = re.compile(r'<(script|style)\b.*?(?:</\1\s*>|\Z)',
    96	                             re.IGNORECASE | re.DOTALL)
    97	
    98	_body_regex = re.compile(r'<body[\s/>]', re.IGNORECASE)
    99	
   100	# elements which used to close an open <p> element
   101	_block_tags = frozenset((
   102	    'address', 'article', 'aside', 'blockquote', 'center', 'dd', 'details',
   103	    'dialog', 'dir', 'div', 'dl', 'dt', 'fieldset', 'figcaption', 'figure',
   104	    'footer', 'form', 'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'head', 'header',
   105	    'hgroup', 'hr', 'li', 'main', 'menu', 'nav', 'noframes', 'ol', 'p',
   106	    'pre', 'section', 'table', 'ul',
   107	))
   108	
   109	
   110	def _escape_html5_entities(html):
   111	    """keep the entities libxml2 used to ignore as plain text"""
   112	    def repl(match):
   113	        if match.group(1) in _html5_entities:
   114	            return '&amp;%s;' % match.group(1)
   115	        return match.group(0)
   116	
   117	    result = []
   118	    position = 0
   119	    # entities are not resolved in raw text elements
   120	    for raw_text in _raw_text_regex.finditer(html):
   121	        result.append(_entity_regex.sub(repl, html[position:raw_text.start()]))
   122	        result.append(raw_text.group(0))
   123	        position = raw_text.end()
   124	    result.append(_entity_regex.sub(repl, html[position:]))
   125	    return ''.join(result)

   128	def _restore_textarea_markup(element):
   129	    """parse the raw content of the textareas as markup"""
   130	    for tag in element.iter('textarea'):
   131	        text = tag.text
   132	        if not text or '<' not in text or len(tag):
   133	            continue
   134	        try:
   135	            root = lxml.html.fragment_fromstring(text, create_parent='div')
   136	        except Exception:
   137	            continue
   138	        children = root.getchildren()
   139	        if not children:
   140	            continue
   141	        tag.text = root.text
   142	        tag.extend(children)

   145	def _wrap_leading_text(container):
   146	    """move the leading character data of container in a <p> element"""
   147	    text = container.text
   148	    if not text or not text.strip():
   149	        return None
   150	    paragraph = container.makeelement('p')
   151	    paragraph.text = text
   152	    container.text = None
   153	    for child in list(container):
   154	        tag = child.tag
   155	        if isinstance(tag, basestring) and tag.lower() in _block_tags:
   156	            # the paragraph used to be closed by such elements
   157	            break
   158	        # removing the child keeps its tail attached to it
   159	        container.remove(child)
   160	        paragraph.append(child)
   161	    container.insert(0, paragraph)
   162	    return paragraph
   163	
   164	
   165	def _restore_implied_paragraph(element):
   166	    """open an implied <p> element for top level character data"""
   167	    if element.tag == 'html':
   168	        body = element.find('body')
   169	        if body is not None:
   170	            _wrap_leading_text(body)
   171	        return element
   172	    parent = element.getparent()
   173	    if (element.tag not in ('span', 'div') or
   174	            parent is None or parent.tag != 'html'):
   175	        # not one of the containers lxml.html builds for html fragments
   176	        return element
   177	    paragraph = _wrap_leading_text(element)
   178	    if paragraph is None:
   179	        return element
   180	    if len(element) == 1 and not (paragraph.tail or '').strip():
   181	        # lxml.html would have returned the paragraph itself
   182	        element.tag = 'body'
   183	        return paragraph
   184	    # the container now holds a block level element
   185	    element.tag = 'div'
   186	    return element

   189	def _html_parser(meth):
   190	    """``lxml.html`` parser with the historical behaviours restored"""
   191	    parse = getattr(lxml.html, meth)
   192	
   193	    def parser(context):
   194	        # bytes may carry an encoding declaration. leave them alone
   195	        source = context if isinstance(context, str) else None
   196	        if source is not None:
   197	            context = _escape_html5_entities(source)
   198	        result = parse(context)
   199	        if isinstance(result, etree._ElementTree):
   200	            elements = [result.getroot()]
   201	        elif isinstance(result, list):
   202	            elements = result
   203	        elif result is None:
   204	            elements = []
   205	        else:
   206	            elements = [result]
   207	        elements = [e for e in elements if e is not None]
   208	        for element in elements:
   209	            if not isinstance(element, basestring):
   210	                _restore_textarea_markup(element)
   211	        if source is None or _body_regex.search(source):
   212	            return elements
   213	        return [element if isinstance(element, basestring)
   214	                else _restore_implied_paragraph(element)
   215	                for element in elements]
   216	
   217	    return parser

   220	def fromstring(context, parser=None, custom_parser=None):
   221	    """use html parser if we don't have clean xml
   222	    """
   223	    if hasattr(context, 'read') and hasattr(context.read, '__call__'):
   224	        meth = 'parse'
   225	    else:
   226	        meth = 'fromstring'
   227	    if custom_parser is None:
   228	        if parser is None:
   229	            try:
   230	                result = getattr(etree, meth)(context)
   231	            except etree.XMLSyntaxError:
   232	                if hasattr(context, 'seek'):
   233	                    context.seek(0)
   234	                return _html_parser(meth)(context)
   235	            if isinstance(result, etree._ElementTree):
   236	                return [result.getroot()]
   237	            else:
   238	                return [result]
   239	        elif parser == 'xml':
   240	            custom_parser = getattr(etree, meth)
   241	        elif parser == 'html':
   242	            custom_parser = _html_parser(meth)
   243	        elif parser == 'html5':
   244	            from lxml.html import html5parser
   245	            custom_parser = getattr(html5parser, meth)
   246	        elif parser == 'soup':
   247	            from lxml.html import soupparser
   248	            custom_parser = getattr(soupparser, meth)
   249	        elif parser == 'html_fragments':
   250	            custom_parser = _html_parser('fragments_fromstring')
   251	        else:
   252	            raise ValueError('No such parser: "%s"' % parser)
   253	
   254	    result = custom_parser(context)
   255	    if isinstance(result, list):
   256	        return result
   257	    elif isinstance(result, etree._ElementTree):
   258	        return [result.getroot()]
   259	    elif result is not None:
   260	        return [result]
   261	    else:
   262	        return []

   279	class FlexibleElement(object):
   280	    """property to allow a flexible api"""
   281	    def __init__(self, pget, pset=no_default, pdel=no_default):
   282	        self.pget = pget
   283	        self.pset = pset
   284	        self.pdel = pdel
   285	
   286	    def __get__(self, instance, klass):
   287	        class _element(object):
   288	            """real element to support set/get/del attr and item and js call
   289	            style"""
   290	            def __call__(prop, *args, **kwargs):
   291	                return self.pget(instance, *args, **kwargs)
   292	            __getattr__ = __getitem__ = __setattr__ = __setitem__ = __call__
   293	
   294	            def __delitem__(prop, name):
   295	                if self.pdel is not no_default:
   296	                    return self.pdel(instance, name)
   297	                else:
   298	                    raise NotImplementedError()
   299	            __delattr__ = __delitem__
   300	
   301	            def __repr__(prop):
   302	                return '<flexible_element %s>' % self.pget.__name__
   303	        return _element()
   304	
   305	    def __set__(self, instance, value):
   306	        if self.pset is not no_default:
   307	            self.pset(instance, value)
   308	        else:
   309	            raise NotImplementedError()

   312	class PyQuery(list):
   313	    """The main class
   314	    """
   315	
   316	    _translator_class = JQueryTranslator
   317	
   318	    def __init__(self, *args, **kwargs):
   319	        html = None
   320	        elements = []
   321	        self._base_url = None
   322	        self.parser = kwargs.pop('parser', None)
   323	
   324	        if 'parent' in kwargs:
   325	            self._parent = kwargs.pop('parent')
   326	        else:
   327	            self._parent = no_default
   328	
   329	        if 'css_translator' in kwargs:
   330	            self._translator = kwargs.pop('css_translator')
   331	        elif self.parser in ('xml',):
   332	            self._translator = self._translator_class(xhtml=True)
   333	        elif self._parent is not no_default:
   334	            self._translator = self._parent._translator
   335	        else:
   336	            self._translator = self._translator_class(xhtml=False)
   337	
   338	        self.namespaces = kwargs.pop('namespaces', None)
   339	
   340	        if kwargs:
   341	            # specific case to get the dom
   342	            if 'filename' in kwargs:
   343	                html = open(kwargs['filename'],
   344	                            encoding=kwargs.get('encoding'))
   345	            elif 'url' in kwargs:
   346	                url = kwargs.pop('url')
   347	                if 'opener' in kwargs:
   348	                    opener = kwargs.pop('opener')
   349	                    html = opener(url, **kwargs)
   350	                else:
   351	                    html = url_opener(url, kwargs)
   352	                if not self.parser:
   353	                    self.parser = 'html'
   354	                self._base_url = url
   355	            else:
   356	                raise ValueError('Invalid keyword arguments %s' % kwargs)
   357	
   358	            elements = fromstring(html, self.parser)
   359	            # close open descriptor if possible
   360	            if hasattr(html, 'close'):
   361	                try:
   362	                    html.close()
   363	                except Exception:
   364	                    pass
   365	
   366	        else:
   367	            # get nodes
   368	
   369	            # determine context and selector if any
   370	            selector = context = no_default
   371	            length = len(args)
   372	            if length == 1:
   373	                context = args[0]
   374	            elif length == 2:
   375	                selector, context = args
   376	            else:
   377	                raise ValueError(
   378	                    "You can't do that. Please, provide arguments")
   379	
   380	            # get context
   381	            if isinstance(context, basestring):
   382	                try:
   383	                    elements = fromstring(context, self.parser)
   384	                except Exception:
   385	                    raise
   386	            elif isinstance(context, self.__class__):
   387	                # copy
   388	                elements = context[:]
   389	            elif isinstance(context, list):
   390	                elements = context
   391	            elif isinstance(context, etree._Element):
   392	                elements = [context]
   393	            else:
   394	                raise TypeError(context)
   395	
   396	            # select nodes
   397	            if elements and selector is not no_default:
   398	                xpath = self._css_to_xpath(selector)
   399	                results = []
   400	                for tag in elements:
   401	                    results.extend(
   402	                        tag.xpath(xpath, namespaces=self.namespaces))
   403	                elements = results
   404	
   405	        list.__init__(self, elements)

   407	    def _css_to_xpath(self, selector, prefix='descendant-or-self::'):
   408	        selector = selector.replace('[@', '[')
   409	        return self._translator.css_to_xpath(selector, prefix)

   411	    def _copy(self, *args, **kwargs):
   412	        kwargs.setdefault('namespaces', self.namespaces)
   413	        return self.__class__(*args, **kwargs)
   414	
   415	    def __call__(self, *args, **kwargs):
   416	        """return a new PyQuery instance
   417	        """
   418	        length = len(args)
   419	        if length == 0:
   420	            raise ValueError('You must provide at least a selector')
   421	        if args[0] == '':
   422	            return self._copy([])
   423	        if (len(args) == 1 and
   424	                isinstance(args[0], str) and
   425	                not args[0].startswith('<')):
   426	            args += (self,)
   427	        result = self._copy(*args, parent=self, **kwargs)
   428	        return result

   445	    def items(self, selector=None):
   446	        """Iter over elements. Return PyQuery objects:
   447	
   448	            >>> d = PyQuery('<div><span>foo</span><span>bar</span></div>')
   449	            >>> [i.text() for i in d.items('span')]
   450	            ['foo', 'bar']
   451	            >>> [i.text() for i in d('span').items()]
   452	            ['foo', 'bar']
   453	            >>> list(d.items('a')) == list(d('a').items())
   454	            True
   455	        """
   456	        if selector:
   457	            elems = self(selector) or []
   458	        else:
   459	            elems = self
   460	        for elem in elems:
   461	            yield self._copy(elem, parent=self)

   463	    def xhtml_to_html(self):
   464	        """Remove xhtml namespace:
   465	
   466	            >>> doc = PyQuery(
   467	            ...         '<html xmlns="http://www.w3.org/1999/xhtml"></html>')
   468	            >>> doc
   469	            [<{http://www.w3.org/1999/xhtml}html>]
   470	            >>> doc.xhtml_to_html()
   471	            [<html>]
   472	        """
   473	        try:
   474	            root = self[0].getroottree()
   475	        except IndexError:
   476	            pass
   477	        else:
   478	            lxml.html.xhtml_to_html(root)
   479	        return self

   481	    def remove_namespaces(self):
   482	        """Remove all namespaces:
   483	
   484	            >>> doc = PyQuery('<foo xmlns="http://example.com/foo"></foo>')
   485	            >>> doc
   486	            [<{http://example.com/foo}foo>]
   487	            >>> doc.remove_namespaces()
   488	            [<foo>]
   489	        """
   490	        try:
   491	            root = self[0].getroottree()
   492	        except IndexError:
   493	            pass
   494	        else:
   495	            for el in root.iter('{*}*'):
   496	                if el.tag.startswith('{'):
   497	                    el.tag = el.tag.split('}', 1)[1]
   498	        return self

   529	        r = []
   530	        try:
   531	            for el in self:

   561	    def _filter_only(self, selector, elements, reverse=False, unique=False):
   562	        """Filters the selection set only, as opposed to also including
   563	           descendants.
   564	        """
   565	        if selector is None:
   566	            results = elements
   567	        else:
   568	            xpath = self._css_to_xpath(selector, 'self::')
   569	            results = []
   570	            for tag in elements:
   571	                results.extend(tag.xpath(xpath, namespaces=self.namespaces))
   572	        if reverse:
   573	            results.reverse()
   574	        if unique:
   575	            result_list = results
   576	            results = []
   577	            for item in result_list:
   578	                if item not in results:
   579	                    results.append(item)
   580	        return self._copy(results, parent=self)

   582	    def parent(self, selector=None):
   583	        return self._filter_only(
   584	            selector,
   585	            [e.getparent() for e in self if e.getparent() is not None],
   586	            unique=True)

   588	    def prev(self, selector=None):
   589	        return self._filter_only(
   590	            selector,
   591	            [e.getprevious() for e in self if e.getprevious() is not None])

   593	    def next(self, selector=None):
   594	        return self._filter_only(
   595	            selector,
   596	            [e.getnext() for e in self if e.getnext() is not None])

   598	    def _traverse(self, method):
   599	        for e in self:
   600	            current = getattr(e, method)()
   601	            while current is not None:
   602	                yield current
   603	                current = getattr(current, method)()
   604	
   605	    def _traverse_parent_topdown(self):
   606	        for e in self:
   607	            this_list = []
   608	            current = e.getparent()
   609	            while current is not None:
   610	                this_list.append(current)
   611	                current = current.getparent()
   612	            this_list.reverse()
   613	            for j in this_list:
   614	                yield j
   615	
   616	    def _next_all(self):
   617	        return [e for e in self._traverse('getnext')]

   619	    @with_camel_case_alias
   620	    def next_all(self, selector=None):
   621	        """
   622	        >>> h = '<span><p class="hello">Hi</p><p>Bye</p><img scr=""/></span>'
   623	        >>> d = PyQuery(h)
   624	        >>> d('p:last').next_all()
   625	        [<img>]
   626	        >>> d('p:last').nextAll()
   627	        [<img>]
   628	        """
   629	        return self._filter_only(selector, self._next_all())

   631	    @with_camel_case_alias
   632	    def next_until(self, selector, filter_=None):
   633	        """
   634	        >>> h = '''
   635	        ... <h2>Greeting 1</h2>
   636	        ... <p>Hello!</p><p>World!</p>
   637	        ... <h2>Greeting 2</h2><p>Bye!</p>
   638	        ... '''
   639	        >>> d = PyQuery(h)
   640	        >>> d('h2:first').nextUntil('h2')
   641	        [<p>, <p>]
   642	        """
   643	        return self._filter_only(
   644	            filter_, [
   645	                e
   646	                for q in itertools.takewhile(
   647	                    lambda q: not q.is_(selector), self.next_all().items())
   648	                for e in q
   649	            ]
   650	        )

   695	    def children(self, selector=None):
   696	        """Filter elements that are direct children of self using optional
   697	        selector:
   698	
   699	            >>> d = PyQuery('<span><p class="hello">Hi</p><p>Bye</p></span>')
   700	            >>> d
   701	            [<span>]
   702	            >>> d.children()
   703	            [<p.hello>, <p>]
   704	            >>> d.children('.hello')
   705	            [<p.hello>]
   706	        """
   707	        elements = [child for tag in self for child in tag.getchildren()]
   708	        return self._filter_only(selector, elements)

   710	    def closest(self, selector=None):
   711	        """
   712	        >>> d = PyQuery(
   713	        ...  '<div class="hello"><p>This is a '
   714	        ...  '<strong class="hello">test</strong></p></div>')
   715	        >>> d('strong').closest('div')
   716	        [<div.hello>]
   717	        >>> d('strong').closest('.hello')
   718	        [<strong.hello>]
   719	        >>> d('strong').closest('form')
   720	        []
   721	        """
   722	        result = []
   723	        for current in self:
   724	            while (current is not None and
   725	                    not self._copy(current).is_(selector)):
   726	                current = current.getparent()
   727	            if current is not None:
   728	                result.append(current)
   729	        return self._copy(result, parent=self)

   731	    def contents(self):
   732	        """
   733	        Return contents (with text nodes):
   734	
   735	            >>> d = PyQuery('hello <b>bold</b>')
   736	            >>> d.contents()  # doctest: +ELLIPSIS
   737	            ['hello ', <Element b at ...>]
   738	        """
   739	        results = []
   740	        for elem in self:
   741	            results.extend(elem.xpath('child::text()|child::*',
   742	                           namespaces=self.namespaces))
   743	        return self._copy(results, parent=self)

   745	    def filter(self, selector):
   746	        """Filter elements in self using selector (string or function):
   747	
   748	            >>> d = PyQuery('<p class="hello">Hi</p><p>Bye</p>')
   749	            >>> d('p')
   750	            [<p.hello>, <p>]
   751	            >>> d('p').filter('.hello')
   752	            [<p.hello>]
   753	            >>> d('p').filter(lambda i: i == 1)
   754	            [<p>]
   755	            >>> d('p').filter(lambda i: PyQuery(this).text() == 'Hi')
   756	            [<p.hello>]
   757	            >>> d('p').filter(lambda i, this: PyQuery(this).text() == 'Hi')
   758	            [<p.hello>]
   759	        """
   760	        if not hasattr(selector, '__call__'):
   761	            return self._filter_only(selector, self)
   762	        else:
   763	            elements = []
   764	            args = getargspec(callback)
   765	            try:
   766	                for i, this in enumerate(self):
   767	                    if len(args) == 1:
   768	                        selector.__globals__['this'] = this
   769	                    if callback(selector, i, this):
   770	                        elements.append(this)
   771	            finally:
   772	                f_globals = selector.__globals__
   773	                if 'this' in f_globals:
   774	                    del f_globals['this']
   775	            return self._copy(elements, parent=self)

   777	    def not_(self, selector):
   778	        """Return elements that don't match the given selector:
   779	
   780	            >>> d = PyQuery('<p class="hello">Hi</p><p>Bye</p><div></div>')
   781	            >>> d('p').not_('.hello')
   782	            [<p>]
   783	        """
   784	        exclude = set(self._copy(selector, self))
   785	        return self._copy([e for e in self if e not in exclude],
   786	                          parent=self)

   788	    def is_(self, selector):
   789	        """Returns True if selector matches at least one current element, else
   790	        False:
   791	
   792	            >>> d = PyQuery('<p class="hello"><span>Hi</span></p><p>Bye</p>')
   793	            >>> d('p').eq(0).is_('.hello')
   794	            True
   795	
   796	            >>> d('p').eq(0).is_('span')
   797	            False
   798	
   799	            >>> d('p').eq(1).is_('.hello')
   800	            False
   801	
   802	        ..
   803	        """
   804	        return bool(self._filter_only(selector, self))

   806	    def find(self, selector):
   807	        """Find elements using selector traversing down from self:
   808	
   809	            >>> m = '<p><span><em>Whoah!</em></span></p><p><em> there</em></p>'
   810	            >>> d = PyQuery(m)
   811	            >>> d('p').find('em')
   812	            [<em>, <em>]
   813	            >>> d('p').eq(1).find('em')
   814	            [<em>]
   815	        """
   816	        xpath = self._css_to_xpath(selector)
   817	        results = [child.xpath(xpath, namespaces=self.namespaces)
   818	                   for tag in self
   819	                   for child in tag.getchildren()]
   820	        # Flatten the results
   821	        elements = []
   822	        for r in results:
   823	            elements.extend(r)
   824	        return self._copy(elements, parent=self)

   826	    def eq(self, index):
   827	        """Return PyQuery of only the element with the provided index::
   828	
   829	            >>> d = PyQuery('<p class="hello">Hi</p><p>Bye</p><div></div>')
   830	            >>> d('p').eq(0)
   831	            [<p.hello>]
   832	            >>> d('p').eq(1)
   833	            [<p>]
   834	            >>> d('p').eq(2)
   835	            []
   836	
   837	        ..
   838	        """
   839	        # Slicing will return empty list when index=-1
   840	        # we should handle out of bound by ourselves
   841	        try:
   842	            items = self[index]
   843	        except IndexError:
   844	            items = []
   845	        return self._copy(items, parent=self)

   847	    def each(self, func):
   848	        """apply func on each nodes
   849	        """
   850	        try:
   851	            for i, element in enumerate(self):
   852	                func.__globals__['this'] = element
   853	                if callback(func, i, element) is False:
   854	                    break
   855	        finally:
   856	            f_globals = func.__globals__
   857	            if 'this' in f_globals:
   858	                del f_globals['this']
   859	        return self

   861	    def map(self, func):
   862	        """Returns a new PyQuery after transforming current items with func.
   863	
   864	        func should take two arguments - 'index' and 'element'.  Elements can
   865	        also be referred to as 'this' inside of func::
   866	
   867	            >>> d = PyQuery('<p class="hello">Hi there</p><p>Bye</p><br />')
   868	            >>> d('p').map(lambda i, e: PyQuery(e).text())
   869	            ['Hi there', 'Bye']
   870	
   871	            >>> d('p').map(lambda i, e: len(PyQuery(this).text()))
   872	            [8, 3]
   873	
   874	            >>> d('p').map(lambda i, e: PyQuery(this).text().split())
   875	            ['Hi', 'there', 'Bye']
   876	
   877	        """
   878	        items = []
   879	        try:
   880	            for i, element in enumerate(self):
   881	                func.__globals__['this'] = element
   882	                result = callback(func, i, element)
   883	                if result is not None:
   884	                    if not isinstance(result, list):
   885	                        items.append(result)
   886	                    else:
   887	                        items.extend(result)
   888	        finally:
   889	            f_globals = func.__globals__
   890	            if 'this' in f_globals:
   891	                del f_globals['this']
   892	        return self._copy(items, parent=self)

   914	    def attr(self, *args, **kwargs):
   915	        """Attributes manipulation
   916	        """
   917	
   918	        mapping = {'class_': 'class', 'for_': 'for'}
   919	
   920	        attr = value = no_default
   921	        length = len(args)
   922	        if length == 1:
   923	            attr = args[0]
   924	            attr = mapping.get(attr, attr)
   925	        elif length == 2:
   926	            attr, value = args
   927	            attr = mapping.get(attr, attr)
   928	        elif kwargs:
   929	            attr = {}
   930	            for k, v in kwargs.items():
   931	                attr[mapping.get(k, k)] = v
   932	        else:
   933	            raise ValueError('Invalid arguments %s %s' % (args, kwargs))
   934	
   935	        if not self:
   936	            return None
   937	        elif isinstance(attr, dict):
   938	            for tag in self:
   939	                for key, value in attr.items():
   940	                    tag.set(key, value)
   941	        elif value is no_default:
   942	            return self[0].get(attr)
   943	        elif value is None:
   944	            return self.remove_attr(attr)
   945	        else:
   946	            for tag in self:
   947	                tag.set(attr, value)
   948	        return self

   957	            >>> d.removeAttr('id')

   964	                del tag.attrib[name]

   999	    def add_class(self, value):

  1001	
  1002	            >>> d = PyQuery('<div></div>')

  1009	        """

  1017	    @with_camel_case_alias
  1018	    def remove_class(self, value):
  1019	        """Remove a css class to elements::
  1020	
  1021	            >>> d = PyQuery('<div class="myclass"></div>')
  1022	            >>> d.remove_class('myclass')
  1023	            [<div>]
  1024	            >>> d.removeClass('myclass')
  1025	            [<div>]
  1026	
  1027	        ..
  1028	        """
  1029	        for tag in self:
  1030	            values = value.split(' ')
  1031	            classes = set((tag.get('class') or '').split())
  1032	            classes.difference_update(values)
  1033	            classes.difference_update([''])
  1034	            classes = ' '.join(classes)
  1035	            if classes.strip():
  1036	                tag.set('class', classes)
  1037	            elif tag.get('class'):
  1038	                tag.set('class', classes)
  1039	        return self

  1044	
  1045	            >>> d = PyQuery('<div></div>')
  1046	            >>> d.toggle_class('myclass')
  1047	            [<div.myclass>]
  1048	            >>> d.toggleClass('myclass')
  1049	            [<div>]
  1050	
  1051	        """
  1052	        for tag in self:
  1053	            values = value.split(' ')
  1054	            classes = (tag.get('class') or '').split()
  1055	            values_to_add = [v for v in values if v not in classes]
  1056	            values_to_del = [v for v in values if v in classes]
  1057	            classes = [v for v in classes if v not in values_to_del]
  1058	            classes += values_to_add
  1059	            tag.set('class', ' '.join(classes))
  1060	        return self

  1062	    def css(self, *args, **kwargs):
  1063	        """css attributes manipulation
  1064	        """
  1065	
  1066	        attr = value = no_default
  1067	        length = len(args)
  1068	        if length == 1:
  1069	            attr = args[0]
  1070	        elif length == 2:
  1071	            attr, value = args
  1072	        elif kwargs:
  1073	            attr = kwargs
  1074	        else:
  1075	            raise ValueError('Invalid arguments %s %s' % (args, kwargs))
  1076	
  1077	        if isinstance(attr, dict):
  1078	            for tag in self:
  1079	                stripped_keys = [key.strip().replace('_', '-')
  1080	                                 for key in attr.keys()]
  1081	                current = [el.strip()
  1082	                           for el in (tag.get('style') or '').split(';')
  1083	                           if el.strip()
  1084	                           and el.split(':')[0].strip() not in stripped_keys]
  1085	                for key, value in attr.items():
  1086	                    key = key.replace('_', '-')
  1087	                    current.append('%s: %s' % (key, value))
  1088	                tag.set('style', '; '.join(current))
  1089	        elif isinstance(value, basestring):
  1090	            attr = attr.replace('_', '-')
  1091	            for tag in self:
  1092	                current = [
  1093	                    el.strip()
  1094	                    for el in (tag.get('style') or '').split(';')
  1095	                    if (el.strip() and
  1096	                        not el.split(':')[0].strip() == attr.strip())]
  1097	                current.append('%s: %s' % (attr, value))
  1098	                tag.set('style', '; '.join(current))
  1099	        return self

  1127	    def val(self, value=no_default):
  1128	        """Set the attribute value::
  1129	
  1130	            >>> d = PyQuery('<input />')
  1131	            >>> d.val('Youhou')
  1132	            [<input>]
  1133	
  1134	        Get the attribute value::
  1135	
  1136	            >>> d.val()
  1137	            'Youhou'
  1138	
  1139	        Set the selected values for a `select` element with the `multiple`
  1140	        attribute::
  1141	
  1142	            >>> d = PyQuery('''
  1143	            ...             <select multiple>
  1144	            ...                 <option value="you"><option value="hou">
  1145	            ...             </select>
  1146	            ...             ''')
  1147	            >>> d.val(['you', 'hou'])
  1148	            [<select>]
  1149	
  1150	        Get the selected values for a `select` element with the `multiple`
  1151	        attribute::
  1152	
  1153	            >>> d.val()
  1154	            ['you', 'hou']
  1155	
  1156	        """
  1157	        def _get_value(tag):
  1158	            # <textarea>
  1159	            if tag.tag == 'textarea':
  1160	                return self._copy(tag).html()
  1161	            # <select>
  1162	            elif tag.tag == 'select':
  1163	                if 'multiple' in tag.attrib:
  1164	                    # Only extract value if selected
  1165	                    selected = self._copy(tag)('option[selected]')
  1166	                    # Rebuild list to avoid serialization error
  1167	                    return list(selected.map(
  1168	                        lambda _, o: self._copy(o).attr('value')
  1169	                    ))
  1170	                selected_option = self._copy(tag)('option[selected]:last')
  1171	                if selected_option:
  1172	                    return selected_option.attr('value')
  1173	                else:
  1174	                    return self._copy(tag)('option').attr('value')
  1175	            # <input type="checkbox"> or <input type="radio">
  1176	            elif self.is_(':checkbox,:radio'):
  1177	                val = self._copy(tag).attr('value')
  1178	                if val is None:
  1179	                    return 'on'
  1180	                else:
  1181	                    return val
  1182	            # <input>
  1183	            elif tag.tag == 'input':
  1184	                val = self._copy(tag).attr('value')
  1185	                return val.replace('\n', '') if val else ''
  1186	            # everything else.
  1187	            return self._copy(tag).attr('value') or ''
  1188	
  1189	        def _set_value(pq, value):
  1190	            for tag in pq:
  1191	                # <select>
  1192	                if tag.tag == 'select':
  1193	                    if not isinstance(value, list):
  1194	                        value = [value]
  1195	
  1196	                    def _make_option_selected(_, elem):
  1197	                        pq = self._copy(elem)
  1198	                        if pq.attr('value') in value:
  1199	                            pq.attr('selected', 'selected')
  1200	                            if 'multiple' not in tag.attrib:
  1201	                                del value[:]  # Ensure it toggles first match
  1202	                        else:
  1203	                            pq.removeAttr('selected')
  1204	
  1205	                    self._copy(tag)('option').each(_make_option_selected)
  1206	                    continue
  1207	                # Stringify array
  1208	                if isinstance(value, list):
  1209	                    value = ','.join(value)
  1210	                # <textarea>
  1211	                if tag.tag == 'textarea':
  1212	                    self._copy(tag).text(value)
  1213	                    continue
  1214	                # <input> and everything else.
  1215	                self._copy(tag).attr('value', value)
  1216	
  1217	        if value is no_default:
  1218	            if len(self):
  1219	                return _get_value(self[0])
  1220	        else:
  1221	            _set_value(self, value)
  1222	            return self

  1224	    def html(self, value=no_default, **kwargs):
  1225	        """Get or set the html representation of sub nodes.
  1226	
  1227	        Get the text value::
  1228	
  1229	            >>> d = PyQuery('<div><span>toto</span></div>')
  1230	            >>> print(d.html())
  1231	            <span>toto</span>
  1232	
  1233	        Extra args are passed to ``lxml.etree.tostring``::
  1234	
  1235	            >>> d = PyQuery('<div><span></span></div>')

  1705	    @with_camel_case_alias
  1706	    def serialize_pairs(self):
  1707	        """Serialize form elements as an array of 2-tuples conventional for
  1708	        typical URL-parsing operations in Python.
  1709	
  1710	            >>> d = PyQuery('<form><input name="order" value="spam"></form>')
  1711	            >>> d.serialize_pairs()
  1712	            [('order', 'spam')]
  1713	            >>> d.serializePairs()
  1714	            [('order', 'spam')]
  1715	        """
  1716	        # https://github.com/jquery/jquery/blob
  1717	        # /2d4f53416e5f74fa98e0c1d66b6f3c285a12f0ce/src/serialize.js#L14
  1718	        _submitter_types = ['submit', 'button', 'image', 'reset', 'file']
  1719	
  1720	        controls = self._copy([])
  1721	        # Expand list of form controls
  1722	        for el in self.items():
  1723	            if el[0].tag == 'form':
  1724	                form_id = el.attr('id')
  1725	                if form_id:
  1726	                    # Include inputs outside of their form owner
  1727	                    root = self._copy(el.root.getroot())
  1728	                    controls.extend(root(
  1729	                        '#%s :not([form]):input, [form="%s"]:input'
  1730	                        % (form_id, form_id)))
  1731	                else:
  1732	                    controls.extend(el(':not([form]):input'))
  1733	            elif el[0].tag == 'fieldset':
  1734	                controls.extend(el(':input'))
  1735	            else:
  1736	                controls.extend(el)
  1737	        # Filter controls
  1738	        selector = '[name]:enabled:not(button)'  # Not serializing image button
  1739	        selector += ''.join(map(
  1740	            lambda s: ':not([type="%s"])' % s,
  1741	            _submitter_types))
  1742	        controls = controls.filter(selector)
  1743	
  1744	        def _filter_out_unchecked(_, el):
  1745	            el = controls._copy(el)
  1746	            return not el.is_(':checkbox:not(:checked)') and \
  1747	                not el.is_(':radio:not(:checked)')
  1748	        controls = controls.filter(_filter_out_unchecked)
  1749	
  1750	        # jQuery serializes inputs with the datalist element as an ancestor
  1751	        # contrary to WHATWG spec as of August 2018
  1752	        #
  1753	        # xpath = 'self::*[not(ancestor::datalist)]'
  1754	        # results = []
  1755	        # for tag in controls:
  1756	        #     results.extend(tag.xpath(xpath, namespaces=controls.namespaces))
  1757	        # controls = controls._copy(results)
  1758	
  1759	        # Serialize values
  1760	        ret = []
  1761	        for field in controls:
  1762	            val = self._copy(field).val() or ''
  1763	            if isinstance(val, list):
  1764	                ret.extend(map(
  1765	                    lambda v: (field.attrib['name'], v.replace('\n', '\r\n')),
  1766	                    val
  1767	                ))
  1768	            else:
  1769	                ret.append((field.attrib['name'], val.replace('\n', '\r\n')))
  1770	        return ret
