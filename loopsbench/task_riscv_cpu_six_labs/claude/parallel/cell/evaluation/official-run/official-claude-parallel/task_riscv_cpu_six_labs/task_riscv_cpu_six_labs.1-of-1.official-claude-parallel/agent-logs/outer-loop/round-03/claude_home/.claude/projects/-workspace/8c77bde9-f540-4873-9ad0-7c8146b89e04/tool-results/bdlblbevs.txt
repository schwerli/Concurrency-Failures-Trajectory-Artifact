     4	module CtrlUnit(
     5	    input clk,
     6	    input rst,
     7	
     8	    input[31:0] PC,
     9	    input[31:0] inst,
    10	    input[31:0] imm,
    11	    
    12	    input ALU_done,
    13	    input MEM_done,
    14	    input MUL_done,
    15	    input DIV_done,
    16	    input JUMP_done,
    17	    input is_jump,
    18	
    19	    // IF
    20	    output IS_en,
    21	
    22	    // IS
    23	    output[2:0] ImmSel,
    24	
    25	    // RO/FU
    26	    output reg ALU_en, MEM_en, MUL_en, DIV_en, JUMP_en,
    27	    output reg[31:0] PC_ctrl,
    28	    output reg[31:0] imm_ctrl,
    29	    output reg[4:0] rs1_ctrl, rs2_ctrl, 
    30	    
    31	    // FU
    32	    output reg[4:0] JUMP_op,
    33	    output reg[3:0] ALU_op,
    34	    output reg ALU_use_PC,
    35	    output reg ALU_use_imm,
    36	    output reg MEM_we,
    37	    output reg[2:0] MEM_bhw,
    38	    output reg[2:0] MUL_op,
    39	    output reg[1:0] DIV_op,
    40	    
    41	    // WB
    42	    output reg[2:0] write_sel,
    43	    output reg reg_write,
    44	    output reg[4:0] rd_ctrl
    45	);
    46	    // used in for loop
    47	    integer i;
    48	
    49	    // Function Unit Status
    50	    // fields of FUS are defined at CtrlDefine.vh
    51	    reg[31:0] FUS[1:5];
    52	    reg[31:0] IMM[1:5];
    53	
    54	    // Register Status
    55	    // records which FU will write corresponding reg at WB
    56	    reg[2:0] RRS[0:31];
    57	
    58	    // sometimes an instruction needs PC to execute
    59	    // pc record
    60	    reg[31:0] PCR[1:5];
    61	
    62	    wire RO_en;
    63	    wire normal_stall;
    64	    reg ctrl_stall = 0, IS_flush = 0;
    65	
    66	    // a jump (JAL/JALR/BXX) instr has issued,
    67	    // stop issuing until result is known
    68	    wire jump_stall;
    69	
    70	    // instruction field
    71	    wire[6:0] funct7 = inst[31:25];
    72	    wire[2:0] funct3 = inst[14:12];
    73	    wire[6:0] opcode = inst[6:0];
    74	    wire[4:0] rd = inst[11:7];
    75	    wire[4:0] rs1 = inst[19:15];
    76	    wire[4:0] rs2 = inst[24:20];
    77	
    78	    // type specification
    79	    wire Rop = opcode == 7'b0110011;
    80	    wire Iop = opcode == 7'b0010011;
    81	    wire Bop = opcode == 7'b1100011;
    82	    wire Lop = opcode == 7'b0000011;
    83	    wire Sop = opcode == 7'b0100011;
    84	
    85	    wire funct7_0  = funct7 == 7'h0;
    86	    wire funct7_1  = funct7 == 7'h1;
    87	    wire funct7_32 = funct7 == 7'h20;
    88	
    89	    wire funct3_0 = funct3 == 3'h0;
    90	    wire funct3_1 = funct3 == 3'h1;
    91	    wire funct3_2 = funct3 == 3'h2;
    92	    wire funct3_3 = funct3 == 3'h3;
    93	    wire funct3_4 = funct3 == 3'h4;
    94	    wire funct3_5 = funct3 == 3'h5;
    95	    wire funct3_6 = funct3 == 3'h6;
    96	    wire funct3_7 = funct3 == 3'h7;
    97	
    98	    wire ADD  = Rop & funct3_0 & funct7_0;
    99	    wire SUB  = Rop & funct3_0 & funct7_32;
   100	    wire SLL  = Rop & funct3_1 & funct7_0;
   101	    wire SLT  = Rop & funct3_2 & funct7_0;
   102	    wire SLTU = Rop & funct3_3 & funct7_0;
   103	    wire XOR  = Rop & funct3_4 & funct7_0;
   104	    wire SRL  = Rop & funct3_5 & funct7_0;
   105	    wire SRA  = Rop & funct3_5 & funct7_32;
   106	    wire OR   = Rop & funct3_6 & funct7_0;
   107	    wire AND  = Rop & funct3_7 & funct7_0;
   108	
   109	    wire MUL    = Rop & funct3_0 & funct7_1;
   110	    wire MULH   = Rop & funct3_1 & funct7_1;
   111	    wire MULHSU = Rop & funct3_2 & funct7_1;
   112	    wire MULHU  = Rop & funct3_3 & funct7_1;
   113	    wire DIV    = Rop & funct3_4 & funct7_1;
   114	    wire DIVU   = Rop & funct3_5 & funct7_1;
   115	    wire REM    = Rop & funct3_6 & funct7_1;
   116	    wire REMU    = Rop & funct3_7 & funct7_1;
   117	
   118	    wire ADDI  = Iop & funct3_0;	
   119	    wire SLTI  = Iop & funct3_2;
   120	    wire SLTIU = Iop & funct3_3;
   121	    wire XORI  = Iop & funct3_4;
   122	    wire ORI   = Iop & funct3_6;
   123	    wire ANDI  = Iop & funct3_7;
   124	    wire SLLI  = Iop & funct3_1 & funct7_0;
   125	    wire SRLI  = Iop & funct3_5 & funct7_0;
   126	    wire SRAI  = Iop & funct3_5 & funct7_32;
   127	
   128	    wire BEQ = Bop & funct3_0;
   129	    wire BNE = Bop & funct3_1;
   130	    wire BLT = Bop & funct3_4;
   131	    wire BGE = Bop & funct3_5;
   132	    wire BLTU = Bop & funct3_6;
   133	    wire BGEU = Bop & funct3_7;
   134	
   135	    wire LB =  Lop & funct3_0;
   136	    wire LH =  Lop & funct3_1;
   137	    wire LW =  Lop & funct3_2;
   138	    wire LBU = Lop & funct3_4;
   139	    wire LHU = Lop & funct3_5;
   140	
   141	    wire SB = Sop & funct3_0;
   142	    wire SH = Sop & funct3_1;
   143	    wire SW = Sop & funct3_2;
   144	
   145	    wire LUI   = opcode == 7'b0110111;
   146	    wire AUIPC = opcode == 7'b0010111;
   147	
   148	    wire JAL  =  opcode == 7'b1101111;
   149	    wire JALR = (opcode == 7'b1100111) && funct3_0;
   150	
   151	    wire R_valid = AND | OR | ADD | XOR | SLL | SRL | SRA | SUB | SLT | SLTU 
   152	        | MUL | MULH | MULHSU | MULHU | DIV | DIVU | REM | REMU;
   153	    wire I_valid = ANDI | ORI | ADDI | XORI | SLLI | SRLI | SRAI | SLTI | SLTIU;
   154	    wire B_valid = BEQ | BNE | BLT | BGE | BLTU | BGEU;
   155	    wire L_valid = LW | LH | LB | LHU | LBU;
   156	    wire S_valid = SW | SH | SB;
   157	
   158	    // function unit specification
   159	    wire use_ALU = AND | OR | ADD | XOR | SLL | SRL | SRA | SUB | SLT | SLTU
   160	        | I_valid | LUI | AUIPC;
   161	    wire use_MEM = L_valid | S_valid;
   162	    wire use_MUL = MUL | MULH | MULHSU | MULHU;
   163	    wire use_DIV = DIV | DIVU | REM | REMU;
   164	    wire use_JUMP = B_valid | JAL | JALR;
   165	
   166	    wire[2:0] use_FU = {3{use_ALU}}  & `FU_ALU  |
   167	                       {3{use_MEM}}  & `FU_MEM  |
   168	                       {3{use_MUL}}  & `FU_MUL  |
   169	                       {3{use_DIV}}  & `FU_DIV  |
   170	                       {3{use_JUMP}} & `FU_JUMP ;
   171	
   172	    // registers to be used
   173	    wire[4:0] dst = {5{R_valid | I_valid | L_valid | LUI | AUIPC | JAL | JALR}} & rd;
   174	    wire[4:0] src1 = {5{R_valid | I_valid | S_valid | L_valid | B_valid | JALR}} & rs1;
   175	    wire[4:0] src2 = {5{R_valid | S_valid | B_valid}} & rs2;
   176	    // initial values of some FUS fields
   177	    wire[2:0] fu1 = RRS[src1];
   178	    wire[2:0] fu2 = RRS[src2];
   179	    wire rdy1 = ~|fu1;
   180	    wire rdy2 = ~|fu2;
   181	
   182	    // normal stall: structural hazard or WAW
   183	    assign normal_stall = ...;           //fill sth. here
   184	
   185	    assign IS_en = IS_flush | ~normal_stall & ~ctrl_stall;
   186	    assign RO_en = ~IS_flush & ~normal_stall & ~ctrl_stall;
   187	
   188	    always @ (posedge clk or posedge rst) begin
   189	        if (rst) begin
   190	            ctrl_stall <= 0;
   191	        end
   192	        else begin
   193	            // IS
   194	            if (RO_en & (use_FU == `FU_JUMP)) begin
   195	                ctrl_stall <= 1;
   196	            end
   197	            else if (JUMP_done) begin
   198	                ctrl_stall <= 0;
   199	            end
   200	        end
   201	    end
   202	
   203	    always @ (posedge clk or posedge rst) begin
   204	        if (rst) begin
   205	            IS_flush <= 0;
   206	        end
   207	        else if (JUMP_done & is_jump) begin
   208	            IS_flush <= 1;
   209	        end
   210	        else begin
   211	            IS_flush <= 0;
   212	        end
   213	    end
   214	
   215	    assign jump_stall = FUS[`FU_JUMP][`BUSY] & (~JUMP_done);
   216	
   217	    wire[4:0] op = {5{ADD}}          & `ALU_ADD    |
   218	                   {5{SUB}}          & `ALU_SUB    |
   219	                   {5{AND}}          & `ALU_AND    |
   220	                   {5{OR}}           & `ALU_OR     |
   221	                   {5{XOR}}          & `ALU_XOR    |
   222	                   {5{SLL}}          & `ALU_SLL    |
   223	                   {5{SRL}}          & `ALU_SRL    |
   224	                   {5{SLT}}          & `ALU_SLT    |
   225	                   {5{SLTU}}         & `ALU_SLTU   |
   226	                   {5{SRA}}          & `ALU_SRA    |
   227	                   {5{LUI}}          & `ALU_LUI    |
   228	                   {5{AUIPC}}        & `ALU_AUIPC  |
   229	                   {5{ADDI}}         & `ALU_ADDI   |
   230	                   {5{ANDI}}         & `ALU_ANDI   |
   231	                   {5{ORI}}          & `ALU_ORI    |
   232	                   {5{XORI}}         & `ALU_XORI   |
   233	                   {5{SLLI}}         & `ALU_SLLI   |
   234	                   {5{SRLI}}         & `ALU_SRLI   |
   235	                   {5{SLTI}}         & `ALU_SLTI   |
   236	                   {5{SLTIU}}        & `ALU_SLTIU  |
   237	                   {5{SRAI}}         & `ALU_SRAI   |
   238	                   {5{LB}}           & `MEM_LB     |
   239	                   {5{LH}}           & `MEM_LH     |
   240	                   {5{LW}}           & `MEM_LW     |
   241	                   {5{LBU}}          & `MEM_LBU    |
   242	                   {5{LHU}}          & `MEM_LHU    |
   243	                   {5{SB}}           & `MEM_SB     |
   244	                   {5{SH}}           & `MEM_SH     |
   245	                   {5{SW}}           & `MEM_SW     |
   246	                   {5{MUL}}          & `MUL_MUL    |
   247	                   {5{MULH}}         & `MUL_MULH   |
   248	                   {5{MULHSU}}       & `MUL_MULHU  |
   249	                   {5{MULHU}}        & `MUL_MULHSU |
   250	                   {5{DIV}}          & `DIV_DIV    |
   251	                   {5{DIVU}}         & `DIV_DIVU   |
   252	                   {5{REM}}          & `DIV_REM    |
   253	                   {5{REMU}}         & `DIV_REMU   |
   254	                   {5{BEQ}}          & `JUMP_BEQ   |
