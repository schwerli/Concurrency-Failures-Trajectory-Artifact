   143	    render(
   144	        dataZoomModel: SliderZoomModel,
   145	        ecModel: GlobalModel,
   146	        api: ExtensionAPI,
   147	        payload: Payload & {
   148	            from: string
   149	            type: string
   150	        }
   151	    ) {
   152	        super.render.apply(this, arguments as any);
   153	
   154	        throttle.createOrUpdate(
   155	            this,
   156	            '_dispatchZoomAction',
   157	            dataZoomModel.get('throttle'),
   158	            'fixRate'
   159	        );
   160	
   161	        this._orient = dataZoomModel.getOrient();
   162	
   163	        if (dataZoomModel.get('show') === false) {
   164	            this.group.removeAll();
   165	            return;
   166	        }
   167	
   168	        if (dataZoomModel.noTarget()) {
   169	            this._clear();
   170	            this.group.removeAll();
   171	            return;
   172	        }
   173	
   174	        // Notice: this._resetInterval() should not be executed when payload.type
   175	        // is 'dataZoom', origin this._range should be maintained, otherwise 'pan'
   176	        // or 'zoom' info will be missed because of 'throttle' of this.dispatchAction,
   177	        if (!payload || payload.type !== 'dataZoom' || payload.from !== this.uid) {
   178	            this._buildView();
   179	        }
   180	
   181	        this._updateView();
   182	    }
   183	
   184	    dispose() {
   185	        this._clear();
   186	        super.dispose.apply(this, arguments as any);
   187	    }

   221	    private _resetLocation() {
   222	        const dataZoomModel = this.dataZoomModel;
   223	        const api = this.api;
   224	        const showMoveHandle = dataZoomModel.get('brushSelect');
   225	        const moveHandleSize = showMoveHandle ? DEFAULT_MOVE_HANDLE_SIZE : 0;
   226	
   227	        // If some of x/y/width/height are not specified,
   228	        // auto-adapt according to target grid.
   229	        const coordRect = this._findCoordRect();
   230	        const ecSize = {width: api.getWidth(), height: api.getHeight()};
   231	        // Default align by coordinate system rect.
   232	        const positionInfo = this._orient === HORIZONTAL
   233	            ? {
   234	                // Why using 'right', because right should be used in vertical,
   235	                // and it is better to be consistent for dealing with position param merge.
   236	                right: ecSize.width - coordRect.x - coordRect.width,
   237	                top: (ecSize.height - DEFAULT_FILLER_SIZE - DEFAULT_LOCATION_EDGE_GAP - moveHandleSize),
   238	                width: coordRect.width,
   239	                height: DEFAULT_FILLER_SIZE
   240	            }
   241	            : { // vertical
   242	                right: DEFAULT_LOCATION_EDGE_GAP,
   243	                top: coordRect.y,
   244	                width: DEFAULT_FILLER_SIZE,
   245	                height: coordRect.height
   246	            };
   247	
   248	        // Do not write back to option and replace value 'ph', because
   249	        // the 'ph' value should be recalculated when resize.
   250	        const layoutParams = layout.getLayoutParams(dataZoomModel.option);
   251	
   252	        // Replace the placeholder value.
   253	        each(['right', 'top', 'width', 'height'] as const, function (name) {
   254	            if (layoutParams[name] === 'ph') {
   255	                layoutParams[name] = positionInfo[name];
   256	            }
   257	        });
   258	
   259	        const layoutRect = layout.getLayoutRect(
   260	            layoutParams,
   261	            ecSize
   262	        );
   263	
   264	        this._location = {x: layoutRect.x, y: layoutRect.y};
   265	        this._size = [layoutRect.width, layoutRect.height];
   266	        this._orient === VERTICAL && this._size.reverse();
   267	    }

   269	    private _positionGroup() {
   270	        const thisGroup = this.group;
   271	        const location = this._location;
   272	        const orient = this._orient;
   273	
   274	        // Just use the first axis to determine mapping.
   275	        const targetAxisModel = this.dataZoomModel.getFirstTargetAxisModel();
   276	        const inverse = targetAxisModel && targetAxisModel.get('inverse');
   277	
   278	        const sliderGroup = this._displayables.sliderGroup;
   279	        const otherAxisInverse = (this._dataShadowInfo || {}).otherAxisInverse;
   280	
   281	        // Transform barGroup.
   282	        sliderGroup.attr(
   283	            (orient === HORIZONTAL && !inverse)
   284	            ? {scaleY: otherAxisInverse ? 1 : -1, scaleX: 1 }
   285	            : (orient === HORIZONTAL && inverse)
   286	            ? {scaleY: otherAxisInverse ? 1 : -1, scaleX: -1 }
   287	            : (orient === VERTICAL && !inverse)
   288	            ? {scaleY: otherAxisInverse ? -1 : 1, scaleX: 1, rotation: Math.PI / 2}
   289	            // Don't use Math.PI, considering shadow direction.
   290	            : {scaleY: otherAxisInverse ? -1 : 1, scaleX: -1, rotation: Math.PI / 2}
   291	        );
   292	
   293	        // Position barGroup
   294	        const rect = thisGroup.getBoundingRect([sliderGroup]);
   295	        thisGroup.x = location.x - rect.x;
   296	        thisGroup.y = location.y - rect.y;
   297	        thisGroup.markRedraw();
   298	    }

   349	    private _renderDataShadow() {
   350	        const info = this._dataShadowInfo = this._prepareDataShadowInfo();
   351	
   352	        this._displayables.dataShadowSegs = [];
   353	
   354	        if (!info) {
   355	            return;
   356	        }
   357	
   358	        const size = this._size;
   359	        const oldSize = this._shadowSize || [];
   360	        const seriesModel = info.series;
   361	        const data = seriesModel.getRawData();
   362	        const candlestickDim = seriesModel.getShadowDim && seriesModel.getShadowDim();
   363	        const otherDim: string = candlestickDim && data.getDimensionInfo(candlestickDim)
   364	            ? seriesModel.getShadowDim() // @see candlestick
   365	            : info.otherDim;
   366	
   367	        if (otherDim == null) {
   368	            return;
   369	        }
   370	
   371	        let polygonPts = this._shadowPolygonPts;
   372	        let polylinePts = this._shadowPolylinePts;
   373	        // Not re-render if data doesn't change.
   374	        if (
   375	            data !== this._shadowData || otherDim !== this._shadowDim
   376	            || size[0] !== oldSize[0] || size[1] !== oldSize[1]
   377	        ) {
   378	            let otherDataExtent = data.getDataExtent(otherDim);
   379	            // Nice extent.
   380	            const otherOffset = (otherDataExtent[1] - otherDataExtent[0]) * 0.3;
   381	            otherDataExtent = [
   382	                otherDataExtent[0] - otherOffset,
   383	                otherDataExtent[1] + otherOffset
   384	            ];
   385	            const otherShadowExtent = [0, size[1]];
   386	            const thisShadowExtent = [0, size[0]];
   387	
   388	            const areaPoints = [[size[0], 0], [0, 0]];
   389	            const linePoints: number[][] = [];
   390	            const step = thisShadowExtent[1] / (data.count() - 1);
   391	            let thisCoord = 0;
   392	
   393	            // Optimize for large data shadow
   394	            const stride = Math.round(data.count() / size[0]);
   395	            let lastIsEmpty: boolean;
   396	            data.each([otherDim], function (value: ParsedValue, index) {
   397	                if (stride > 0 && (index % stride)) {
   398	                    thisCoord += step;
   399	                    return;
   400	                }
   401	
   402	                // FIXME
   403	                // Should consider axis.min/axis.max when drawing dataShadow.
   404	
   405	                // FIXME
   406	                // 应该使用统一的空判断？还是在list里进行空判断？
   407	                const isEmpty = value == null || isNaN(value as number) || value === '';
   408	                // See #4235.
   409	                const otherCoord = isEmpty
   410	                    ? 0 : linearMap(value as number, otherDataExtent, otherShadowExtent, true);
   411	
   412	                // Attempt to draw data shadow precisely when there are empty value.
   413	                if (isEmpty && !lastIsEmpty && index) {
   414	                    areaPoints.push([areaPoints[areaPoints.length - 1][0], 0]);
   415	                    linePoints.push([linePoints[linePoints.length - 1][0], 0]);
   416	                }
   417	                else if (!isEmpty && lastIsEmpty) {
   418	                    areaPoints.push([thisCoord, 0]);
   419	                    linePoints.push([thisCoord, 0]);
   420	                }
   421	
   422	                areaPoints.push([thisCoord, otherCoord]);
   423	                linePoints.push([thisCoord, otherCoord]);
   424	
   425	                thisCoord += step;
   426	                lastIsEmpty = isEmpty;
   427	            });
   428	
   429	            polygonPts = this._shadowPolygonPts = areaPoints;
   430	            polylinePts = this._shadowPolylinePts = linePoints;
   431	
   432	        }
   433	        this._shadowData = data;
   434	        this._shadowDim = otherDim;
   435	        this._shadowSize = [size[0], size[1]];
   436	
   437	        const dataZoomModel = this.dataZoomModel;
   438	
   439	        function createDataShadowGroup(isSelectedArea?: boolean) {
   440	            const model = dataZoomModel.getModel(isSelectedArea ? 'selectedDataBackground' : 'dataBackground');
   441	            const group = new graphic.Group();
   442	            const polygon = new graphic.Polygon({
   443	                shape: {points: polygonPts},
   444	                segmentIgnoreThreshold: 1,
   445	                style: model.getModel('areaStyle').getAreaStyle(),
   446	                silent: true,
   447	                z2: -20
   448	            });
   449	            const polyline = new graphic.Polyline({
   450	                shape: {points: polylinePts},
   451	                segmentIgnoreThreshold: 1,
   452	                style: model.getModel('lineStyle').getLineStyle(),
   453	                silent: true,
   454	                z2: -19
   455	            });
   456	            group.add(polygon);
   457	            group.add(polyline);
   458	            return group;
   459	        }
   460	
   461	        // let dataBackgroundModel = dataZoomModel.getModel('dataBackground');
   462	        for (let i = 0; i < 3; i++) {
   463	            const group = createDataShadowGroup(i === 1);
   464	            this._displayables.sliderGroup.add(group);
   465	            this._displayables.dataShadowSegs.push(group);
   466	        }
   467	    }

   469	    private _prepareDataShadowInfo() {
   470	        const dataZoomModel = this.dataZoomModel;
   471	        const showDataShadow = dataZoomModel.get('showDataShadow');
   472	
   473	        if (showDataShadow === false) {
   474	            return;
   475	        }
   476	
   477	        // Find a representative series.
   478	        let result: SliderZoomView['_dataShadowInfo'];
   479	        const ecModel = this.ecModel;
   480	
   481	        dataZoomModel.eachTargetAxis(function (axisDim, axisIndex) {
   482	            const seriesModels = dataZoomModel
   483	                .getAxisProxy(axisDim, axisIndex)
   484	                .getTargetSeriesModels();
   485	
   486	            each(seriesModels, function (seriesModel) {
   487	                if (result) {
   488	                    return;
   489	                }
   490	
   491	                if (showDataShadow !== true && indexOf(
   492	                        SHOW_DATA_SHADOW_SERIES_TYPE, seriesModel.get('type')
   493	                    ) < 0
   494	                ) {
   495	                    return;
   496	                }
   497	
   498	                const thisAxis = (
   499	                    ecModel.getComponent(getAxisMainType(axisDim), axisIndex) as AxisBaseModel
   500	                ).axis;
   501	                let otherDim = getOtherDim(axisDim);
   502	                let otherAxisInverse;
   503	                const coordSys = seriesModel.coordinateSystem;
   504	
   505	                if (otherDim != null && coordSys.getOtherAxis) {
   506	                    otherAxisInverse = coordSys.getOtherAxis(thisAxis).inverse;
   507	                }
   508	
   509	                otherDim = seriesModel.getData().mapDimension(otherDim);
   510	
   511	                result = {
   512	                    thisAxis: thisAxis,
   513	                    series: seriesModel,
   514	                    thisDim: axisDim,
   515	                    otherDim: otherDim,
   516	                    otherAxisInverse: otherAxisInverse
   517	                };
   518	
   519	            }, this);
   520	
   521	        }, this);
   522	
   523	        return result;
   524	    }

   526	    private _renderHandle() {
   527	        const thisGroup = this.group;
   528	        const displayables = this._displayables;
   529	        const handles: [graphic.Path, graphic.Path] = displayables.handles = [null, null];
   530	        const handleLabels: [graphic.Text, graphic.Text] = displayables.handleLabels = [null, null];
   531	        const sliderGroup = this._displayables.sliderGroup;
   532	        const size = this._size;
   533	        const dataZoomModel = this.dataZoomModel;
   534	        const api = this.api;
   535	
   536	        const borderRadius = dataZoomModel.get('borderRadius') || 0;
   537	
   538	        const brushSelect = dataZoomModel.get('brushSelect');
   539	
   540	        const filler = displayables.filler = new Rect({
   541	            silent: brushSelect,
   542	            style: {
   543	                fill: dataZoomModel.get('fillerColor')
   544	            },
   545	            textConfig: {
   546	                position: 'inside'
   547	            }
   548	        });
   549	
   550	        sliderGroup.add(filler);
   551	
   552	        // Frame border.
   553	        sliderGroup.add(new Rect({
   554	            silent: true,
   555	            subPixelOptimize: true,
   556	            shape: {
   557	                x: 0,
   558	                y: 0,
   559	                width: size[0],
   560	                height: size[1],
   561	                r: borderRadius
   562	            },
   563	            style: {
   564	                // deprecated option
   565	                stroke: dataZoomModel.get('dataBackgroundColor' as any)
   566	                    || dataZoomModel.get('borderColor'),
   567	                lineWidth: DEFAULT_FRAME_BORDER_WIDTH,
   568	                fill: 'rgba(0,0,0,0)'
   569	            }
   570	        }));
   571	
   572	        // Left and right handle to resize
   573	        each([0, 1] as const, function (handleIndex) {
   574	            let iconStr = dataZoomModel.get('handleIcon');
   575	            if (
   576	                !symbolBuildProxies[iconStr]
   577	                && iconStr.indexOf('path://') < 0
   578	                && iconStr.indexOf('image://') < 0
   579	            ) {
   580	                // Compatitable with the old icon parsers. Which can use a path string without path://
   581	                iconStr = 'path://' + iconStr;
   582	                if (__DEV__) {
   583	                    deprecateLog('handleIcon now needs \'path://\' prefix when using a path string');
   584	                }
   585	            }
   586	            const path = createSymbol(
   587	                iconStr,
   588	                -1, 0, 2, 2, null, true
   589	            ) as graphic.Path;
   590	            path.attr({
   591	                cursor: getCursor(this._orient),
   592	                draggable: true,
   593	                drift: bind(this._onDragMove, this, handleIndex),
   594	                ondragend: bind(this._onDragEnd, this),
   595	                onmouseover: bind(this._showDataInfo, this, true),
   596	                onmouseout: bind(this._showDataInfo, this, false),
   597	                z2: 5
   598	            });
   599	
   600	            const bRect = path.getBoundingRect();
   601	            const handleSize = dataZoomModel.get('handleSize');
   602	
   603	            this._handleHeight = parsePercent(handleSize, this._size[1]);
   604	            this._handleWidth = bRect.width / bRect.height * this._handleHeight;
   605	
   606	            path.setStyle(dataZoomModel.getModel('handleStyle').getItemStyle());
   607	            path.style.strokeNoScale = true;
   608	            path.rectHover = true;
   609	
   610	            path.ensureState('emphasis').style = dataZoomModel.getModel(['emphasis', 'handleStyle']).getItemStyle();
   611	            enableHoverEmphasis(path);
   612	
   613	            const handleColor = dataZoomModel.get('handleColor' as any); // deprecated option
   614	            // Compatitable with previous version
   615	            if (handleColor != null) {
   616	                path.style.fill = handleColor;
   617	            }
   618	
   619	            sliderGroup.add(handles[handleIndex] = path);
   620	
   621	            const textStyleModel = dataZoomModel.getModel('textStyle');
   622	
   623	            thisGroup.add(
   624	                handleLabels[handleIndex] = new graphic.Text({
   625	                silent: true,
   626	                invisible: true,
   627	                style: createTextStyle(textStyleModel, {
   628	                    x: 0, y: 0, text: '',
   629	                    verticalAlign: 'middle',
   630	                    align: 'center',
   631	                    fill: textStyleModel.getTextColor(),
   632	                    font: textStyleModel.getFont()
   633	                }),
   634	                z2: 10
   635	            }));
   636	
   637	        }, this);
   638	
   639	        // Handle to move. Only visible when brushSelect is set true.
   640	        let actualMoveZone: Displayable = filler;
   641	        if (brushSelect) {
   642	            const moveHandleHeight = parsePercent(dataZoomModel.get('moveHandleSize'), size[1]);
   643	            const moveHandle = displayables.moveHandle = new graphic.Rect({
   644	                style: dataZoomModel.getModel('moveHandleStyle').getItemStyle(),
   645	                silent: true,
   646	                shape: {
   647	                    r: [0, 0, 2, 2],
   648	                    y: size[1] - 0.5,
   649	                    height: moveHandleHeight
   650	                }
   651	            });
   652	            const iconSize = moveHandleHeight * 0.8;
   653	            const moveHandleIcon = displayables.moveHandleIcon = createSymbol(
   654	                dataZoomModel.get('moveHandleIcon'),
   655	                -iconSize / 2, -iconSize / 2, iconSize, iconSize,
   656	                '#fff',
   657	                true
   658	            );
   659	            moveHandleIcon.silent = true;
   660	            moveHandleIcon.y = size[1] + moveHandleHeight / 2 - 0.5;
   661	
   662	            moveHandle.ensureState('emphasis').style = dataZoomModel.getModel(
   663	                ['emphasis', 'moveHandleStyle']
   664	            ).getItemStyle();
   665	
   666	            const moveZoneExpandSize = Math.min(size[1] / 2, Math.max(moveHandleHeight, 10));
   667	            actualMoveZone = displayables.moveZone = new graphic.Rect({
   668	                invisible: true,
   669	                shape: {
   670	                    y: size[1] - moveZoneExpandSize,
   671	                    height: moveHandleHeight + moveZoneExpandSize
   672	                }
   673	            });
   674	
   675	            actualMoveZone.on('mouseover', () => {
   676	                    api.enterEmphasis(moveHandle);
   677	                })
   678	                .on('mouseout', () => {
   679	                    api.leaveEmphasis(moveHandle);
   680	                });
   681	
   682	            sliderGroup.add(moveHandle);
   683	            sliderGroup.add(moveHandleIcon);
   684	            sliderGroup.add(actualMoveZone);
   685	        }
   686	
   687	        actualMoveZone.attr({
   688	            draggable: true,
   689	            cursor: getCursor(this._orient),
   690	            drift: bind(this._onDragMove, this, 'all'),
   691	            ondragstart: bind(this._showDataInfo, this, true),
   692	            ondragend: bind(this._onDragEnd, this),
   693	            onmouseover: bind(this._showDataInfo, this, true),
   694	            onmouseout: bind(this._showDataInfo, this, false)
   695	        });
   696	    }

   698	    private _resetInterval() {
   699	        const range = this._range = this.dataZoomModel.getPercentRange();
   700	        const viewExtent = this._getViewExtent();
   701	
   702	        this._handleEnds = [
   703	            linearMap(range[0], [0, 100], viewExtent, true),
   704	            linearMap(range[1], [0, 100], viewExtent, true)
   705	        ];
   706	    }

   708	    private _updateInterval(handleIndex: 0 | 1 | 'all', delta: number): boolean {
   709	        const dataZoomModel = this.dataZoomModel;
   710	        const handleEnds = this._handleEnds;
   711	        const viewExtend = this._getViewExtent();
   712	        const minMaxSpan = dataZoomModel.findRepresentativeAxisProxy().getMinMaxSpan();
   713	        const percentExtent = [0, 100];
   714	
   715	        sliderMove(
   716	            delta,
   717	            handleEnds,
   718	            viewExtend,
   719	            dataZoomModel.get('zoomLock') ? 'all' : handleIndex,
   720	            minMaxSpan.minSpan != null
   721	                ? linearMap(minMaxSpan.minSpan, percentExtent, viewExtend, true) : null,
   722	            minMaxSpan.maxSpan != null
   723	                ? linearMap(minMaxSpan.maxSpan, percentExtent, viewExtend, true) : null
   724	        );
   725	
   726	        const lastRange = this._range;
   727	        const range = this._range = asc([
   728	            linearMap(handleEnds[0], viewExtend, percentExtent, true),
   729	            linearMap(handleEnds[1], viewExtend, percentExtent, true)
   730	        ]);
   731	
   732	        return !lastRange || lastRange[0] !== range[0] || lastRange[1] !== range[1];
   733	    }

   735	    private _updateView(nonRealtime?: boolean) {
   736	        const displaybles = this._displayables;
   737	        const handleEnds = this._handleEnds;
   738	        const handleInterval = asc(handleEnds.slice());
   739	        const size = this._size;
   740	
   741	        each([0, 1] as const, function (handleIndex) {
   742	            // Handles
   743	            const handle = displaybles.handles[handleIndex];
   744	            const handleHeight = this._handleHeight;
   745	            (handle as graphic.Path).attr({
   746	                scaleX: handleHeight / 2,
   747	                scaleY: handleHeight / 2,
   748	                // This is a trick, by adding an extra tiny offset to let the default handle's end point align to the drag window.
   749	                // NOTE: It may affect some custom shapes a bit. But we prefer to have better result by default.
   750	                x: handleEnds[handleIndex] + (handleIndex ? -1 : 1),
   751	                y: size[1] / 2 - handleHeight / 2
   752	            });
   753	        }, this);
   754	
   755	        // Filler
   756	        displaybles.filler.setShape({
   757	            x: handleInterval[0],
   758	            y: 0,
   759	            width: handleInterval[1] - handleInterval[0],
   760	            height: size[1]
   761	        });
   762	
   763	        const viewExtent = {
   764	            x: handleInterval[0],
   765	            width: handleInterval[1] - handleInterval[0]
   766	        };
   767	        // Move handle
   768	        if (displaybles.moveHandle) {
   769	            displaybles.moveHandle.setShape(viewExtent);
   770	            displaybles.moveZone.setShape(viewExtent);
   771	            // Force update path on the invisible object
   772	            displaybles.moveZone.getBoundingRect();
   773	            displaybles.moveHandleIcon && displaybles.moveHandleIcon.attr('x', viewExtent.x + viewExtent.width / 2);
   774	        }
   775	
   776	        // update clip path of shadow.
   777	        const dataShadowSegs = displaybles.dataShadowSegs;
   778	        const segIntervals = [0, handleInterval[0], handleInterval[1], size[0]];
   779	
   780	        for (let i = 0; i < dataShadowSegs.length; i++) {
   781	            const segGroup = dataShadowSegs[i];
   782	            let clipPath = segGroup.getClipPath();
   783	            if (!clipPath) {
   784	                clipPath = new graphic.Rect();
   785	                segGroup.setClipPath(clipPath);
   786	            }
   787	            clipPath.setShape({
   788	                x: segIntervals[i],
   789	                y: 0,
   790	                width: segIntervals[i + 1] - segIntervals[i],
   791	                height: size[1]
   792	            });
   793	        }
   794	
   795	        this._updateDataInfo(nonRealtime);
   796	    }
   797	
   798	    private _updateDataInfo(nonRealtime?: boolean) {
   799	        const dataZoomModel = this.dataZoomModel;
   800	        const displaybles = this._displayables;
   801	        const handleLabels = displaybles.handleLabels;
   802	        const orient = this._orient;
   803	        let labelTexts = ['', ''];
   804	
   805	        // FIXME
   806	        // date型，支持formatter，autoformatter（ec2 date.getAutoFormatter）
   807	        if (dataZoomModel.get('showDetail')) {
   808	            const axisProxy = dataZoomModel.findRepresentativeAxisProxy();
   809	
   810	            if (axisProxy) {
   811	                const axis = axisProxy.getAxisModel().axis;
   812	                const range = this._range;
   813	
   814	                const dataInterval = nonRealtime
   815	                    // See #4434, data and axis are not processed and reset yet in non-realtime mode.
   816	                    ? axisProxy.calculateDataWindow({
   817	                        start: range[0], end: range[1]
   818	                    }).valueWindow
   819	                    : axisProxy.getDataValueWindow();
   820	
   821	                labelTexts = [
   822	                    this._formatLabel(dataInterval[0], axis),
   823	                    this._formatLabel(dataInterval[1], axis)
   824	                ];
   825	            }
   826	        }
   827	
   828	        const orderedHandleEnds = asc(this._handleEnds.slice());
   829	
   830	        setLabel.call(this, 0);
   831	        setLabel.call(this, 1);
   832	
   833	        function setLabel(this: SliderZoomView, handleIndex: 0 | 1) {
   834	            // Label
   835	            // Text should not transform by barGroup.
   836	            // Ignore handlers transform
   837	            const barTransform = graphic.getTransform(
   838	                displaybles.handles[handleIndex].parent, this.group
   839	            );
   840	            const direction = graphic.transformDirection(
   841	                handleIndex === 0 ? 'right' : 'left', barTransform
   842	            );
   843	            const offset = this._handleWidth / 2 + LABEL_GAP;
   844	            const textPoint = graphic.applyTransform(
   845	                [
   846	                    orderedHandleEnds[handleIndex] + (handleIndex === 0 ? -offset : offset),
   847	                    this._size[1] / 2
   848	                ],
   849	                barTransform
   850	            );
   851	            handleLabels[handleIndex].setStyle({
   852	                x: textPoint[0],
   853	                y: textPoint[1],
   854	                verticalAlign: orient === HORIZONTAL ? 'middle' : direction as ZRTextVerticalAlign,
   855	                align: orient === HORIZONTAL ? direction as ZRTextAlign : 'center',
   856	                text: labelTexts[handleIndex]
   857	            });
   858	        }
   859	    }
   860	
   861	    private _formatLabel(value: ParsedValue, axis: Axis) {
   862	        const dataZoomModel = this.dataZoomModel;
   863	        const labelFormatter = dataZoomModel.get('labelFormatter');
   864	
   865	        let labelPrecision = dataZoomModel.get('labelPrecision');
   866	        if (labelPrecision == null || labelPrecision === 'auto') {
   867	            labelPrecision = axis.getPixelPrecision();
   868	        }
   869	
   870	        const valueStr = (value == null || isNaN(value as number))
   871	            ? ''
   872	            // FIXME Glue code
   873	            : (axis.type === 'category' || axis.type === 'time')
   874	                ? axis.scale.getLabel({
   875	                    value: Math.round(value as number)
   876	                })
   877	                // param of toFixed should less then 20.
   878	                : (value as number).toFixed(Math.min(labelPrecision as number, 20));
   879	
   880	        return isFunction(labelFormatter)
   881	            ? labelFormatter(value as number, valueStr)
   882	            : isString(labelFormatter)
   883	            ? labelFormatter.replace('{value}', valueStr)
   884	            : valueStr;
   885	    }

   887	    /**
   888	     * @param showOrHide true: show, false: hide
   889	     */
   890	    private _showDataInfo(showOrHide?: boolean) {
   891	        // Always show when drgging.
   892	        showOrHide = this._dragging || showOrHide;
   893	        const displayables = this._displayables;
   894	        const handleLabels = displayables.handleLabels;
   895	        handleLabels[0].attr('invisible', !showOrHide);
   896	        handleLabels[1].attr('invisible', !showOrHide);
   897	
   898	        // Highlight move handle
   899	        displayables.moveHandle
   900	            && this.api[showOrHide ? 'enterEmphasis' : 'leaveEmphasis'](displayables.moveHandle, 1);
   901	    }

   903	    private _onDragMove(handleIndex: 0 | 1 | 'all', dx: number, dy: number, event: ZRElementEvent) {
   904	        this._dragging = true;
   905	
   906	        // For mobile device, prevent screen slider on the button.
   907	        eventTool.stop(event.event);
   908	
   909	        // Transform dx, dy to bar coordination.
   910	        const barTransform = this._displayables.sliderGroup.getLocalTransform();
   911	        const vertex = graphic.applyTransform([dx, dy], barTransform, true);
   912	
   913	        const changed = this._updateInterval(handleIndex, vertex[0]);
   914	
   915	        const realtime = this.dataZoomModel.get('realtime');
   916	
   917	        this._updateView(!realtime);
   918	
   919	        // Avoid dispatch dataZoom repeatly but range not changed,
   920	        // which cause bad visual effect when progressive enabled.
   921	        changed && realtime && this._dispatchZoomAction(true);
   922	    }
   923	
   924	    private _onDragEnd() {
   925	        this._dragging = false;
   926	        this._showDataInfo(false);
   927	
   928	        // While in realtime mode and stream mode, dispatch action when
   929	        // drag end will cause the whole view rerender, which is unnecessary.
   930	        const realtime = this.dataZoomModel.get('realtime');
   931	        !realtime && this._dispatchZoomAction(false);
   932	    }

   934	    private _onClickPanel(e: ZRElementEvent) {
   935	        const size = this._size;
   936	        const localPoint = this._displayables.sliderGroup.transformCoordToLocal(e.offsetX, e.offsetY);
   937	
   938	        if (localPoint[0] < 0 || localPoint[0] > size[0]
   939	            || localPoint[1] < 0 || localPoint[1] > size[1]
   940	        ) {
   941	            return;
   942	        }
   943	
   944	        const handleEnds = this._handleEnds;
   945	        const center = (handleEnds[0] + handleEnds[1]) / 2;
   946	
   947	        const changed = this._updateInterval('all', localPoint[0] - center);
   948	        this._updateView();
   949	        changed && this._dispatchZoomAction(false);
   950	    }

   952	    private _onBrushStart(e: ZRElementEvent) {
   953	        const x = e.offsetX;
   954	        const y = e.offsetY;
   955	        this._brushStart = new graphic.Point(x, y);
   956	
   957	        this._brushing = true;
   958	
   959	        this._brushStartTime = +new Date();
   960	        // this._updateBrushRect(x, y);
   961	    }
   962	
   963	    private _onBrushEnd(e: ZRElementEvent) {
   964	        if (!this._brushing) {
   965	            return;
   966	        }
   967	
   968	        const brushRect = this._displayables.brushRect;
   969	        this._brushing = false;
   970	
   971	        if (!brushRect) {
   972	            return;
   973	        }
   974	
   975	        brushRect.attr('ignore', true);
   976	
   977	        const brushShape = brushRect.shape;
   978	
   979	        const brushEndTime = +new Date();
   980	        // console.log(brushEndTime - this._brushStartTime);
   981	        if (brushEndTime - this._brushStartTime < 200 && Math.abs(brushShape.width) < 5) {
   982	            // Will treat it as a click
   983	            return;
   984	        }
   985	
   986	        const viewExtend = this._getViewExtent();
   987	        const percentExtent = [0, 100];
   988	
   989	        this._range = asc([
   990	            linearMap(brushShape.x, viewExtend, percentExtent, true),
   991	            linearMap(brushShape.x + brushShape.width, viewExtend, percentExtent, true)
   992	        ]);
   993	
   994	        this._handleEnds = [brushShape.x, brushShape.x + brushShape.width];
   995	
   996	        this._updateView();
   997	
   998	        this._dispatchZoomAction(false);
   999	    }

  1001	    private _onBrush(e: ZRElementEvent) {
  1002	        if (this._brushing) {
  1003	            // For mobile device, prevent screen slider on the button.
  1004	            eventTool.stop(e.event);
  1005	
  1006	            this._updateBrushRect(e.offsetX, e.offsetY);
  1007	        }
  1008	    }

  1010	    private _updateBrushRect(mouseX: number, mouseY: number) {
  1011	        const displayables = this._displayables;
  1012	        const dataZoomModel = this.dataZoomModel;
  1013	        let brushRect = displayables.brushRect;
  1014	        if (!brushRect) {
  1015	            brushRect = displayables.brushRect = new Rect({
  1016	                silent: true,
  1017	                style: dataZoomModel.getModel('brushStyle').getItemStyle()
  1018	            });
  1019	            displayables.sliderGroup.add(brushRect);
  1020	        }
  1021	
  1022	        brushRect.attr('ignore', false);
  1023	
  1024	        const brushStart = this._brushStart;
  1025	
  1026	        const sliderGroup = this._displayables.sliderGroup;
  1027	
  1028	        const endPoint = sliderGroup.transformCoordToLocal(mouseX, mouseY);
  1029	        const startPoint = sliderGroup.transformCoordToLocal(brushStart.x, brushStart.y);
  1030	
  1031	        const size = this._size;
  1032	
  1033	        endPoint[0] = Math.max(Math.min(size[0], endPoint[0]), 0);
  1034	
  1035	        brushRect.setShape({
  1036	            x: startPoint[0], y: 0,
  1037	            width: endPoint[0] - startPoint[0], height: size[1]
  1038	        });
  1039	    }

  1041	    /**
  1042	     * This action will be throttled.
  1043	     */
  1044	    _dispatchZoomAction(realtime: boolean) {
  1045	        const range = this._range;
  1046	
  1047	        this.api.dispatchAction({
  1048	            type: 'dataZoom',
  1049	            from: this.uid,
  1050	            dataZoomId: this.dataZoomModel.id,
  1051	            animation: realtime ? REALTIME_ANIMATION_CONFIG : null,
  1052	            start: range[0],
  1053	            end: range[1]
  1054	        });
  1055	    }

  1057	    private _findCoordRect() {
  1058	        // Find the grid corresponding to the first axis referred by dataZoom.
  1059	        let rect: RectLike;
  1060	        const coordSysInfoList = collectReferCoordSysModelInfo(this.dataZoomModel).infoList;
  1061	
  1062	        if (!rect && coordSysInfoList.length) {
  1063	            const coordSys = coordSysInfoList[0].model.coordinateSystem;
  1064	            rect = coordSys.getRect && coordSys.getRect();
  1065	        }
  1066	
  1067	        if (!rect) {
  1068	            const width = this.api.getWidth();
  1069	            const height = this.api.getHeight();
  1070	            rect = {
  1071	                x: width * 0.2,
  1072	                y: height * 0.2,
  1073	                width: width * 0.6,
  1074	                height: height * 0.6
  1075	            };
  1076	        }
  1077	
  1078	        return rect;
  1079	    }
