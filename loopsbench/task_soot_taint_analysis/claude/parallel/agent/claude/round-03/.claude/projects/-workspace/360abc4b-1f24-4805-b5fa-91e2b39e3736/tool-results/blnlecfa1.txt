     1	"""
     2	soot_analysis.py -- Jimple-like intermediate representation and core
     3	whole-program analysis utilities (a Soot work-alike).
     4	
     5	This is layer 1 of the analysis stack.  It converts the neutral
     6	:mod:`program_model` representation into a Jimple-like three-address IR,
     7	registers everything in a global ``Scene``, builds a Class Hierarchy
     8	Analysis (CHA) call graph, and provides classic intra-procedural
     9	optimisations (copy propagation, dead-assignment elimination,
    10	unreachable-code elimination).
    11	
    12	Layer 2 (``flowdroid_analysis``) and layer 3 (``stubdroid_analysis``)
    13	build on the IR, the ``Scene`` and the ``CallGraph`` exposed here.
    14	
    15	Vocabulary follows Soot as closely as is reasonable in Python:
    16	
    17	    Value        -- Local, Constant, FieldRef, InvokeExpr, NewExpr, ...
    18	    Unit / Stmt  -- AssignStmt, IdentityStmt, InvokeStmt, ReturnStmt, ...
    19	    JimpleBody   -- locals + units of one method
    20	    SootField / SootMethod / SootClass
    21	    Scene        -- the global registry (``Scene.v()``)
    22	    Hierarchy    -- sub/super class queries and virtual dispatch
    23	    CallGraph    -- Edge set produced by ``CHATransformer``
    24	    UnitGraph    -- intra-procedural CFG
    25	    CopyPropagator / DeadAssignmentEliminator / UnreachableCodeEliminator
    26	
    27	Run standalone::
    28	
    29	    python3 soot_analysis.py
    30	"""

    74	def make_type(name: str) -> Type:
    75	    """Build a :class:`program_model.Type` from a type name string."""
    76	    if name is None:
    77	        name = OBJECT_CLASS
    78	    is_array = name.endswith("[]")
    79	    base = name
    80	    while base.endswith("[]"):
    81	        base = base[:-2]
    82	    return Type(
    83	        name=name,
    84	        is_primitive=base in PRIMITIVE_TYPE_NAMES and not is_array,
    85	        is_array=is_array,
    86	        element_type=base if is_array else None,
    87	    )
    88	
    89	
    90	VOID_TYPE = make_type("void")
    91	INT_TYPE = make_type("int")
    92	BOOLEAN_TYPE = make_type("boolean")
    93	OBJECT_TYPE = make_type(OBJECT_CLASS)
    94	STRING_TYPE = make_type("java.lang.String")
    95	NULL_TYPE = make_type("null_type")
    96	UNKNOWN_TYPE = make_type("unknown")

    99	def is_reference_type(t: Type | None) -> bool:
   100	    """True for object/array types (i.e. things that can carry taint)."""
   101	    if t is None:
   102	        return True
   103	    return not t.is_primitive or t.is_array

   111	class Value:
   112	    """Base class of the Jimple value hierarchy."""
   113	
   114	    __slots__ = ()
   115	
   116	    def get_type(self) -> Type:  # pragma: no cover - overridden
   117	        return UNKNOWN_TYPE
   118	
   119	    def get_use_boxes(self) -> list["Value"]:
   120	        """Values used by this value (transitively, excluding ``self``)."""
   121	        return []
   122	
   123	    def used_locals(self) -> list["Local"]:
   124	        out: list[Local] = []
   125	        for v in self.get_use_boxes():
   126	            if isinstance(v, Local):
   127	                out.append(v)
   128	        if isinstance(self, Local):
   129	            out.append(self)
   130	        return out
   131	
   132	    # -- value identity is structural, which keeps analyses deterministic --
   133	    def _key(self) -> tuple:
   134	        return (type(self).__name__, str(self))
   135	
   136	    def __eq__(self, other: object) -> bool:
   137	        if not isinstance(other, Value):
   138	            return NotImplemented
   139	        return self._key() == other._key()
   140	
   141	    def __hash__(self) -> int:
   142	        return hash(self._key())
   143	
   144	    def __repr__(self) -> str:
   145	        return str(self)

   306	class FieldSignature:
   307	    """``<declaring.Class: type name>``."""
   308	
   309	    __slots__ = ("declaring_class", "name", "type")
   310	
   311	    def __init__(self, declaring_class: str, name: str, field_type: Type | None = None) -> None:
   312	        self.declaring_class = declaring_class

   317	        return "<%s: %s %s>" % (self.declaring_class, self.type.name, self.name)

   320	        return "%s %s" % (self.type.name, self.name)

   324	
   325	    def __repr__(self) -> str:
   326	        return self.full_signature()

   328	    def __eq__(self, other: object) -> bool:
   329	        if not isinstance(other, FieldSignature):
   330	            return NotImplemented
   331	        return self.full_signature() == other.full_signature()

   333	    def __hash__(self) -> int:

   691	    return "%s %s(%s)" % (

   860	class Unit:
   861	    """Base class of the Jimple unit (statement) hierarchy.
   862	
   863	    Units carry a deterministic, globally unique ``uid`` so that they can be
   864	    used as dictionary/set keys without relying on object identity hashing.
   865	    """
   866	
   867	    _next_uid = 0
   868	
   869	    __slots__ = (
   870	        "uid",
   871	        "stmt_id",
   872	        "line_number",
   873	        "source_stmt",
   874	        "owner",
   875	        "_succ_ids",
   876	        "metadata",
   877	    )
   878	
   879	    def __init__(
   880	        self,
   881	        stmt_id: int = -1,
   882	        line_number: int = 0,
   883	        source_stmt: Statement | None = None,
   884	    ) -> None:
   885	        Unit._next_uid += 1
   886	        self.uid = Unit._next_uid
   887	        self.stmt_id = stmt_id
   888	        self.line_number = line_number
   889	        self.source_stmt = source_stmt
   890	        self.owner: "SootMethod | None" = None
   891	        self._succ_ids: list[int] = []
   892	        self.metadata: dict[str, Any] = dict(source_stmt.metadata) if source_stmt else {}
   893	
   894	    # -- classification ------------------------------------------------------
   895	    def branches(self) -> bool:
   896	        """True if control may transfer somewhere other than the next unit."""
   897	        return False
   898	
   899	    def falls_through(self) -> bool:
   900	        """True if control may continue with the textually next unit."""
   901	        return True
   902	
   903	    def is_return(self) -> bool:
   904	        return False
   905	
   906	    # -- def/use -------------------------------------------------------------
   907	    def get_def_boxes(self) -> list[Value]:
   908	        return []
   909	
   910	    def get_use_boxes(self) -> list[Value]:
   911	        return []
   912	
   913	    def get_defined_local(self) -> Local | None:
   914	        for v in self.get_def_boxes():
   915	            if isinstance(v, Local):
   916	                return v
   917	        return None
   918	
   919	    def get_used_locals(self) -> list[Local]:
   920	        out: list[Local] = []
   921	        seen: set[str] = set()
   922	        for v in self.get_use_boxes():
   923	            for loc in v.used_locals():
   924	                if loc.name not in seen:
   925	                    seen.add(loc.name)
   926	                    out.append(loc)
   927	        return out
   928	
   929	    # -- invoke helpers ------------------------------------------------------
   930	    def contains_invoke_expr(self) -> bool:
   931	        return self.get_invoke_expr() is not None
   932	
   933	    def get_invoke_expr(self) -> InvokeExpr | None:
   934	        return None
   935	
   936	    def contains_field_ref(self) -> bool:
   937	        return self.get_field_ref() is not None
   938	
   939	    def get_field_ref(self) -> FieldRefValue | None:
   940	        return None
   941	
   942	    def contains_array_ref(self) -> bool:
   943	        return self.get_array_ref() is not None
   944	
   945	    def get_array_ref(self) -> ArrayRef | None:
   946	        return None
   947	
   948	    # -- taint metadata forwarded from the neutral program model -------------
   949	    @property
   950	    def is_source(self) -> bool:
   951	        return bool(self.source_stmt.is_source) if self.source_stmt else False
   952	
   953	    @property
   954	    def is_sink(self) -> bool:
   955	        return bool(self.source_stmt.is_sink) if self.source_stmt else False
   956	
   957	    @property
   958	    def source_category(self) -> str | None:
   959	        return self.source_stmt.source_category if self.source_stmt else None
   960	
   961	    @property
   962	    def sink_category(self) -> str | None:
   963	        return self.source_stmt.sink_category if self.source_stmt else None
   964	
   965	    @property
   966	    def kind(self) -> str:
   967	        return self.source_stmt.kind if self.source_stmt else "nop"
   968	
   969	    def get_java_source_start_line_number(self) -> int:
   970	        return self.line_number
   971	
   972	    def replace_use(self, old: Value, new: Value) -> int:  # pragma: no cover - overridden
   973	        return 0
   974	
   975	    # -- identity ------------------------------------------------------------
   976	    def __eq__(self, other: object) -> bool:
   977	        if not isinstance(other, Unit):
   978	            return NotImplemented
   979	        return self.uid == other.uid
   980	
   981	    def __hash__(self) -> int:
   982	        return hash(self.uid)
   983	
   984	    def __repr__(self) -> str:
   985	        return str(self)
   986	
   987	    def __str__(self) -> str:  # pragma: no cover - overridden
   988	        return "nop"

  1004	class DefinitionStmt(Stmt):
  1005	    """Base class of statements that define something (``left = right``)."""
  1006	
  1007	    __slots__ = ("left_op", "right_op")
  1008	
  1009	    def __init__(
  1010	        self,
  1011	        left_op: Value,
  1012	        right_op: Value,
  1013	        stmt_id: int = -1,
  1014	        line_number: int = 0,
  1015	        source_stmt: Statement | None = None,
  1016	    ) -> None:
  1017	        super().__init__(stmt_id, line_number, source_stmt)
  1018	        self.left_op = left_op
  1019	        self.right_op = right_op
  1020	
  1021	    def get_left_op(self) -> Value:
  1022	        return self.left_op
  1023	
  1024	    def get_right_op(self) -> Value:
  1025	        return self.right_op
  1026	
  1027	    def get_def_boxes(self) -> list[Value]:
  1028	        return [self.left_op]
  1029	
  1030	    def get_use_boxes(self) -> list[Value]:
  1031	        uses: list[Value] = [self.right_op]
  1032	        uses.extend(self.right_op.get_use_boxes())
  1033	        # ``a.f = x`` and ``a[i] = x`` *use* the base (and index) too.
  1034	        if isinstance(self.left_op, (InstanceFieldRef, ArrayRef)):
  1035	            uses.extend(self.left_op.get_use_boxes())
  1036	        return uses
  1037	
  1038	    def get_def_locals(self) -> list[Local]:
  1039	        return [self.left_op] if isinstance(self.left_op, Local) else []
  1040	
  1041	    def get_invoke_expr(self) -> InvokeExpr | None:
  1042	        return self.right_op if isinstance(self.right_op, InvokeExpr) else None
  1043	
  1044	    def get_field_ref(self) -> FieldRefValue | None:
  1045	        if isinstance(self.left_op, FieldRefValue):
  1046	            return self.left_op
  1047	        if isinstance(self.right_op, FieldRefValue):
  1048	            return self.right_op
  1049	        return None
  1050	
  1051	    def get_array_ref(self) -> ArrayRef | None:
  1052	        if isinstance(self.left_op, ArrayRef):
  1053	            return self.left_op
  1054	        if isinstance(self.right_op, ArrayRef):
  1055	            return self.right_op
  1056	        return None
  1057	
  1058	    def replace_use(self, old: Value, new: Value) -> int:
  1059	        replaced = 0
  1060	        if self.right_op == old:
  1061	            self.right_op = new
  1062	            replaced += 1
  1063	        else:
  1064	            replaced += _replace_in_value(self.right_op, old, new)
  1065	        if isinstance(self.left_op, (InstanceFieldRef, ArrayRef)):
  1066	            replaced += _replace_in_value(self.left_op, old, new)
  1067	        return replaced
  1068	
  1069	    def __str__(self) -> str:
  1070	        return "%s = %s" % (self.left_op, self.right_op)
  1071

  1289	        if self.condition == old:
  1290	            self.condition = new
  1291	            return 1
  1292	        return _replace_in_value(self.condition, old, new)
  1293	
  1294	    def __str__(self) -> str:
  1295	        t = self.target
  1296	        return "if %s goto %s" % (
  1297	            self.condition,
  1298	            "stmt%d" % t.stmt_id if t is not None else "?",
  1299	        )

  1302	def _replace_in_value(value: Value, old: Value, new: Value) -> int:
  1303	    """Structurally replace ``old`` with ``new`` inside ``value``."""
  1304	    replaced = 0
  1305	    if isinstance(value, InstanceInvokeExpr):
  1306	        if value.base == old:
  1307	            value.base = new
  1308	            replaced += 1
  1309	    if isinstance(value, InvokeExpr):
  1310	        for i, a in enumerate(value.args):
  1311	            if a == old:
  1312	                value.args[i] = new
  1313	                replaced += 1
  1314	    elif isinstance(value, InstanceFieldRef):
  1315	        if value.base == old:
  1316	            value.base = new
  1317	            replaced += 1
  1318	    elif isinstance(value, ArrayRef):
  1319	        if value.base == old:
  1320	            value.base = new
  1321	            replaced += 1
  1322	        if value.index == old:
  1323	            value.index = new
  1324	            replaced += 1
  1325	    elif isinstance(value, CastExpr):
  1326	        if value.op == old:
  1327	            value.op = new
  1328	            replaced += 1
  1329	    elif isinstance(value, InstanceOfExpr):
  1330	        if value.op == old:
  1331	            value.op = new
  1332	            replaced += 1
  1333	    elif isinstance(value, PhiExpr):
  1334	        for i, o in enumerate(value.operands):
  1335	            if o == old:
  1336	                value.operands[i] = new
  1337	                replaced += 1
  1338	    elif isinstance(value, BinopExpr):
  1339	        if value.op1 == old:
  1340	            value.op1 = new
  1341	            replaced += 1
  1342	        if value.op2 == old:
  1343	            value.op2 = new
  1344	            replaced += 1
  1345	    elif isinstance(value, NewArrayExpr):
  1346	        if value.size == old:
  1347	            value.size = new
  1348	            replaced += 1
  1349	    return replaced

  1374	class JimpleBody:
  1375	    """The Jimple body of one method: locals plus an ordered unit chain."""
  1376	
  1377	    def __init__(
  1378	        self,

  1380	        locals_: list[Local] | None = None,

  1382	        this_local: Local | None = None,
  1383	        param_locals: list[Local] | None = None,
  1384	    ) -> None:
  1385	        self.method = method
  1386	        self.locals: list[Local] = list(locals_ or [])
  1387	        self.units: list[Unit] = list(units or [])
  1388	        self.this_local = this_local
  1389	        self.param_locals: list[Local] = list(param_locals or [])
  1390	        self.traps: list[dict] = []
  1391	
  1392	    # -- Soot-style accessors ------------------------------------------------
  1393	    def get_method(self) -> "SootMethod | None":
  1394	        return self.method
  1395	
  1396	    def get_units(self) -> list[Unit]:
  1397	        return self.units
  1398	
  1399	    def get_locals(self) -> list[Local]:
  1400	        return self.locals
  1401	
  1402	    def get_this_local(self) -> Local | None:
  1403	        return self.this_local
  1404	
  1405	    def get_parameter_local(self, index: int) -> Local | None:
  1406	        if 0 <= index < len(self.param_locals):
  1407	            return self.param_locals[index]
  1408	        return None
  1409	
  1410	    def get_parameter_locals(self) -> list[Local]:
  1411	        return list(self.param_locals)
  1412	
  1413	    def get_local_by_name(self, name: str) -> Local | None:
  1414	        for loc in self.locals:
  1415	            if loc.name == name:
  1416	                return loc
  1417	        return None
  1418	
  1419	    def get_unit_by_stmt_id(self, stmt_id: int) -> Unit | None:
  1420	        for u in self.units:
  1421	            if u.stmt_id == stmt_id:
  1422	                return u
  1423	        return None
  1424	
  1425	    def get_head(self) -> Unit | None:
  1426	        return self.units[0] if self.units else None
  1427	
  1428	    def get_tails(self) -> list[Unit]:
  1429	        return [u for u in self.units if not u.falls_through() and not u.branches()]
  1430	
  1431	    def get_return_units(self) -> list[Unit]:
  1432	        return [u for u in self.units if u.is_return()]
  1433	
  1434	    def index_of(self, unit: Unit) -> int:
  1435	        for i, u in enumerate(self.units):
  1436	            if u is unit or u == unit:
  1437	                return i
  1438	        return -1
  1439	
  1440	    def get_successor_of(self, unit: Unit) -> Unit | None:
  1441	        i = self.index_of(unit)
  1442	        if i >= 0 and i + 1 < len(self.units):
  1443	            return self.units[i + 1]
  1444	        return None
  1445	
  1446	    def __len__(self) -> int:
  1447	        return len(self.units)
  1448	
  1449	    def __iter__(self) -> Iterator[Unit]:
  1450	        return iter(self.units)
  1451	
  1452	    def to_string(self) -> str:
  1453	        lines = []
  1454	        sig = self.method.get_signature() if self.method else "<unknown>"
  1455	        lines.append("%s {" % sig)
  1456	        for loc in self.locals:
  1457	            lines.append("    %s %s;" % (loc.type.name, loc.name))
  1458	        if self.locals:
  1459	            lines.append("")
  1460	        for u in self.units:
  1461	            lines.append("    stmt%-3d %s;" % (u.stmt_id, u))
  1462	        lines.append("}")
  1463	        return "\n".join(lines)
  1464	
  1465	    def __str__(self) -> str:
  1466	        return self.to_string()
  1467	
  1468	    def clone(self) -> "JimpleBody":
  1469	        """Deep-ish clone: fresh units, shared method reference."""
  1470	        if self.method is None:
  1471	            return JimpleBody()
  1472	        builder = IRBuilder(Scene.v())
  1473	        return builder.rebuild_body(self)
