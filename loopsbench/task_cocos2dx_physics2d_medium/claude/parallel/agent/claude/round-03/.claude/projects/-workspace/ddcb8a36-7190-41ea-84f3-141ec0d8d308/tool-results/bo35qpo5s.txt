   114	void PhysicsShape::setScale(float scaleX, float scaleY)
   115	{
   116	    if (std::abs(_scaleX - scaleX) > FLT_EPSILON || std::abs(_scaleY - scaleY) > FLT_EPSILON)
   117	    {
   118	        if (_type == Type::CIRCLE && scaleX != scaleY)
   119	        {
   120	            CCLOG("PhysicsShapeCircle WARNING: CANNOT support setScale with different x and y");
   121	            return;
   122	        }
   123	        _newScaleX = scaleX;
   124	        _newScaleY = scaleY;
   125	        
   126	        updateScale();
   127	        
   128	        // re-calculate area and mass
   129	        _area = calculateArea();
   130	        _mass = _material.density * _area;
   131	        _moment = calculateDefaultMoment();
   132	    }
   133	}

   221	void PhysicsShape::setDensity(float density)
   222	{
   223	    if (density < 0)
   224	    {
   225	        return;
   226	    }
   227	    
   228	    _material.density = density;
   229	    
   230	    if (_material.density == PHYSICS_INFINITY)
   231	    {
   232	        setMass(PHYSICS_INFINITY);
   233	    }else if (_area > 0)
   234	    {
   235	        setMass(_material.density * _area);
   236	    }
   237	}

   271	void PhysicsShape::recenterPoints(Vec2* points, int count, const Vec2& center)
   272	{
   273	    cpVect* cpvs = new cpVect[count];
   274	    PhysicsHelper::points2cpvs(points, cpvs, count);
   275	    cpVect centroid = cpCentroidForPoly(count, cpvs);
   276	    for(int i=0; i<count; i++){
   277	        cpvs[i] = cpvsub(cpvs[i], centroid);
   278	    }
   279	    PhysicsHelper::cpvs2points(cpvs, points, count);
   280	    delete[] cpvs;
   281	    
   282	    if (center != Vec2::ZERO)
   283	    {
   284	        for (int i = 0; i < count; ++i)
   285	        {
   286	            points[i] += center;
   287	        }
   288	    }
   289	}

   301	void PhysicsShape::setBody(PhysicsBody *body)
   302	{
   303	    // already added
   304	    if (body && _body == body)
   305	    {
   306	        return;
   307	    }
   308	    
   309	    if (_body)
   310	    {
   311	        _body->removeShape(this);
   312	    }
   313	    
   314	    for (auto shape : _cpShapes)
   315	    {
   316	        cpShapeSetBody(shape, body == nullptr ? s_sharedBody : body->_cpBody);
   317	    }
   318	    _body = body;
   319	}

   335	bool PhysicsShapeCircle::init(float radius, const PhysicsMaterial& material/* = MaterialDefault*/, const Vec2& offset /*= Vec2(0, 0)*/)
   336	{
   337	    do
   338	    {
   339	        _type = Type::CIRCLE;
   340	        
   341	        auto shape = cpCircleShapeNew(s_sharedBody, radius, PhysicsHelper::point2cpv(offset));
   342	        CC_BREAK_IF(shape == nullptr);
   343	        cpShapeSetUserData(shape, this);
   344	        
   345	        addShape(shape);
   346	        
   347	        _area = calculateArea();
   348	        _mass = material.density == PHYSICS_INFINITY ? PHYSICS_INFINITY : material.density * _area;
   349	        _moment = calculateDefaultMoment();
   350	        
   351	        setMaterial(material);
   352	        return true;
   353	    } while (false);
   354	    
   355	    return false;
   356	}

   398	void PhysicsShapeCircle::updateScale()
   399	{
   400	    cpFloat factor = std::abs(_newScaleX / _scaleX);
   401	
   402	    cpShape* shape = _cpShapes.front();
   403	    cpVect v = cpCircleShapeGetOffset(shape);
   404	    v = cpvmult(v, factor);
   405	    cpCircleShapeSetOffset(shape, v);
   406	
   407	    cpCircleShapeSetRadius(shape, cpCircleShapeGetRadius(shape) * factor);
   408	
   409	    PhysicsShape::updateScale();
   410	}

   469	void PhysicsShapeEdgeSegment::updateScale()
   470	{
   471	    cpFloat factorX = _newScaleX / _scaleX;
   472	    cpFloat factorY = _newScaleY / _scaleY;
   473	
   474	    cpShape* shape = _cpShapes.front();
   475	    cpVect a = cpSegmentShapeGetA(shape);
   476	    a.x *= factorX;
   477	    a.y *= factorY;
   478	    cpVect b = cpSegmentShapeGetB(shape);
   479	    b.x *= factorX;
   480	    b.y *= factorY;
   481	    cpSegmentShapeSetEndpoints(shape, a, b);
   482	
   483	    PhysicsShape::updateScale();
   484	}

   500	bool PhysicsShapeBox::init(const Size& size, const PhysicsMaterial& material/* = MaterialDefault*/, const Vec2& offset /*= Vec2(0, 0)*/, float radius/* = 0.0f*/)
   501	{
   502	    do
   503	    {
   504	        _type = Type::BOX;
   505	        
   506	        auto wh = PhysicsHelper::size2cpv(size);
   507	        cpVect vec[4] =
   508	        {
   509	            {-wh.x/2.0f, -wh.y/2.0f}, {-wh.x/2.0f, wh.y/2.0f}, {wh.x/2.0f, wh.y/2.0f}, {wh.x/2.0f, -wh.y/2.0f}
   510	        };
   511	        
   512	        cpTransform transform = cpTransformTranslate(PhysicsHelper::point2cpv(offset));
   513	        
   514	        auto shape = cpPolyShapeNew(s_sharedBody, 4, vec, transform, radius);
   515	        CC_BREAK_IF(shape == nullptr);
   516	        cpShapeSetUserData(shape, this);
   517	        
   518	        addShape(shape);
   519	        
   520	        _area = calculateArea();
   521	        _mass = material.density == PHYSICS_INFINITY ? PHYSICS_INFINITY : material.density * _area;
   522	        _moment = calculateDefaultMoment();
   523	        
   524	        setMaterial(material);
   525	        
   526	        return true;
   527	    } while (false);
   528	    
   529	    return false;
   530	}

   592	float PhysicsShapePolygon::calculateMoment(float mass, const Vec2* points, int count, const Vec2& offset, float radius)
   593	{

   595	    PhysicsHelper::points2cpvs(points, vecs, count);
   596	    float moment = mass == PHYSICS_INFINITY ? PHYSICS_INFINITY
   597	    : PhysicsHelper::cpfloat2float(cpMomentForPoly(mass, count, vecs, PhysicsHelper::point2cpv(offset), radius));

   599

   602	
   603	float PhysicsShapePolygon::calculateArea()
   604	{
   605	    auto shape = _cpShapes.front();
   606	    int count = cpPolyShapeGetCount(shape);
   607	    cpVect* vecs = new cpVect[count];
   608	    for(int i=0;i<count;++i)
   609	        vecs[i] = cpPolyShapeGetVert(shape, i);
   610	    float area = PhysicsHelper::cpfloat2float(cpAreaForPoly(count, vecs, cpPolyShapeGetRadius(shape)));
   611	    CC_SAFE_DELETE_ARRAY(vecs);
   612	    return area;
   613	}
   614	
   615	float PhysicsShapePolygon::calculateDefaultMoment()
   616	{
   617	    if(_mass == PHYSICS_INFINITY)
   618	    {
   619	        return PHYSICS_INFINITY;
   620	    }
   621	    else
   622	    {
   623	        auto shape = _cpShapes.front();
   624	        int count = cpPolyShapeGetCount(shape);
   625	        cpVect* vecs = new cpVect[count];
   626	        for(int i=0;i<count;++i)
   627	            vecs[i] = cpPolyShapeGetVert(shape, i);
   628	        float moment = PhysicsHelper::cpfloat2float(cpMomentForPoly(_mass, count, vecs, cpvzero, cpPolyShapeGetRadius(shape)));
   629	        CC_SAFE_DELETE_ARRAY(vecs);
   630	        return moment;
   631	    }
   632	}

   657	    auto shape = _cpShapes.front();
   658	    int count = cpPolyShapeGetCount(shape);
   659	    cpVect* vecs = new cpVect[count];
   660	    for(int i=0;i<count;++i)
   661	        vecs[i] = cpPolyShapeGetVert(shape, i);
   662	    
   663	    Vec2 center =  PhysicsHelper::cpv2point(cpCentroidForPoly(count, vecs));
   664	    CC_SAFE_DELETE_ARRAY(vecs);
   665	    
   666	    return center;
   667	}

   669	void PhysicsShapePolygon::updateScale()
   670	{
   671	    cpFloat factorX = _newScaleX / _scaleX;
   672	    cpFloat factorY = _newScaleY / _scaleY;
   673	
   674	    auto shape = _cpShapes.front();
   675	    int count = cpPolyShapeGetCount(shape);
   676	    cpVect* vecs = new cpVect[count];
   677	    for(int i=0;i<count;++i)
   678	    {
   679	        cpVect v = cpPolyShapeGetVert(shape, i);
   680	        v.x *= factorX;
   681	        v.y *= factorY;
   682	        vecs[i] = v;
   683	    }
   684	
   685	    // convert hole to clockwise
   686	    if (factorX * factorY < 0)
   687	    {
   688	        for (int i = 0; i < count / 2; ++i)
   689	        {
   690	            cpVect v = vecs[i];
   691	            vecs[i] = vecs[count - i - 1];
   692	            vecs[count - i - 1] = v;
   693	        }
   694	    }
   695	
   696	    cpPolyShapeSetVerts(shape, count, vecs, cpTransformIdentity);
   697	    CC_SAFE_DELETE_ARRAY(vecs);
   698	
   699	    PhysicsShape::updateScale();
   700	}

   716	bool PhysicsShapeEdgeBox::init(const Size& size, const PhysicsMaterial& material/* = MaterialDefault*/, float border/* = 1*/, const Vec2& offset/*= Vec2(0, 0)*/)
   717	{
   718	    do
   719	    {
   720	        _type = Type::EDGEBOX;
   721	        
   722	        cpVect vec[4] = {};
   723	        vec[0] = PhysicsHelper::point2cpv(Vec2(-size.width/2+offset.x, -size.height/2+offset.y));
   724	        vec[1] = PhysicsHelper::point2cpv(Vec2(+size.width/2+offset.x, -size.height/2+offset.y));
   725	        vec[2] = PhysicsHelper::point2cpv(Vec2(+size.width/2+offset.x, +size.height/2+offset.y));
   726	        vec[3] = PhysicsHelper::point2cpv(Vec2(-size.width/2+offset.x, +size.height/2+offset.y));
   727	        
   728	        int i = 0;
   729	        for (; i < 4; ++i)
   730	        {
   731	            auto shape = cpSegmentShapeNew(s_sharedBody, vec[i], vec[(i + 1) % 4], border);
   732	            CC_BREAK_IF(shape == nullptr);
   733	            cpShapeSetUserData(shape, this);
   734	            addShape(shape);
   735	        }
   736	        CC_BREAK_IF(i < 4);
   737	        
   738	        _mass = PHYSICS_INFINITY;
   739	        _moment = PHYSICS_INFINITY;
   740	        
   741	        setMaterial(material);
   742	        
   743	        return true;
   744	    } while (false);
   745	    
   746	    return false;
   747	}
