    87	    do {                                    \
    88	    if(UNLIKELY(_initDirty))                \
    89	    {                                       \
    90	        return _writeCache->field.get();    \
    91	    }                                       \
    92	    }while(0)

    96	#define CC_PJOINT_CACHE_WRITE2(field, method, arg, convertedArg)    \
    97	    do {                                                            \
    98	    if(UNLIKELY(_initDirty))                                        \
    99	    {                                                               \
   100	        _writeCache->field.set(arg);                                \
   101	        delay([this, arg]() {                                       \
   102	            method(_cpConstraints.front(), convertedArg);           \
   103	        });                                                         \
   104	    }                                                               \
   105	    else                                                            \
   106	    {                                                               \
   107	        method(_cpConstraints.front(), convertedArg);               \
   108	    }                                                               \
   109	    }while(0)

   128	PhysicsJoint::~PhysicsJoint()
   129	{
   130	    // reset the shapes collision group
   131	    setCollisionEnable(true);
   132	
   133	    for (cpConstraint* joint : _cpConstraints)
   134	    {
   135	        cpConstraintFree(joint);
   136	    }
   137	    _cpConstraints.clear();
   138	
   139	    delete _writeCache;
   140	}

   142	bool PhysicsJoint::init(cocos2d::PhysicsBody *a, cocos2d::PhysicsBody *b)
   143	{
   144	    do
   145	    {
   146	        CCASSERT(a != nullptr && b != nullptr, "the body passed in is nil");
   147	        CCASSERT(a != b, "the two bodies are equal");
   148	
   149	        _bodyA = a;
   150	        _bodyB = b;
   151	        _bodyA->_joints.push_back(this);
   152	        _bodyB->_joints.push_back(this);
   153	
   154	        return true;
   155	    } while (false);
   156	
   157	    return false;
   158	}
   159	
   160	bool PhysicsJoint::initJoint()
   161	{
   162	    bool ret = !_initDirty;
   163	    while (_initDirty)
   164	    {
   165	        ret = createConstraints();
   166	        CC_BREAK_IF(!ret);
   167	
   168	        for (auto subjoint : _cpConstraints)
   169	        {
   170	            cpConstraintSetMaxForce(subjoint, _maxForce);
   171	            cpConstraintSetErrorBias(subjoint, cpfpow(1.0f - 0.15f, 60.0f));
   172	            cpSpaceAddConstraint(_world->_cpSpace, subjoint);
   173	        }
   174	        _initDirty = false;
   175	        ret = true;
   176	    }
   177	
   178	    return ret;
   179	}

   190	void PhysicsJoint::setEnable(bool enable)
   191	{
   192	    if (_enable != enable)
   193	    {
   194	        _enable = enable;
   195	
   196	        if (_world)
   197	        {
   198	            if (enable)
   199	            {
   200	                _world->addJoint(this);
   201	            }
   202	            else
   203	            {
   204	                _world->removeJoint(this, false);
   205	            }
   206	        }
   207	    }
   208	}

   226	void PhysicsJoint::setMaxForce(float force)
   227	{
   228	    if(_initDirty)
   229	    {
   230	        delay([this, force]() {
   231	            _maxForce = force;
   232	            for (auto joint : _cpConstraints)
   233	            {
   234	                cpConstraintSetMaxForce(joint, force);
   235	            }
   236	        });
   237	    }
   238	    else
   239	    {
   240	        _maxForce = force;
   241	        for (auto joint : _cpConstraints)
   242	        {
   243	            cpConstraintSetMaxForce(joint, force);
   244	        }
   245	    }
   246	}

   262	bool PhysicsJointFixed::createConstraints()
   263	{
   264	    do
   265	    {
   266	        _bodyA->getNode()->setPosition(_anchr);
   267	        _bodyB->getNode()->setPosition(_anchr);
   268	
   269	        // add a pivot joint to fixed two body together
   270	        auto constraint = cpPivotJointNew(_bodyA->getCPBody(), _bodyB->getCPBody(),
   271	            PhysicsHelper::point2cpv(_anchr));
   272	        CC_BREAK_IF(constraint == nullptr);
   273	        _cpConstraints.push_back(constraint);
   274	
   275	        // add a gear joint to make two body have the same rotation.
   276	        constraint = cpGearJointNew(_bodyA->getCPBody(), _bodyB->getCPBody(), 0, 1);
   277	        CC_BREAK_IF(constraint == nullptr);
   278	        _cpConstraints.push_back(constraint);
   279	
   280	        setCollisionEnable(false);
   281	
   282	        return true;
   283	    } while (false);
   284	
   285	    return false;
   286	}

   320	bool PhysicsJointPin::createConstraints()
   321	{
   322	    do
   323	    {
   324	        cpConstraint* constraint = nullptr;
   325	        if (_useSpecificAnchr)
   326	        {
   327	            constraint = cpPivotJointNew2(_bodyA->getCPBody(), _bodyB->getCPBody(),
   328	                PhysicsHelper::point2cpv(_anchr1),
   329	                PhysicsHelper::point2cpv(_anchr2));
   330	        }
   331	        else
   332	        {
   333	            constraint = cpPivotJointNew(_bodyA->getCPBody(), _bodyB->getCPBody(),
   334	                PhysicsHelper::point2cpv(_anchr1));
   335	        }
   336	
   337	        CC_BREAK_IF(constraint == nullptr);
   338	
   339	        _cpConstraints.push_back(constraint);
   340	
   341	        return true;
   342	    } while (false);
   343	
   344	    return false;
   345	}
