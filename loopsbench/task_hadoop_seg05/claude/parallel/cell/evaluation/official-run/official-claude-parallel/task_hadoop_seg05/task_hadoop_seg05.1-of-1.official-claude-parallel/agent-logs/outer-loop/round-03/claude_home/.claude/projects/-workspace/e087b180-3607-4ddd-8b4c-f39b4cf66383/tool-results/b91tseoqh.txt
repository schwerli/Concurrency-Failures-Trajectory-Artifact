   259	   * YARN Router forwards every getNewApplication requests to any RM. During
   260	   * this operation there will be no communication with the State Store. The
   261	   * Router will forward the requests to any SubCluster. The Router will retry
   262	   * to submit the request on #numSubmitRetries different SubClusters. The
   263	   * SubClusters are randomly chosen from the active ones.
   264	   *
   265	   * Possible failures and behaviors:
   266	   *
   267	   * Client: identical behavior as {@code ClientRMService}.
   268	   *
   269	   * Router: the Client will timeout and resubmit.
   270	   *
   271	   * ResourceManager: the Router will timeout and contacts another RM.
   272	   *
   273	   * StateStore: not in the execution.
   274	   */
   275	  @Override
   276	  public GetNewApplicationResponse getNewApplication(
   277	      GetNewApplicationRequest request) throws YarnException, IOException {
   278	
   279	    long startTime = clock.getTime();
   280	
   281	    Map<SubClusterId, SubClusterInfo> subClustersActive =
   282	        federationFacade.getSubClusters(true);
   283	
   284	    for (int i = 0; i < numSubmitRetries; ++i) {
   285	      SubClusterId subClusterId = getRandomActiveSubCluster(subClustersActive);
   286	      LOG.debug(
   287	          "getNewApplication try #{} on SubCluster {}", i, subClusterId);
   288	      ApplicationClientProtocol clientRMProxy =
   289	          getClientRMProxyForSubCluster(subClusterId);
   290	      GetNewApplicationResponse response = null;
   291	      try {
   292	        response = clientRMProxy.getNewApplication(request);
   293	      } catch (Exception e) {
   294	        LOG.warn("Unable to create a new ApplicationId in SubCluster "
   295	            + subClusterId.getId(), e);
   296	      }
   297	
   298	      if (response != null) {
   299	
   300	        long stopTime = clock.getTime();
   301	        routerMetrics.succeededAppsCreated(stopTime - startTime);
   302	        return response;
   303	      } else {
   304	        // Empty response from the ResourceManager.
   305	        // Blacklist this subcluster for this request.
   306	        subClustersActive.remove(subClusterId);
   307	      }
   308	
   309	    }
   310	
   311	    routerMetrics.incrAppsFailedCreated();
   312	    String errMsg = "Fail to create a new application.";
   313	    LOG.error(errMsg);
   314	    throw new YarnException(errMsg);
   315	  }

   317	  /**
   318	   * Today, in YARN there are no checks of any applicationId submitted.
   319	   *
   320	   * Base scenarios:
   321	   *
   322	   * The Client submits an application to the Router. The Router selects one
   323	   * SubCluster to forward the request. The Router inserts a tuple into
   324	   * StateStore with the selected SubCluster (e.g. SC1) and the appId. The
   325	   * State Store replies with the selected SubCluster (e.g. SC1). The Router
   326	   * submits the request to the selected SubCluster.
   327	   *
   328	   * In case of State Store failure:
   329	   *
   330	   * The client submits an application to the Router. The Router selects one
   331	   * SubCluster to forward the request. The Router inserts a tuple into State
   332	   * Store with the selected SubCluster (e.g. SC1) and the appId. Due to the
   333	   * State Store down the Router times out and it will retry depending on the
   334	   * FederationFacade settings. The Router replies to the client with an error
   335	   * message.
   336	   *
   337	   * If State Store fails after inserting the tuple: identical behavior as
   338	   * {@code ClientRMService}.
   339	   *
   340	   * In case of Router failure:
   341	   *
   342	   * Scenario 1 – Crash before submission to the ResourceManager
   343	   *
   344	   * The Client submits an application to the Router. The Router selects one
   345	   * SubCluster to forward the request. The Router inserts a tuple into State
   346	   * Store with the selected SubCluster (e.g. SC1) and the appId. The Router
   347	   * crashes. The Client timeouts and resubmits the application. The Router
   348	   * selects one SubCluster to forward the request. The Router inserts a tuple
   349	   * into State Store with the selected SubCluster (e.g. SC2) and the appId.
   350	   * Because the tuple is already inserted in the State Store, it returns the
   351	   * previous selected SubCluster (e.g. SC1). The Router submits the request
   352	   * to the selected SubCluster (e.g. SC1).
   353	   *
   354	   * Scenario 2 – Crash after submission to the ResourceManager
   355	   *
   356	   * The Client submits an application to the Router. The Router selects one
   357	   * SubCluster to forward the request. The Router inserts a tuple into State
   358	   * Store with the selected SubCluster (e.g. SC1) and the appId. The Router
   359	   * submits the request to the selected SubCluster. The Router crashes. The
   360	   * Client timeouts and resubmit the application. The Router selects one
   361	   * SubCluster to forward the request. The Router inserts a tuple into State
   362	   * Store with the selected SubCluster (e.g. SC2) and the appId. The State
   363	   * Store replies with the selected SubCluster (e.g. SC1). The Router submits
   364	   * the request to the selected SubCluster (e.g. SC1). When a client re-submits
   365	   * the same application to the same RM, it does not raise an exception and
   366	   * replies with operation successful message.
   367	   *
   368	   * In case of Client failure: identical behavior as {@code ClientRMService}.
   369	   *
   370	   * In case of ResourceManager failure:
   371	   *
   372	   * The Client submits an application to the Router. The Router selects one
   373	   * SubCluster to forward the request. The Router inserts a tuple into State
   374	   * Store with the selected SubCluster (e.g. SC1) and the appId. The Router
   375	   * submits the request to the selected SubCluster. The entire SubCluster is
   376	   * down – all the RMs in HA or the master RM is not reachable. The Router
   377	   * times out. The Router selects a new SubCluster to forward the request.
   378	   * The Router update a tuple into State Store with the selected SubCluster
   379	   * (e.g. SC2) and the appId. The State Store replies with OK answer. The
   380	   * Router submits the request to the selected SubCluster (e.g. SC2).
   381	   */
   382	  @Override
   383	  public SubmitApplicationResponse submitApplication(
   384	      SubmitApplicationRequest request) throws YarnException, IOException {
   385	
   386	    long startTime = clock.getTime();
   387	
   388	    if (request == null || request.getApplicationSubmissionContext() == null
   389	        || request.getApplicationSubmissionContext()
   390	            .getApplicationId() == null) {
   391	      routerMetrics.incrAppsFailedSubmitted();
   392	      RouterServerUtil
   393	          .logAndThrowException("Missing submitApplication request or "
   394	              + "applicationSubmissionContex information.", null);
   395	    }
   396	
   397	    ApplicationId applicationId =
   398	        request.getApplicationSubmissionContext().getApplicationId();
   399	
   400	    List<SubClusterId> blacklist = new ArrayList<SubClusterId>();
   401	
   402	    for (int i = 0; i < numSubmitRetries; ++i) {
   403	
   404	      SubClusterId subClusterId = policyFacade.getHomeSubcluster(
   405	          request.getApplicationSubmissionContext(), blacklist);
   406	      LOG.info("submitApplication appId" + applicationId + " try #" + i
   407	          + " on SubCluster " + subClusterId);
   408	
   409	      ApplicationHomeSubCluster appHomeSubCluster =
   410	          ApplicationHomeSubCluster.newInstance(applicationId, subClusterId);
   411	
   412	      if (i == 0) {
   413	        try {
   414	          // persist the mapping of applicationId and the subClusterId which has
   415	          // been selected as its home
   416	          subClusterId =
   417	              federationFacade.addApplicationHomeSubCluster(appHomeSubCluster);
   418	        } catch (YarnException e) {
   419	          routerMetrics.incrAppsFailedSubmitted();
   420	          String message = "Unable to insert the ApplicationId " + applicationId
   421	              + " into the FederationStateStore";
   422	          RouterServerUtil.logAndThrowException(message, e);
   423	        }
   424	      } else {
   425	        try {
   426	          // update the mapping of applicationId and the home subClusterId to
   427	          // the new subClusterId we have selected
   428	          federationFacade.updateApplicationHomeSubCluster(appHomeSubCluster);
   429	        } catch (YarnException e) {
   430	          String message = "Unable to update the ApplicationId " + applicationId
   431	              + " into the FederationStateStore";
   432	          SubClusterId subClusterIdInStateStore =
   433	              federationFacade.getApplicationHomeSubCluster(applicationId);
   434	          if (subClusterId == subClusterIdInStateStore) {
   435	            LOG.info("Application " + applicationId
   436	                + " already submitted on SubCluster " + subClusterId);
   437	          } else {
   438	            routerMetrics.incrAppsFailedSubmitted();
   439	            RouterServerUtil.logAndThrowException(message, e);
   440	          }
   441	        }
   442	      }
   443	
   444	      ApplicationClientProtocol clientRMProxy =
   445	          getClientRMProxyForSubCluster(subClusterId);
   446	
   447	      SubmitApplicationResponse response = null;
   448	      try {
   449	        response = clientRMProxy.submitApplication(request);
   450	      } catch (Exception e) {
   451	        LOG.warn("Unable to submit the application " + applicationId
   452	            + "to SubCluster " + subClusterId.getId(), e);
   453	      }
   454	
   455	      if (response != null) {
   456	        LOG.info("Application "
   457	            + request.getApplicationSubmissionContext().getApplicationName()
   458	            + " with appId " + applicationId + " submitted on " + subClusterId);
   459	        long stopTime = clock.getTime();
   460	        routerMetrics.succeededAppsSubmitted(stopTime - startTime);
   461	        return response;
   462	      } else {
   463	        // Empty response from the ResourceManager.
   464	        // Blacklist this subcluster for this request.
   465	        blacklist.add(subClusterId);
   466	      }
   467	    }
   468	
   469	    routerMetrics.incrAppsFailedSubmitted();
   470	    String errMsg = "Application "
   471	        + request.getApplicationSubmissionContext().getApplicationName()
   472	        + " with appId " + applicationId + " failed to be submitted.";
   473	    LOG.error(errMsg);
   474	    throw new YarnException(errMsg);
   475	  }

   684	  <R> Map<SubClusterId, R> invokeConcurrent(ArrayList<SubClusterId> clusterIds,
   685	      ClientMethod request, Class<R> clazz) throws YarnException, IOException {
   686	    List<Callable<Object>> callables = new ArrayList<>();
   687	    List<Future<Object>> futures = new ArrayList<>();
   688	    Map<SubClusterId, IOException> exceptions = new TreeMap<>();
   689	    for (SubClusterId subClusterId : clusterIds) {
   690	      callables.add(new Callable<Object>() {
   691	        @Override
   692	        public Object call() throws Exception {
   693	          ApplicationClientProtocol protocol =
   694	              getClientRMProxyForSubCluster(subClusterId);
   695	          Method method = ApplicationClientProtocol.class
   696	              .getMethod(request.getMethodName(), request.getTypes());
   697	          return method.invoke(protocol, request.getParams());
   698	        }
   699	      });
   700	    }
   701	    Map<SubClusterId, R> results = new TreeMap<>();
   702	    try {
   703	      futures.addAll(executorService.invokeAll(callables));
   704	      for (int i = 0; i < futures.size(); i++) {
   705	        SubClusterId subClusterId = clusterIds.get(i);
   706	        try {
   707	          Future<Object> future = futures.get(i);
   708	          Object result = future.get();
   709	          results.put(subClusterId, clazz.cast(result));
   710	        } catch (ExecutionException ex) {
   711	          Throwable cause = ex.getCause();
   712	          LOG.debug("Cannot execute {} on {}: {}", request.getMethodName(),
   713	              subClusterId.getId(), cause.getMessage());
   714	          IOException ioe;
   715	          if (cause instanceof IOException) {
   716	            ioe = (IOException) cause;
   717	          } else if (cause instanceof YarnException) {
   718	            throw (YarnException) cause;
   719	          } else {
   720	            ioe = new IOException(
   721	                "Unhandled exception while calling " + request.getMethodName()
   722	                    + ": " + cause.getMessage(), cause);
   723	          }
   724	          // Store the exceptions
   725	          exceptions.put(subClusterId, ioe);
   726	        }
   727	      }
   728	      if (results.isEmpty() && !clusterIds.isEmpty()) {
   729	        SubClusterId subClusterId = clusterIds.get(0);
   730	        IOException ioe = exceptions.get(subClusterId);
   731	        if (ioe != null) {
   732	          throw ioe;
   733	        }
   734	      }
   735	    } catch (InterruptedException e) {
   736	      throw new YarnException(e);
   737	    }
   738	    return results;
   739	  }

   744	    return invokeConcurrent(clusterIdList, request, clazz);

   820	  /**
   821	   * The YARN Router will forward to the respective YARN RM in which the AM is
   822	   * running.
   823	   *
   824	   * Possible failure:
   825	   *
   826	   * Client: identical behavior as {@code ClientRMService}.
   827	   *
   828	   * Router: the Client will timeout and resubmit the request.
   829	   *
   830	   * ResourceManager: the Router will timeout and the call will fail.
   831	   *
   832	   * State Store: the Router will timeout and it will retry depending on the
   833	   * FederationFacade settings - if the failure happened before the select
   834	   * operation.
   835	   */
   836	  @Override
   837	  public GetApplicationAttemptReportResponse getApplicationAttemptReport(
   838	      GetApplicationAttemptReportRequest request)
   839	      throws YarnException, IOException {
   840	
   841	    long startTime = clock.getTime();
   842	
   843	    if (request == null || request.getApplicationAttemptId() == null
   844	            || request.getApplicationAttemptId().getApplicationId() == null) {
   845	      routerMetrics.incrAppAttemptsFailedRetrieved();
   846	      RouterServerUtil.logAndThrowException(
   847	              "Missing getApplicationAttemptReport " +
   848	                      "request or applicationId " +
   849	                      "or applicationAttemptId information.",
   850	              null);
   851	    }
   852	
   853	    SubClusterId subClusterId = null;
   854	
   855	    try {
   856	      subClusterId = federationFacade
   857	              .getApplicationHomeSubCluster(
   858	                      request.getApplicationAttemptId().getApplicationId());
   859	    } catch (YarnException e) {
   860	      routerMetrics.incrAppAttemptsFailedRetrieved();
   861	      RouterServerUtil
   862	              .logAndThrowException("ApplicationAttempt " +
   863	                      request.getApplicationAttemptId() +
   864	                      "belongs to Application " +
   865	                      request.getApplicationAttemptId().getApplicationId() +
   866	                      " does not exist in FederationStateStore", e);
   867	    }
   868	
   869	    ApplicationClientProtocol clientRMProxy =
   870	            getClientRMProxyForSubCluster(subClusterId);
   871	
   872	    GetApplicationAttemptReportResponse response = null;
   873	    try {
   874	      response = clientRMProxy.getApplicationAttemptReport(request);
   875	    } catch (Exception e) {
   876	      routerMetrics.incrAppAttemptsFailedRetrieved();
   877	      LOG.error("Unable to get the applicationAttempt report for "
   878	              + request.getApplicationAttemptId() + "to SubCluster "
   879	              + subClusterId.getId(), e);
   880	      throw e;
   881	    }
   882	
   883	    if (response == null) {
   884	      LOG.error("No response when attempting to retrieve the report of "
   885	              + "the applicationAttempt "
   886	              + request.getApplicationAttemptId() + " to SubCluster "
   887	              + subClusterId.getId());
   888	    }
   889	
   890	    long stopTime = clock.getTime();
   891	    routerMetrics.succeededAppAttemptsRetrieved(stopTime - startTime);
   892	    return response;
   893	  }
