    71	/**
    72	 * Thread management for the active audit.
    73	 * This should be created by whatever wants to have active
    74	 * audit span tracking.
    75	 *
    76	 * It creates and starts the actual
    77	 * {@link OperationAuditor} for auditing.
    78	 * It then stores the thread-local span and returns a wrapping
    79	 * span.
    80	 *
    81	 * When the wrapper is closed/deactivated it
    82	 * will deactivate the wrapped span and then
    83	 * switch the active span to the unbounded span.
    84	 *
    85	 * The inner class {@link AWSAuditEventCallbacks} is returned
    86	 * as a request handler in {@link #createRequestHandlers()};
    87	 * this forwards all requests to the outer {@code ActiveAuditManagerS3A},
    88	 * which then locates the active span and forwards the request.
    89	 * If any such invocation raises an {@link AuditFailureException}
    90	 * then the IOStatistics counter for {@code AUDIT_FAILURE}
    91	 * is incremented.
    92	 *
    93	 * Uses a {@link WeakReferenceThreadMap} to store the active span
    94	 * of each thread, rather than a {@code ThreadLocal} reference.
    95	 * This means that the spans -and hence the audit manager and
    96	 * filesystem instances they refer to- are not retained by
    97	 * long-lived threads once all other references to a span have been
    98	 * released (HADOOP-18091).
    99	 * The cost of this is that if garbage collection takes place before
   100	 * a thread switches to a new span, the reference is lost; the
   101	 * unbonded span is then handed back and the auditor notified through
   102	 * {@code OperationAuditor.noteSpanReferenceLost()}.
   103	 */
   104	@InterfaceAudience.Private
   105	public final class ActiveAuditManagerS3A
   106	    extends CompositeService
   107	    implements AuditManagerS3A {

   172	            .withConfiguration(conf)

   283	        new WrappingAuditSpan(span, span.isValidSpan()));

   286	  /**
   287	   * Switch to a given span. If it is null, use the
   288	   * unbounded span.
   289	   * @param span to switch to; may be null
   290	   * @return the span switched to
   291	   */
   292	  private WrappingAuditSpan switchToActiveSpan(WrappingAuditSpan span) {
   293	    if (span != null && span.isValidSpan()) {
   294	      activeSpanMap.setForCurrentThread(span);
   295	    } else {
   296	      activeSpanMap.removeForCurrentThread();
   297	    }
   298	    return activeSpan();
   299	  }

   331	  @Override
   332	  public AuditSpanS3A createSpan(final String operation,
   333	      @Nullable final String path1,
   334	      @Nullable final String path2) throws IOException {
   335	    // must be started
   336	    Preconditions.checkState(isInState(STATE.STARTED),
   337	        "Audit Manager %s is in wrong state: %s",
   338	        this, getServiceState());
   339	    ioStatisticsStore.incrementCounter(
   340	        Statistic.AUDIT_SPAN_CREATION.getSymbol());
   341	    return setActiveThreadSpan(auditor.createSpan(
   342	        operation, path1, path2));

   345	  /**
   346	   * Return a request handler for the AWS SDK which
   347	   * relays to this class.
   348	   * @return a request handler.
   349	   */
   350	  @Override
   351	  public List<RequestHandler2> createRequestHandlers()
   352	      throws IOException {
   353	
   354	    // wire up the AWS SDK To call back into this class when
   355	    // preparing to make S3 calls.
   356	    List<RequestHandler2> requestHandlers = new ArrayList<>();
   357	    requestHandlers.add(new SdkRequestHandler());
   358	    // now look for any more handlers
   359	    final Class<?>[] handlers = getConfig().getClasses(AUDIT_REQUEST_HANDLERS);
   360	    if (handlers != null) {
   361	      for (Class<?> handler : handlers) {
   362	        try {
   363	          Constructor<?> ctor = handler.getConstructor();
   364	          requestHandlers.add((RequestHandler2)ctor.newInstance());
   365	        } catch (ExceptionInInitializerError e) {
   366	          throw FutureIO.unwrapInnerException(e);
   367	        } catch (Exception e) {
   368	          throw new IOException(e);
   369	        }
   370	      }
   371	    }
   372	    return requestHandlers;
   373	  }

   395	  /**
   396	   * Attach a reference to the active thread span, then
   397	   * invoke the same callback on that active thread.
   398	   */
   399	  @Override
   400	  public <T extends AmazonWebServiceRequest> T requestCreated(
   401	      final T request) {
   402	    AuditSpanS3A span = getActiveAuditSpan();
   403	    if (LOG.isTraceEnabled()) {
   404	      LOG.trace("Created Request {} in span {}",
   405	          analyzer.analyze(request), span);
   406	    }
   407	    attachSpanToRequest(request, span);
   408	    try {
   409	      return span.requestCreated(request);
   410	    } catch (AuditFailureException e) {
   411	      ioStatisticsStore.incrementCounter(AUDIT_FAILURE.getSymbol());
   412	      throw e;
   413	    }
   414	  }
   415	
   416	  /**
   417	   * Forward to the active span.
   418	   * All invocations increment the statistics counter for
   419	   * {@link Statistic#AUDIT_REQUEST_EXECUTION};
   420	   * failures will also increment
   421	   * {@link Statistic#AUDIT_FAILURE};
   422	   * {@inheritDoc}
   423	   */
   424	  @Override
   425	  public <T extends AmazonWebServiceRequest> T beforeExecution(
   426	      final T request) {
   427	    ioStatisticsStore.incrementCounter(AUDIT_REQUEST_EXECUTION.getSymbol());
   428	
   429	    // identify the span and invoke the callback
   430	    try {
   431	      return extractAndActivateSpanFromRequest(request)
   432	          .beforeExecution(request);
   433	    } catch (AuditFailureException e) {
   434	      ioStatisticsStore.incrementCounter(AUDIT_FAILURE.getSymbol());
   435	      throw e;
   436	    }
   437	  }

   443	   */
   444	  @Override
   445	  public void afterResponse(final Request<?> request,
   446	      final Response<?> response)
   447	      throws AuditFailureException, SdkBaseException {
   448	    try {
   449	      extractAndActivateSpanFromRequest(request)
   450	          .afterResponse(request, response);
   451	    } catch (AuditFailureException e) {
   452	      ioStatisticsStore.incrementCounter(AUDIT_FAILURE.getSymbol());
   453	      throw e;
   454	    }
   455	  }

   457	  /**
   458	   * Get the active span from the handler context,
   459	   * falling back to the active thread span if there
   460	   * is nothing in the context.
   461	   * Provided the span is a wrapped span, the
   462	   * @param request request
   463	   * @param <T> type of request.
   464	   * @return the callbacks
   465	   */
   466	  private <T extends HandlerContextAware> AWSAuditEventCallbacks
   467	      extractAndActivateSpanFromRequest(final T request) {
   468	    AWSAuditEventCallbacks span;
   469	    span = retrieveAttachedSpan(request);
   470	    if (span == null) {
   471	      // no span is attached. Not unusual for the copy operations,
   472	      // or for calls to GetBucketLocation made by the AWS client
   473	      LOG.debug("No audit span attached to request {}",
   474	          request);
   475	      // fall back to the active thread span.
   476	      // this will be the unbonded span if the thread is unbonded.
   477	      span = getActiveAuditSpan();
   478	    } else {
   479	      if (span instanceof WrappingAuditSpan) {
   480	        switchToActiveSpan((WrappingAuditSpan) span);
   481	      } else {
   482	        // warn/log and continue without switching.
   483	        WARN_OF_SPAN_TYPE.warn(NOT_A_WRAPPED_SPAN + ": {}", span);
   484	        LOG.debug(NOT_A_WRAPPED_SPAN + ": {}", span);
   485	      }
   486	    }
   487	    return span;
   488	  }

   626	  /**
   627	   * Wraps the plugged in spans with management of the active thread
   628	   * span, including switching to the unbounded span when a valid
   629	   * span is deactivated.
   630	   * Package-private for testing.
   631	   */
   632	  private final class WrappingAuditSpan extends AbstractAuditSpanImpl {
   633	
   634	    /**
   635	     * Inner span.
   636	     */
   637	    private final AuditSpanS3A span;
   638	
   639	    /**
   640	     * Is this span considered valid?
   641	     */
   642	    private final boolean isValid;
   643	
   644	    /**
   645	     * Create, wrapped.
   646	     * The spanID, name, timestamp etc copied from the span being wrapped.
   647	     * Why not the isValid state? We want to set our unbonded span without
   648	     * relying on the auditor doing the right thing.
   649	     * @param span inner span.
   650	     * @param isValid is the span valid
   651	     */
   652	    private WrappingAuditSpan(
   653	        final AuditSpanS3A span, final boolean isValid) {
   654	      super(span.getSpanId(), span.getTimestamp(), span.getOperationName());
   655	      this.span = requireNonNull(span);
   656	      this.isValid = isValid;
   657	    }
   658	
   659	    /**
   660	     * Is the span active?
   661	     * @return true if this span is the active one for the current thread.
   662	     */
   663	    private boolean isActive() {
   664	      return this == getActiveAuditSpan();
   665	    }
   666	
   667	    /**
   668	     * Makes this the thread's active span and activate.
   669	     * If the span was already active: no-op.
   670	     */
   671	    @Override
   672	    public AuditSpanS3A activate() {
   673	      if (!isActive()) {
   674	        switchToActiveSpan(this);
   675	        span.activate();
   676	      }
   677	      return this;
   678	    }
   679	
   680	    /**
   681	     * Switch to the unbounded span and then deactivate this span.
   682	     * No-op for invalid spans,
   683	     * so as to prevent the unbounded span from being closed
   684	     * and everything getting very confused.
   685	     */
   686	    @Override
   687	    public void deactivate() {
   688	      // no-op for invalid spans,
   689	      // so as to prevent the unbounded span from being closed
   690	      // and everything getting very confused.
   691	      if (!isValid || !isActive()) {
   692	        return;
   693	      }
   694	      // deactivate the span
   695	      span.deactivate();
   696	      // and go to the unbounded one.
   697	      switchToActiveSpan(getUnbondedSpan());
   698	    }
   699	
   700	    /**
   701	     * Forward to the wrapped span.
   702	     * {@inheritDoc}
   703	     */
   704	    @Override
   705	    public <T extends AmazonWebServiceRequest> T requestCreated(
   706	        final T request) {
   707	      return span.requestCreated(request);
   708	    }
   709	
   710	    /**
   711	     * This span is valid if the span isn't closed and the inner
   712	     * span is valid.
   713	     * @return true if the span is considered valid.
   714	     */
   715	    @Override
   716	    public boolean isValidSpan() {
   717	      return isValid && span.isValidSpan();
   718	    }
   719	
   720	    /**
   721	     * Forward to the inner span.
   722	     * {@inheritDoc}
   723	     */
   724	    @Override
   725	    public void set(final String key, final String value) {
   726	      span.set(key, value);
   727	    }
   728	
   729	    /**
   730	     * Forward to the inner span.
   731	     * @param request request
   732	     * @param <T> type of request
   733	     * @return an updated request.
   734	     */
   735	    @Override
   736	    public <T extends AmazonWebServiceRequest> T beforeExecution(
   737	        final T request) {
   738	      return span.beforeExecution(request);
   739	    }
   740	
   741	    /**
   742	     * Forward to the inner span.
   743	     * @param request request
   744	     * @param response response.
   745	     */
   746	    @Override
   747	    public void afterResponse(final Request<?> request,
   748	        final Response<?> response) {
   749	      span.afterResponse(request, response);
   750	    }
   751	
   752	    /**
   753	     * Forward to the inner span.
   754	     * @param request request
   755	     * @param response response.
   756	     * @param exception exception raised.
   757	     */
   758	    @Override
   759	    public void afterError(final Request<?> request,
   760	        final Response<?> response,
   761	        final Exception exception) {
   762	      span.afterError(request, response, exception);
   763	    }
   764	
   765	    /**
   766	     * Forward to the inner span.
   767	     * @param request request
   768	     * @return request to marshall
   769	     */
   770	    @Override
   771	    public AmazonWebServiceRequest beforeMarshalling(
   772	        final AmazonWebServiceRequest request) {
   773	      return span.beforeMarshalling(request);
   774	    }
   775	
   776	    /**
   777	     * Forward to the inner span.
   778	     * @param request request
   779	     */
   780	    @Override
   781	    public void beforeRequest(final Request<?> request) {
   782	      span.beforeRequest(request);
   783	    }
   784	
   785	    /**
   786	     * Forward to the inner span.
   787	     * @param context full context, including the request.
   788	     */
   789	    @Override
   790	    public void beforeAttempt(
   791	        final HandlerBeforeAttemptContext context) {
   792	      span.beforeAttempt(context);
   793	    }
   794	
   795	    /**
   796	     * Forward to the inner span.
   797	     *
   798	     * @param context full context, including the request.
   799	     */
   800	    @Override
   801	    public void afterAttempt(
   802	        final HandlerAfterAttemptContext context) {
   803	      span.afterAttempt(context);
   804	    }
   805	
   806	    @Override
   807	    public String toString() {
   808	      final StringBuilder sb = new StringBuilder(
   809	          "WrappingAuditSpan{");
   810	      sb.append("span=").append(span);
   811	      sb.append(", valid=").append(isValidSpan());
   812	      sb.append('}');
   813	      return sb.toString();
   814	    }
   815	  }
   816	
   817	}
