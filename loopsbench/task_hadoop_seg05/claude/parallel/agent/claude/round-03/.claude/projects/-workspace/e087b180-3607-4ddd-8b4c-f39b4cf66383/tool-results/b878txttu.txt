    71	/**
    72	 * Channel to a remote JournalNode using Hadoop IPC.
    73	 * All of the calls are run on a separate thread, and return
    74	 * {@link ListenableFuture} instances to wait for their result.
    75	 * This allows calls to be bound together using the {@link QuorumCall}
    76	 * class.
    77	 */
    78	@InterfaceAudience.Private
    79	public class IPCLoggerChannel implements AsyncLogger {

   144	   * the writer sets this flag to true to avoid sending useless RPCs.
   145	   */
   146	  private boolean outOfSync = false;

   181	    this.addr = addr;

   185	    
   186	    singleThreadExecutor = MoreExecutors.listeningDecorator(
   187	        createSingleThreadExecutor());

   199	  @Override
   200	  public synchronized void setCommittedTxId(long txid) {
   201	    Preconditions.checkArgument(txid >= committedTxId,
   202	        "Trying to move committed txid backwards in client " +
   203	         "old: %s new: %s", committedTxId, txid);
   204	    this.committedTxId = txid;
   205	    this.lastCommitNanos = System.nanoTime();
   206	  }
   207	  
   208	  @Override
   209	  public void close() {
   210	    // No more tasks may be submitted after this point.
   211	    singleThreadExecutor.shutdown();
   212	    parallelExecutor.shutdown();
   213	    if (proxy != null) {
   214	      // TODO: this can hang for quite some time if the client
   215	      // is currently in the middle of a call to a downed JN.
   216	      // We should instead do this asynchronously, and just stop
   217	      // making any more calls after this point (eg clear the queue)
   218	      RPC.stopProxy(proxy);
   219	    }
   220	  }
   221	  
   222	  protected QJournalProtocol getProxy() throws IOException {
   223	    if (proxy != null) return proxy;
   224	    proxy = createProxy();
   225	    return proxy;
   226	  }
   227	  
   228	  protected QJournalProtocol createProxy() throws IOException {
   229	    final Configuration confCopy = new Configuration(conf);
   230	    
   231	    // Need to set NODELAY or else batches larger than MTU can trigger 
   232	    // 40ms nagling delays.
   233	    confCopy.setBoolean(
   234	        CommonConfigurationKeysPublic.IPC_CLIENT_TCPNODELAY_KEY,
   235	        true);
   236	    
   237	    RPC.setProtocolEngine(confCopy,
   238	        QJournalProtocolPB.class, ProtobufRpcEngine2.class);
   239	    return SecurityUtil.doAsLoginUser(
   240	        new PrivilegedExceptionAction<QJournalProtocol>() {
   241	          @Override
   242	          public QJournalProtocol run() throws IOException {
   243	            RPC.setProtocolEngine(confCopy,
   244	                QJournalProtocolPB.class, ProtobufRpcEngine2.class);
   245	            QJournalProtocolPB pbproxy = RPC.getProxy(
   246	                QJournalProtocolPB.class,
   247	                RPC.getProtocolVersion(QJournalProtocolPB.class),
   248	                addr, confCopy);
   249	            return new QJournalProtocolTranslatorPB(pbproxy);
   250	          }
   251	        });
   252	  }

   381	  @Override
   382	  public ListenableFuture<Void> sendEdits(
   383	      final long segmentTxId, final long firstTxnId,
   384	      final int numTxns, final byte[] data) {
   385	    try {
   386	      reserveQueueSpace(data.length);
   387	    } catch (LoggerTooFarBehindException e) {
   388	      return Futures.immediateFailedFuture(e);
   389	    }
   390	    
   391	    // When this batch is acked, we use its submission time in order
   392	    // to calculate how far we are lagging.
   393	    final long submitNanos = System.nanoTime();
   394	    
   395	    ListenableFuture<Void> ret = null;
   396	    try {
   397	      ret = singleThreadExecutor.submit(new Callable<Void>() {
   398	        @Override
   399	        public Void call() throws IOException {
   400	          throwIfOutOfSync();
   401	
   402	          long rpcSendTimeNanos = System.nanoTime();
   403	          try {
   404	            getProxy().journal(createReqInfo(),
   405	                segmentTxId, firstTxnId, numTxns, data);
   406	          } catch (IOException e) {
   407	            QuorumJournalManager.LOG.warn(
   408	                "Remote journal " + IPCLoggerChannel.this + " failed to " +
   409	                "write txns " + firstTxnId + "-" + (firstTxnId + numTxns - 1) +
   410	                ". Will try to write to this JN again after the next " +
   411	                "log roll.", e); 
   412	            synchronized (IPCLoggerChannel.this) {
   413	              outOfSync = true;
   414	            }
   415	            throw e;
   416	          } finally {
   417	            long now = System.nanoTime();
   418	            long rpcTime = TimeUnit.MICROSECONDS.convert(
   419	                now - rpcSendTimeNanos, TimeUnit.NANOSECONDS);
   420	            long endToEndTime = TimeUnit.MICROSECONDS.convert(
   421	                now - submitNanos, TimeUnit.NANOSECONDS);
   422	            metrics.addWriteEndToEndLatency(endToEndTime);
   423	            metrics.addWriteRpcLatency(rpcTime);
   424	            if (rpcTime / 1000 > WARN_JOURNAL_MILLIS_THRESHOLD) {
   425	              QuorumJournalManager.LOG.warn(
   426	                  "Took " + (rpcTime / 1000) + "ms to send a batch of " +
   427	                  numTxns + " edits (" + data.length + " bytes) to " +
   428	                  "remote journal " + IPCLoggerChannel.this);
   429	            }
   430	          }
   431	          synchronized (IPCLoggerChannel.this) {
   432	            highestAckedTxId = firstTxnId + numTxns - 1;
   433	            lastAckNanos = submitNanos;
   434	          }
   435	          return null;
   436	        }
   437	      });
   438	    } finally {
   439	      if (ret == null) {
   440	        // it didn't successfully get submitted,
   441	        // so adjust the queue size back down.
   442	        unreserveQueueSpace(data.length);
   443	      } else {
   444	        // It was submitted to the queue, so adjust the length
   445	        // once the call completes, regardless of whether it
   446	        // succeeds or fails.
   447	        Futures.addCallback(ret, new FutureCallback<Void>() {
   448	          @Override
   449	          public void onFailure(Throwable t) {
   450	            unreserveQueueSpace(data.length);
   451	          }
   452	
   453	          @Override
   454	          public void onSuccess(Void t) {
   455	            unreserveQueueSpace(data.length);
   456	          }
   457	        }, MoreExecutors.directExecutor());
   458	      }
   459	    }
   460	    return ret;
   461	  }

   463	  private void throwIfOutOfSync()
   464	      throws JournalOutOfSyncException, IOException {
   465	    if (isOutOfSync()) {
   466	      // Even if we're out of sync, it's useful to send an RPC
   467	      // to the remote node in order to update its lag metrics, etc.
   468	      heartbeatIfNecessary();
   469	      throw new JournalOutOfSyncException(
   470	          "Journal disabled until next roll");
   471	    }
   472	  }
   473	
   474	  /**
   475	   * When we've entered an out-of-sync state, it's still useful to periodically
   476	   * send an empty RPC to the server, such that it has the up to date
   477	   * committedTxId. This acts as a sanity check during recovery, and also allows
   478	   * that node's metrics to be up-to-date about its lag.
   479	   * 
   480	   * In the future, this method may also be used in order to check that the
   481	   * current node is still the current writer, even if no edits are being
   482	   * written.
   483	   */
   484	  private void heartbeatIfNecessary() throws IOException {
   485	    if (lastHeartbeatStopwatch.now(TimeUnit.MILLISECONDS)
   486	        > HEARTBEAT_INTERVAL_MILLIS || !lastHeartbeatStopwatch.isRunning()) {
   487	      try {
   488	        getProxy().heartbeat(createReqInfo());
   489	      } finally {
   490	        // Don't send heartbeats more often than the configured interval,
   491	        // even if they fail.
   492	        lastHeartbeatStopwatch.reset().start();
   493	      }
   494	    }
   495	  }

   497	  private synchronized void reserveQueueSpace(int size)
   498	      throws LoggerTooFarBehindException {
   499	    Preconditions.checkArgument(size >= 0);
   500	    if (queuedEditsSizeBytes + size > queueSizeLimitBytes &&
   501	        queuedEditsSizeBytes > 0) {
   502	      QuorumJournalManager.LOG.warn("Pending edits to " + IPCLoggerChannel.this
   503	          + " is going to exceed limit size: " + queueSizeLimitBytes
   504	          + ", current queued edits size: " + queuedEditsSizeBytes
   505	          + ", will silently drop " + size + " bytes of edits!");
   506	      throw new LoggerTooFarBehindException();
   507	    }
   508	    queuedEditsSizeBytes += size;
   509	  }
   510	  
   511	  private synchronized void unreserveQueueSpace(int size) {
   512	    Preconditions.checkArgument(size >= 0);
   513	    queuedEditsSizeBytes -= size;
   514	  }

   528	  @Override
   529	  public ListenableFuture<Void> startLogSegment(final long txid,
   530	      final int layoutVersion) {
   531	    return singleThreadExecutor.submit(new Callable<Void>() {
   532	      @Override
   533	      public Void call() throws IOException {
   534	        getProxy().startLogSegment(createReqInfo(), txid, layoutVersion);
   535	        synchronized (IPCLoggerChannel.this) {
   536	          if (outOfSync) {
   537	            outOfSync = false;
   538	            QuorumJournalManager.LOG.info(
   539	                "Restarting previously-stopped writes to " +
   540	                IPCLoggerChannel.this + " in segment starting at txid " +
   541	                txid);
   542	          }
   543	        }
   544	        return null;
   545	      }
   546	    });
   547	  }

   574	  @Override
   575	  public ListenableFuture<GetJournaledEditsResponseProto> getJournaledEdits(
   576	      long fromTxnId, int maxTransactions) {
   577	    return parallelExecutor.submit(
   578	        new Callable<GetJournaledEditsResponseProto>() {
   579	          @Override
   580	          public GetJournaledEditsResponseProto call() throws IOException {

   716	  @Override
   717	  public synchronized void appendReport(StringBuilder sb) {
   718	    sb.append("Written txid ").append(highestAckedTxId);
   719	    long behind = getLagTxns();
   720	    if (behind > 0) {
   721	      if (lastAckNanos != 0) {
   722	        long lagMillis = getLagTimeMillis();
   723	        sb.append(" (" + behind + " txns/" + lagMillis + "ms behind)");
   724	      } else {
   725	        sb.append(" (never written");
   726	      }
   727	    }
   728	    if (outOfSync) {
   729	      sb.append(" (will try to re-sync on next segment)");
   730	    }
   731	  }
   732	  
   733	  public synchronized long getLagTxns() {
   734	    return Math.max(committedTxId - highestAckedTxId, 0);
   735	  }
   736	  
   737	  public synchronized long getLagTimeMillis() {
   738	    return TimeUnit.MILLISECONDS.convert(
   739	        Math.max(lastCommitNanos - lastAckNanos, 0),
   740	        TimeUnit.NANOSECONDS);
   741	  }

   743	  private void constructHttpServerURI(GetEditLogManifestResponseProto ret) {
   744	    if (ret.hasFromURL()) {
   745	      URI uri = URI.create(ret.getFromURL());
   746	      httpServerURL = getHttpServerURI(uri.getScheme(), uri.getPort());
   747	    } else {
   748	      httpServerURL = getHttpServerURI("http", ret.getHttpPort());;
   749	    }
   750	  }

   752	  private void constructHttpServerURI(GetJournalStateResponseProto ret) {
   753	    if (ret.hasFromURL()) {
   754	      URI uri = URI.create(ret.getFromURL());
   755	      httpServerURL = getHttpServerURI(uri.getScheme(), uri.getPort());
   756	    } else {
   757	      httpServerURL = getHttpServerURI("http", ret.getHttpPort());;
   758	    }
   759	  }
   760	
   761	  /**
   762	   * Construct the http server based on the response.
   763	   *
   764	   * The fromURL field in the response specifies the endpoint of the http
   765	   * server. However, the address might not be accurate since the server can
   766	   * bind to multiple interfaces. Here the client plugs in the address specified
   767	   * in the configuration and generates the URI.
   768	   */
   769	  private URL getHttpServerURI(String scheme, int port) {
   770	    try {
   771	      return new URL(scheme, addr.getHostName(), port, "");
   772	    } catch (MalformedURLException e) {
   773	      // Unreachable
   774	      throw new RuntimeException(e);
   775	    }
   776	  }
   777	
   778	  private boolean hasHttpServerEndPoint() {
   779	   return httpServerURL != null;
   780	  }
