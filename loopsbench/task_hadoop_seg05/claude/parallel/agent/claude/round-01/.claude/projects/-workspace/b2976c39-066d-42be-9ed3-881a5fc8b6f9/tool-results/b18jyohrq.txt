   131	  public RBFMetrics(Router router) throws IOException {
   132	    this.router = router;
   133	
   134	    try {
   135	      StandardMBean bean = new StandardMBean(this, RouterMBean.class);
   136	      this.routerBeanName = MBeans.register("Router", "Router", bean);
   137	      LOG.info("Registered Router MBean: {}", this.routerBeanName);
   138	    } catch (NotCompliantMBeanException e) {
   139	      throw new RuntimeException("Bad Router MBean setup", e);
   140	    }
   141	
   142	    try {
   143	      StandardMBean bean = new StandardMBean(this, FederationMBean.class);
   144	      this.federationBeanName = MBeans.register("Router", "FederationState",
   145	          bean);
   146	      LOG.info("Registered FederationState MBean: {}", this.federationBeanName);
   147	    } catch (NotCompliantMBeanException e) {
   148	      throw new RuntimeException("Bad FederationState MBean setup", e);
   149	    }
   150	
   151	    // Resolve namenode for each nameservice
   152	    this.namenodeResolver = this.router.getNamenodeResolver();
   153	
   154	    // State store interfaces
   155	    this.stateStore = this.router.getStateStore();
   156	    if (this.stateStore == null) {
   157	      LOG.error("State store not available");
   158	    } else {
   159	      this.membershipStore = stateStore.getRegisteredRecordStore(
   160	          MembershipStore.class);
   161	      this.mountTableStore = stateStore.getRegisteredRecordStore(
   162	          MountTableStore.class);
   163	      this.routerStore = stateStore.getRegisteredRecordStore(
   164	          RouterStore.class);
   165	    }
   166	
   167	    // Initialize the cache for the DN reports
   168	    Configuration conf = router.getConfig();
   169	    this.timeOut = conf.getTimeDuration(RBFConfigKeys.DN_REPORT_TIME_OUT,
   170	        RBFConfigKeys.DN_REPORT_TIME_OUT_MS_DEFAULT, TimeUnit.MILLISECONDS);
   171	    this.topTokenRealOwners = conf.getInt(
   172	        RBFConfigKeys.DFS_ROUTER_METRICS_TOP_NUM_TOKEN_OWNERS_KEY,
   173	        RBFConfigKeys.DFS_ROUTER_METRICS_TOP_NUM_TOKEN_OWNERS_KEY_DEFAULT);
   174	  }

   188	  @Override
   189	  public String getNamenodes() {
   190	    final Map<String, Map<String, Object>> info = new LinkedHashMap<>();
   191	    if (membershipStore == null) {
   192	      return "{}";
   193	    }
   194	
   195	    try {
   196	      // Get the values from the store
   197	      GetNamenodeRegistrationsRequest request =
   198	          GetNamenodeRegistrationsRequest.newInstance();
   199	      GetNamenodeRegistrationsResponse response =
   200	          membershipStore.getNamenodeRegistrations(request);
   201	
   202	      // Order the namenodes
   203	      final List<MembershipState> namenodes = response.getNamenodeMemberships();
   204	      if (namenodes == null || namenodes.size() == 0) {
   205	        return JSON.toString(info);
   206	      }
   207	      List<MembershipState> namenodesOrder = new ArrayList<>(namenodes);
   208	      Collections.sort(namenodesOrder, MembershipState.NAME_COMPARATOR);
   209	
   210	      // Dump namenodes information into JSON
   211	      for (MembershipState namenode : namenodesOrder) {
   212	        Map<String, Object> innerInfo = new HashMap<>();
   213	        Map<String, Object> map = getJson(namenode);
   214	        innerInfo.putAll(map);
   215	        long dateModified = namenode.getDateModified();
   216	        long lastHeartbeat = getSecondsSince(dateModified);
   217	        innerInfo.put("lastHeartbeat", lastHeartbeat);
   218	        MembershipStats stats = namenode.getStats();
   219	        long used = stats.getTotalSpace() - stats.getAvailableSpace();
   220	        innerInfo.put("used", used);
   221	        info.put(namenode.getNamenodeKey(),
   222	            Collections.unmodifiableMap(innerInfo));
   223	      }
   224	    } catch (IOException e) {
   225	      LOG.error("Enable to fetch json representation of namenodes {}",
   226	          e.getMessage());
   227	      return "{}";
   228	    }
   229	    return JSON.toString(info);
   230	  }

   261	  @Override
   262	  public String getMountTable() {
   263	    final List<Map<String, Object>> info = new LinkedList<>();
   264	    if (mountTableStore == null) {
   265	      return "[]";
   266	    }
   267	
   268	    try {
   269	      // Get all the mount points in order
   270	      GetMountTableEntriesRequest request =
   271	          GetMountTableEntriesRequest.newInstance("/");
   272	      GetMountTableEntriesResponse response =
   273	          mountTableStore.getMountTableEntries(request);
   274	      final List<MountTable> mounts = response.getEntries();
   275	      List<MountTable> orderedMounts = new ArrayList<>(mounts);
   276	      Collections.sort(orderedMounts, MountTable.SOURCE_COMPARATOR);
   277	
   278	      // Dump mount table entries information into JSON
   279	      for (MountTable entry : orderedMounts) {
   280	        // Sumarize destinations
   281	        Set<String> nameservices = new LinkedHashSet<>();
   282	        Set<String> paths = new LinkedHashSet<>();
   283	        for (RemoteLocation location : entry.getDestinations()) {
   284	          nameservices.add(location.getNameserviceId());
   285	          paths.add(location.getDest());
   286	        }
   287	
   288	        Map<String, Object> map = getJson(entry);
   289	        // We add some values with a cleaner format
   290	        map.put("dateCreated", getDateString(entry.getDateCreated()));
   291	        map.put("dateModified", getDateString(entry.getDateModified()));
   292	
   293	        Map<String, Object> innerInfo = new HashMap<>();
   294	        innerInfo.putAll(map);
   295	        innerInfo.put("nameserviceId", StringUtils.join(",", nameservices));
   296	        innerInfo.put("path", StringUtils.join(",", paths));
   297	        if (nameservices.size() > 1) {
   298	          innerInfo.put("order", entry.getDestOrder().toString());
   299	        } else {
   300	          innerInfo.put("order", "");
   301	        }
   302	        innerInfo.put("readonly", entry.isReadOnly());
   303	        innerInfo.put("faulttolerant", entry.isFaultTolerant());
   304	        info.add(Collections.unmodifiableMap(innerInfo));
   305	      }
   306	    } catch (IOException e) {
   307	      LOG.error(
   308	          "Cannot generate JSON of mount table from store: {}", e.getMessage());
   309	      return "[]";
   310	    }
   311	    return JSON.toString(info);
   312	  }

   314	  @Override
   315	  public String getRouters() {
   316	    final Map<String, Map<String, Object>> info = new LinkedHashMap<>();
   317	    if (routerStore == null) {
   318	      return "{}";
   319	    }
   320	    try {
   321	      // Get all the routers in order
   322	      GetRouterRegistrationsRequest request =
   323	          GetRouterRegistrationsRequest.newInstance();
   324	      GetRouterRegistrationsResponse response =
   325	          routerStore.getRouterRegistrations(request);
   326	      final List<RouterState> routers = response.getRouters();
   327	      List<RouterState> routersOrder = new ArrayList<>(routers);
   328	      Collections.sort(routersOrder);
   329	
   330	      // Dump router information into JSON
   331	      for (RouterState record : routersOrder) {
   332	        Map<String, Object> innerInfo = new HashMap<>();
   333	        Map<String, Object> map = getJson(record);
   334	        innerInfo.putAll(map);
   335	        long dateModified = record.getDateModified();
   336	        long lastHeartbeat = getSecondsSince(dateModified);
   337	        innerInfo.put("lastHeartbeat", lastHeartbeat);
   338	
   339	        StateStoreVersion stateStoreVersion = record.getStateStoreVersion();
   340	        if (stateStoreVersion == null) {
   341	          LOG.error("Cannot get State Store versions");
   342	        } else {
   343	          setStateStoreVersions(innerInfo, stateStoreVersion);
   344	        }
   345	
   346	        info.put(record.getPrimaryKey(),
   347	            Collections.unmodifiableMap(innerInfo));
   348	      }
   349	    } catch (IOException e) {
   350	      LOG.error("Cannot get Routers JSON from the State Store", e);
   351	      return "{}";
   352	    }
   353	    return JSON.toString(info);
   354	  }

   513	  @Override // NameNodeMXBean
   514	  public String getNodeUsage() {
   515	    float median = 0;
   516	    float max = 0;
   517	    float min = 0;
   518	    float dev = 0;
   519	
   520	    final Map<String, Map<String, Object>> info = new HashMap<>();
   521	    try {
   522	      RouterRpcServer rpcServer = this.router.getRpcServer();
   523	      DatanodeInfo[] live = rpcServer.getDatanodeReport(
   524	          DatanodeReportType.LIVE, false, timeOut);
   525	
   526	      if (live.length > 0) {
   527	        float totalDfsUsed = 0;
   528	        float[] usages = new float[live.length];
   529	        int i = 0;
   530	        for (DatanodeInfo dn : live) {
   531	          usages[i++] = dn.getDfsUsedPercent();
   532	          totalDfsUsed += dn.getDfsUsedPercent();
   533	        }
   534	        totalDfsUsed /= live.length;
   535	        Arrays.sort(usages);
   536	        median = usages[usages.length / 2];
   537	        max = usages[usages.length - 1];
   538	        min = usages[0];
   539	
   540	        for (i = 0; i < usages.length; i++) {
   541	          dev += (usages[i] - totalDfsUsed) * (usages[i] - totalDfsUsed);
   542	        }
   543	        dev = (float) Math.sqrt(dev / usages.length);
   544	      }
   545	    } catch (IOException e) {
   546	      LOG.error("Cannot get the live nodes: {}", e.getMessage());
   547	    }
   548	
   549	    final Map<String, Object> innerInfo = new HashMap<>();
   550	    innerInfo.put("min", StringUtils.format("%.2f%%", min));
   551	    innerInfo.put("median", StringUtils.format("%.2f%%", median));
   552	    innerInfo.put("max", StringUtils.format("%.2f%%", max));
   553	    innerInfo.put("stdDev", StringUtils.format("%.2f%%", dev));
   554	    info.put("nodeUsage", innerInfo);
   555	
   556	    return JSON.toString(info);
   557	  }

   750	   * Build a set of unique values found in all namespaces.
   751	   *
   752	   * @param f Method reference of the appropriate FederationNamespaceInfo
   753	   *          getter function
   754	   * @return Set of unique string values found in all discovered namespaces.
   755	   * @throws IOException if the query could not be executed.
   756	   */
   757	  private Collection<String> getNamespaceInfo(
   758	      Function<FederationNamespaceInfo, String> f) throws IOException {
   759	    if (membershipStore == null) {
   760	      return new HashSet<>();
   761	    }
   762	    GetNamespaceInfoRequest request = GetNamespaceInfoRequest.newInstance();
   763	    GetNamespaceInfoResponse response =
   764	        membershipStore.getNamespaceInfo(request);
   765	    return response.getNamespaceInfo().stream()
   766	      .map(f)
   767	      .collect(Collectors.toSet());
   768	  }

   770	  /**
   771	   * Get the aggregated value for a method for all nameservices.
   772	   * @param f Method reference
   773	   * @return Aggregated integer.
   774	   */
   775	  private int getNameserviceAggregatedInt(ToIntFunction<MembershipStats> f) {
   776	    try {
   777	      return getActiveNamenodeRegistrations().stream()
   778	               .map(MembershipState::getStats)
   779	               .collect(Collectors.summingInt(f));
   780	    } catch (IOException e) {
   781	      LOG.error("Unable to extract metrics: {}", e.getMessage());
   782	      return 0;
   783	    }
   784	  }

   789	   * @return Aggregated long.
   790	   */
   791	  private long getNameserviceAggregatedLong(ToLongFunction<MembershipStats> f) {
   792	    try {
   793	      return getActiveNamenodeRegistrations().stream()
   794	               .map(MembershipState::getStats)
   795	               .collect(Collectors.summingLong(f));
   796	    } catch (IOException e) {
   797	      LOG.error("Unable to extract metrics: {}", e.getMessage());
   798	      return 0;
   799	    }

   802	  private BigInteger getNameserviceAggregatedBigInt(
   803	      ToLongFunction<MembershipStats> f) {
   804	    try {
   805	      List<MembershipState> states = getActiveNamenodeRegistrations();
   806	      BigInteger sum = BigInteger.valueOf(0);
   807	      for (MembershipState state : states) {
   808	        long lvalue = f.applyAsLong(state.getStats());
   809	        sum = sum.add(BigInteger.valueOf(lvalue));
   810	      }
   811	      return sum;
   812	    } catch (IOException e) {
   813	      LOG.error("Unable to extract metrics: {}", e.getMessage());
   814	      return new BigInteger("0");
   815	    }
   816	  }

   818	  /**
   819	   * Fetches the most active namenode memberships for all known nameservices.
   820	   * The fetched membership may not or may not be active. Excludes expired
   821	   * memberships.
   822	   * @throws IOException if the query could not be performed.
   823	   * @return List of the most active NNs from each known nameservice.
   824	   */
   825	  private List<MembershipState> getActiveNamenodeRegistrations()
   826	      throws IOException {
   827	    List<MembershipState> resultList = new ArrayList<>();
   828	    if (membershipStore == null) {
   829	      return resultList;
   830	    }
   831	
   832	    GetNamespaceInfoRequest request = GetNamespaceInfoRequest.newInstance();
   833	    GetNamespaceInfoResponse response =
   834	        membershipStore.getNamespaceInfo(request);
   835	    for (FederationNamespaceInfo nsInfo : response.getNamespaceInfo()) {
   836	      // Fetch the most recent namenode registration
   837	      String nsId = nsInfo.getNameserviceId();
   838	      List<? extends FederationNamenodeContext> nns =
   839	          namenodeResolver.getNamenodesForNameserviceId(nsId);
   840	      if (nns != null) {
   841	        FederationNamenodeContext nn = nns.get(0);
   842	        if (nn instanceof MembershipState) {
   843	          resultList.add((MembershipState) nn);
   844	        }
   845	      }
   846	    }
   847	    return resultList;
   848	  }

   879	  /**
   880	   * Get JSON for this record.
   881	   *
   882	   * @return Map representing the data for the JSON representation.
   883	   */
   884	  private static Map<String, Object> getJson(BaseRecord record) {
   885	    Map<String, Object> json = new HashMap<>();
   886	    Map<String, Class<?>> fields = getFields(record);
   887	
   888	    for (String fieldName : fields.keySet()) {
   889	      if (!fieldName.equalsIgnoreCase("proto")) {
   890	        try {
   891	          Object value = getField(record, fieldName);
   892	          if (value instanceof BaseRecord) {
   893	            BaseRecord recordField = (BaseRecord) value;
   894	            json.putAll(getJson(recordField));
   895	          } else {
   896	            json.put(fieldName, value == null ? JSONObject.NULL : value);
   897	          }
   898	        } catch (Exception e) {
   899	          throw new IllegalArgumentException(
   900	              "Cannot serialize field " + fieldName + " into JSON");
   901	        }
   902	      }
   903	    }
   904	    return json;
   905	  }

   907	  /**
   908	   * Returns all serializable fields in the object.
   909	   *
   910	   * @return Map with the fields.
   911	   */
   912	  private static Map<String, Class<?>> getFields(BaseRecord record) {
   913	    Map<String, Class<?>> getters = new HashMap<>();
   914	    for (Method m : record.getClass().getDeclaredMethods()) {
   915	      if (m.getName().startsWith("get")) {
   916	        try {
   917	          Class<?> type = m.getReturnType();
   918	          char[] c = m.getName().substring(3).toCharArray();
   919	          c[0] = Character.toLowerCase(c[0]);
   920	          String key = new String(c);
   921	          getters.put(key, type);
   922	        } catch (Exception e) {
   923	          LOG.error("Cannot execute getter {} on {}", m.getName(), record);
   924	        }
   925	      }
   926	    }
   927	    return getters;
   928	  }

   930	  /**
   931	   * Fetches the value for a field name.
   932	   *
   933	   * @param fieldName the legacy name of the field.
   934	   * @return The field data or null if not found.
   935	   */
   936	  private static Object getField(BaseRecord record, String fieldName) {
   937	    Object result = null;
   938	    Method m = locateGetter(record, fieldName);
   939	    if (m != null) {
   940	      try {
   941	        result = m.invoke(record);
   942	      } catch (Exception e) {
   943	        LOG.error("Cannot get field {} on {}", fieldName, record);
   944	      }
   945	    }
   946	    return result;
   947	  }

   949	  /**
   950	   * Finds the appropriate getter for a field name.
   951	   *
   952	   * @param fieldName The legacy name of the field.
   953	   * @return The matching getter or null if not found.
   954	   */
   955	  private static Method locateGetter(BaseRecord record, String fieldName) {
   956	    for (Method m : record.getClass().getMethods()) {
   957	      if (m.getName().equalsIgnoreCase("get" + fieldName)) {
   958	        return m;
   959	      }
   960	    }
   961	    return null;
   962	  }
