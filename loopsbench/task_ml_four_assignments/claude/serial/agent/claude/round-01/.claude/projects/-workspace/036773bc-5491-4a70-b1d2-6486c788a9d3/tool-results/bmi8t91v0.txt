    49	def silo_analysis(X, y, clusterer, n_clusters, random_state, filename):
    50	    if filename == "Chess":
    51	        col1, col2 = 0, 5
    52	    else:
    53	        col1, col2 = 13, 11
    54	
    55	    cluster_labels = clusterer.fit_predict(X)
    56	
    57	    # The silhouette_score gives the average value for all the samples.
    58	    # This gives a perspective into the density and separation of the formed
    59	    # clusters
    60	    silhouette_avg = silhouette_score(X, cluster_labels, metric='euclidean', sample_size=400, random_state=random_state)
    61	
    62	    plt.figure()
    63	
    64	    # The 1st subplot is the silhouette plot
    65	    # The silhouette coefficient can range from -1, 1 but in this example all
    66	    # lie within [-0.1, 1]
    67	    plt.xlim([-0.1, 1])
    68	    # The (n_clusters+1)*10 is for inserting blank space between silhouette
    69	    # plots of individual clusters, to demarcate them clearly.
    70	    plt.ylim([0, len(X) + (n_clusters + 1) * 10])
    71	
    72	    # Compute the silhouette scores for each sample
    73	    sample_silhouette_values = silhouette_samples(X, cluster_labels)
    74	
    75	    y_lower = 10
    76	    for i in range(n_clusters):
    77	        # Aggregate the silhouette scores for samples belonging to
    78	        # cluster i, and sort them
    79	        ith_cluster_silhouette_values = \
    80	            sample_silhouette_values[cluster_labels == i]
    81	
    82	        ith_cluster_silhouette_values.sort()
    83	
    84	        size_cluster_i = ith_cluster_silhouette_values.shape[0]
    85	        y_upper = y_lower + size_cluster_i
    86

    90	                          0, ith_cluster_silhouette_values,
    91	                          facecolor=color, edgecolor=color, alpha=0.7)
    92	
    93	        # Label the silhouette plots with their cluster numbers at the middle
    94	        plt.text(-0.05, y_lower + 0.5 * size_cluster_i, str(i))
    95	
    96	        # Compute the new y_lower for next plot
    97	        y_lower = y_upper + 10  # 10 for the 0 samples
    98

   100	
   101	    plt.xlabel("The silhouette coefficient values")
   102	    plt.ylabel("Cluster label")
   103	
   104	    # The vertical line for average silhouette score of all the values
   105	    plt.axvline(x=silhouette_avg, color="red", linestyle="--")

   140	# Citation: https://scikit-learn.org/stable/auto_examples/cluster/plot_kmeans_stability_low_dim_dense.html#sphx-glr-auto-examples-cluster-plot-kmeans-stability-low-dim-dense-py
   141	def init_analysis(X, y, n_clusters, filename):
   142	    # Number of run (with randomly generated dataset) for each strategy so as
   143	    # to be able to compute an estimate of the standard deviation
   144	    n_runs = 5
   145	
   146	    # k-means models can do several random inits so as to be able to trade
   147	    # CPU time for convergence robustness
   148	    n_init_range = np.array([1, 5, 10, 15, 20, 50, 75, 100])
   149	
   150	    # Part 1: Quantitative evaluation of various init methods
   151	
   152	    plt.figure()
   153	    plots = []
   154	    legends = []
   155	
   156	    cases = [
   157	        (KMeans, 'k-means++', {}),
   158	        (KMeans, 'random', {}),
   159	        (MiniBatchKMeans, 'k-means++', {'max_no_improvement': 3}),
   160	        (MiniBatchKMeans, 'random', {'max_no_improvement': 3, 'init_size': 500}),
   161	    ]
   162	
   163	    for factory, init, params in cases:
   164	        print("Evaluation of %s with %s init" % (factory.__name__, init))
   165	        inertia = np.empty((len(n_init_range), n_runs))
   166	
   167	        for run_id in range(n_runs):
   168	            for i, n_init in enumerate(n_init_range):
   169	                km = factory(n_clusters=n_clusters, init=init, random_state=run_id,
   170	                             n_init=n_init, **params).fit(X)
   171	                inertia[i, run_id] = km.inertia_
   172	        p = plt.errorbar(n_init_range, inertia.mean(axis=1), inertia.std(axis=1))
   173	        plots.append(p[0])
   174	        legends.append("%s with %s init" % (factory.__name__, init))
   175	
   176	    plt.xlabel('n_init')
   177	    plt.ylabel('inertia')
   178	    plt.legend(plots, legends)
   179	    plt.title("Mean inertia for various k-means init across %d runs" % n_runs)
   180	    plt.savefig(
   181	            "Images\\" + filename + "Mean Interia across " + str(n_runs) + " runs and " + str(n_clusters) + " clusters")
   182	
   183	    plt.show()
