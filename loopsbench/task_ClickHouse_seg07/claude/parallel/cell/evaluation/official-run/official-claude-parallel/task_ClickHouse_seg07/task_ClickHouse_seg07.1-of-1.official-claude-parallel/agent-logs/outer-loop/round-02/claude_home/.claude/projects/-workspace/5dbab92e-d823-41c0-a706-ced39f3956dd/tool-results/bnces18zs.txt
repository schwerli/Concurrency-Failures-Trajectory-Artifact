   101	    std::vector<String> global_flags(command_pos);
   102	    command_arguments.resize(common_arguments.size() - command_pos);
   103	    copy(common_arguments.begin(), common_arguments.begin() + command_pos, global_flags.begin());
   104	    copy(common_arguments.begin() + command_pos, common_arguments.end(), command_arguments.begin());
   105	
   106	    parseAndCheckOptions(options_description, positional_options_description, global_flags);
   107	
   108	    po::notify(options);
   109	
   110	    if (options.count("help"))
   111	    {
   112	        printHelpMessage(options_description);
   113	        exit(0);
   114	    }
   115	
   116	    if (!supported_commands.contains(command_name))
   117	    {
   118	        std::cerr << "Unknown command name:  " << command_name << "\n";
   119	        printHelpMessage(options_description);
   120	        throw DB::Exception("Bad Arguments", DB::ErrorCodes::BAD_ARGUMENTS);
   121	    }
   122	
   123	    processOptions();
   124	}

   126	void DisksApp::parseAndCheckOptions(
   127	    ProgramOptionsDescription & options_description_,
   128	    boost::program_options::positional_options_description & positional_options_description,
   129	    std::vector<String> & arguments)
   130	{
   131	    auto parser = po::command_line_parser(arguments)
   132	        .options(options_description_)
   133	        .positional(positional_options_description)
   134	        .allow_unregistered();
   135	    po::parsed_options parsed = parser.run();
   136	    po::store(parsed, options);
   137	
   138	    auto positional_arguments = po::collect_unrecognized(parsed.options, po::collect_unrecognized_mode::include_positional);
   139	    for (const auto & arg : positional_arguments)
   140	    {
   141	        if (command_descriptions.contains(arg))
   142	        {
   143	            command_name = arg;
   144	            break;
   145	        }
   146	    }
   147	}

   149	int DisksApp::main(const std::vector<String> & /*args*/)
   150	{
   151	    if (config().has("send-logs"))
   152	    {
   153	        auto log_level = config().getString("log-level", "trace");
   154	        Poco::Logger::root().setLevel(Poco::Logger::parseLevel(log_level));
   155	
   156	        auto log_path = config().getString("logger.clickhouse-disks", "/var/log/clickhouse-server/clickhouse-disks.log");
   157	        Poco::Logger::root().setChannel(Poco::AutoPtr<Poco::FileChannel>(new Poco::FileChannel(log_path)));
   158	    }
   159	
   160	    if (config().has("config-file") || fs::exists(getDefaultConfigFileName()))
   161	    {
   162	        String config_path = config().getString("config-file", getDefaultConfigFileName());
   163	        ConfigProcessor config_processor(config_path, false, false);
   164	        config_processor.setConfigPath(fs::path(config_path).parent_path());
   165	        auto loaded_config = config_processor.loadConfig();
   166	        config().add(loaded_config.configuration.duplicate(), false, false);
   167	    }
   168	    else
   169	    {
   170	        throw Exception(ErrorCodes::BAD_ARGUMENTS, "No config-file specifiged");
   171	    }
   172	
   173	    registerDisks();
   174	    registerFormats();
   175	
   176	    shared_context = Context::createShared();
   177	    global_context = Context::createGlobal(shared_context.get());
   178	
   179	    global_context->makeGlobalContext();
   180	    global_context->setApplicationType(Context::ApplicationType::DISKS);
   181	
   182	    String path = config().getString("path", DBMS_DEFAULT_PATH);
   183	    global_context->setPath(path);
   184	
   185	    auto & command = command_descriptions[command_name];
   186	
   187	    auto command_options = command->getCommandOptions();
   188	    std::vector<String> args;
   189	    if (command_options)
   190	    {
   191	        auto parser = po::command_line_parser(command_arguments).options(*command_options).allow_unregistered();
   192	        po::parsed_options parsed = parser.run();
   193	        po::store(parsed, options);
   194	        po::notify(options);
   195	        args = po::collect_unrecognized(parsed.options, po::collect_unrecognized_mode::include_positional);
   196	
   197	        command->processOptions(config(), options);
   198	    }
   199	    else
   200	    {
   201	        auto parser = po::command_line_parser(command_arguments).options({}).allow_unregistered();
   202	        po::parsed_options parsed = parser.run();
   203	        args = po::collect_unrecognized(parsed.options, po::collect_unrecognized_mode::include_positional);
   204	    }
   205	    command->execute(args, global_context, config());
   206	
   207	    return Application::EXIT_OK;
   208	}

    33	String ICommand::fullPathWithValidate(const DiskPtr & disk, const String & path)
    34	{
    35	    if (fs::path(path).lexically_normal().string() != path)
    36	        throw DB::Exception(DB::ErrorCodes::BAD_ARGUMENTS, "Path {} is not normalized", path);
    37	
    38	    String disk_path = fs::canonical(fs::path(disk->getPath())) / "";
    39	    String full_path = (fs::absolute(disk_path) / path).lexically_normal();
    40	
    41	    if (!full_path.starts_with(disk_path))
    42	        throw DB::Exception(
    43	            DB::ErrorCodes::BAD_ARGUMENTS, "Path {} must be inside disk path {}", full_path, disk_path);
    44	
    45	    return path;
    46	}
