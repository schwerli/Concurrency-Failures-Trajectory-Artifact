+def _build(cfg):
+    """Instantiate and start a network described by a config entry."""
+    link_opts = {"bw": cfg.get("link_bw", 1000), "delay": cfg.get("link_delay", 0.0)}
+    if "link_loss" in cfg:
+        link_opts["loss"] = cfg["link_loss"]
+    if cfg["type"] == "linear":

+    net.start()
+    return net

+"""Interfaces and links.
+
+A :class:`Link` models the virtual Ethernet pair that Mininet uses to connect two
+nodes; :class:`TCLink` adds the ``tc``-style shaping parameters (bandwidth,
+delay and loss) that the simulation engine reads when computing throughput,
+latency and reachability.
+"""
+
+
+class Intf:
+    """A network interface belonging to a node."""
+
+    def __init__(self, name, node=None, port=None, link=None, **params):
+        self.name = name
+        self.node = node
+        self.port = port
+        self.link = link
+        self.params = dict(params)
+        self.ip = params.get("ip")
+        self.mac = params.get("mac")
+        if node is not None:
+            node.addIntf(self, port=port)
+
+    def setIP(self, ip):
+        self.ip = ip
+        return ip
+
+    def IP(self):
+        return self.ip
+
+    def setMAC(self, mac):
+        self.mac = mac
+        return mac
+
+    def MAC(self):
+        return self.mac
+
+    def config(self, **kwargs):
+        self.params.update(kwargs)
+        return self.params
+
+    def __repr__(self):
+        return "<Intf %s>" % self.name
+
+    def __str__(self):
+        return self.name
+
+
+class Link:
+    """A bidirectional link between two nodes.
+
+    Creates one :class:`Intf` on each side and stores the link characteristics
+    used by the simulation engine: ``bw`` in Mbps, ``delay`` in milliseconds and
+    ``loss`` as a percentage.
+    """
+
+    #: Bandwidth used when a link is created without an explicit value (Mbps).
+    DEFAULT_BW = 1000.0
+    #: One-way propagation delay used when none is given (milliseconds).
+    DEFAULT_DELAY = 0.0
+    #: Loss percentage used when none is given.
+    DEFAULT_LOSS = 0.0
+
+    def __init__(self, node1, node2, port1=None, port2=None,
+                 intfName1=None, intfName2=None, cls1=Intf, cls2=Intf,
+                 bw=None, delay=None, loss=None, **params):
+        self.node1 = node1
+        self.node2 = node2
+        self.bw = float(bw) if bw is not None else float(self.DEFAULT_BW)
+        self.delay = float(delay) if delay is not None else float(self.DEFAULT_DELAY)
+        self.loss = float(loss) if loss is not None else float(self.DEFAULT_LOSS)
+        self.params = dict(params)
+
+        if port1 is None:
+            port1 = node1.newPort()
+        if port2 is None:
+            port2 = node2.newPort()
+        if intfName1 is None:
+            intfName1 = self.intfName(node1, port1)
+        if intfName2 is None:
+            intfName2 = self.intfName(node2, port2)
+
+        self.intf1 = cls1(intfName1, node=node1, port=port1, link=self)
+        self.intf2 = cls2(intfName2, node=node2, port=port2, link=self)
+        self.port1 = self.intf1.port
+        self.port2 = self.intf2.port
+
+    @staticmethod
+    def intfName(node, port):
+        """Mininet-style interface name, e.g. ``s1-eth0``."""
+        return "%s-eth%d" % (node.name, port)
+
+    # ------------------------------------------------------------------
+    @property
+    def loss_fraction(self):
+        """Loss expressed on a 0-1 scale (``loss`` is a percentage)."""
+        return self.loss / 100.0
+
+    def endpoints(self):
+        return (self.node1, self.node2)
+
+    def other(self, node):
+        """Return the node at the far end of the link."""
+        name = getattr(node, "name", node)
+        if self.node1.name == name:
+            return self.node2
+        if self.node2.name == name:
+            return self.node1
+        return None
+
+    def intf(self, node):
+        """Return the interface belonging to ``node``."""
+        name = getattr(node, "name", node)
+        if self.node1.name == name:
+            return self.intf1
+        if self.node2.name == name:
+            return self.intf2
+        return None
+
+    def otherIntf(self, intf):
+        """Return the interface on the opposite side of ``intf``."""
+        if intf is self.intf1:
+            return self.intf2
+        if intf is self.intf2:
+            return self.intf1
+        return None
+
+    def connects(self, node1, node2):
+        n1 = getattr(node1, "name", node1)
+        n2 = getattr(node2, "name", node2)
+        pair = {self.node1.name, self.node2.name}
+        return pair == {n1, n2}
+
+    def attrs(self):
+        """Link characteristics as a dict (the payload stored on graph edges)."""
+        d = {"bw": self.bw, "delay": self.delay, "loss": self.loss, "link": self}
+        d.update(self.params)
+        return d
+
+    def status(self):
+        return "(OK OK)"
+
+    def delete(self):
+        for node, intf in ((self.node1, self.intf1), (self.node2, self.intf2)):
+            if intf.port in node.intfs and node.intfs[intf.port] is intf:
+                del node.intfs[intf.port]
+        self.intf1.link = None
+        self.intf2.link = None
+
+    def __repr__(self):
+        return "<%s %s<->%s>" % (self.__class__.__name__,
+                                 self.intf1.name, self.intf2.name)
+
+    def __str__(self):
+        return "%s<->%s" % (self.intf1.name, self.intf2.name)
+
+
+class TCLink(Link):
+    """Traffic-controlled link with explicit bandwidth, delay and loss.
+
+    Parameters
+    ----------
+    bw : float
+        Bandwidth in Mbps.
+    delay : float
+        One-way delay in milliseconds.
+    loss : float
+        Loss rate as a percentage (0-100).
+    """
+
+    def __init__(self, node1, node2, bw=1000.0, delay=0.0, loss=0.0,
+                 max_queue_size=None, use_htb=True, **params):
+        self.max_queue_size = max_queue_size
+        self.use_htb = use_htb
+        Link.__init__(self, node1, node2, bw=bw, delay=delay, loss=loss,
+                      **params)
+
+    def config(self, **kwargs):
+        """Update shaping parameters after creation."""
+        if "bw" in kwargs:
+            self.bw = float(kwargs.pop("bw"))
+        if "delay" in kwargs:
+            self.delay = float(kwargs.pop("delay"))
+        if "loss" in kwargs:
+            self.loss = float(kwargs.pop("loss"))
+        self.params.update(kwargs)
+        return self.attrs()

+"""The Mininet simulation engine.
+
+:class:`Mininet` owns the instantiated nodes and links and answers the questions
+an experiment asks of an emulated network: is a pair of hosts reachable, what
+throughput does ``iperf`` report between them, and what is the round-trip time.
+
+Reachability and paths are resolved on a :class:`netsim.graph.WeightedGraph`
+built from the links, so higher layers (WiFi, containers, P4, the SDN
+controller) can reuse the same routing substrate.
+"""
+
+from .graph import WeightedGraph
+from .link import Link, TCLink
+from .node import Controller, Host, OVSSwitch, Switch
+from .packet import Packet, PacketTrace
+from .utils import (compute_bottleneck_bw, compute_path_delay,
+                    compute_path_loss_rate, generate_ip)
+
+
+class Mininet:
+    """Network emulation engine built from a :class:`netsim.topo.Topo`."""
+
+    def __init__(self, topo=None, switch=OVSSwitch, host=Host,
+                 controller=Controller, link=TCLink, ipBase="10.0.0.0/8",
+                 build=True, autoSetMacs=False, autoStaticArp=False,
+                 waitConnected=False, **params):
+        self.topo = topo
+        self.switch = switch
+        self.host = host
+        self.controller = controller
+        self.link = link
+        self.ipBase = ipBase
+        self.ipBaseNum = ".".join(ipBase.split("/")[0].split(".")[:2])
+        self.autoSetMacs = autoSetMacs
+        self.autoStaticArp = autoStaticArp
+        self.waitConnected = waitConnected
+        self.params = dict(params)
+
+        self.hosts = []
+        self.switches = []
+        self.controllers = []
+        self.links = []
+        self.nameToNode = {}
+        self.graph = WeightedGraph()
+        self.built = False
+        self.started = False
+        self.nextIP = 0
+
+        if topo is not None and build:
+            self.build()
+
+    # ------------------------------------------------------------------
+    # construction
+    # ------------------------------------------------------------------
+    def addHost(self, name, cls=None, **params):
+        """Instantiate a host, register it and return it."""
+        cls = cls or self.host
+        host = cls(name, **params)
+        self.hosts.append(host)
+        self._register(host)
+        return host
+
+    def addSwitch(self, name, cls=None, **params):
+        """Instantiate a switch, register it and return it."""
+        cls = cls or self.switch
+        switch = cls(name, **params)
+        self.switches.append(switch)
+        self._register(switch)
+        return switch
+
+    def addController(self, name="c0", cls=None, **params):
+        """Instantiate a controller, register it and return it."""
+        cls = cls or self.controller
+        ctrl = cls(name, **params) if not isinstance(name, Controller) else name
+        self.controllers.append(ctrl)
+        self._register(ctrl)
+        return ctrl
+
+    def addLink(self, node1, node2, port1=None, port2=None, cls=None,
+                **params):
+        """Create a link between two nodes (names or objects) and return it."""
+        node1 = self._resolve(node1)
+        node2 = self._resolve(node2)
+        cls = cls or self.link
+        params.pop("isSwitch", None)
+        link = cls(node1, node2, port1=port1, port2=port2, **params)
+        self.links.append(link)
+        self.graph.add_edge(node1.name, node2.name, **link.attrs())
+        return link
+
+    def _register(self, node):
+        self.nameToNode[node.name] = node
+        self.graph.add_node(node.name)
+        return node
+
+    def _resolve(self, node):
+        if isinstance(node, str):
+            return self.nameToNode[node]
+        return node
+
+    def build(self):
+        """Instantiate the topology: controller, switches, hosts and links."""
+        if self.topo is not None:
+            self.buildFromTopo(self.topo)
+        self.built = True
+        return self
+
+    def buildFromTopo(self, topo):
+        """Create nodes and links described by ``topo``."""
+        if self.controller is not None and not self.controllers:
+            self.addController("c0")
+        for name in topo.switches():
+            opts = topo.nodeInfo(name)
+            opts.pop("isSwitch", None)
+            cls = opts.pop("cls", None)
+            self.addSwitch(name, cls=cls, **opts)
+        for name in topo.hosts():
+            opts = topo.nodeInfo(name)
+            opts.pop("isSwitch", None)
+            cls = opts.pop("cls", None)
+            self.addHost(name, cls=cls, **opts)
+        for node1, node2, opts in topo.links(sort=True, withInfo=True):
+            opts = dict(opts)
+            cls = opts.pop("cls", None)
+            self.addLink(node1, node2, cls=cls, **opts)
+        for switch in self.switches:
+            for ctrl in self.controllers:
+                switch.attach(ctrl)
+                if hasattr(ctrl, "add_switch"):
+                    ctrl.add_switch(switch)
+        return self
+
+    # ------------------------------------------------------------------
+    # lifecycle
+    # ------------------------------------------------------------------
+    def start(self):
+        """Start the network: assign host IP addresses."""
+        if not self.built:
+            self.build()
+        for index, host in enumerate(self.hosts):
+            if host.ip is None:
+                host.setIP(generate_ip(index, base=self.ipBaseNum))
+            if self.autoSetMacs and host.mac is None:
+                host.setMAC("00:00:00:00:%02x:%02x" %
+                            ((index + 1) // 256, (index + 1) % 256))
+        for switch in self.switches:
+            switch.start(self.controllers)
+        self.nextIP = len(self.hosts)

+            if len(self.hosts) < 2:

+        path = self.path(src, dst)

+        hosts = self._host_list(hosts)

+        pairs = 0
+        for i, src in enumerate(hosts):
+            for dst in hosts[i + 1:]:
+                rtt = self.rtt(src, dst)
+                if rtt is None:
+                    continue
+                total += rtt
+                pairs += 1
+        if pairs == 0:
+            return 0.0
+        return total / pairs
+
+    def avg_hops(self, hosts=None):
+        """Average hop count over all distinct host pairs."""
+        hosts = self._host_list(hosts)
+        total = 0
+        pairs = 0
+        for i, src in enumerate(hosts):
+            for dst in hosts[i + 1:]:
+                path = self.path(src, dst)
+                if not path:
+                    continue
+                total += len(path) - 1
+                pairs += 1
+        return (total / pairs) if pairs else 0.0
+
+    def send_packet(self, src, dst, size_bytes=1500, protocol="ICMP"):
+        """Trace a packet from ``src`` to ``dst`` and return its PacketTrace."""
+        src = self._resolve(src)
+        dst = self._resolve(dst)
+        packet = Packet(src_ip=src.ip or "0.0.0.0", dst_ip=dst.ip or "0.0.0.0",
+                        size_bytes=size_bytes, protocol=protocol)
+        trace = PacketTrace(packet=packet)
+        path = self.path(src, dst)
+        if not path:
+            return trace
+        trace.add_hop(path[0], 0.0)
+        for u, v in zip(path, path[1:]):
+            attrs = self.graph.get_edge(u, v) or {}
+            trace.add_hop(v, attrs.get("delay", 0.0))
+        trace.delivered = True
+        return trace

+"""Topology templates.
+
+A :class:`Topo` is a *blueprint*: it records hosts, switches and links without
+instantiating any node object.  :class:`netsim.net.Mininet` consumes the
+blueprint in :meth:`netsim.net.Mininet.build` and creates the real objects.
+"""
+
+from .graph import WeightedGraph
+
+
+class Topo:
+    """Base topology blueprint.
+
+    Subclasses override :meth:`build`; the constructor forwards its arguments to
+    it, exactly like Mininet's ``Topo``.
+    """
+
+    def __init__(self, *args, **params):
+        self._hosts = []
+        self._switches = []
+        self._links = []
+        self.node_info = {}
+        self.link_info = {}
+        self.ports = {}
+        self.hopts = dict(params.pop("hopts", {}))
+        self.sopts = dict(params.pop("sopts", {}))
+        self.lopts = dict(params.pop("lopts", {}))
+        self.g = WeightedGraph()
+        self.build(*args, **params)
+
+    # ------------------------------------------------------------------
+    # build hook
+    # ------------------------------------------------------------------
+    def build(self, *args, **params):
+        """Override to describe the topology."""
+        return None
+
+    # ------------------------------------------------------------------
+    # node / link creation
+    # ------------------------------------------------------------------
+    def addNode(self, name, **opts):
+        self.node_info[name] = dict(opts)
+        self.g.add_node(name)
+        self.ports.setdefault(name, {})
+        return name
+
+    def addHost(self, name, **opts):
+        """Register a host in the blueprint and return its name."""
+        merged = dict(self.hopts)
+        merged.update(opts)
+        merged.setdefault("isSwitch", False)
+        self.addNode(name, **merged)
+        if name not in self._hosts:
+            self._hosts.append(name)
+        return name
+
+    def addSwitch(self, name, **opts):
+        """Register a switch in the blueprint and return its name."""
+        merged = dict(self.sopts)
+        merged.update(opts)
+        merged["isSwitch"] = True
+        self.addNode(name, **merged)
+        if name not in self._switches:
+            self._switches.append(name)
+        return name
+
+    def addLink(self, node1, node2, port1=None, port2=None, **opts):
+        """Register a link between two already-added nodes."""
+        merged = dict(self.lopts)
+        merged.update(opts)
+        if port1 is None:
+            port1 = self._newPort(node1)
+        if port2 is None:
+            port2 = self._newPort(node2)
+        self.ports.setdefault(node1, {})[port1] = (node2, port2)
+        self.ports.setdefault(node2, {})[port2] = (node1, port1)
+        merged["port1"] = port1
+        merged["port2"] = port2
+        self._links.append((node1, node2, merged))
+        self.link_info[(node1, node2)] = merged
+        self.g.add_edge(node1, node2, **merged)
+        return (node1, node2)
+
+    def _newPort(self, node):
+        used = self.ports.setdefault(node, {})
+        return len(used)
+
+    # ------------------------------------------------------------------
+    # queries
+    # ------------------------------------------------------------------
+    def hosts(self, sort=True):
+        """Host names in the blueprint."""
+        return sorted(self._hosts, key=self._nameKey) if sort else list(self._hosts)
+
+    def switches(self, sort=True):
+        """Switch names in the blueprint."""
+        return sorted(self._switches, key=self._nameKey) if sort else list(self._switches)
+
+    def nodes(self, sort=True):
+        """All node names (switches first, mirroring Mininet)."""
+        return self.switches(sort) + self.hosts(sort)
+
+    def links(self, sort=True, withKeys=False, withInfo=False):
+        """Links as ``(node1, node2)`` pairs (or with their option dicts)."""
+        links = list(self._links)
+        if sort:
+            links = sorted(links, key=lambda l: (self._nameKey(l[0]),
+                                                 self._nameKey(l[1])))
+        if withInfo:
+            return [(a, b, dict(o)) for a, b, o in links]
+        return [(a, b) for a, b, _ in links]
+
+    def isSwitch(self, name):
+        return bool(self.node_info.get(name, {}).get("isSwitch", False))
+
+    def nodeInfo(self, name):
+        return dict(self.node_info.get(name, {}))
+
+    def linkInfo(self, node1, node2):
+        info = self.link_info.get((node1, node2))
+        if info is None:
+            info = self.link_info.get((node2, node1), {})
+        return dict(info)
+
+    def port(self, src, dst):
+        """Return ``(src_port, dst_port)`` of the link between two nodes."""
+        for port, (peer, peer_port) in self.ports.get(src, {}).items():
+            if peer == dst:
+                return (port, peer_port)
+        return (None, None)
+
+    @staticmethod
+    def _nameKey(name):
+        """Sort ``h2`` before ``h10`` (natural ordering)."""
+        prefix = "".join(ch for ch in name if not ch.isdigit())
+        digits = "".join(ch for ch in name if ch.isdigit())
+        return (prefix, int(digits) if digits else 0)
+
+    def __repr__(self):
+        return "<%s hosts=%d switches=%d links=%d>" % (
+            self.__class__.__name__, len(self._hosts), len(self._switches),
+            len(self._links))
+
+
+class SingleSwitchTopo(Topo):
+    """A single switch connected to ``k`` hosts."""
+
+    def build(self, k=2, **opts):
+        self.k = k
+        switch = self.addSwitch("s1")
+        for i in range(1, k + 1):
+            host = self.addHost("h%d" % i)
+            self.addLink(host, switch, **opts)
+
+
+class LinearTopo(Topo):
+    """``k`` switches in a chain, each carrying ``n`` hosts.
+
+    ``LinearTopo(k=3)`` builds ``h1-s1-s2-s3-h3`` with h2 hanging off s2.
+    """
+
+    def build(self, k=2, n=1, **opts):
+        self.k = k
+        self.n = n
+        last_switch = None
+        for i in range(1, k + 1):
+            switch = self.addSwitch("s%d" % i)
+            for j in range(n):
+                host_num = (i - 1) * n + j + 1
+                host = self.addHost("h%d" % host_num)
+                self.addLink(host, switch, **opts)
+            if last_switch is not None:
+                self.addLink(last_switch, switch, **opts)
+            last_switch = switch
+
+
+class TreeTopo(Topo):
+    """Tree of switches of the given ``depth`` and ``fanout``, hosts at leaves.
+
+    ``TreeTopo(depth=d, fanout=f)`` has ``f**d`` hosts and
+    ``(f**d - 1) / (f - 1)`` switches (``d`` switches when ``fanout == 1``).
+    """
+
+    def build(self, depth=1, fanout=2, **opts):
+        self.depth = depth
+        self.fanout = fanout
+        self._lopts_build = dict(opts)
+        self.hostNum = 1
+        self.switchNum = 1
+        self.addTree(depth, fanout)
+
+    def addTree(self, depth, fanout):
+        """Recursively add a subtree and return the name of its root node."""
+        isSwitch = depth > 0
+        if isSwitch:
+            node = self.addSwitch("s%d" % self.switchNum)
+            self.switchNum += 1
+            for _ in range(fanout):
+                child = self.addTree(depth - 1, fanout)
+                self.addLink(node, child, **self._lopts_build)
+        else:
+            node = self.addHost("h%d" % self.hostNum)
+            self.hostNum += 1
+        return node
+
+
+class MinimalTopo(Topo):
+    """Two hosts on one switch (Mininet's default topology)."""
+
+    def build(self, **opts):
+        switch = self.addSwitch("s1")
+        for i in (1, 2):
+            self.addLink(self.addHost("h%d" % i), switch, **opts)
+
+
+class RingTopo(Topo):
+    """``k`` switches in a ring, each carrying one host.
+
+    A ring gives every switch pair a redundant path, which makes it the natural
+    topology for link-failure-recovery experiments.
+    """
+
+    def build(self, k=4, n=1, **opts):
+        self.k = k
+        self.n = n
+        switches = []
+        for i in range(1, k + 1):
+            switch = self.addSwitch("s%d" % i)
+            switches.append(switch)
+            for j in range(n):
+                host_num = (i - 1) * n + j + 1
+                self.addLink(self.addHost("h%d" % host_num), switch, **opts)
+        for i in range(k):
+            if k > 2 or i == 0:
+                self.addLink(switches[i], switches[(i + 1) % k], **opts)

+"""WiFi access points."""
+
+from ..node import Switch
+from ..utils import distance_2d
+from .propagation import (DEFAULT_PATH_LOSS_EXP, DEFAULT_REF_DISTANCE_M,
+                          ghz_to_hz, log_distance_path_loss, max_range,
+                          received_signal_strength)
+
+
+class AccessPoint(Switch):
+    """A WiFi access point: an OpenFlow switch with a radio.
+
+    The AP carries the parameters of its radio (transmit power, frequency, path
+    loss exponent and the RSSI floor below which a station cannot associate) and
+    tracks the stations currently attached to it.
+    """
+
+    def __init__(self, name, position=(0.0, 0.0), ssid="netsim-ssid", channel=1,
+                 tx_power_dbm=20.0, freq_ghz=2.4,
+                 path_loss_exp=DEFAULT_PATH_LOSS_EXP,
+                 min_rssi_dbm=-80.0, antenna_gain_dbi=0.0,
+                 ref_distance_m=DEFAULT_REF_DISTANCE_M, mode="g", **params):
+        Switch.__init__(self, name, **params)
+        self.position = tuple(float(c) for c in position)
+        self.ssid = ssid
+        self.channel = channel
+        self.tx_power_dbm = tx_power_dbm
+        self.freq_ghz = freq_ghz
+        self.path_loss_exp = path_loss_exp
+        self.min_rssi_dbm = min_rssi_dbm
+        self.antenna_gain_dbi = antenna_gain_dbi
+        self.ref_distance_m = ref_distance_m
+        self.mode = mode
+        self.associated_stations = []
+
+    # ------------------------------------------------------------------
+    @property
+    def freq_hz(self):
+        """Radio frequency in Hz."""
+        return ghz_to_hz(self.freq_ghz)
+
+    def set_position(self, x, y=None):
+        if y is None and isinstance(x, (tuple, list)):
+            self.position = tuple(float(c) for c in x[:2])
+        else:
+            self.position = (float(x), float(y))
+        return self.position
+
+    def distance_to(self, other):
+        """Euclidean distance to a node or an ``(x, y)`` position."""
+        pos = getattr(other, "position", other)
+        return distance_2d(self.position, pos)
+
+    # ------------------------------------------------------------------
+    def path_loss_to(self, other):
+        """Path loss in dB towards a station or position."""
+        return log_distance_path_loss(self.freq_hz, self.distance_to(other),
+                                      self.path_loss_exp, self.ref_distance_m)
+
+    def signal_strength(self, other):
+        """RSSI in dBm seen by a station (or at an ``(x, y)`` position)."""
+        rx_gain = getattr(other, "antenna_gain_dbi", 0.0)
+        return received_signal_strength(
+            self.tx_power_dbm, self.freq_hz, self.distance_to(other),
+            path_loss_exp=self.path_loss_exp,
+            ref_distance_m=self.ref_distance_m,
+            tx_gain_dbi=self.antenna_gain_dbi, rx_gain_dbi=rx_gain)
+
+    #: Alias used by the association logic.
+    rssi_to = signal_strength
+
+    @property
+    def coverage_radius(self):
+        """Distance at which the RSSI falls to ``min_rssi_dbm`` (metres)."""
+        return max_range(self.tx_power_dbm, self.min_rssi_dbm, self.freq_hz,
+                         path_loss_exp=self.path_loss_exp,
+                         ref_distance_m=self.ref_distance_m,
+                         tx_gain_dbi=self.antenna_gain_dbi)
+
+    #: Alias kept for readability at call sites.
+    max_range_m = coverage_radius
+
+    def in_range(self, other, min_rssi_dbm=None):

+        floor = self.min_rssi_dbm if min_rssi_dbm is None else min_rssi_dbm

+"""Mobility models.
+
+Position traces are generated with ``numpy.random.RandomState`` so that a given
+seed always produces the same trace, which is what makes the mobility-driven
+experiments reproducible.
+"""
+
+import math
+
+import numpy as np
+
+
+class MobilityModel:
+    """Base class for mobility models over a rectangular area."""
+
+    def __init__(self, area_width=100.0, area_height=100.0, seed=None,
+                 time_step=1.0):
+        self.area_width = float(area_width)
+        self.area_height = float(area_height)
+        self.seed = seed
+        self.time_step = float(time_step)
+        self.rng = np.random.RandomState(seed)
+
+    def reset(self, seed=None):
+        """Re-seed the generator so a trace can be reproduced."""
+        if seed is not None:
+            self.seed = seed
+        self.rng = np.random.RandomState(self.seed)
+        return self
+
+    def random_position(self):
+        """Uniformly distributed position inside the area."""
+        x = float(self.rng.uniform(0.0, self.area_width))
+        y = float(self.rng.uniform(0.0, self.area_height))
+        return (x, y)
+
+    def center(self):
+        return (self.area_width / 2.0, self.area_height / 2.0)
+
+    def generate_trace(self, start_pos=None, num_steps=100):
+        raise NotImplementedError
+
+
+class StaticMobility(MobilityModel):
+    """A node that never moves (useful as a control in experiments)."""
+
+    def generate_trace(self, start_pos=None, num_steps=100):
+        pos = tuple(start_pos) if start_pos is not None else self.center()
+        return [pos for _ in range(num_steps)]
+
+
+class RandomWaypointMobility(MobilityModel):
+    """Classic random waypoint model.
+
+    The node repeatedly picks a uniformly random waypoint inside the area and a
+    uniformly random speed in ``speed_range``, then walks towards the waypoint.
+    On arrival it optionally pauses for ``pause_time`` before picking the next
+    waypoint.
+    """
+
+    def __init__(self, area_width=100.0, area_height=100.0,
+                 speed_range=(1.0, 5.0), pause_time=0.0, seed=None,
+                 time_step=1.0):
+        MobilityModel.__init__(self, area_width, area_height, seed, time_step)
+        self.speed_range = (float(speed_range[0]), float(speed_range[1]))
+        self.pause_time = float(pause_time)
+        self.position = self.center()
+        self.target = None
+        self.speed = None
+        self._pause_left = 0.0
+
+    # ------------------------------------------------------------------
+    def random_speed(self):
+        """Uniformly distributed speed in ``speed_range`` (m/s)."""
+        low, high = self.speed_range
+        if high <= low:
+            return low
+        return float(self.rng.uniform(low, high))
+
+    def new_waypoint(self):
+        """Pick the next waypoint and the speed used to reach it."""
+        self.target = self.random_position()
+        self.speed = self.random_speed()
+        return self.target, self.speed
+
+    def step(self):
+        """Advance the node by one ``time_step`` and return its new position."""
+        if self.target is None:
+            self.new_waypoint()
+        if self._pause_left > 0.0:
+            self._pause_left -= self.time_step
+            return self.position
+        x, y = self.position
+        dx = self.target[0] - x
+        dy = self.target[1] - y
+        distance = math.hypot(dx, dy)
+        travel = self.speed * self.time_step
+        if distance <= travel or distance == 0.0:
+            # Waypoint reached: sit on it and choose the next one.
+            self.position = self.target
+            self.new_waypoint()
+            self._pause_left = self.pause_time
+        else:
+            self.position = (x + dx / distance * travel,
+                             y + dy / distance * travel)
+        return self.position
+
+    def generate_trace(self, start_pos=None, num_steps=100):
+        """Return ``num_steps`` successive ``(x, y)`` positions."""
+        self.position = (tuple(float(c) for c in start_pos)
+                         if start_pos is not None else self.center())
+        self.target = None
+        self.speed = None
+        self._pause_left = 0.0
+        return [self.step() for _ in range(int(num_steps))]
+
+    @staticmethod
+    def trace_distance(trace):
+        """Total distance walked along a trace (metres)."""
+        return sum(math.hypot(b[0] - a[0], b[1] - a[1])
+                   for a, b in zip(trace, trace[1:]))

+"""Radio propagation models.
+
+The models follow Mininet-WiFi (Fontes et al., CNSM 2015): a free-space (Friis)
+reference loss extended by a log-distance exponent that captures indoor/outdoor
+attenuation.  All powers are in dBm, all gains and losses in dB.
+"""
+
+import math
+
+#: Speed of light in m/s, as used by Mininet-WiFi's propagation models.
+SPEED_OF_LIGHT = 3e8
+
+#: Default log-distance path loss exponent (free space = 2, indoor >= 3).
+DEFAULT_PATH_LOSS_EXP = 3.0
+
+#: Default reference distance d0 in metres.
+DEFAULT_REF_DISTANCE_M = 1.0
+
+
+def ghz_to_hz(freq_ghz):
+    """Convert a frequency in GHz to Hz."""
+    return float(freq_ghz) * 1e9
+
+
+def channel_to_freq_hz(channel, band_ghz=2.4):
+    """Centre frequency of an 802.11 channel (2.4GHz band) in Hz."""
+    if band_ghz == 2.4:
+        return (2407 + 5 * int(channel)) * 1e6
+    return (5000 + 5 * int(channel)) * 1e6
+
+
+def friis_free_space_loss(freq_hz, distance_m):
+    """Free-space path loss in dB: ``20*log10(4*pi*d*f/c)``."""
+    if distance_m <= 0:
+        return 0.0
+    return 20.0 * math.log10(4.0 * math.pi * distance_m * freq_hz / SPEED_OF_LIGHT)
+
+
+def log_distance_path_loss(freq_hz, distance_m,
+                           path_loss_exp=DEFAULT_PATH_LOSS_EXP,
+                           ref_distance_m=DEFAULT_REF_DISTANCE_M):
+    """Log-distance path loss in dB.
+
+    ``PL(d) = PL(d0) + 10 * n * log10(d / d0)`` where ``PL(d0)`` is the Friis
+    free-space loss at the reference distance ``d0``.
+    """
+    pl0 = friis_free_space_loss(freq_hz, ref_distance_m)
+    if distance_m <= ref_distance_m:
+        return pl0
+    return pl0 + 10.0 * path_loss_exp * math.log10(distance_m / ref_distance_m)
+
+
+def received_signal_strength(tx_power_dbm, freq_hz, distance_m,
+                            path_loss_exp=DEFAULT_PATH_LOSS_EXP,
+                            ref_distance_m=DEFAULT_REF_DISTANCE_M,
+                            tx_gain_dbi=0.0, rx_gain_dbi=0.0):
+    """Received signal strength in dBm: ``tx_power + gains - path_loss``."""
+    path_loss = log_distance_path_loss(freq_hz, distance_m, path_loss_exp,
+                                       ref_distance_m)
+    return tx_power_dbm + tx_gain_dbi + rx_gain_dbi - path_loss
+
+
+#: Backwards-compatible alias.
+rssi = received_signal_strength
+
+
+def max_range(tx_power_dbm, min_rssi_dbm, freq_hz,
+              path_loss_exp=DEFAULT_PATH_LOSS_EXP,
+              ref_distance_m=DEFAULT_REF_DISTANCE_M,
+              tx_gain_dbi=0.0, rx_gain_dbi=0.0):
+    """Distance in metres at which the RSSI drops to ``min_rssi_dbm``.
+
+    Inverts :func:`received_signal_strength` for ``d``:
+    ``d = d0 * 10 ** ((budget - PL(d0)) / (10 * n))``.
+    """
+    budget = tx_power_dbm + tx_gain_dbi + rx_gain_dbi - min_rssi_dbm
+    pl0 = friis_free_space_loss(freq_hz, ref_distance_m)
+    if path_loss_exp <= 0:
+        return float("inf")
+    exponent = (budget - pl0) / (10.0 * path_loss_exp)
+    return ref_distance_m * (10.0 ** exponent)
+
+
+def path_loss_at_range(tx_power_dbm, min_rssi_dbm, tx_gain_dbi=0.0,
+                       rx_gain_dbi=0.0):
+    """Path loss budget (dB) available before the RSSI floor is reached."""
+    return tx_power_dbm + tx_gain_dbi + rx_gain_dbi - min_rssi_dbm
+
+
+def snr_db(rssi_dbm, noise_floor_dbm=-91.0):
+    """Signal-to-noise ratio in dB for a given RSSI."""
+    return rssi_dbm - noise_floor_dbm

+"""Signal-aware wireless links."""
+
+from ..link import TCLink
+
+
+class WirelessLink(TCLink):
+    """A link whose bandwidth follows the measured RSSI.
+
+    Rates come from a simplified 802.11g rate table::
+
+        RSSI >= -50 dBm -> 54 Mbps      RSSI >= -70 dBm -> 24 Mbps
+        RSSI >= -60 dBm -> 48 Mbps      RSSI >= -75 dBm -> 12 Mbps
+        RSSI >= -65 dBm -> 36 Mbps      RSSI >= -80 dBm ->  6 Mbps
+
+    Below -80 dBm the link cannot carry traffic (0 Mbps).
+    """
+
+    #: ``(rssi_threshold_dbm, bandwidth_mbps)`` in descending signal order.
+    RATE_TABLE = (
+        (-50.0, 54.0),
+        (-60.0, 48.0),
+        (-65.0, 36.0),
+        (-70.0, 24.0),
+        (-75.0, 12.0),
+        (-80.0, 6.0),
+    )
+
+    #: RSSI below which no rate is available.
+    MIN_RSSI_DBM = -80.0
+
+    def __init__(self, station, ap, rssi_dbm=None, bw=None, delay=0.0,
+                 loss=0.0, **params):
+        if rssi_dbm is None and hasattr(ap, "signal_strength"):
+            rssi_dbm = ap.signal_strength(station)
+        self.rssi_dbm = rssi_dbm
+        if bw is None:
+            bw = self.bandwidth_for_rssi(rssi_dbm)
+        TCLink.__init__(self, station, ap, bw=bw, delay=delay, loss=loss,
+                        **params)
+        self.station = station
+        self.ap = ap
+
+    # ------------------------------------------------------------------
+    @classmethod
+    def bandwidth_for_rssi(cls, rssi_dbm):
+        """Bandwidth in Mbps supported at the given RSSI."""
+        if rssi_dbm is None:
+            return 0.0
+        for threshold, rate in cls.RATE_TABLE:
+            if rssi_dbm >= threshold:
+                return rate
+        return 0.0
+
+    #: Readable alias.
+    rate_for_rssi = bandwidth_for_rssi
+
+    def update_rssi(self, rssi_dbm=None):
+        """Recompute the RSSI (and therefore the rate) after a station moved."""
+        if rssi_dbm is None and hasattr(self.ap, "signal_strength"):
+            rssi_dbm = self.ap.signal_strength(self.station)
+        self.rssi_dbm = rssi_dbm
+        self.bw = self.bandwidth_for_rssi(rssi_dbm)
+        return self.bw
+
+    @property
+    def is_usable(self):
+        return self.bw > 0.0
+
+    def __repr__(self):
+        return "<WirelessLink %s<->%s rssi=%s bw=%s>" % (
+            self.node1.name, self.node2.name, self.rssi_dbm, self.bw)

+"""Mininet-WiFi style network engine.
+
+:class:`MininetWifi` extends :class:`netsim.net.Mininet` with stations, access
+points and the association logic that links them: a station attaches to the
+access point that gives it the strongest signal, provided that signal is above
+the association floor.
+"""
+
+from .net import Mininet
+from .wifi.accesspoint import AccessPoint
+from .wifi.station import Station
+from .wifi.wireless_link import WirelessLink
+
+
+class MininetWifi(Mininet):
+    """Wireless-capable network engine."""
+
+    def __init__(self, topo=None, station=Station, accessPoint=AccessPoint,
+                 wirelessLink=WirelessLink, min_rssi_dbm=None, **params):
+        self.stations = []
+        self.accessPoints = []
+        self.station = station
+        self.accessPoint = accessPoint
+        self.wirelessLink = wirelessLink
+        self.min_rssi_dbm = min_rssi_dbm
+        Mininet.__init__(self, topo=topo, **params)
+
+    # ------------------------------------------------------------------
+    # node creation
+    # ------------------------------------------------------------------
+    def addStation(self, name, cls=None, **params):
+        """Add a WiFi station (also registered as a host)."""
+        cls = cls or self.station
+        sta = cls(name, **params)
+        self.stations.append(sta)
+        self.hosts.append(sta)
+        self._register(sta)
+        return sta
+
+    def addAccessPoint(self, name, cls=None, **params):
+        """Add an access point (also registered as a switch)."""
+        cls = cls or self.accessPoint
+        ap = cls(name, **params)
+        self.accessPoints.append(ap)
+        self.switches.append(ap)
+        self._register(ap)
+        for ctrl in self.controllers:
+            ap.attach(ctrl)

+    addSta = addStation

+        if min_rssi_dbm is not None:
+            return min_rssi_dbm
+        if self.min_rssi_dbm is not None:
+            return self.min_rssi_dbm
+        return ap.min_rssi_dbm
+

+        """Return ``(ap, rssi)`` for the strongest in-range AP, else ``(None, None)``."""
+        best, best_rssi = None, None
+        for ap in self.accessPoints:
+            rssi = ap.signal_strength(station)
+            if rssi < self._floor(ap, min_rssi_dbm):
+                continue
+            if best_rssi is None or rssi > best_rssi:
+                best, best_rssi = ap, rssi
+        return best, best_rssi

+        if rssi_dbm is None and ap is not None:
+            rssi_dbm = ap.signal_strength(station)

+            station.rssi_dbm = rssi_dbm
+            station.wireless_link.update_rssi(rssi_dbm)
+            self.graph.add_edge(station.name, ap.name,

+            return station.wireless_link
+        self._drop_wireless_link(station)

+            return None

+        self._drop_wireless_link(station)
+        return station.disassociate()

+        link = station.wireless_link
+        if link is None:

+        for sta in (stations if stations is not None else self.stations):
+            ap, rssi = self.best_ap(sta, min_rssi_dbm)
+            if ap is None:

+            else:
+                self.associate(sta, ap, rssi)
+        return self.count_associated()
+
+    def update_associations(self, min_rssi_dbm=None):
+        """Re-evaluate associations after movement; returns handovers observed."""
+        handovers = 0
+        for sta in self.stations:
+            previous = sta.associated_ap
+            ap, rssi = self.best_ap(sta, min_rssi_dbm)
+            if ap is None:
+                self.disassociate(sta)
+            else:
+                self.associate(sta, ap, rssi)
+                if previous is not None and previous is not ap:
+                    handovers += 1
+        return handovers

+        return sum(1 for sta in self.stations if sta.associated_ap is not None)

+    def associated_stations(self):
+        return [sta for sta in self.stations if sta.associated_ap is not None]

+    # mobility helpers
+    # ------------------------------------------------------------------
+    def association_trace(self, station, positions, min_rssi_dbm=None):
+        """Walk a station through ``positions``; return the serving AP per step.
+
+        Each entry is the AP name serving the station at that step, or ``None``
+        while the station is outside every AP's coverage.
+        """
+        serving = []
+        for pos in positions:
+            station.set_position(pos)
+            ap, rssi = self.best_ap(station, min_rssi_dbm)
+            self.associate(station, ap, rssi) if ap is not None \
+                else self.disassociate(station)
+            serving.append(ap.name if ap is not None else None)
+        return serving
+
+    @staticmethod
+    def count_handovers(serving):
+        """Number of times a station roams directly from one AP to another.
+
+        A handover is an association that *replaces* a live one, which is the
+        same event :attr:`netsim.wifi.station.Station.handover_count` records.
+        Losing coverage tears the association down, so ``ap1 -> None -> ap2`` is
+        a fresh association rather than a handover and is not counted.
+        """
+        handovers = 0
+        previous = None
+        for ap in serving:
+            if ap is not None and previous is not None and ap != previous:
+                handovers += 1
+            previous = ap
+        return handovers
+
+    @staticmethod
+    def coverage_percentage(serving):
+        """Percentage of steps during which the station had a serving AP."""
+        if not serving:
+            return 0.0
+        covered = sum(1 for ap in serving if ap is not None)
+        return 100.0 * covered / len(serving)
+
+    def total_nodes(self):
+        """Hosts (including stations) plus switches (including APs)."""
+        return len(self.hosts) + len(self.switches)
