+def fd_grads_wrt_params(model, inputs, grads_wrt_outputs, eps=1e-6):
+    """Finite difference estimate of grads wrt all model params."""
+    fd = []
+    for param in model.params:
+        grads = np.zeros_like(param)
+        for index in np.ndindex(*param.shape):
+            original = param[index]
+            param[index] = original + eps
+            value_p = (model.fprop(inputs)[-1] * grads_wrt_outputs).sum()
+            param[index] = original - eps
+            value_m = (model.fprop(inputs)[-1] * grads_wrt_outputs).sum()
+            param[index] = original
+            grads[index] = (value_p - value_m) / (2. * eps)
+        fd.append(grads)
+    return fd

+def make_regression_data(rng, num_data=40, batch_size=10):
+    inputs = rng.normal(size=(num_data, 2))
+    targets = (inputs.sum(-1) * 0.5)[:, None]
+    return DataProvider(inputs, targets, batch_size=batch_size, rng=rng,
+                        shuffle_order=False)
+
+
+def make_model(rng, output_dim=1):
+    init_w = init.UniformInit(-0.5, 0.5, rng)
+    return MultipleLayerModel([
+        AffineLayer(2, 5, init_w, init_w),
+        SigmoidLayer(),
+        AffineLayer(5, output_dim, init_w, init_w),
+    ])
+
+
+def make_optimiser(rng=None, data_monitors=None, valid=True, **kwargs):
+    rng = rng if rng is not None else np.random.RandomState(0)
+    train_data = make_regression_data(rng)
+    valid_data = make_regression_data(rng) if valid else None
+    model = make_model(rng)
+    error = SumOfSquaredDiffsError()
+    rule = GradientDescentLearningRule(0.1)
+    return Optimiser(model, error, rule, train_data, valid_data,
+                     data_monitors, **kwargs), model, train_data, valid_data
+
