+# -- Poisson sampling -------------------------------------------------------

+def test_poisson_matches_the_distribution():
+    rng = random.Random(42)
+    draws = [poisson(1.0, rng) for _ in range(100000)]
+    mean = sum(draws) / len(draws)
+    variance = sum(d * d for d in draws) / len(draws) - mean * mean
+    # Poisson(1) has mean 1 and variance 1
+    assert abs(mean - 1.0) < 0.02
+    assert abs(variance - 1.0) < 0.05
+    assert min(draws) == 0
+    # P(k = 0) = P(k = 1) = e^-1 = 0.3679
+    assert abs(draws.count(0) / len(draws) - 0.3679) < 0.01
+    assert abs(draws.count(1) / len(draws) - 0.3679) < 0.01
+
+
+def test_poisson_is_reproducible_from_the_seed():
+    assert ([poisson(1.0, random.Random(7)) for _ in range(5)]
+            == [poisson(1.0, random.Random(7)) for _ in range(5)])
+    first = random.Random(1)
+    second = random.Random(1)
+    assert [poisson(1.0, first) for _ in range(50)] == \
+           [poisson(1.0, second) for _ in range(50)]

+"""MOA: Massive Online Analysis - ensembles and evaluation tasks.
+
+This module provides the parts of the MOA framework used to benchmark the
+online learners of this package:
+
+* :class:`OzaBagging` - online bagging with Poisson(1) resampling
+  (Oza & Russell, "Experimental comparisons of online and batch versions of
+  bagging and boosting", KDD 2001);
+* :class:`EvaluateInterleavedTestThenTrain` - the standard MOA evaluation
+  task: every example is first used to test the classifier and then to train
+  it, with accuracy snapshots taken at a fixed sampling frequency;
+* :class:`PerformanceStats` - a small collector that reports the metrics of
+  several experiments in the ``EXPERIMENT_NAME: VALUE`` format expected by
+  ``run.sh``.
+"""

+# ---------------------------------------------------------------------------
+# Poisson sampling
+# ---------------------------------------------------------------------------
+
+
+def poisson(lam: float = 1.0, rng: Optional[random.Random] = None) -> int:
+    """Draw ``k ~ Poisson(lam)`` with Knuth's product-of-uniforms method.
+
+    ``k`` is the number of times a base learner of an online bagging ensemble
+    trains on the current example.  ``lam = 1`` reproduces the multiplicity of
+    a bootstrap sample in the limit of an infinite training set.
+
+    Args:
+        lam: Rate of the Poisson distribution.
+        rng: Source of randomness; a module level ``random`` is used when None.
+
+    Returns:
+        A non negative integer drawn from ``Poisson(lam)``.
+    """
+    if lam <= 0.0:
+        return 0
+    source = rng.random if rng is not None else random.random
+    limit = math.exp(-lam)
+    k = 0
+    product = 1.0
+    while True:
+        k += 1
+        product *= source()
+        if product <= limit:
+            break
+    return k - 1
+
+
+# ---------------------------------------------------------------------------
+# online bagging
+# ---------------------------------------------------------------------------
+
+
+class OzaBagging(AbstractClassifier):
+    """Online bagging of Hoeffding Trees with Poisson(1) resampling.
+
+    Each of the ``num_learners`` base learners trains on every arriving example
+    ``k ~ Poisson(1)`` times, which simulates drawing a bootstrap replicate of
+    the stream.  Predictions are combined by a majority vote over the base
+    learners.
+
+    Args:
+        num_learners: Size ``N`` of the ensemble.
+        seed: Seed of the ``random.Random`` used for the Poisson draws.
+        num_classes: Number of classes (passed on to the base learners).
+        num_attributes: Number of attributes (passed on to the base learners).
+        grace_period: ``n_min`` of the base Hoeffding Trees.
+        split_confidence: ``delta`` of the base Hoeffding Trees.
+        tie_threshold: ``tau`` of the base Hoeffding Trees.
+        max_depth: Maximum depth of the base Hoeffding Trees.
+        num_bins: Bins of the numeric attribute observers.
+        lam: Rate of the Poisson resampling distribution (1.0 in Oza bagging).
+        base_learner_factory: Optional ``callable(index) -> classifier`` used
+            instead of the default Hoeffding Tree factory.
+    """
+
+    def __init__(self,
+                 num_learners: int = 10,
+                 seed: int = 42,
+                 num_classes: Optional[int] = None,
+                 num_attributes: Optional[int] = None,
+                 grace_period: int = 200,
+                 split_confidence: float = 1e-7,
+                 tie_threshold: float = 0.05,
+                 max_depth: int = 20,
+                 num_bins: int = 10,
+                 lam: float = 1.0,
+                 base_learner_factory: Optional[Callable[[int], object]] = None):
+        super().__init__()
+        self.num_learners = int(num_learners)
+        self.seed = int(seed)
+        self.num_classes = num_classes if num_classes else 0
+        self.num_attributes = num_attributes if num_attributes else 0
+        self.grace_period = int(grace_period)
+        self.split_confidence = float(split_confidence)
+        self.tie_threshold = float(tie_threshold)
+        self.max_depth = int(max_depth)
+        self.num_bins = int(num_bins)
+        self.lam = float(lam)
+        self.base_learner_factory = base_learner_factory
+
+        self.rng = random.Random(self.seed)
+        #: number of examples each base learner was trained on
+        self.learner_weights: List[float] = [0.0] * self.num_learners
+        self.learners: List[object] = [self._new_learner(index)
+                                       for index in range(self.num_learners)]
+
+    # -- construction ------------------------------------------------------
+    def _new_learner(self, index: int):
+        if self.base_learner_factory is not None:
+            return self.base_learner_factory(index)
+        return HoeffdingTree(num_attributes=self.num_attributes or None,
+                             num_classes=self.num_classes or None,
+                             grace_period=self.grace_period,
+                             split_confidence=self.split_confidence,
+                             tie_threshold=self.tie_threshold,
+                             max_depth=self.max_depth,
+                             num_bins=self.num_bins)
+
+    # -- training ----------------------------------------------------------
+    def train(self, instance: Instance):
+        """Train every base learner ``k ~ Poisson(lam)`` times on ``instance``."""
+        if instance.label is not None and instance.label + 1 > self.num_classes:
+            self.num_classes = instance.label + 1
+        if instance.num_attributes > self.num_attributes:
+            self.num_attributes = instance.num_attributes
+        self.instances_seen += 1
+        for index, learner in enumerate(self.learners):
+            k = poisson(self.lam, self.rng)
+            if k <= 0:
+                continue
+            self.learner_weights[index] += k
+            for _ in range(k):
+                learner.train(instance)
+
+    def partial_fit(self, instance: Instance):
+        """Alias for :meth:`train`."""
+        self.train(instance)
+
+    # -- prediction --------------------------------------------------------
+    def get_votes(self, instance: Instance) -> Dict[int, float]:
+        """Majority vote: one vote per base learner that has an opinion."""
+        votes: Dict[int, float] = {}
+        for learner in self.learners:
+            learner_votes = learner.get_votes(instance)
+            if not learner_votes:
+                continue
+            best_class = -1
+            best_weight = float("-inf")
+            for cls in sorted(learner_votes):
+                weight = learner_votes[cls]
+                if weight > best_weight:
+                    best_weight = weight
+                    best_class = cls
+            if best_class < 0:
+                continue
+            votes[best_class] = votes.get(best_class, 0.0) + 1.0
+        return votes
+
+    def get_soft_votes(self, instance: Instance) -> Dict[int, float]:
+        """Sum of the normalised class distributions of the base learners."""
+        votes: Dict[int, float] = {}
+        for learner in self.learners:
+            learner_votes = learner.get_votes(instance)
+            total = sum(learner_votes.values())
+            if total <= 0:
+                continue
+            for cls, weight in learner_votes.items():
+                votes[cls] = votes.get(cls, 0.0) + weight / total
+        return votes
+
+    def predict(self, instance: Instance) -> int:
+        votes = self.get_votes(instance)
+        if not votes:
+            return 0
+        best_class = -1
+        best_weight = float("-inf")
+        for cls in sorted(votes):
+            weight = votes[cls]
+            if weight > best_weight:
+                best_weight = weight
+                best_class = cls
+        return best_class if best_class >= 0 else 0
+
+    def predict_proba(self, instance: Instance) -> Dict[int, float]:
+        votes = self.get_votes(instance)
+        total = sum(votes.values())
+        if total <= 0:
+            return {}
+        return {cls: weight / total for cls, weight in votes.items()}
+
+    # -- introspection -----------------------------------------------------
+    def get_ensemble_size(self) -> int:
+        return len(self.learners)
+
+    def get_tree_sizes(self) -> List[Dict[str, int]]:
+        sizes = []
+        for learner in self.learners:
+            getter = getattr(learner, "get_tree_size", None)
+            sizes.append(getter() if getter is not None else {})
+        return sizes
+
+    def get_total_nodes(self) -> int:
+        return sum(size.get("total_nodes", 0) for size in self.get_tree_sizes())
+
+    def get_average_tree_nodes(self) -> float:
+        sizes = self.get_tree_sizes()
+        if not sizes:
+            return 0.0
+        return self.get_total_nodes() / len(sizes)
+
+    def reset(self):
+        self.instances_seen = 0
+        self.rng = random.Random(self.seed)
+        self.learner_weights = [0.0] * self.num_learners
+        self.learners = [self._new_learner(index)
+                         for index in range(self.num_learners)]
+
+    def __str__(self) -> str:  # pragma: no cover - debug helper
+        return ("OzaBagging(num_learners=%d, nodes=%d)"
+                % (len(self.learners), self.get_total_nodes()))
+

+# ---------------------------------------------------------------------------
+# evaluation task
+# ---------------------------------------------------------------------------
+
+
+class EvaluateInterleavedTestThenTrain:
+    """The MOA ``EvaluateInterleavedTestThenTrain`` task.
+
+    Every example of the stream is first classified (test) and then handed to
+    the learner (train), so no example is ever used for testing after it has
+    been learnt.  Accuracy is reported cumulatively and over a sliding window,
+    with a snapshot recorded every ``sample_frequency`` examples.
+
+    Args:
+        classifier: Learner implementing ``predict`` and ``train``.
+        stream: Stream generator to draw the examples from.
+        num_instances: Number of examples to process.
+        window_size: Length of the sliding accuracy window.
+        sample_frequency: Number of examples between two accuracy snapshots.
+        name: Optional label used in the returned metadata.
+    """
+
+    def __init__(self, classifier, stream: StreamGenerator,
+                 num_instances: int = 100000, window_size: int = 1000,
+                 sample_frequency: int = 10000, name: Optional[str] = None):
+        self.classifier = classifier
+        self.stream = stream
+        self.num_instances = int(num_instances)
+        self.window_size = int(window_size)
+        self.sample_frequency = max(1, int(sample_frequency))
+        self.name = name or type(classifier).__name__
+        self.evaluator = PrequentialEvaluator(window_size=self.window_size)
+        #: accuracy snapshots taken every ``sample_frequency`` examples
+        self.snapshots: List[Dict[str, float]] = []
+        self.result: Dict[str, object] = {}
+
+    # -- task --------------------------------------------------------------
+    def run(self) -> Dict[str, object]:
+        """Run the task and return the collected measurements."""
+        evaluator = self.evaluator
+        evaluator.reset()
+        self.snapshots = []
+        classifier = self.classifier
+        stream = self.stream
+        start = time.time()
+        for index in range(1, self.num_instances + 1):
+            instance = stream.next_instance()
+            prediction = classifier.predict(instance)
+            evaluator.add_result(prediction, instance.label)
+            classifier.train(instance)
+            if index % self.sample_frequency == 0:
+                self.snapshots.append({
+                    "instances": float(index),
+                    "cumulative_accuracy": evaluator.get_cumulative_accuracy(),
+                    "window_accuracy": evaluator.get_window_accuracy(),
+                })
+        elapsed = time.time() - start
+
+        self.result = {
+            "cumulative_accuracy": evaluator.get_cumulative_accuracy(),
+            "window_accuracy": evaluator.get_window_accuracy(),
+            "snapshots": self.snapshots,
+            "learning_curve": [(int(s["instances"]), s["window_accuracy"])
+                               for s in self.snapshots],
+            "metadata": self.get_metadata(elapsed),
+        }
+        return self.result
+
+    def get_metadata(self, elapsed: float = 0.0) -> Dict[str, object]:
+        """Describe the run: learner, stream, sizes and timing."""
+        metadata: Dict[str, object] = {
+            "task": type(self).__name__,
+            "name": self.name,
+            "classifier": type(self.classifier).__name__,
+            "stream": type(self.stream).__name__,
+            "num_instances": self.num_instances,
+            "instances_processed": self.evaluator.total_seen,
+            "correct": self.evaluator.total_correct,
+            "window_size": self.window_size,
+            "sample_frequency": self.sample_frequency,
+            "evaluation_time_seconds": elapsed,
+        }
+        if isinstance(self.classifier, OzaBagging):
+            metadata["ensemble_size"] = self.classifier.get_ensemble_size()
+            metadata["total_nodes"] = self.classifier.get_total_nodes()
+        else:
+            getter = getattr(self.classifier, "get_tree_size", None)
+            if getter is not None:
+                metadata["total_nodes"] = getter().get("total_nodes", 0)
+        return metadata
+
+    # -- convenience -------------------------------------------------------
+    def get_cumulative_accuracy(self) -> float:
+        return self.evaluator.get_cumulative_accuracy()
+
+    def get_window_accuracy(self) -> float:
+        return self.evaluator.get_window_accuracy()
+
+
+# ---------------------------------------------------------------------------
+# metric collection
+# ---------------------------------------------------------------------------
+
+
+class PerformanceStats:
+    """Collect the metrics of several experiments and report them uniformly.
+
+    Values are printed as ``EXPERIMENT_NAME: VALUE`` lines, integers without a
+    fractional part and floating point metrics with ``precision`` decimals.
+    """
+
+    def __init__(self, precision: int = 4):
+        self.precision = int(precision)
+        self.metrics: Dict[str, float] = {}
+        self.order: List[str] = []
+        self.metadata: Dict[str, Dict[str, object]] = {}
+
+    # -- collection --------------------------------------------------------
+    def add(self, name: str, value: float,
+            metadata: Optional[Dict[str, object]] = None):
+        """Record a single metric (overwriting a previous value)."""
+        if name not in self.metrics:
+            self.order.append(name)
+        self.metrics[name] = value
+        if metadata:
+            self.metadata[name] = dict(metadata)
+        return self
+
+    def add_all(self, metrics: Dict[str, float]):
+        for name in metrics:
+            self.add(name, metrics[name])
+        return self
+
+    def add_result(self, prefix: str, result: Dict[str, object],
+                   metrics: Sequence[str] = ("cumulative_accuracy",
+                                             "window_accuracy")):
+        """Record selected metrics of an evaluation task result.
+
+        ``prefix`` is combined with the upper cased metric name, e.g.
+        ``add_result("MOA_ENSEMBLE", result, ("cumulative_accuracy",))``
+        records ``MOA_ENSEMBLE_CUMULATIVE_ACCURACY``.
+        """
+        metadata = result.get("metadata") if isinstance(result, dict) else None
+        for metric in metrics:
+            if metric in result:
+                name = "%s_%s" % (prefix, metric.upper()) if prefix else metric.upper()
+                self.add(name, float(result[metric]),  # type: ignore[arg-type]
+                         metadata if isinstance(metadata, dict) else None)
+        return self
+
+    # -- access ------------------------------------------------------------
+    def get(self, name: str, default: Optional[float] = None) -> Optional[float]:
+        return self.metrics.get(name, default)
+
+    def names(self) -> List[str]:
+        return list(self.order)
+
+    def as_dict(self) -> Dict[str, float]:
+        return {name: self.metrics[name] for name in self.order}
+
+    def __contains__(self, name: str) -> bool:
+        return name in self.metrics
+
+    def __len__(self) -> int:
+        return len(self.metrics)
+
+    # -- reporting ---------------------------------------------------------
+    def format_value(self, value: float) -> str:
+        if isinstance(value, bool):
+            return str(int(value))
+        if isinstance(value, int):
+            return str(value)
+        if isinstance(value, float) and value == int(value) and abs(value) >= 1000:
+            # node counts and instance counts read better without decimals
+            return str(int(value))
+        return "%.*f" % (self.precision, value)
+
+    def to_lines(self) -> List[str]:
+        """Return the ``EXPERIMENT_NAME: VALUE`` lines of every metric."""
+        return ["%s: %s" % (name, self.format_value(self.metrics[name]))
+                for name in self.order]
+
+    def report(self) -> str:
+        return "\n".join(self.to_lines())
+
+    def print_report(self):
+        for line in self.to_lines():
+            print(line)
+
+    def __str__(self) -> str:  # pragma: no cover - debug helper
+        return self.report()
+
+

+# experiments
+# ---------------------------------------------------------------------------

+def run_moa_experiments(num_instances: int = 100000, num_learners: int = 10,
+                        seed: int = 42, window_size: int = 1000,
+                        sample_frequency: int = 10000) -> Dict[str, float]:
+    """MOA_ENSEMBLE_ACCURACY / MOA_WINDOW_ACCURACY on the waveform stream."""
+    from streams import WaveformGenerator

+    stream = WaveformGenerator(seed=seed)

+        "MOA_ENSEMBLE_ACCURACY": float(result["cumulative_accuracy"]),

+++"""CVFDT: Concept-adapting Very Fast Decision Tree.
+++
+++Implementation of the time-changing stream learner of Hulten, Spencer &
+++Domingos, "Mining Time-Changing Data Streams", KDD 2001.
+++
+++CVFDT keeps the VFDT growth procedure but adds the machinery needed to track
+++a moving concept:
+++
+++* a FIFO window ``W`` of the ``w`` most recent examples.  When an example
+++  leaves the window its contribution is *forgotten*, i.e. subtracted from the
+++  sufficient statistics of every node it updated when it arrived;
+++* internal (split) nodes keep sufficient statistics as well, so the tree can
+++  periodically ask whether the attribute it split on is still the best one
+++  (``CheckSplitValidity``);
+++* whenever the current split is no longer the best choice an *alternate*
+++  subtree is grown in the background.  Alternates learn from the same stream
+++  and are periodically compared against the subtree they shadow; a more
+++  accurate alternate replaces it, a hopeless one is pruned.
+++"""
+++
+++from collections import deque
+++import math
+++from typing import Deque, Dict, List, Optional, Sequence, Tuple

+++                                 NumericAttributeObserver)

+++from hoeffding_tree import (HoeffdingTree, HoeffdingTreeLeaf, HoeffdingTreeNode,
+++                            HoeffdingTreeSplit, SplitSuggestion,

+++# statistics helpers shared by leaves and internal nodes

+++def _learn_stats(node, instance: Instance, tree: Optional["CVFDT"],
+++                 weight: float = 1.0):
+++    """Add ``instance`` to the sufficient statistics held by ``node``."""
+++    label = instance.label
+++    node.class_counts[label] = node.class_counts.get(label, 0.0) + weight
+++    node.weight_seen += weight
+++    observers = node.attribute_observers
+++    for attr_index, value in enumerate(instance.features):
+++        observer = observers.get(attr_index)
+++        if observer is None:
+++            if tree is not None:
+++                observer = tree.new_attribute_observer(attr_index)
+++            else:
+++                observer = NumericAttributeObserver()
+++            observers[attr_index] = observer
+++        observer.observe(value, label, weight)
+++
+++
+++def _forget_stats(node, instance: Instance, weight: float = 1.0):
+++    """Subtract ``instance`` from the sufficient statistics held by ``node``."""
+++    label = instance.label
+++    remaining = node.class_counts.get(label, 0.0) - weight
+++    if remaining > 0.0:
+++        node.class_counts[label] = remaining
+++    else:
+++        node.class_counts.pop(label, None)
+++    node.weight_seen = max(0.0, node.weight_seen - weight)
+++    observers = node.attribute_observers
+++    if not observers:
+++        return
+++    for attr_index, value in enumerate(instance.features):
+++        observer = observers.get(attr_index)
+++        if observer is not None:
+++            # negative weight decrements the stored counts
+++            observer.observe(value, label, -weight)

+++def gain_of_existing_split(observer: AttributeObserver, num_classes: int,
+++                           split_value: float) -> float:
+++    """Information gain the *current* split of a node would have today.
+++
+++    The candidate cut points of a :class:`NumericAttributeObserver` are the
+++    ``num_bins - 1`` bin boundaries; the gain of the existing split is the gain
+++    of the cut point closest to ``split_value``, recomputed from the statistics
+++    of the examples that are currently inside the sliding window.
+++    """
+++    from classifiers import compute_info_gain
+++
+++    if isinstance(observer, NominalAttributeObserver):
+++        gain, _value, _dists = observer.get_best_split_suggestion(num_classes)
+++        return gain
+++    if not isinstance(observer, NumericAttributeObserver):
+++        return 0.0
+++    if observer.total_count <= 0 or observer.max_val <= observer.min_val:
+++        return 0.0
+++
+++    n_classes = max(2, num_classes)
+++    pre_counts = [0.0] * n_classes
+++    for cls, count in observer.total_per_class.items():
+++        if cls < n_classes:
+++            pre_counts[cls] = count
+++
+++    span = observer.max_val - observer.min_val
+++    left_counts = [0.0] * n_classes
+++    best_distance = float("inf")
+++    gain_at_split = 0.0
+++    for i in range(observer.num_bins - 1):
+++        for cls, count in observer.bin_counts[i].items():
+++            if cls < n_classes:
+++                left_counts[cls] += count
+++        cut_point = observer.min_val + (i + 1) * span / observer.num_bins
+++        distance = abs(cut_point - split_value)
+++        if distance > best_distance:
+++            break
+++        right_counts = [pre_counts[c] - left_counts[c] for c in range(n_classes)]
+++        if sum(left_counts) > 0 and sum(right_counts) > 0:
+++            best_distance = distance
+++            gain_at_split = compute_info_gain(pre_counts,
+++                                              [left_counts[:], right_counts])
+++        else:
+++            best_distance = distance
+++            gain_at_split = 0.0
+++    return gain_at_split

+++class CVFDTLeaf(HoeffdingTreeLeaf):
+++    """A learning leaf that can also *forget* examples leaving the window."""
+++
+++    def __init__(self, depth: int = 0,
+++                 class_counts: Optional[Dict[int, float]] = None,
+++                 tree: Optional["CVFDT"] = None):
+++        super().__init__(depth=depth, class_counts=class_counts, tree=tree)
+++        # Examples seen since the last split attempt.  This counter only ever
+++        # grows (forgetting does not cancel split attempts), which is what the
+++        # CVFDT pseudo code means by "the number of examples seen at l".
+++        self.examples_since_split_attempt: float = 0.0
+++
+++    def learn_from_instance(self, instance: Instance, tree=None,
+++                            weight: float = 1.0):
+++        if tree is not None:
+++            self._tree = tree
+++        _learn_stats(self, instance, self._tree, weight)
+++        self.examples_since_split_attempt += weight
+++
+++    def forget(self, instance: Instance, weight: float = 1.0):
+++        """Undo the effect of ``instance`` on this leaf."""
+++        _forget_stats(self, instance, weight)
+++        if self.weight_seen_at_last_split_attempt > self.weight_seen:
+++            self.weight_seen_at_last_split_attempt = self.weight_seen
+++
+++    def reset_split_attempt_counter(self):
+++        self.weight_seen_at_last_split_attempt = self.weight_seen
+++        self.examples_since_split_attempt = 0.0

+++    def should_attempt_split(self, grace_period: float) -> bool:
+++        return self.examples_since_split_attempt >= grace_period

+++    def __init__(self,
+++                 num_attributes: Optional[int] = None,
+++                 num_classes: Optional[int] = None,

+++                 tie_threshold: float = 0.05,

+++        self.num_classes = num_classes if num_classes else 0

+++        self.tie_threshold = float(tie_threshold)

+++    def _observe_instance_meta(self, instance: Instance):
+++        if instance.num_attributes > self.num_attributes:
+++            self.num_attributes = instance.num_attributes
+++        if instance.label is not None and instance.label + 1 > self.num_classes:
+++            self.num_classes = instance.label + 1

+++    def train(self, instance: Instance):
+++        """Train on one example: window maintenance, growth, periodic checks."""
+++        self._observe_instance_meta(instance)
+++        self.instances_seen += 1
+++
+++        # 1. make room in the window, forgetting the oldest example
+++        if len(self.window) >= self.window_size:
+++            old_instance, old_nodes = self.window.popleft()
+++            self._forget_instance(old_instance, old_nodes)

+++        # 2. register the example first: the list of touched nodes is shared
+++        #    with the window so that a split happening while this very example
+++        #    is being learnt also updates its path.
+++        touched: List[HoeffdingTreeNode] = []
+++        self.window.append((instance, touched))
+++        label = instance.label
+++        self.window_class_counts[label] = self.window_class_counts.get(label, 0.0) + 1.0
+++
+++        # 3. grow the main tree (and every alternate on the path)
+++        self._learn_instance(instance, touched)
+++
+++        # 4. every ``check_frequency`` examples revisit the split decisions
+++        if self.check_frequency > 0 and self.instances_seen % self.check_frequency == 0:
+++            self.check_split_validity()

+++        self.train(instance)

+++    def _forget_instance(self, instance: Instance,
+++                         nodes: Sequence[HoeffdingTreeNode]):
+++        label = instance.label
+++        remaining = self.window_class_counts.get(label, 0.0) - 1.0
+++        if remaining > 0.0:
+++            self.window_class_counts[label] = remaining
+++        else:
+++            self.window_class_counts.pop(label, None)
+++        for node in nodes:
+++            if node.alive:
+++                node.forget(instance)

+++    def _learn_instance(self, instance: Instance,
+++                        touched: Optional[List[HoeffdingTreeNode]] = None
+++                        ) -> List[HoeffdingTreeNode]:

+++
+++        Returns the list of nodes whose statistics were updated so that the
+++        example can be forgotten later on.
+++        """
+++        if touched is None:
+++            touched = []
+++        self._learn_subtree(self.root, instance, touched, owner=None,
+++                            owner_branch=-1, alt_owner=None, alt_index=-1,
+++                            allow_alternates=True)
+++        return touched
+++
+++    def _learn_subtree(self, root: HoeffdingTreeNode, instance: Instance,
+++                       touched: List[HoeffdingTreeNode],
+++                       owner: Optional[CVFDTSplit], owner_branch: int,
+++                       alt_owner: Optional[CVFDTSplit], alt_index: int,
+++                       allow_alternates: bool):
+++        """Route ``instance`` through one (sub)tree, learning on the way.
+++
+++        ``owner``/``owner_branch`` identify where ``root`` hangs in the main
+++        tree, ``alt_owner``/``alt_index`` identify it as an alternate subtree.
+++        Exactly one of the two is used to re-attach a new root after a split.
+++        """
+++        node = root
+++        parent: Optional[HoeffdingTreeSplit] = None
+++        parent_branch = -1
+++        while True:
+++            touched.append(node)
+++            if node.is_leaf():
+++                leaf = node  # type: CVFDTLeaf
+++                leaf.learn_from_instance(instance, self)
+++                if leaf.should_attempt_split(self.grace_period):
+++                    leaf.reset_split_attempt_counter()
+++                    if not (self.stop_on_pure_leaf
+++                            and leaf.observed_class_distribution_is_pure()):
+++                        new_node = self.attempt_to_split(leaf, parent, parent_branch)
+++                        if new_node is not None and parent is None:
+++                            self._reattach_root(new_node, owner, owner_branch,
+++                                                alt_owner, alt_index)
+++                return
+++
+++            split = node  # type: CVFDTSplit
+++            split.learn_from_instance(instance, self)
+++            if allow_alternates and split.alternate_trees:
+++                self._train_alternates(split, instance, touched)
+++
+++            branch = split.branch_for_instance(instance)
+++            child = split.get_child(branch)
+++            if child is None:
+++                child = self._new_leaf(depth=split.depth + 1)
+++                split.set_child(branch, child)
+++            parent = split
+++            parent_branch = branch
+++            node = child
+++
+++    def _reattach_root(self, new_root: HoeffdingTreeNode,
+++                       owner: Optional[CVFDTSplit], owner_branch: int,
+++                       alt_owner: Optional[CVFDTSplit], alt_index: int):
+++        """Install ``new_root`` where the previous subtree root was hanging."""
+++        if alt_owner is not None and 0 <= alt_index < len(alt_owner.alternate_trees):
+++            alt_owner.alternate_trees[alt_index] = new_root
+++        elif owner is not None:
+++            owner.set_child(owner_branch, new_root)
+++        else:
+++            self.root = new_root
+++
+++    def _train_alternates(self, split: CVFDTSplit, instance: Instance,
+++                          touched: List[HoeffdingTreeNode]):
+++        """Grow every alternate of ``split`` and score it against the main."""
+++        label = instance.label
+++        main_pred = self._subtree_predict(split, instance)
+++        split.main_seen += 1.0
+++        if main_pred == label:
+++            split.main_correct += 1.0
+++        for index in range(len(split.alternate_trees)):
+++            alt_root = split.alternate_trees[index]
+++            alt_pred = self._subtree_predict(alt_root, instance)
+++            split.alt_seen[index] += 1.0
+++            if alt_pred == label:
+++                split.alt_correct[index] += 1.0
+++            # alternates never spawn alternates of their own
+++            self._learn_subtree(alt_root, instance, touched, owner=None,
+++                                owner_branch=-1, alt_owner=split,
+++                                alt_index=index, allow_alternates=False)
+++
+++    def attempt_to_split(self, leaf: CVFDTLeaf,
+++                         parent: Optional[HoeffdingTreeSplit],
+++                         parent_branch: int) -> Optional[HoeffdingTreeSplit]:
+++        """Try to turn ``leaf`` into a split node; return the new node or None."""
+++        if leaf.depth >= self.max_depth:
+++            return None
+++        suggestions = compute_split_suggestions(leaf, self.num_classes)
+++        chosen, _epsilon = select_split_suggestion(
+++            suggestions, self.num_classes, leaf.weight_seen,
+++            self.split_confidence, self.tie_threshold)
+++        if chosen is None:
+++            return None
+++
+++        split_node = self._new_split_node(chosen, depth=leaf.depth)
+++        num_branches = max(2, len(chosen.post_split_dists))
+++        for branch in range(num_branches):
+++            split_node.set_child(branch, self._new_leaf(depth=leaf.depth + 1))
+++        leaf.alive = False
+++        if parent is not None:
+++            parent.set_child(parent_branch, split_node)
+++        self.num_splits_performed += 1
+++        # Re-play the window examples that belonged to ``leaf`` so that the new
+++        # subtree carries exactly the statistics of the current window.
+++        self._reseed_after_split(leaf, split_node)
+++        return split_node
+++
+++    def _reseed_after_split(self, leaf: CVFDTLeaf, split_node: CVFDTSplit):
+++        """Move ``leaf``'s window examples into the freshly created subtree."""
+++        for _instance, nodes in self.window:
+++            for position, node in enumerate(nodes):
+++                if node is leaf:
+++                    branch = split_node.branch_for_instance(_instance)
+++                    child = split_node.get_child(branch)
+++                    if child is None:
+++                        child = self._new_leaf(depth=split_node.depth + 1)
+++                        split_node.set_child(branch, child)
+++                    split_node.learn_from_instance(_instance, self)
+++                    child.learn_from_instance(_instance, self)
+++                    nodes[position] = split_node
+++                    nodes.append(child)
+++                    break
+++        # the replayed examples must not immediately trigger another split
+++        for child in split_node.children_nodes():
+++            if child.is_leaf():
+++                child.reset_split_attempt_counter()  # type: ignore[attr-defined]
+++
+++    def _seed_alternate(self, split: CVFDTSplit, alternate: CVFDTLeaf):
+++        """Give a new alternate the window statistics of the node it shadows."""
+++        for _instance, nodes in self.window:
+++            for node in nodes:
+++                if node is split:
+++                    alternate.learn_from_instance(_instance, self)
+++                    nodes.append(alternate)
+++                    break
+++        alternate.reset_split_attempt_counter()

+++    # -- CheckSplitValidity ------------------------------------------------
+++    def check_split_validity(self):
+++        """Re-examine every split, start alternates and promote/prune them."""
+++        self.num_split_checks += 1
+++        stack: List[Tuple[HoeffdingTreeNode, Optional[CVFDTSplit], int]] = [
+++            (self.root, None, -1)]
+++        while stack:
+++            node, parent, branch = stack.pop()
+++            if node.is_leaf():
+++                continue
+++            split = node  # type: CVFDTSplit
+++            replaced = self._test_alternates(split, parent, branch)
+++            if replaced is not None:
+++                # the alternate took over: revisit the promoted subtree later
+++                stack.append((replaced, parent, branch))
+++                continue
+++            self._check_node_split(split)
+++            for child_branch, child in enumerate(split.children):
+++                if child is not None:
+++                    stack.append((child, split, child_branch))
+++
+++    def _check_node_split(self, split: CVFDTSplit):
+++        """Start an alternate when the current split decision is stale.
+++
+++        The split of a node is stale when some other candidate split of the
+++        window statistics is better than the split currently installed, by more
+++        than the Hoeffding bound (capped by the tie threshold).  This covers
+++        both a change of the best *attribute* (the case discussed in the CVFDT
+++        paper for nominal attributes) and a drifting *threshold* on the very
+++        same numeric attribute.
+++        """
+++        if len(split.alternate_trees) >= self.max_alternates:
+++            return
+++        if split.weight_seen < self.grace_period:
+++            return
+++        suggestions = compute_split_suggestions(split, self.num_classes)
+++        if not suggestions:
+++            return
+++        best = suggestions[-1]
+++        if best.merit <= 0.0:
+++            return
+++
+++        observer = split.attribute_observers.get(split.split_attribute)
+++        if observer is None:
+++            return
+++        current_gain = gain_of_existing_split(observer, self.num_classes,
+++                                              split.split_value)
+++        range_val = math.log2(max(2, self.num_classes))
+++        epsilon = compute_hoeffding_bound(range_val, self.split_confidence,
+++                                          split.weight_seen)
+++        # Mirror the VFDT tie rule: while the bound is loose the improvement has
+++        # to beat it, once enough examples support the estimate (epsilon below
+++        # the tie threshold) any material improvement is trusted.
+++        required = epsilon if epsilon >= self.tie_threshold else self.stale_margin
+++        if (best.merit - current_gain) <= required:
+++            return
+++
+++        # the installed split is no longer the best one -> grow an alternate
+++        alternate = self._new_leaf(depth=split.depth)
+++        split.add_alternate(alternate)
+++        self._seed_alternate(split, alternate)
+++        split.main_correct = 0.0
+++        split.main_seen = 0.0
+++        self.num_alternates_started += 1
+++
+++    def _test_alternates(self, split: CVFDTSplit,
+++                         parent: Optional[CVFDTSplit],
+++                         parent_branch: int) -> Optional[HoeffdingTreeNode]:
+++        """Promote a winning alternate, prune hopeless ones.
+++
+++        Returns the promoted subtree root when a replacement happened.
+++        """
+++        if not split.alternate_trees:
+++            return None
+++        main_seen = split.main_seen
+++        main_acc = (split.main_correct / main_seen) if main_seen > 0 else 0.0
+++        best_index = -1
+++        best_acc = main_acc
+++        for index in range(len(split.alternate_trees)):
+++            seen = split.alt_seen[index]
+++            if seen < self.min_alt_examples:
+++                continue
+++            acc = split.alt_correct[index] / seen
+++            if acc > best_acc:
+++                best_acc = acc
+++                best_index = index
+++        if best_index >= 0:
+++            return self._promote_alternate(split, best_index, parent,
+++                                           parent_branch)
+++        # prune alternates that had enough time and still lose
+++        for index in range(len(split.alternate_trees) - 1, -1, -1):
+++            if split.alt_seen[index] >= self.max_alt_examples:
+++                dropped = split.drop_alternate(index)
+++                if dropped is not None:
+++                    self._mark_dead(dropped)
+++                    self.num_alternates_pruned += 1
+++        return None
+++
+++    def _promote_alternate(self, split: CVFDTSplit, index: int,
+++                           parent: Optional[CVFDTSplit],
+++                           parent_branch: int) -> HoeffdingTreeNode:
+++        """Replace ``split``'s subtree by its ``index``-th alternate."""
+++        alternate = split.drop_alternate(index)
+++        assert alternate is not None
+++        self._mark_dead(split)          # kills the old subtree + other alternates
+++        alternate.alive = True
+++        self._refresh_depths(alternate, split.depth)
+++        if parent is None:
+++            self.root = alternate
+++        else:
+++            parent.set_child(parent_branch, alternate)
+++        self.num_alternates_promoted += 1
+++        return alternate
+++
+++    def _mark_dead(self, node: HoeffdingTreeNode):
+++        stack = [node]
+++        while stack:
+++            current = stack.pop()
+++            current.alive = False
+++            for child in current.children_nodes():
+++                stack.append(child)
+++            for alt in getattr(current, "alternate_trees", ()):  # type: ignore[arg-type]
+++                stack.append(alt)
+++
+++    def _refresh_depths(self, node: HoeffdingTreeNode, depth: int):
+++        stack = [(node, depth)]
+++        while stack:
+++            current, current_depth = stack.pop()
+++            current.depth = current_depth
+++            current.alive = True
+++            for child in current.children_nodes():
+++                stack.append((child, current_depth + 1))
+++            for alt in getattr(current, "alternate_trees", ()):  # type: ignore[arg-type]
+++                stack.append((alt, current_depth))

+++    def _subtree_predict(self, root: HoeffdingTreeNode,

+++            weight = votes[cls]
+++            if weight > best_weight:
+++                best_weight = weight
+++                best_class = cls

+++            child = node.get_child(branch)  # type: ignore[attr-defined]
+++            if child is None:
+++                break
+++            node = child

+++    def get_votes(self, instance: Instance) -> Dict[int, float]:

+++        best_weight = float("-inf")
+++        for cls in sorted(votes):

+++            if weight > best_weight:
+++                best_weight = weight
+++                best_class = cls

+++    def predict_proba(self, instance: Instance) -> Dict[int, float]:

+++        total = sum(votes.values())
+++        if total <= 0:
+++            return {}

+++
+++    # -- introspection -----------------------------------------------------
+++    def get_tree_size(self) -> Dict[str, int]:

++"""ADWIN: ADaptive WINdowing for concept drift detection.
++
++Implementation of the change detector of Bifet & Gavalda, "Learning from
++Time-Changing Data with Adaptive Windowing", SDM 2007.
++
++ADWIN keeps a variable length window of the most recent observations, stored in
++compressed form as an *exponential histogram*: a list of
++:class:`BucketRow` objects where row ``i`` holds buckets summarising ``2^i``
++observations each.  At most ``max_buckets_per_row`` buckets live in a row; when
++a row overflows its two oldest buckets are merged into a single bucket of the
++next row, so a window of ``n`` items needs only ``O(M log n)`` memory.
++
++After every insertion the window is scanned for a split point ``W = W0 . W1``
++such that the means of the two sub-windows differ by more than the statistical
++bound
++
++    epsilon = sqrt(2 * m * v * ln(2 ln n / delta')) + (2/3) * m * ln(2 ln n / delta')
++
++(the variance-sensitive bound of ADWIN2, with ``1/m = 1/n0 + 1/n1`` the
++harmonic mean of the sub-window lengths and ``v`` the observed variance) or,
++when ``use_variance_bound`` is disabled, the plain Hoeffding cut
++
++    epsilon = sqrt((1 / (2 m)) * ln(4 / delta'))     with delta' = delta / n.
++
++Whenever such a split exists the distribution has changed: the oldest bucket is
++dropped and the test is repeated, so the window shrinks to the portion of the
++stream that is consistent with the new concept.
++"""
++
++import math
++import os
++import sys
++from typing import Iterator, List, Optional
++
++_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
++if _ROOT not in sys.path:
++    sys.path.insert(0, _ROOT)
++
++__all__ = ["Bucket", "BucketRow", "ADWIN"]
++
++
++class Bucket:
++    """Summary statistics of one segment of the window.
++
++    Attributes:
++        total: Sum of the observations contained in the bucket.
++        variance: Sum of squared deviations from the bucket mean.
++        count: Number of observations summarised by the bucket.
++    """
++
++    __slots__ = ("total", "variance", "count", "previous", "next")
++
++    def __init__(self, total: float = 0.0, variance: float = 0.0,
++                 count: int = 0):
++        self.total = float(total)
++        self.variance = float(variance)
++        self.count = int(count)
++        self.previous: Optional["Bucket"] = None
++        self.next: Optional["Bucket"] = None
++
++    # -- statistics --------------------------------------------------------
++    @property
++    def mean(self) -> float:
++        if self.count <= 0:
++            return 0.0
++        return self.total / self.count
++
++    def clear(self):
++        self.total = 0.0
++        self.variance = 0.0
++        self.count = 0
++        self.previous = None
++        self.next = None
++
++    def copy(self) -> "Bucket":
++        return Bucket(self.total, self.variance, self.count)
++
++    # -- merging -----------------------------------------------------------
++    def merge(self, other: "Bucket") -> "Bucket":
++        """Merge the adjacent bucket ``other`` into this one.
++
++        The variance of the union of two samples follows from
++
++            V = V_a + V_b + n_a n_b (mean_a - mean_b)^2 / (n_a + n_b)
++
++        Returns ``self`` so that merges can be chained.
++        """
++        if other is None or other.count == 0:
++            return self
++        if self.count == 0:
++            self.total = other.total
++            self.variance = other.variance
++            self.count = other.count
++            return self
++        n_a = self.count
++        n_b = other.count
++        mean_diff = self.total / n_a - other.total / n_b
++        self.variance = (self.variance + other.variance
++                         + n_a * n_b * mean_diff * mean_diff / (n_a + n_b))
++        self.total += other.total
++        self.count = n_a + n_b
++        return self
++
++    @staticmethod
++    def merged(first: "Bucket", second: "Bucket") -> "Bucket":
++        """Return a new bucket holding the union of ``first`` and ``second``."""
++        return first.copy().merge(second)
++
++    def __repr__(self) -> str:  # pragma: no cover - debug helper
++        return ("Bucket(total=%.4f, variance=%.4f, count=%d)"
++                % (self.total, self.variance, self.count))
