   262	LONG EmacPsSetupBD(XEmacPs *EmacPsInstancePtr)
   263	{
   264	    /* Allocate Rx and Tx BD space each */
   265	    // TODO:
   266	    RxBdSpacePtr = &(bd_space[0]);
   267		TxBdSpacePtr = &(bd_space[0x10000]);
   268	
   269	    /*
   270	     * Setup RxBD space And Create the RxBD ring
   271	     */
   272	    BdRxPtr = (XEmacPs_Bd*)RxBdSpacePtr;
   273	    XEmacPs_Bd* CurPtr = BdRxPtr;
   274	    for(int i = 0; i < RXBD_CNT; i++){
   275	        XEmacPs_BdSetRxNew(CurPtr);
   276	        XEmacPs_BdSetAddressRx(CurPtr, kva2pa((uintptr_t)&rx_buffers[i]));
   277	        CurPtr++;
   278	    }
   279	
   280	    /*
   281	     * Setup TxBD space And Create the TxBD ring
   282	     */
   283	    BdTxPtr = (XEmacPs_Bd*)TxBdSpacePtr;
   284	    XEmacPs_BdSetStatus(BdTxPtr, XEMACPS_TXBUF_USED_MASK);
   285	    XEmacPs_BdSetAddressTx(BdTxPtr, kva2pa((uintptr_t)&tx_buffer));
   286	    XEmacPs_BdSetLast(BdTxPtr);
   287	    // At least 2 BDs is needed
   288	    XEmacPs_BdSetStatus(BdTxPtr + 1, XEMACPS_TXBUF_USED_MASK);
   289	
   290	    return XST_SUCCESS;
   291	}

   297	LONG EmacPsInitPhy(XEmacPs * EmacPsInstancePtr, u32 *Speed)
   298	{
   299		LONG Status;
   300		u16 PhyIdentity;
   301		u16 PhyIdentity2;
   302		u32 PhyAddr;
   303	
   304		/*
   305		 * Detect the PHY address
   306		 */
   307		PhyAddr = XEmacPsDetectPHY(EmacPsInstancePtr);
   308	
   309		if (PhyAddr >= 32) {
   310			EmacPsUtilErrorTrap("Error detect phy");
   311			return XST_FAILURE;
   312		}
   313	
   314	    /* phy id is stored in {phy reg 2, phy reg 3}
   315	     * PYNQ use RTL 8211E
   316	     */
   317		if (XEmacPs_PhyRead(EmacPsInstancePtr, PhyAddr, PHY_DETECT_REG1, &PhyIdentity) != XST_SUCCESS) {
   318			EmacPsUtilErrorTrap("Phy Read Error");
   319		}
   320	    if (XEmacPs_PhyRead(EmacPsInstancePtr, PhyAddr, PHY_DETECT_REG2, &PhyIdentity2) != XST_SUCCESS) {
   321			EmacPsUtilErrorTrap("Phy Read Error");
   322		}
   323		xil_printf("phy identity: %x\n\r", ((uint32_t) PhyIdentity << 16) | PhyIdentity2);
   324	
   325		if (PhyIdentity == PHY_ID_RTL) {
   326			Status = EmacPsUtilRtlPhyInit(EmacPsInstancePtr, Speed, PhyAddr);
   327		} else {
   328			EmacPsUtilErrorTrap("Unknown phy identity");
   329		}
   330	
   331		if (Status != XST_SUCCESS) {
   332			EmacPsUtilErrorTrap("Error setup phy loopback");
   333			return XST_FAILURE;
   334		}
   335	
   336		return XST_SUCCESS;
   337	}

   339	LONG EmacPsSetupDevice(XEmacPs *EmacPsInstancePtr)
   340	{
   341	    LONG Status;
   342	    XEmacPs_Config *Config = &xemacps_config; 
   343	
   344	    /*************************************/
   345	    /* Setup device for first-time usage */
   346	    /*************************************/
   347	
   348	    Status =
   349	        XEmacPs_CfgInitialize(EmacPsInstancePtr, Config, Config->BaseAddress);
   350	
   351	    if (Status != XST_SUCCESS) {
   352	        EmacPsUtilErrorTrap("Error in initialize");
   353	        return XST_FAILURE;
   354	    } else {
   355	        xil_printf("XEmacPs_CfgInitialize success! \r\n");
   356	    }
   357	
   358	    GemVersion = ((Xil_In32(Config->BaseAddress + 0xFC)) >> 16) & 0xFFF;
   359		xil_printf("GemVersion: %x\n\r", GemVersion);
   360	
   361		Platform = PLATFORM_VERSALEMU;
   362	
   363	    XEmacPsClkSetup(EmacPsInstancePtr);
   364	    xil_printf("XEmacPsClkSetup success! \r\n");
   365	
   366	    XEmacPs_Reset(EmacPsInstancePtr);
   367	
   368	    /*
   369	     * Set the MAC address
   370	     */
   371	    Status = XEmacPs_SetMacAddress(EmacPsInstancePtr, EmacPsMAC, 1);
   372	    if (Status != XST_SUCCESS) {
   373	        EmacPsUtilErrorTrap("Error setting MAC address");
   374	        return XST_FAILURE;
   375	    } else {
   376	        //char testMacAddr[6]
   377	        //void XEmacPs_GetMacAddress(EmacPsInstancePtr, , u8 Index)
   378			xil_printf("MAC address success! \r\n");
   379		}
   380	    /*
   381	     * Setup callbacks
   382	     */
   383	    Status = XEmacPs_SetHandler(
   384	        EmacPsInstancePtr, XEMACPS_HANDLER_DMASEND, (void *)XEmacPsSendHandler,
   385	        EmacPsInstancePtr);
   386	    Status |= XEmacPs_SetHandler(
   387	        EmacPsInstancePtr, XEMACPS_HANDLER_DMARECV, (void *)XEmacPsRecvHandler,
   388	        EmacPsInstancePtr);
   389	    Status |= XEmacPs_SetHandler(
   390	        EmacPsInstancePtr, XEMACPS_HANDLER_ERROR, (void *)XEmacPsErrorHandler,
   391	        EmacPsInstancePtr);
   392	    if (Status != XST_SUCCESS) {
   393	        EmacPsUtilErrorTrap("Error assigning handlers");
   394	        return XST_FAILURE;
   395	    } else {
   396			xil_printf("Assigning handlers success! \r\n");
   397		}
   398	
   399	    return XST_SUCCESS;
   400	}

   402	LONG EmacPsInit(XEmacPs *EmacPsInstancePtr)
   403	{
   404	    LONG status = XST_SUCCESS;
   405	    status = EmacPsSetupDevice(EmacPsInstancePtr);
   406	    if (status != XST_SUCCESS) {
   407	        xil_printf("setup device error!\n\r");
   408	    }
   409	
   410	    status = EmacPsSetupBD(EmacPsInstancePtr);
   411	    if (status != XST_SUCCESS) {
   412	        xil_printf("setup BD error!\n\r");
   413	    }
   414	
   415	    /*
   416	     * Set emacps to phy loopback or init phy for link
   417	     */
   418	    if (GemVersion == 2) {
   419	        XEmacPs_SetMdioDivisor(EmacPsInstancePtr, MDC_DIV_224);
   420	        EmacpsDelay(1);
   421	        // NOTE: for convenience you can enter loopback mode to test your code.
   422	        // EmacPsUtilEnterLocalLoopback(EmacPsInstancePtr);
   423	        // EmacPsUtilEnterLoopback(EmacPsInstancePtr, EMACPS_LOOPBACK_SPEED_1G);
   424	        u32 speed = EMACPS_LOOPBACK_SPEED_1G;
   425	        if (EmacPsInitPhy(EmacPsInstancePtr, &speed) != XST_SUCCESS) {
   426	            EmacPsUtilErrorTrap("init phy error!\n\r");
   427	        }
   428	        XEmacPs_SetOperatingSpeed(EmacPsInstancePtr, EMACPS_LOOPBACK_SPEED_1G);
   429	    }
   430	
   431	    /* clear any existed int status */
   432		XEmacPs_WriteReg(EmacPsInstancePtr->Config.BaseAddress, XEMACPS_ISR_OFFSET,
   433				   XEMACPS_IXR_ALL_MASK);
   434	
   435	    /* Enable TX and RX interrupts */
   436	    /* TODO: 
   437	     * NOTE: you can use XEmacPs_IntEnable and XEmacPs_IntDisable
   438	     *       enable these bits: 
   439	     *          (XEMACPS_IXR_TX_ERR_MASK | XEMACPS_IXR_RX_ERR_MASK |
   440	     *          (u32)XEMACPS_IXR_FRAMERX_MASK | (u32)XEMACPS_IXR_TXCOMPL_MASK)
   441	     */
   442	    
   443	    return status;
   444	}

   446	LONG EmacPsSend(XEmacPs *EmacPsInstancePtr, EthernetFrame *TxFrame, size_t length)
   447	{
   448	    LONG Status = XST_SUCCESS;
   449	
   450	    // Setup TxBD
   451	    // set length, clear tx used bit
   452	    XEmacPs_BdSetLength(BdTxPtr, length);
   453	    XEmacPs_BdClearTxUsed(BdTxPtr);
   454	
   455	    // remember to flush dcache
   456	    Xil_DCacheFlushRange(0, 64);
   457	
   458	    // set tx queue base
   459	    XEmacPs_SetQueuePtr(EmacPsInstancePtr, (UINTPTR)BdTxPtr, 0, XEMACPS_SEND);
   460	
   461	    /* Enable transmitter if not already enabled */
   462		if ((EmacPsInstancePtr->Options & (u32)XEMACPS_TRANSMITTER_ENABLE_OPTION)!=0x00000000U) {
   463			u32 Reg = XEmacPs_ReadReg(EmacPsInstancePtr->Config.BaseAddress,
   464						XEMACPS_NWCTRL_OFFSET);
   465			if ((!(Reg & XEMACPS_NWCTRL_TXEN_MASK))==TRUE) {
   466				XEmacPs_WriteReg(EmacPsInstancePtr->Config.BaseAddress,
   467						   XEMACPS_NWCTRL_OFFSET,
   468					   Reg | (u32)XEMACPS_NWCTRL_TXEN_MASK);
   469			}
   470		}
   471	
   472		// start transmit
   473	    XEmacPs_Transmit(EmacPsInstancePtr);
   474	
   475	    return Status;
   476	}

   478	LONG EmacPsWaitSend(XEmacPs *EmacPsInstancePtr)
   479	{
   480	    while(!XEmacPs_BdIsTxUsed(BdTxPtr)) {
   481			Xil_DCacheFlushRange(0, 64);
   482	    }
   483	    return XST_SUCCESS;
   484	}

   486	LONG EmacPsRecv(XEmacPs *EmacPsInstancePtr, EthernetFrame *RxFrame, int num_packet)
   487	{
   488	    LONG Status = XST_SUCCESS;
   489	
   490	    /* disable receiver */
   491		if ((EmacPsInstancePtr->Options & XEMACPS_RECEIVER_ENABLE_OPTION) != 0x00000000U) {
   492			u32 Reg = XEmacPs_ReadReg(EmacPsInstancePtr->Config.BaseAddress,
   493						XEMACPS_NWCTRL_OFFSET);
   494			if ((Reg & XEMACPS_NWCTRL_RXEN_MASK)) {
   495				XEmacPs_WriteReg(EmacPsInstancePtr->Config.BaseAddress,
   496						   XEMACPS_NWCTRL_OFFSET,
   497					   Reg & (~(u32)XEMACPS_NWCTRL_RXEN_MASK));
   498	            XEmacPs_WriteReg(EmacPsInstancePtr->Config.BaseAddress,
   499						   XEMACPS_RXSR_OFFSET, 0xf);
   500			}
   501		}
   502	    
   503	    /*
   504	     * Setup buffer address to associated BD.
   505	     * Remember to set wrap bit and clear owner bit.
   506	     */
   507	    XEmacPs_Bd* CurPtr = BdRxPtr;
   508	    for(int i = 0; i < num_packet; i++){
   509	        XEmacPs_BdWrite(CurPtr, XEMACPS_BD_STAT_OFFSET, 0);
   510	        XEmacPs_BdClearRxNew(CurPtr);
   511	        XEmacPs_BdClearRxWrap(CurPtr);
   512	        CurPtr++;
   513	    }
   514	    CurPtr--;
   515	    XEmacPs_BdSetRxWrap((UINTPTR)CurPtr);
   516	    
   517	    // flush again!
   518	    Xil_DCacheFlushRange(0, 64);
   519	
   520	    /*
   521	     * Set the Queue pointers
   522	     */
   523	    // TODO: set rx queue base
   524	    XEmacPs_SetQueuePtr(EmacPsInstancePtr, (UINTPTR)BdRxPtr, 0, XEMACPS_RECV);
   525	
   526		/* Enable receiver if not already enabled */
   527		if ((EmacPsInstancePtr->Options & XEMACPS_RECEIVER_ENABLE_OPTION) != 0x00000000U) {
   528			u32 Reg = XEmacPs_ReadReg(EmacPsInstancePtr->Config.BaseAddress,
   529						XEMACPS_NWCTRL_OFFSET);
   530			if ((!(Reg & XEMACPS_NWCTRL_RXEN_MASK))==TRUE) {
   531				XEmacPs_WriteReg(EmacPsInstancePtr->Config.BaseAddress,
   532						   XEMACPS_NWCTRL_OFFSET,
   533					   Reg | (u32)XEMACPS_NWCTRL_RXEN_MASK);
   534			}
   535		}
   536	
   537	    return Status;
   538	}

   541	{

   699	/****************************************************************************/
   700	/**
   701	 * This function resets the device but preserves the options set by
   702	 *the user.
   703	 *
   704	 * The descriptor list could be reinitialized with the same calls
   705	 *to XEmacPs_BdRingClone() as used in main(). Doing this is a
   706	 *matter of preference. In many cases, an OS may have resources
   707	 *tied up in the descriptors. Reinitializing in this case may bad
   708	 *for the OS since its resources may be permamently lost.
   709	 *
   710	 * @param	EmacPsInstancePtr is a pointer to the instance of the
   711	 *EmacPs driver.
   712	 *
   713	 * @return	XST_SUCCESS if successful, else XST_FAILURE.
   714	 *
   715	 * @note		None.
   716	 *
   717	 *****************************************************************************/
   718	static LONG EmacPsResetDevice(XEmacPs *EmacPsInstancePtr)
   719	{
   720	    LONG Status = 0;
   721	    u8 MacSave[6];
   722	    u32 Options;
   723	    XEmacPs_Bd BdTemplate;
   724	
   725	    /*
   726	     * Stop device
   727	     */
   728	    XEmacPs_Stop(EmacPsInstancePtr);
   729	
   730	    /*
   731	     * Save the device state
   732	     */
   733	    XEmacPs_GetMacAddress(EmacPsInstancePtr, &MacSave, 1);
   734	    Options = XEmacPs_GetOptions(EmacPsInstancePtr);
   735	
   736	    /*
   737	     * Stop and reset the device
   738	     */
   739	    XEmacPs_Reset(EmacPsInstancePtr);
   740	
   741	    /*
   742	     * Restore the state
   743	     */
   744	    XEmacPs_SetMacAddress(EmacPsInstancePtr, &MacSave, 1);
   745	    Status |= XEmacPs_SetOptions(EmacPsInstancePtr, Options);
   746	    Status |= XEmacPs_ClearOptions(EmacPsInstancePtr, ~Options);
   747	    if (Status != XST_SUCCESS) {
   748	        EmacPsUtilErrorTrap("Error restoring state after reset");
   749	        return XST_FAILURE;
   750	    }
   751	
   752	    /*
   753	     * Setup callbacks
   754	     */
   755	    Status = XEmacPs_SetHandler(
   756	        EmacPsInstancePtr, XEMACPS_HANDLER_DMASEND, (void *)XEmacPsSendHandler,
   757	        EmacPsInstancePtr);
   758	    Status |= XEmacPs_SetHandler(
   759	        EmacPsInstancePtr, XEMACPS_HANDLER_DMARECV, (void *)XEmacPsRecvHandler,
   760	        EmacPsInstancePtr);
   761	    Status |= XEmacPs_SetHandler(
   762	        EmacPsInstancePtr, XEMACPS_HANDLER_ERROR, (void *)XEmacPsErrorHandler,
   763	        EmacPsInstancePtr);
   764	    if (Status != XST_SUCCESS) {
   765	        EmacPsUtilErrorTrap("Error assigning handlers");
   766	        return XST_FAILURE;
   767	    }
   768	
   769	    /*
   770	     * Setup RxBD space.
   771	     *
   772	     * We have already defined a properly aligned area of memory to
   773	     * store RxBDs at the beginning of this source code file so
   774	     * just pass its address into the function. No MMU is being
   775	     * used so the physical and virtual addresses are the same.
   776	     *
   777	     * Setup a BD template for the Rx channel. This template will
   778	     * be copied to every RxBD. We will not have to explicitly set
   779	     * these again.
   780	     */
   781	    XEmacPs_BdClear(&BdTemplate);
   782	
   783	    /*
   784	     * Create the RxBD ring
   785	     */
   786	    Status = XEmacPs_BdRingCreate(
   787	        &(XEmacPs_GetRxRing(EmacPsInstancePtr)), kva2pa((UINTPTR)RxBdSpacePtr),
   788	        (UINTPTR)RxBdSpacePtr, XEMACPS_BD_ALIGNMENT, RXBD_CNT, &(XEmacPs_GetTxRing(EmacPsInstancePtr)));
   789	    if (Status != XST_SUCCESS) {
   790	        EmacPsUtilErrorTrap("Error setting up RxBD space, BdRingCreate");
   791	        return XST_FAILURE;
   792	    }
   793	
   794	    Status = XEmacPs_BdRingClone(
   795	        &(XEmacPs_GetRxRing(EmacPsInstancePtr)), &BdTemplate, XEMACPS_RECV);
   796	    if (Status != XST_SUCCESS) {
   797	        EmacPsUtilErrorTrap("Error setting up RxBD space, BdRingClone");
   798	        return XST_FAILURE;
   799	    }
   800	
   801	    /*
   802	     * Setup TxBD space.
   803	     *
   804	     * Like RxBD space, we have already defined a properly aligned
   805	     * area of memory to use.
   806	     *
   807	     * Also like the RxBD space, we create a template. Notice we
   808	     * don't set the "last" attribute. The examples will be
   809	     * overriding this attribute so it does no good to set it up
   810	     * here.
   811	     */
   812	    XEmacPs_BdClear(&BdTemplate);
   813	    XEmacPs_BdSetStatus(&BdTemplate, XEMACPS_TXBUF_USED_MASK);
   814	
   815	    /*
   816	     * Create the TxBD ring
   817	     */
   818	    Status = XEmacPs_BdRingCreate(
   819	        &(XEmacPs_GetTxRing(EmacPsInstancePtr)), kva2pa((UINTPTR)TxBdSpacePtr),
   820	        (UINTPTR)TxBdSpacePtr, XEMACPS_BD_ALIGNMENT, TXBD_CNT, &(XEmacPs_GetRxRing(EmacPsInstancePtr)));
   821	    if (Status != XST_SUCCESS) {
   822	        EmacPsUtilErrorTrap("Error setting up TxBD space, BdRingCreate");
   823	        return XST_FAILURE;
   824	    }
   825	    Status = XEmacPs_BdRingClone(
   826	        &(XEmacPs_GetTxRing(EmacPsInstancePtr)), &BdTemplate, XEMACPS_SEND);
   827	    if (Status != XST_SUCCESS) {
   828	        EmacPsUtilErrorTrap("Error setting up TxBD space, BdRingClone");
   829	        return XST_FAILURE;
   830	    }
   831	
   832	    /*
   833	     * Restart the device
   834	     */
   835	    XEmacPs_Start(EmacPsInstancePtr);
   836	
   837	    return XST_SUCCESS;
   838	}

   840	void handle_xemacps_irq()
   841	{
   842	    int send_flag = 0, recv_flag = 0;
   843	    Xil_DCacheFlushRange(0, 64);
   844	    u32 reg;
   845	    reg = XEmacPs_ReadReg(EmacPsInstance.Config.BaseAddress, XEMACPS_TXSR_OFFSET);
   846	    if(reg & XEMACPS_TXSR_TXCOMPL_MASK) {
   847			send_flag = 1;
   848	    }
   849	    reg = XEmacPs_ReadReg(EmacPsInstance.Config.BaseAddress, XEMACPS_RXSR_OFFSET);
   850	    if(reg & XEMACPS_RXSR_FRAMERX_MASK) {
   851			recv_flag = 1;
   852	    }
   853	    assert(send_flag || recv_flag);
   854	
   855	    if(send_flag && send_transaction){
   856	        XEmacPsSendHandler((void*)&EmacPsInstance);
   857	    }
   858	    else if(recv_flag && recv_transaction){
   859	        XEmacPsRecvHandler((void*)&EmacPsInstance);
   860	    }
   861	}

   901	/**
   902	 *
   903	 * This is the Receive handler callback function and will increment
   904	 *a shared counter that can be shared by the main thread of
   905	 *operation.
   906	 *
   907	 * @param	Callback is a pointer to the instance of the EmacPs
   908	 *device.

   910	 * @return	None.
   911	 *
   912	 * @note		None.

   915	static void XEmacPsRecvHandler(void *Callback)
   916	{
   917	    // TODO: you finish receiving packet now
   918	    XEmacPs *EmacPsInstancePtr = (XEmacPs *)Callback;
   919	
   920	    /*
   921	     * Disable the transmit related interrupts
   922	     */
   923	    /*XEmacPs_IntDisable(
   924	        EmacPsInstancePtr,
   925	        (XEMACPS_IXR_FRAMERX_MASK | XEMACPS_IXR_RX_ERR_MASK));*/
   926	
   927	    /*
   928	     * Increment the counter so that main thread knows something
   929	     * happened.
   930	     */
   931	    
   932	    XEmacPs_Bd* CurPtr = BdRxPtr;
   933	    for(int i = 0; i < _num_packet; i++){
   934	        while(!XEmacPs_BdIsRxNew(CurPtr)) {
   935	            Xil_DCacheFlushRange(0, 64);
   936	        }
   937	        rx_len[i] = XEmacPs_BdGetLength(CurPtr);
   938	        CurPtr++;
   939	    }
   940	    uintptr_t rx_curr = _addr;
   941	    for(int i = 0; i < _num_packet; i++){

   946	    do_unblock(net_recv_queue.next);
   947	    u32 Reg = XEmacPs_ReadReg(EmacPsInstancePtr->Config.BaseAddress, XEMACPS_ISR_OFFSET);
   948	    Reg = Reg | XEMACPS_IXR_FRAMERX_MASK;
   949		XEmacPs_WriteReg(EmacPsInstancePtr->Config.BaseAddress, XEMACPS_ISR_OFFSET, Reg);
   950	    recv_transaction = 0;
   951	}

   954	/****************************************************************************/
   955	/**
   956	 *
   957	 * This is the Error handler callback function and this function
   958	 *increments the error counter so that the main thread knows the
   959	 *number of errors.
   960	 *
   961	 * @param	Callback is the callback function for the driver. This
   962	 *		parameter is not used in this example.
   963	 * @param	Direction is passed in from the driver specifying which
   964	 *		direction error has occurred.
   965	 * @param	ErrorWord is the status register value passed in.
   966	 *
   967	 * @return	None.
   968	 *
   969	 * @note		None.
   970	 *
   971	 *****************************************************************************/
   972	static void XEmacPsErrorHandler(void *Callback, u8 Direction, u32 ErrorWord)
   973	{
   974	    XEmacPs *EmacPsInstancePtr = (XEmacPs *)Callback;
   975	
   976	    /*
   977	     * Increment the counter so that main thread knows something
   978	     * happened. Reset the device and reallocate resources ...
   979	     */
   980	    DeviceErrors++;
   981	
   982	    switch (Direction) {
   983	        case XEMACPS_RECV:
   984	            if (ErrorWord & XEMACPS_RXSR_HRESPNOK_MASK) {
   985	                EmacPsUtilErrorTrap("Receive DMA error");
   986	            }
   987	            if (ErrorWord & XEMACPS_RXSR_RXOVR_MASK) {
   988	                EmacPsUtilErrorTrap("Receive over run");
   989	            }
   990	            if (ErrorWord & XEMACPS_RXSR_BUFFNA_MASK) {
   991	                EmacPsUtilErrorTrap("Receive buffer not available");
   992	            }
   993	            break;
   994	        case XEMACPS_SEND:
   995	            if (ErrorWord & XEMACPS_TXSR_HRESPNOK_MASK) {
   996	                EmacPsUtilErrorTrap("Transmit DMA error");
   997	            }
   998	            if (ErrorWord & XEMACPS_TXSR_URUN_MASK) {
   999	                EmacPsUtilErrorTrap("Transmit under run");
  1000	            }
  1001	            if (ErrorWord & XEMACPS_TXSR_BUFEXH_MASK) {
  1002	                EmacPsUtilErrorTrap("Transmit buffer exhausted");
  1003	            }
  1004	            if (ErrorWord & XEMACPS_TXSR_RXOVR_MASK) {
  1005	                EmacPsUtilErrorTrap("Transmit retry excessed limits");
  1006	            }
  1007	            if (ErrorWord & XEMACPS_TXSR_FRAMERX_MASK) {
  1008	                EmacPsUtilErrorTrap("Transmit collision");
  1009	            }
  1010	            if (ErrorWord & XEMACPS_TXSR_USEDREAD_MASK) {
  1011	                EmacPsUtilErrorTrap("Transmit buffer not available");
  1012	            }
  1013	            break;
  1014	    }
  1015	    /*
  1016	     * Bypassing the reset functionality as the default tx status
  1017	     * for q0 is USED BIT READ. so, the first interrupt will be tx
  1018	     * used bit and it resets the core always.
  1019	     */
  1020	    if (GemVersion == 2) {
  1021	        EmacPsResetDevice(EmacPsInstancePtr);
  1022	    }
  1023	}

  1025	/****************************************************************************/
  1026	/**
  1027	 *
  1028	 * This function sets up the clock divisors for 1000Mbps.
  1029	 *
  1030	 * @param	EmacPsInstancePtr is a pointer to the instance of the
  1031	 *EmacPs driver.
  1032	 * @param	EmacPsIntrId is the Interrupt ID and is typically
  1033	 *			XPAR_<EMACPS_instance>_INTR value from xparameters.h.
  1034	 * @return	None.
  1035	 *
  1036	 * @note		None.
  1037	 *
  1038	 *****************************************************************************/
  1039	void XEmacPsClkSetup(XEmacPs *EmacPsInstancePtr)
  1040	{
  1041	    u32 ClkCntrl;
  1042	    if (GemVersion == 2) {
  1043	        /*************************************/
  1044	        /* Setup device for first-time usage */
  1045	        /*************************************/
  1046	
  1047	        /* SLCR unlock */
  1048	        *(volatile unsigned int *)(SLCR_UNLOCK_ADDR) = SLCR_UNLOCK_KEY_VALUE;
  1049	        /* GEM0 1G clock configuration*/
  1050	        ClkCntrl = *(volatile unsigned int *)(SLCR_GEM0_CLK_CTRL_ADDR);
  1051	        ClkCntrl &= EMACPS_SLCR_DIV_MASK;
  1052	        ClkCntrl |= (XPAR_PS7_ETHERNET_0_ENET_SLCR_1000MBPS_DIV1 << 20);
  1053	        ClkCntrl |= (XPAR_PS7_ETHERNET_0_ENET_SLCR_1000MBPS_DIV0 << 8);
  1054	        *(volatile unsigned int *)(SLCR_GEM0_CLK_CTRL_ADDR) = ClkCntrl;
  1055	
  1056	        /* SLCR lock */
  1057	        *(unsigned int *)(SLCR_LOCK_ADDR) = SLCR_LOCK_KEY_VALUE;
  1058	
  1059	        unsigned long count = 0;
  1060	        while (count < 0xffff) {
  1061	            count++;
  1062	        }
  1063	    }
  1064	}
