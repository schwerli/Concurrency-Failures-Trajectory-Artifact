     9	class CoverageBitmap:
    10	    """Simulates AFL's coverage bitmap using a set of edge IDs.
    11	
    12	    In real AFL, this is a 64KB shared memory region where each byte
    13	    represents a hit count for an edge (hashed to bucket). We simplify
    14	    to exact edge tracking for determinism.
    15	    """
    16	
    17	    def __init__(self):
    18	        self.global_coverage: Set[int] = set()
    19	        self.virgin_bits: Set[int] = set()  # edges never seen before
    20	
    21	    def has_new_bits(self, edges: Set[int]) -> bool:
    22	        """Check if the given edges contain any new coverage.
    23	
    24	        Returns True if at least one edge has never been seen before.
    25	        """
    26	        new_edges = edges - self.global_coverage
    27	        return len(new_edges) > 0
    28	
    29	    def update(self, edges: Set[int]) -> Set[int]:
    30	        """Update the global bitmap with new edges.
    31	
    32	        Returns the set of newly discovered edges.
    33	        """
    34	        new_edges = edges - self.global_coverage
    35	        self.global_coverage.update(edges)
    36	        self.virgin_bits.update(new_edges)
    37	        return new_edges
    38	
    39	    def get_coverage_count(self) -> int:
    40	        """Return the number of unique edges covered so far."""
    41	        return len(self.global_coverage)
    42	
    43	    def get_coverage_percentage(self, total_edges: int) -> float:
    44	        """Return coverage as a percentage of total edges."""
    45	        if total_edges == 0:
    46	            return 0.0
    47	        return len(self.global_coverage) / total_edges * 100.0
    48	
    49	    def reset(self):
    50	        """Reset the bitmap to empty state."""
    51	        self.global_coverage.clear()
    52	        self.virgin_bits.clear()

    11	INTERESTING_8 = [-128, -1, 0, 1, 16, 32, 64, 100, 127]
    12	INTERESTING_16 = [-32768, -129, 128, 255, 256, 512, 1000, 1024, 4096, 32767]
    13	INTERESTING_32 = [-2147483648, -100663046, -32769, 32768, 65535, 65536,

    17	def bit_flip(data: bytes, pos: int, width: int = 1) -> bytes:
    18	    """Flip 'width' consecutive bits starting at bit position 'pos'."""
    19	    buf = bytearray(data)
    20	    for i in range(width):
    21	        bit_pos = pos + i
    22	        byte_idx = bit_pos // 8
    23	        bit_idx = bit_pos % 8
    24	        if byte_idx < len(buf):
    25	            buf[byte_idx] ^= (1 << (7 - bit_idx))
    26	    return bytes(buf)
    27	
    28	
    29	def byte_flip(data: bytes, pos: int, width: int = 1) -> bytes:
    30	    """Flip 'width' consecutive bytes starting at position 'pos'."""
    31	    buf = bytearray(data)
    32	    for i in range(width):
    33	        if pos + i < len(buf):
    34	            buf[pos + i] ^= 0xFF
    35	    return bytes(buf)
    36	
    37	
    38	def arith_add(data: bytes, pos: int, value: int = 1) -> bytes:
    39	    """Add a small value to byte at position 'pos'."""
    40	    buf = bytearray(data)
    41	    if pos < len(buf):
    42	        buf[pos] = (buf[pos] + value) & 0xFF
    43	    return bytes(buf)
    44	
    45	
    46	def arith_sub(data: bytes, pos: int, value: int = 1) -> bytes:
    47	    """Subtract a small value from byte at position 'pos'."""
    48	    buf = bytearray(data)
    49	    if pos < len(buf):
    50	        buf[pos] = (buf[pos] - value) & 0xFF
    51	    return bytes(buf)

    54	def interesting_values(data: bytes, pos: int, rng: np.random.RandomState) -> bytes:
    55	    """Replace byte at position with an interesting value."""
    56	    buf = bytearray(data)
    57	    if pos < len(buf):
    58	        val = INTERESTING_8[rng.randint(0, len(INTERESTING_8))]
    59	        buf[pos] = val & 0xFF

    64	    """Replace byte at position with a random value."""

    71	def clone_bytes(data: bytes, src: int, dst: int, length: int) -> bytes:
    72	    """Clone a chunk of bytes from src to dst position."""
    73	    buf = bytearray(data)
    74	    if src + length <= len(buf) and dst + length <= len(buf):
    75	        chunk = buf[src:src + length]
    76	        buf[dst:dst + length] = chunk
    77	    return bytes(buf)

    80	def delete_bytes(data: bytes, pos: int, length: int) -> bytes:
    81	    """Delete 'length' bytes starting at position 'pos'."""
    82	    buf = bytearray(data)
    83	    if pos < len(buf):
    84	        end = min(pos + length, len(buf))
    85	        del buf[pos:end]
    86	    if len(buf) == 0:
    87	        buf = bytearray(b'\x00')
    88	    return bytes(buf)

    91	def insert_bytes(data: bytes, pos: int, new_bytes: bytes) -> bytes:
    92	    """Insert new bytes at position 'pos'."""
    93	    buf = bytearray(data)
    94	    if pos <= len(buf):
    95	        for i, b in enumerate(new_bytes):
    96	            buf.insert(pos + i, b)
    97	    return bytes(buf)

   100	def havoc(data: bytes, rng: np.random.RandomState, num_mutations: int = 0) -> bytes:
   101	    """Apply a random combination of mutations (AFL's havoc stage).
   102	
   103	    If num_mutations is 0, randomly choose 1-7 mutations to stack.
   104	    """
   105	    if num_mutations == 0:
   106	        num_mutations = rng.randint(1, 8)
   107	
   108	    result = data
   109	    for _ in range(num_mutations):
   110	        if len(result) == 0:
   111	            result = b'\x00'
   112	
   113	        op = rng.randint(0, 8)
   114	        pos = rng.randint(0, max(1, len(result)))
   115	
   116	        if op == 0:
   117	            # bit flip
   118	            bit_pos = rng.randint(0, max(1, len(result) * 8))
   119	            result = bit_flip(result, bit_pos)
   120	        elif op == 1:
   121	            # byte flip
   122	            result = byte_flip(result, pos)
   123	        elif op == 2:
   124	            # arith add
   125	            val = rng.randint(1, 36)
   126	            result = arith_add(result, pos, val)
   127	        elif op == 3:
   128	            # arith sub
   129	            val = rng.randint(1, 36)
   130	            result = arith_sub(result, pos, val)
   131	        elif op == 4:
   132	            # interesting values
   133	            result = interesting_values(result, pos, rng)
   134	        elif op == 5:
   135	            # random byte
   136	            result = random_byte(result, pos, rng)
   137	        elif op == 6:
   138	            # delete bytes (only if input > 4 bytes)
   139	            if len(result) > 4:
   140	                length = rng.randint(1, min(4, len(result) - 1))
   141	                result = delete_bytes(result, pos, length)
   142	        elif op == 7:
   143	            # insert random bytes
   144	            length = rng.randint(1, 5)
   145	            new_data = bytes(rng.randint(0, 256, size=length).astype(np.uint8))
   146	            insert_pos = rng.randint(0, max(1, len(result)))
   147	            result = insert_bytes(result, insert_pos, new_data)
   148	
   149	    return result

   152	def splice(data1: bytes, data2: bytes, rng: np.random.RandomState) -> bytes:
   153	    """Splice two test cases at a random midpoint (AFL's splice stage)."""
   154	    if len(data1) < 2 or len(data2) < 2:

   157	    min_len = min(len(data1), len(data2))

    27	class SimulatedProgram:
    28	    """A simulated target program with a known CFG.
    29	
    30	    The program takes an input (bytes) and traces execution through the CFG
    31	    based on input byte values. Returns which edges were hit.
    32	    """
    33	
    34	    def __init__(self, blocks: List[BasicBlock], edges: List[Edge],
    35	                 entry_block: int = 0, target_blocks: Optional[List[int]] = None):
    36	        self.blocks = {b.block_id: b for b in blocks}
    37	        self.edges = edges
    38	        self.edge_map: Dict[Tuple[int, int], Edge] = {}
    39	        self.adj: Dict[int, List[Edge]] = {b.block_id: [] for b in blocks}
    40	        self.entry_block = entry_block
    41	        self.target_blocks = target_blocks or []
    42	
    43	        for e in edges:
    44	            self.edge_map[(e.src, e.dst)] = e
    45	            self.adj[e.src].append(e)
    46	
    47	    def execute(self, input_data: bytes) -> Tuple[Set[int], bool, float]:
    48	        """Execute the program on the given input.
    49	
    50	        Returns:
    51	            (edges_hit, is_crash, exec_time)
    52	            - edges_hit: set of edge IDs that were traversed
    53	            - is_crash: whether a crash-triggering edge was hit
    54	            - exec_time: simulated execution time (proportional to path length)
    55	        """
    56	        edges_hit = set()
    57	        is_crash = False
    58	        current_block = self.entry_block
    59	        exec_time = 0.0
    60	        input_pos = 0
    61	
    62	        max_steps = 100  # prevent infinite loops
    63	
    64	        for step in range(max_steps):
    65	            outgoing = self.adj.get(current_block, [])
    66	            if not outgoing:
    67	                break
    68	
    69	            # Determine which edge to take based on input byte
    70	            if input_pos < len(input_data):
    71	                byte_val = input_data[input_pos]
    72	                input_pos += 1
    73	            else:
    74	                byte_val = 0
    75	
    76	            # Select edge based on byte value modulo number of outgoing edges
    77	            edge_idx = byte_val % len(outgoing)
    78	            chosen_edge = outgoing[edge_idx]
    79	
    80	            edges_hit.add(chosen_edge.edge_id)
    81	            exec_time += 0.001  # 1ms per edge traversal
    82	
    83	            if chosen_edge.is_crash:
    84	                is_crash = True
    85	
    86	            current_block = chosen_edge.dst
    87	
    88	        return edges_hit, is_crash, exec_time
    89	
    90	    def get_all_edge_ids(self) -> Set[int]:
    91	        """Return all edge IDs in the program."""
    92	        return {e.edge_id for e in self.edges}
    93	
    94	    def get_crash_edges(self) -> Set[int]:
    95	        """Return edge IDs that trigger crashes."""
    96	        return {e.edge_id for e in self.edges if e.is_crash}
    97	
    98	    def get_rare_edges(self) -> Set[int]:
    99	        """Return edge IDs that are hard to reach."""
   100	        return {e.edge_id for e in self.edges if e.is_rare}
   101	
   102	    def get_shortest_distances(self, target_block_ids: List[int]) -> Dict[int, float]:
   103	        """Compute shortest distance from each block to the nearest target block.
   104	
   105	        Uses BFS/Dijkstra (all edges weight 1).
   106	        Returns dict mapping block_id -> distance (inf if unreachable).
   107	        """
   108	        from collections import deque
   109	
   110	        # Build reverse adjacency
   111	        rev_adj: Dict[int, List[int]] = {bid: [] for bid in self.blocks}
   112	        for e in self.edges:
   113	            rev_adj[e.dst].append(e.src)
   114	
   115	        distances: Dict[int, float] = {bid: float('inf') for bid in self.blocks}
   116	
   117	        # BFS from each target block backwards
   118	        queue = deque()
   119	        for tb in target_block_ids:
   120	            if tb in self.blocks:
   121	                distances[tb] = 0.0
   122	                queue.append(tb)
   123	
   124	        while queue:
   125	            node = queue.popleft()
   126	            for pred in rev_adj.get(node, []):
   127	                new_dist = distances[node] + 1.0
   128	                if new_dist < distances[pred]:
   129	                    distances[pred] = new_dist
   130	                    queue.append(pred)
   131	
   132	        return distances

   110	class CrashmeProgram:
   111	    """Wrapper around SimulatedProgram with crashme-specific execution logic.
   112	
   113	    Uses a more sophisticated execution model where specific byte sequences
   114	    are needed to trigger certain paths.
   115	    """
   116	
   117	    def __init__(self):
   118	        self.program = build_crashme()
   119	        self.total_edges = len(self.program.edges)
   120	        self.total_blocks = len(self.program.blocks)
   121	
   122	    def execute(self, input_data: bytes) -> Tuple[Set[int], bool, float]:
   123	        """Execute crashme with input-dependent path selection.
   124	
   125	        Path selection logic:
   126	        - Byte 0: selects initial branch (mod 3)
   127	        - Byte 1: selects second-level branch (mod 2)
   128	        - Byte 2+: selects deeper branches with specific conditions
   129	
   130	        Crash path 1: byte[0]%3==2, byte[1]%2==0, byte[2] in [0x30..0x3F],
   131	                       byte[3] in [0x80..0x9F]
   132	          -> blocks 0,3,8,17,19 (crash edge 25)
   133	        Crash path 2: byte[0]%3==2, byte[1]%2==1, byte[2]==0x4D,
   134	                       byte[3]==0x5A, byte[4]%4==3
   135	          -> blocks 0,3,9,18,17,19 (crash edge 25 via rare 26)
   136	        """
   137	        edges_hit = set()
   138	        is_crash = False
   139	        exec_time = 0.001  # base cost
   140	
   141	        if len(input_data) == 0:
   142	            return edges_hit, False, exec_time
   143	
   144	        # Level 1: entry -> {1, 2, 3}
   145	        b0 = input_data[0] if len(input_data) > 0 else 0
   146	        branch1 = b0 % 3
   147	
   148	        if branch1 == 0:
   149	            edges_hit.add(0)  # 0->1
   150	            current = 1
   151	        elif branch1 == 1:
   152	            edges_hit.add(1)  # 0->2
   153	            current = 2
   154	        else:
   155	            edges_hit.add(2)  # 0->3
   156	            current = 3
   157	
   158	        exec_time += 0.001
   159	
   160	        # Level 2
   161	        b1 = input_data[1] if len(input_data) > 1 else 0
   162	        branch2 = b1 % 2
   163	
   164	        if current == 1:
   165	            if branch2 == 0:
   166	                edges_hit.add(3)  # 1->4
   167	                current = 4
   168	            else:
   169	                edges_hit.add(4)  # 1->5
   170	                current = 5
   171	        elif current == 2:
   172	            if branch2 == 0:
   173	                edges_hit.add(5)  # 2->6
   174	                current = 6
   175	            else:
   176	                edges_hit.add(6)  # 2->7
   177	                current = 7
   178	        elif current == 3:
   179	            if branch2 == 0:
   180	                edges_hit.add(7)  # 3->8
   181	                current = 8
   182	            else:
   183	                edges_hit.add(8)  # 3->9
   184	                current = 9
   185	
   186	        exec_time += 0.001
   187	
   188	        # Level 3
