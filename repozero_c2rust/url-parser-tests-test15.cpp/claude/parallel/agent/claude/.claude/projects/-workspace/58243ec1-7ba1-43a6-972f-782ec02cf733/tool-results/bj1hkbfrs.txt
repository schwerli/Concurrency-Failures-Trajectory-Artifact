     1	//! Byte-level character helpers matching the C++ library's internal utilities.
     2	//!
     3	//! The C++ code calls `::tolower` on each `unsigned char`. Under the default "C"
     4	//! locale that maps `A..Z` only and leaves every byte >= 0x80 untouched, so
     5	//! multi-byte UTF-8 sequences pass through unchanged.

     7	/// ASCII-only lowercase, mirroring `url_parser::to_lower`.
     8	///
     9	/// VERIFIED: `"ÉHTTP"` -> `"Éhttp"` and `"HTTPÿ"` -> `"httpÿ"`; the non-ASCII
    10	/// bytes are preserved, so this must not be a Unicode-aware lowercase.
    11	pub fn to_lower(input: &str) -> String {
    12	    let mut out = input.to_owned();
    13	    out.make_ascii_lowercase();
    14	    out
    15	}
    16	
    17	/// True for the six characters that C's `isspace` accepts in the "C" locale.
    18	pub fn is_space(byte: u8) -> bool {
    19	    matches!(byte, b' ' | b'\t' | b'\n' | 0x0b | 0x0c | b'\r')
    20	}
    21	
    22	/// Strips leading and trailing whitespace, mirroring `url_parser::trim`.
    23	pub fn trim(input: &str) -> &str {
    24	    let bytes = input.as_bytes();
    25	    let mut start = 0;
    26	    let mut end = bytes.len();
    27	    while start < end && is_space(bytes[start]) {
    28	        start += 1;
    29	    }
    30	    while end > start && is_space(bytes[end - 1]) {
    31	        end -= 1;
    32	    }
    33	    // `start`/`end` only ever land on ASCII whitespace boundaries, so the range
    34	    // is always on a UTF-8 char boundary.
    35	    &input[start..end]
    36	}
    37	
    38	/// True for `0-9`, `a-f`, `A-F`, mirroring `url_parser::is_hex`.
    39	pub fn is_hex(byte: u8) -> bool {
    40	    byte.is_ascii_hexdigit()
    41	}
    42	
    43	/// Decodes a single hex digit, mirroring `url_parser::hex_to_char`.
    44	pub fn hex_to_value(byte: u8) -> Option<u8> {
    45	    match byte {
    46	        b'0'..=b'9' => Some(byte - b'0'),
    47	        b'a'..=b'f' => Some(byte - b'a' + 10),
    48	        b'A'..=b'F' => Some(byte - b'A' + 10),
    49	        _ => None,
    50	    }
    51	}

    53	/// Encodes the low nibble of `value` as an uppercase hex digit.
    54	pub fn value_to_hex(value: u8) -> char {
    55	    const DIGITS: &[u8; 16] = b"0123456789ABCDEF";
    56	    DIGITS[(value & 0x0f) as usize] as char
    57	}

     1	//! Faithful emulation of the two C++ standard-library behaviours the harness
     2	//! depends on: `std::stoi` and the runtime's response to an uncaught exception.
     3	//!
     4	//! `test15.cpp` calls `std::stoi(argv[3])` outside any `try` block, so a
     5	//! malformed port argument does not produce a tidy error message — it unwinds
     6	//! into `std::terminate`, which prints a fixed two-line diagnostic and raises
     7	//! `SIGABRT`. Both are reproduced byte for byte here.
     8

    13	/// The two exception types `std::stoi` can throw.
    14	#[derive(Clone, Copy, Debug, PartialEq, Eq)]
    15	pub enum StoiError {
    16	    /// `std::invalid_argument` — no digits could be consumed.
    17	    InvalidArgument,
    18	    /// `std::out_of_range` — the value does not fit in an `int`.
    19	    OutOfRange,
    20	}
    21	
    22	impl StoiError {
    23	    /// The fully-qualified C++ type name, as printed by `std::terminate`.
    24	    pub fn cxx_type_name(self) -> &'static str {
    25	        match self {
    26	            StoiError::InvalidArgument => "std::invalid_argument",
    27	            StoiError::OutOfRange => "std::out_of_range",
    28	        }
    29	    }
    30	}
    31	
    32	/// Once the accumulator passes this bound the exact value stops mattering: the
    33	/// result is already outside `int` range, and clamping keeps the `i64` maths
    34	/// from overflowing on very long digit runs.
    35	const ACCUMULATOR_CEILING: i64 = 10_000_000_000;
    36	
    37	/// Emulates `std::stoi(str)` with the default base of 10.
    38	///
    39	/// Like the underlying `strtol`, this skips leading whitespace, accepts one
    40	/// optional sign, consumes base-10 digits, and then ignores whatever trails.
    41	///
    42	/// VERIFIED against the C++ binary: `"080"` -> 80, `"+80"` -> 80, `" 80"` -> 80,
    43	/// `"80 "` -> 80, `"8 0"` -> 8, `"80abc"` -> 80, `"80.9"` -> 80, `"0x50"` -> 0,
    44	/// `"1e3"` -> 1, `"-0"` -> 0, `"  -42xyz"` -> -42; `""`, `"abc"`, `"-"` and
    45	/// `"+"` raise `invalid_argument`; `"2147483648"`, `"-2147483649"` and
    46	/// `"99999999999999999999"` raise `out_of_range`.
    47	pub fn stoi(input: &str) -> Result<i32, StoiError> {
    48	    let bytes = input.as_bytes();
    49	    let mut index = 0;
    50	
    51	    while index < bytes.len() && is_space(bytes[index]) {
    52	        index += 1;
    53	    }
    54	
    55	    let mut negative = false;
    56	    if index < bytes.len() && (bytes[index] == b'+' || bytes[index] == b'-') {
    57	        negative = bytes[index] == b'-';
    58	        index += 1;
    59	    }
    60	
    61	    let first_digit = index;
    62	    let mut accumulator: i64 = 0;
    63	    while index < bytes.len() && bytes[index].is_ascii_digit() {
    64	        if accumulator < ACCUMULATOR_CEILING {
    65	            accumulator = accumulator * 10 + i64::from(bytes[index] - b'0');
    66	        }
    67	        index += 1;
    68	    }
    69	
    70	    if index == first_digit {
    71	        return Err(StoiError::InvalidArgument);
    72	    }
    73	
    74	    let value = if negative { -accumulator } else { accumulator };
    75	    if value < i64::from(i32::MIN) || value > i64::from(i32::MAX) {
    76	        return Err(StoiError::OutOfRange);
    77	    }
    78	    Ok(value as i32)
    79	}
    80	
    81	/// Reproduces libstdc++'s uncaught-exception death: the fixed diagnostic on
    82	/// stderr followed by `SIGABRT` (exit status 134 as reported by a shell).
    83	///
    84	/// VERIFIED byte for byte, including the two spaces after `what():`:
    85	/// `terminate called after throwing an instance of 'std::invalid_argument'\n  what():  stoi\n`
    86	pub fn terminate_uncaught(cxx_type_name: &str, what: &str) -> ! {
    87	    let mut stderr = std::io::stderr();
    88	    let _ = write!(
    89	        stderr,
    90	        "terminate called after throwing an instance of '{cxx_type_name}'\n  what():  {what}\n"
    91	    );
    92	    let _ = stderr.flush();
    93	    std::process::abort();
    94	}
    95	
    96	/// Parses the harness's port argument, dying exactly as the C++ binary does.
    97	pub fn stoi_or_terminate(input: &str) -> i32 {
    98	    match stoi(input) {
    99	        Ok(value) => value,
   100	        Err(error) => terminate_uncaught(error.cxx_type_name(), "stoi"),
   101	    }
   102	}

     1	//! Percent-encoding helpers mirroring `url_parser::url_encode` / `url_decode`.
     2	//!
     3	//! Note: `URL::from_components` never touches these — every component it is
     4	//! handed is stored with its percent-escapes exactly as given (VERIFIED:
     5	//! `"/a%20b"` stays `"/a%20b"`, `"q=%20&r=a+b"` stays verbatim). They exist for
     6	//! the query-parameter helpers and for callers that want them.
     7	
     8	use super::ascii::{hex_to_value, value_to_hex};
     9	
    10	/// Percent-encodes everything outside the RFC 3986 unreserved set
    11	/// (`ALPHA / DIGIT / "-" / "_" / "." / "~"`), using uppercase hex digits.
    12	pub fn url_encode(input: &str) -> String {
    13	    let mut out = String::with_capacity(input.len());
    14	    for &byte in input.as_bytes() {
    15	        match byte {
    16	            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
    17	                out.push(byte as char);
    18	            }
    19	            _ => {
    20	                out.push('%');
    21	                out.push(value_to_hex(byte >> 4));
    22	                out.push(value_to_hex(byte));
    23	            }
    24	        }
    25	    }
    26	    out
    27	}
    28	
    29	/// Reverses [`url_encode`]. A `%` that is not followed by two hex digits is kept
    30	/// literally, which is what the C++ `is_hex` guard amounts to.
    31	///
    32	/// `+` is *not* treated as a space: `url_encode` emits `%20`, so decoding `+`
    33	/// would not be symmetric.
    34	pub fn url_decode(input: &str) -> String {
    35	    let bytes = input.as_bytes();
    36	    let mut out: Vec<u8> = Vec::with_capacity(bytes.len());
    37	    let mut index = 0;
    38	
    39	    while index < bytes.len() {
    40	        if bytes[index] == b'%' && index + 2 < bytes.len() {
    41	            if let (Some(high), Some(low)) =
    42	                (hex_to_value(bytes[index + 1]), hex_to_value(bytes[index + 2]))
    43	            {
    44	                out.push((high << 4) | low);
    45	                index += 3;
    46	                continue;
    47	            }
    48	        }
    49	        out.push(bytes[index]);
    50	        index += 1;
    51	    }
    52	
    53	    // Decoded bytes are arbitrary; fall back to replacement chars rather than
    54	    // failing, since a decoded octet sequence need not be valid UTF-8.
    55	    String::from_utf8_lossy(&out).into_owned()
    56	}

     1	//! Path helpers mirroring `url_parser::split_path` and `URL::normalize_path`.
     2	//!
     3	//! `URL::from_components` deliberately does *not* normalize the path — it only
     4	//! guarantees a leading slash (see [`crate::url::URL::set_path`]).
     5	
     6	/// Splits a path into its non-empty segments, mirroring `url_parser::split_path`.
     7	///
     8	/// Empty segments (from leading, trailing or doubled slashes) are dropped:
     9	/// `"//a//b//"` yields `["a", "b"]`.
    10	pub fn split_path(path: &str) -> Vec<String> {
    11	    path.split('/')
    12	        .filter(|segment| !segment.is_empty())
    13	        .map(str::to_owned)
    14	        .collect()
    15	}
    16	
    17	/// Resolves `.` and `..` segments and collapses repeated slashes.
    18	///
    19	/// A leading slash and a trailing slash are both preserved when the input had
    20	/// one. `..` at the root is discarded rather than escaping above it.
    21	pub fn normalize_path_string(path: &str) -> String {
    22	    if path.is_empty() {
    23	        return String::new();
    24	    }
    25	
    26	    let absolute = path.starts_with('/');
    27	    let trailing_slash = path.ends_with('/');
    28	
    29	    let mut resolved: Vec<&str> = Vec::new();
    30	    for segment in path.split('/') {
    31	        match segment {
    32	            "" | "." => {}
    33	            ".." => {
    34	                // A relative path has to keep a `..` it cannot resolve, or the
    35	                // path would silently change meaning.
    36	                match resolved.last() {
    37	                    Some(&last) if last != ".." => {
    38	                        resolved.pop();
    39	                    }
    40	                    _ if !absolute => resolved.push(".."),
    41	                    _ => {}
    42	                }
    43	            }
    44	            other => resolved.push(other),
    45	        }
    46	    }
    47	
    48	    let mut out = String::with_capacity(path.len());
    49	    if absolute {
    50	        out.push('/');
    51	    }
    52	    out.push_str(&resolved.join("/"));
    53	
    54	    if trailing_slash && !out.ends_with('/') {
    55	        out.push('/');
    56	    }
    57	    if out.is_empty() {
    58	        out.push_str(if absolute { "/" } else { "." });
    59	    }
    60	    out
    61	}

     1	//! Query-string helpers backing `URL::query_params`, `get_query_param`,
     2	//! `add_query_param`, `remove_query_param` and `normalize_query`.
     3	//!
     4	//! As with the path, `URL::from_components` stores the query verbatim — VERIFIED:
     5	//! `"b=2&a=1"` comes back unsorted, `"&&"` is preserved, and a leading `"?"` is
     6	//! *not* stripped. Nothing here runs unless a caller asks for it.
     7	
     8	use super::encoding::{url_decode, url_encode};
     9	
    10	/// A decoded `name=value` pair.
    11	pub type Param = (String, String);
    12	
    13	/// Splits a raw query string into decoded pairs.
    14	///
    15	/// Empty chunks (from `&&` or a trailing `&`) are skipped. A chunk with no `=`
    16	/// becomes a name with an empty value.
    17	pub fn parse_query_string(query: &str) -> Vec<Param> {
    18	    query
    19	        .split('&')
    20	        .filter(|chunk| !chunk.is_empty())
    21	        .map(|chunk| match chunk.split_once('=') {
    22	            Some((name, value)) => (url_decode(name), url_decode(value)),
    23	            None => (url_decode(chunk), String::new()),
    24	        })
    25	        .collect()
    26	}
    27	
    28	/// Rebuilds a raw query string from decoded pairs, re-encoding both sides.
    29	pub fn build_query_string(params: &[Param]) -> String {
    30	    params
    31	        .iter()
    32	        .map(|(name, value)| format!("{}={}", url_encode(name), url_encode(value)))
    33	        .collect::<Vec<_>>()
    34	        .join("&")
    35	}
    36	
    37	/// Canonical form of a query string: empty chunks dropped, pairs sorted by name
    38	/// (ties keep their original order), everything consistently re-encoded.
    39	pub fn normalize_query_string(query: &str) -> String {
    40	    let mut params = parse_query_string(query);
    41	    params.sort_by(|left, right| left.0.cmp(&right.0));
    42	    build_query_string(&params)
    43	}

    46	/// Equality compares the URL's components. `valid` is a pure function of `host`
    47	/// and `path`, so leaving it out changes nothing.
    48	impl PartialEq for URL {
    49	    fn eq(&self, other: &Self) -> bool {
    50	        self.scheme == other.scheme
    51	            && self.userinfo == other.userinfo
    52	            && self.host == other.host
    53	            && self.port == other.port
    54	            && self.path == other.path
    55	            && self.query == other.query
    56	            && self.fragment == other.fragment
    57	    }
    58	}

    60	impl URL {
    61	    /// An empty URL with no port.
    62	    pub fn new() -> Self {
    63	        URL::default()
    64	    }
    65	
    66	    /// Builds a URL from already-separated components.
    67	    ///
    68	    /// This is the entry point `test15` uses. Each component goes through its
    69	    /// setter, so the scheme is lowercased, the port is range-filtered and the
    70	    /// path picks up a leading slash; the host, query and fragment are stored
    71	    /// exactly as handed over.
    72	    pub fn from_components(
    73	        scheme: &str,
    74	        host: &str,
    75	        port: i32,
    76	        path: &str,
    77	        query: &str,
    78	        fragment: &str,
    79	    ) -> Self {
    80	        let mut url = URL::new();
    81	        url.set_scheme(scheme);
    82	        url.set_host(host);
    83	        url.set_port(port);
    84	        url.set_path(path);
    85	        url.set_query(query);
    86	        url.set_fragment(fragment);
    87	        url
    88	    }
    89	
    90	    /// Parses a URL string of the form
    91	    /// `scheme://userinfo@host:port/path?query#fragment`.
    92	    ///
    93	    /// Every part is optional. `mailto:user@example.com`-style opaque forms are
    94	    /// recognised when the text before the first `:` is a valid scheme.
    95	    pub fn parse(input: &str) -> Self {
    96	        let mut url = URL::new();
    97	        let mut rest = trim(input);
    98	
    99	        // Strip from the right, so a `#` inside the query and a `?` inside the
   100	        // fragment both land where a browser would put them.
   101	        if let Some((head, fragment)) = rest.split_once('#') {
   102	            url.set_fragment(fragment);
   103	            rest = head;
   104	        }
   105	        if let Some((head, query)) = rest.split_once('?') {
   106	            url.set_query(query);
   107	            rest = head;
   108	        }
   109	
   110	        let mut authority = None;
   111	        if let Some(position) = rest.find("://") {
   112	            url.set_scheme(&rest[..position]);
   113	            rest = &rest[position + 3..];
   114	            let (found, remainder) = match rest.find('/') {
   115	                Some(slash) => (&rest[..slash], &rest[slash..]),
   116	                None => (rest, ""),
   117	            };
   118	            authority = Some(found);
   119	            rest = remainder;
   120	        } else if let Some((scheme, remainder)) = rest.split_once(':') {
   121	            if is_scheme_valid(scheme) {
   122	                url.set_scheme(scheme);
   123	                rest = remainder;
   124	            }
   125	        }
   126	
   127	        if let Some(authority) = authority {
   128	            let host_and_port = match authority.rsplit_once('@') {
   129	                Some((userinfo, tail)) => {
   130	                    url.userinfo = userinfo.to_owned();
   131	                    tail
   132	                }
   133	                None => authority,
   134	            };
   135	
   136	            // In `[::1]:8080` only the colon after the closing bracket is a port
   137	            // separator; the ones inside belong to the IPv6 literal.
   138	            let port_separator = match host_and_port.rfind(']') {
   139	                Some(bracket) => host_and_port[bracket..]
   140	                    .find(':')
   141	                    .map(|offset| bracket + offset),
   142	                None => host_and_port.rfind(':'),
   143	            };
   144	
   145	            match port_separator {
   146	                Some(colon) => {
   147	                    url.set_host(&host_and_port[..colon]);
   148	                    url.set_port(stoi(&host_and_port[colon + 1..]).unwrap_or(PORT_UNSET));
   149	                }
   150	                None => url.set_host(host_and_port),
   151	            }
   152	        }
   153	
   154	        url.set_path(rest);
   155	        url
   156	    }
   157	
   158	    /// True when `input` parses into a URL that carries an authority or a path.
   159	    pub fn is_valid_url(input: &str) -> bool {
   160	        URL::parse(input).is_valid()
   161	    }

   193	    /// The `userinfo@host:port` portion, with each piece omitted when absent.
   194	    pub fn authority(&self) -> String {
   195	        let mut out = String::new();
   196	        if self.has_userinfo() {
   197	            out.push_str(&self.userinfo);
   198	            out.push('@');
   199	        }
   200	        out.push_str(&self.host);
   201	        if self.has_port() {
   202	            out.push(':');
   203	            out.push_str(&self.port.to_string());
   204	        }
   205	        out
   206	    }

   208	    // ------------------------------------------------------------------ setters
   209	
   210	    /// Stores the scheme lowercased.
   211	    ///
   212	    /// VERIFIED: `"HTTP"` is stored as `"http"`, but nothing else is touched —
   213	    /// `"http:"` stays `"http:"` and `"ht tp"` stays `"ht tp"`, so there is no
   214	    /// trimming and no separator stripping.
   215	    pub fn set_scheme(&mut self, scheme: &str) {
   216	        self.scheme = to_lower(scheme);
   217	    }
   218	
   219	    /// Stores the host verbatim.
   220	    ///
   221	    /// VERIFIED: the host is *not* lowercased (`"EXAMPLE.COM"` survives intact,
   222	    /// unlike the scheme) and is never split — `"h:9090"` stays one host and
   223	    /// renders as `http://h:9090:80/` when a port is also supplied.
   224	    pub fn set_host(&mut self, host: &str) {
   225	        self.host = host.to_owned();
   226	        self.refresh_valid();
   227	    }
   228	
   229	    /// Stores `port` when it is in range, otherwise [`PORT_UNSET`].
   230	    pub fn set_port(&mut self, port: i32) {
   231	        self.port = if is_port_valid(port) { port } else { PORT_UNSET };
   232	    }
   233	
   234	    /// Stores the path, giving a non-empty path a leading slash.
   235	    ///
   236	    /// VERIFIED: `"p"` becomes `"/p"` and `"user@example.com"` becomes
   237	    /// `"/user@example.com"`, while an empty path stays empty rather than
   238	    /// defaulting to `"/"`. Nothing else is normalized: `"/a/../b"` and
   239	    /// `"//a//b//"` are stored exactly as given.
   240	    pub fn set_path(&mut self, path: &str) {
   241	        self.path = if path.is_empty() || path.starts_with('/') {
   242	            path.to_owned()
   243	        } else {
   244	            format!("/{path}")
   245	        };
   246	        self.refresh_valid();
   247	    }
   248	
   249	    /// Stores the query verbatim.
   250	    ///
   251	    /// VERIFIED: a leading `"?"` is *not* stripped, so a query of `"?q=1"`
   252	    /// renders as `...??q=1`.
   253	    pub fn set_query(&mut self, query: &str) {
   254	        self.query = query.to_owned();
   255	    }
   256	
   257	    /// Stores the fragment verbatim.
   258	    ///
   259	    /// VERIFIED: a leading `"#"` is *not* stripped, so a fragment of `"#f"`
   260	    /// renders as `...##f`.
   261	    pub fn set_fragment(&mut self, fragment: &str) {
   262	        self.fragment = fragment.to_owned();
   263	    }

   299	    /// True when the URL carries an authority or a path.

   301	    /// VERIFIED to be exactly `has_host() || has_path()`, and to be independent
   302	    /// of the *content* of either: a host of `"a b"` or `"!!!"` still reports
   303	    /// valid, and a scheme on its own does not (`("http", "", -1, "", "", "")`

   306	        self.valid
   307	    }

   310	        self.valid = self.has_host() || self.has_path();

   314	
   315	    /// Lowercases the host. The scheme is already lowercased on the way in.
   316	    pub fn normalize_case(&mut self) {
   317	        self.host = to_lower(&self.host);
   318	    }

   320	    /// Resolves `.` and `..` segments and collapses repeated slashes.
   321	    pub fn normalize_path(&mut self) {
   322	        self.path = normalize_path_string(&self.path);
   323	        self.refresh_valid();
   324	    }

   327	    pub fn normalize_query(&mut self) {
   328	        self.query = query::normalize_query_string(&self.query);
   329	    }

   331	    /// Applies every normalization step.
   332	    pub fn normalize(&mut self) {
   333	        self.normalize_case();
   334	        self.normalize_path();
   335	        self.normalize_query();
   336	    }

   340	    /// The query's decoded `name=value` pairs.
   341	    pub fn parse_query_params(&self) -> Vec<Param> {
   342	        query::parse_query_string(&self.query)
   343	    }

   345	    /// Alias of [`URL::parse_query_params`].
   346	    pub fn query_params(&self) -> Vec<Param> {
   347	        self.parse_query_params()
   348	    }

   350	    /// The first value stored under `name`, if any.
   351	    pub fn get_query_param(&self, name: &str) -> Option<String> {
   352	        self.parse_query_params()
   353	            .into_iter()
   354	            .find(|(key, _)| key == name)
   355	            .map(|(_, value)| value)
   356	    }
   357	
   358	    /// Appends `name=value` to the raw query, percent-encoding both sides.
   359	    pub fn add_query_param(&mut self, name: &str, value: &str) {
   360	        let mut params = self.parse_query_params();
   361	        params.push((name.to_owned(), value.to_owned()));
   362	        self.query = query::build_query_string(&params);
   363	    }
   364	
   365	    /// Drops every pair stored under `name`.
   366	    pub fn remove_query_param(&mut self, name: &str) {
   367	        let mut params = self.parse_query_params();
   368	        params.retain(|(key, _)| key != name);
   369	        self.query = query::build_query_string(&params);
   370	    }

   373	impl fmt::Display for URL {
   374	    /// Renders the URL, mirroring `URL::to_string`.
   375	    ///
   376	    /// The scheme gates the whole authority. This looks wrong but is exactly what
   377	    /// the C++ library does, and `test15` observes it:
   378	    ///
   379	    /// - `("", "example.com", 80, "/", "", "")` renders as just `"/"` — a host
   380	    ///   and port with no scheme are dropped entirely.
   381	    /// - `("http", "", 80, "/", "", "")` renders as `"http://:80/"` — the `//`
   382	    ///   and the port survive an empty host.
   383	    /// - the separator is a literal `"://"`, never a bare `":"`, so
   384	    ///   `("file", "", -1, "/etc/hosts", ...)` renders as `"file:///etc/hosts"`.
   385	    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
   386	        if self.has_scheme() {
   387	            write!(formatter, "{}://", self.scheme)?;
   388	            if self.has_userinfo() {
   389	                write!(formatter, "{}@", self.userinfo)?;
   390	            }
   391	            formatter.write_str(&self.host)?;
   392	            if self.has_port() {
   393	                write!(formatter, ":{}", self.port)?;
   394	            }
   395	        }
   396	        formatter.write_str(&self.path)?;
   397	        if self.has_query() {
   398	            write!(formatter, "?{}", self.query)?;
   399	        }
   400	        if self.has_fragment() {
   401	            write!(formatter, "#{}", self.fragment)?;
   402	        }
   403	        Ok(())
   404	    }
   405	}

     7	/// Accepts a scheme of the RFC 3986 shape `ALPHA *( ALPHA / DIGIT / "+" / "-" / "." )`.
     8	pub fn is_scheme_valid(scheme: &str) -> bool {
     9	    let mut bytes = scheme.bytes();
    10	    match bytes.next() {
    11	        Some(first) if first.is_ascii_alphabetic() => {}
    12	        _ => return false,
    13	    }
    14	    bytes.all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.'))
    15	}

    17	/// Accepts a bracketed IPv6 literal, or a dot-separated set of host labels made
    18	/// of alphanumerics, `-` and `_`, with no empty label and no label starting or
    19	/// ending in `-`.
    20	pub fn is_host_valid(host: &str) -> bool {
    21	    if host.is_empty() {
    22	        return false;
    23	    }
    24	
    25	    if let Some(inner) = host.strip_prefix('[') {
    26	        return match inner.strip_suffix(']') {
    27	            Some(address) => {
    28	                !address.is_empty()
    29	                    && address
    30	                        .bytes()
    31	                        .all(|byte| byte.is_ascii_hexdigit() || matches!(byte, b':' | b'.'))
    32	            }
    33	            None => false,
    34	        };
    35	    }
    36	
    37	    host.split('.').all(|label| {
    38	        !label.is_empty()
    39	            && !label.starts_with('-')
    40	            && !label.ends_with('-')
    41	            && label
    42	                .bytes()
    43	                .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_'))
    44	    })
    45	}

    47	/// The range a port must fall in to be stored as given.
    48	///
    49	/// VERIFIED against the C++ binary: 0 and 65535 are kept (`to_string()` renders
    50	/// `":0"` and `":65535"`), while -1, -2, -100, `i32::MIN`, 65536, 70000 and
    51	/// `i32::MAX` are all replaced by the "unset" sentinel -1. So the bound is
    52	/// inclusive on both ends and 0 counts as valid.
    53	pub fn is_port_valid(port: i32) -> bool {
    54	    (0..=65535).contains(&port)
    55	}
