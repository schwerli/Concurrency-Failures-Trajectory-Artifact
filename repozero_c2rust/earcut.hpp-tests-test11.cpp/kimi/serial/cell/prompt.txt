#include <vector>
#include <array>
#include "../earcut.hpp"

int main(int argc, char* argv[]) {
    int n = 11;
    if (argc > 1) {
        n = std::atoi(argv[1]);
    }
    
    std::vector<std::vector<std::array<double, 2>>> polygon;
    std::vector<std::array<double, 2>> gear;
    
    for (int i = 0; i < n; i++) {
        double angle1 = 2.0 * M_PI * i / n;
        double angle2 = 2.0 * M_PI * (i + 0.5) / n;
        
        gear.push_back({std::cos(angle1), std::sin(angle1)});
        gear.push_back({0.7 * std::cos(angle2), 0.7 * std::sin(angle2)});
    }
    
    polygon.push_back(gear);
    
    auto indices = mapbox::earcut<uint32_t>(polygon);
    
    for (auto idx : indices) {
        std::cout << idx << " ";
    }
    return 0;
}
