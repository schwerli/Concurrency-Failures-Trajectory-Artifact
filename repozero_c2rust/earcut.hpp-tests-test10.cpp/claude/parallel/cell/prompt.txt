#include <iostream>
#include <vector>
#include <array>
#include "../earcut.hpp"

int main(int argc, char* argv[]) {
    int n = 10;
    if (argc > 1) {
        n = std::atoi(argv[1]);
    }
    
    std::vector<std::vector<std::array<double, 2>>> polygon;
    
    for (int layer = 0; layer < n; layer++) {
        std::vector<std::array<double, 2>> ring;
        double radius = (layer + 1) * 0.3;
        
        for (int i = 0; i < (layer + 3) * 2; i++) {
            double angle = 2.0 * M_PI * i / ((layer + 3) * 2);
            double offset = layer * 0.1;
            ring.push_back({(radius + offset) * std::cos(angle), (radius + offset) * std::sin(angle)});
        }
        
        polygon.push_back(ring);
    }
    
    auto indices = mapbox::earcut<uint32_t>(polygon);
    
    for (auto idx : indices) {
        std::cout << idx << " ";
    }
    return 0;
}
