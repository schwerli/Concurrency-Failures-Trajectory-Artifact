   112	    ///
   113	    /// The doubled value is returned so the shoelace sum stays exact in integer
   114	    /// arithmetic. Positive when the ring winds clockwise on screen (y growing
   115	    /// downward), negative when it winds the other way.
   116	    ///
   117	    /// Accumulation is in `i128`, which is exact for every coordinate range the
   118	    /// C++ library admits. It is *not* wide enough for arbitrary full-range
   119	    /// `i64` coordinates — a single cross term can reach `2 * i64::MAX²`, which
   120	    /// alone nearly fills an `i128` — so the sum saturates at the `i128` bounds

   122	    /// release builds.

   124	        let n = self.0.len();
   125	        if n < 3 {
   126	            return 0;
   127	        }
   128	        let mut acc: i128 = 0;
   129	        let mut prev = self.0[n - 1];
   130	        for &cur in &self.0 {
   131	            let cross = (prev.x as i128 * cur.y as i128) - (cur.x as i128 * prev.y as i128);
   132	            acc = acc.saturating_add(cross);
   133	            prev = cur;
   134	        }
   135	        acc
   136	    }

    12	///
    13	/// The C++ counterpart is `Clipper2Lib::Point64`, a plain aggregate of two

    16	pub struct Point64 {

    18	    pub y: i64,
    19	}

    21	impl Point64 {
    22	    /// The origin, `(0, 0)` — equivalent to the default-constructed `Point64()`.
    23	    pub const ORIGIN: Self = Self { x: 0, y: 0 };

    25	    /// Constructs a point, matching the C++ `Point64(x, y)` constructor.

    31	    /// Rounds a floating-point pair to the nearest integer point, halves away
    32	    /// from zero (`0.5 -> 1`, `-0.5 -> -1`), matching [`f64::round`].
    33	    ///
    34	    /// Half-away-from-zero is chosen because it is what C's `llround` family
    35	    /// specifies, which is the conventional way to narrow a `double` coordinate
    36	    /// to an integer one. Unlike `llround`, the conversion here is total: a NaN
    37	    /// input yields `0` and out-of-range inputs clamp to `i64::MIN`/`i64::MAX`,
    38	    /// which is Rust's defined behaviour for a float-to-int cast, rather than
    39	    /// C++'s unspecified result.

     2	//!
     3	//! Mirrors `Clipper2Lib::Rect64`. The field naming follows the C++ header's

     5	//! *smallest* y and `bottom` the largest.

    10	/// An axis-aligned rectangle with 64-bit integer bounds — the C++ `Rect64`.
    11	#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
    12	pub struct Rect64 {
    13	    pub left: i64,
    14	    pub top: i64,
    15	    pub right: i64,
    16	    pub bottom: i64,
    17	}
    18	
    19	impl Rect64 {
    20	    /// Constructs a rectangle from explicit edges, matching the C++
    21	    /// `Rect64(left, top, right, bottom)` constructor. Edges are stored as
    22	    /// given; no normalisation is applied.
    23	    #[inline]
    24	    pub const fn new(left: i64, top: i64, right: i64, bottom: i64) -> Self {
    25	        Self {
    26	            left,
    27	            top,
    28	            right,
    29	            bottom,
    30	        }
    31	    }
    32	
    33	    /// Tightest rectangle covering every point, or `None` for an empty slice.
    34	    ///
    35	    /// `Option` rather than a sentinel rectangle: the C++ library's "invalid
    36	    /// rect" value is an implementation detail that the public interface does
    37	    /// not pin down, so this crate makes the empty case explicit instead.
    38	    pub fn from_points(points: &[Point64]) -> Option<Self> {
    39	        let (first, rest) = points.split_first()?;
    40	        let mut r = Self::new(first.x, first.y, first.x, first.y);
    41	        for p in rest {
    42	            r.left = r.left.min(p.x);
    43	            r.right = r.right.max(p.x);
    44	            r.top = r.top.min(p.y);
    45	            r.bottom = r.bottom.max(p.y);
    46	        }
    47	        Some(r)
    48	    }
    49	
    50	    /// Horizontal extent — the C++ `Width()`.
    51	    ///
    52	    /// Saturating, because the span between two extreme `i64` edges is not
    53	    /// representable as an `i64`; the C++ original would simply overflow there.
    54	    #[inline]
    55	    pub fn width(&self) -> i64 {
    56	        self.right.saturating_sub(self.left)
    57	    }
    58	
    59	    /// Vertical extent — the C++ `Height()`.
    60	    ///
    61	    /// Saturating, for the same reason as [`Rect64::width`].
    62	    #[inline]
    63	    pub fn height(&self) -> i64 {
    64	        self.bottom.saturating_sub(self.top)
    65	    }
    66	
    67	    /// Whether the rectangle encloses no area — the C++ `IsEmpty()`.
    68	    #[inline]
    69	    pub fn is_empty(&self) -> bool {
    70	        self.bottom <= self.top || self.right <= self.left
    71	    }
    72	
    73	    /// Centre point, rounded toward zero.
    74	    ///
    75	    /// Averaged through `i128` so that edges near the `i64` limits cannot
    76	    /// overflow the intermediate sum; the halved result always fits in `i64`.
    77	    #[inline]
    78	    pub fn mid_point(&self) -> Point64 {
    79	        let mid = |a: i64, b: i64| ((a as i128 + b as i128) / 2) as i64;
    80	        Point64::new(mid(self.left, self.right), mid(self.top, self.bottom))
    81	    }
    82	
    83	    /// Whether `p` lies inside or on the boundary — the C++ `Contains()`.
    84	    #[inline]
    85	    pub fn contains(&self, p: Point64) -> bool {
    86	        p.x >= self.left && p.x <= self.right && p.y >= self.top && p.y <= self.bottom
    87	    }
    88	
    89	    /// The four corners, clockwise from the top-left — the C++ `AsPath()`.
    90	    pub fn corners(&self) -> [Point64; 4] {
    91	        [
    92	            Point64::new(self.left, self.top),
    93	            Point64::new(self.right, self.top),
    94	            Point64::new(self.right, self.bottom),
    95	            Point64::new(self.left, self.bottom),
    96	        ]
    97	    }
    98	}

   103	            f,
   104	            "({},{},{},{})",
   105	            self.left, self.top, self.right, self.bottom
