#include "../color.hpp"
#include <iostream>
#include <vector>
using namespace ColorSpace;

int main(int argc, char* argv[]) {
    std::vector<RGB> colors;
    
    // 添加一些默认颜色
    colors.push_back(RGB(0.1f, 0.2f, 0.3f));
    colors.push_back(RGB(0.4f, 0.5f, 0.6f));
    colors.push_back(RGB(0.7f, 0.2f, 0.1f));
    colors.push_back(RGB(0.3f, 0.8f, 0.4f));
    colors.push_back(RGB(0.9f, 0.1f, 0.8f));
    
    // 从命令行参数添加更多颜色
    for (int i = 1; i + 2 < argc; i += 3) {
        float r = std::stof(argv[i]);
        float g = std::stof(argv[i+1]);
        float b = std::stof(argv[i+2]);
        colors.push_back(RGB(r, g, b));
    }
    
    // 执行多个转换和操作
    for (size_t i = 0; i < colors.size(); ++i) {
        const RGB& rgb = colors[i];
        
        // 基本转换
        HSL hsl = ColorConverter::rgbToHSL(rgb);
        HSV hsv = ColorConverter::rgbToHSV(rgb);
        CMYK cmyk = ColorConverter::rgbToCMYK(rgb);
        YUV yuv = ColorConverter::rgbToYUV(rgb);
        
        // 亮度调整
        RGB bright = ColorConverter::adjustBrightness(rgb, 1.2f);
        RGB dark = ColorConverter::adjustBrightness(rgb, 0.8f);
        
        // 饱和度调整
        RGB saturated = ColorConverter::adjustSaturation(rgb, 1.5f);
        
        // 灰度化
        RGB gray = ColorConverter::grayscale(rgb);
        
        // 反转
        RGB inverted = ColorConverter::invert(rgb);
        
        // 输出结果
        std::cout << "Color " << i << ": ";
        std::cout << rgb.r << " " << rgb.g << " " << rgb.b << " ";
        std::cout << hsl.h << " " << hsl.s << " " << hsl.l << " ";
        std::cout << hsv.h << " " << hsv.s << " " << hsv.v << " ";
        std::cout << cmyk.c << " " << cmyk.m << " " << cmyk.y << " " << cmyk.k << " ";
        std::cout << yuv.y << " " << yuv.u << " " << yuv.v << " ";
        std::cout << bright.r << " " << bright.g << " " << bright.b << " ";
        std::cout << dark.r << " " << dark.g << " " << dark.b << " ";
        std::cout << saturated.r << " " << saturated.g << " " << saturated.b << " ";
        std::cout << gray.r << " " << gray.g << " " << gray.b << " ";
        std::cout << inverted.r << " " << inverted.g << " " << inverted.b << " ";
        
        // 如果是最后一个颜色，计算与第一个颜色的距离
        if (i == colors.size() - 1 && colors.size() > 1) {
            float dist = ColorConverter::colorDistance(rgb, colors[0]);
            std::cout << dist << " ";
        }
        
        std::cout << std::endl;
    }
    
    return 0;
}
