#include "clipper2/clipper.h"
#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
#include <random>

using namespace Clipper2Lib;

int main(int argc, char* argv[]) {
    // 20个API：终极综合测试
    int complexity = (argc > 1) ? std::min(std::stoi(argv[1]), 5) : 3;
    
    // 创建多层次复杂几何
    std::vector<Paths64> geometryLayers;
    
    // 第1层：基础形状
    Paths64 layer1;
    for (int i = 0; i < complexity; ++i) {
        // 圆形
        Path64 circle;
        for (int j = 0; j < 24; ++j) {
            double angle = 2 * 3.1415926 * j / 24;
            int64_t x = 100 + i * 80 + static_cast<int64_t>(40 * cos(angle));
            int64_t y = 100 + static_cast<int64_t>(40 * sin(angle));
            circle.push_back(Point64(x, y));
        }
        layer1.push_back(circle);
        
        // 正方形
        Path64 square;
        int64_t size = 50 + i * 10;
        square.push_back(Point64(200 + i * 60, 200));
        square.push_back(Point64(200LL + i * 60 + size, 200LL));
        square.push_back(Point64(200 + i * 60 + size, 200 + size));
        square.push_back(Point64(200LL + i * 60, 200LL + size));
        layer1.push_back(square);
    }
    geometryLayers.push_back(layer1);
    
    // 第2层：复杂多边形
    Paths64 layer2;
    for (int i = 0; i < complexity; ++i) {
        // 星形
        Path64 star;
        int points = 5 + i;
        for (int j = 0; j < points * 2; ++j) {
            double angle = 2 * 3.1415926 * j / (points * 2);
            double radius = (j % 2 == 0) ? 35.0 : 20.0;
            int64_t x = 150 + static_cast<int64_t>(radius * cos(angle));
            int64_t y = 150 + i * 100 + static_cast<int64_t>(radius * sin(angle));
            star.push_back(Point64(x, y));
        }
        layer2.push_back(star);
        
        // 螺旋形
        Path64 spiral;
        for (int j = 0; j < 30; ++j) {
            double t = j * 0.3;
            double radius = 10.0 + t * 4.0;
            double angle = t;
            int64_t x = 300 + static_cast<int64_t>(radius * cos(angle));
            int64_t y = 150 + i * 100 + static_cast<int64_t>(radius * sin(angle));
            spiral.push_back(Point64(x, y));
        }
        layer2.push_back(spiral);
    }
    geometryLayers.push_back(layer2);
    
    // 第3层：网格和图案
    Paths64 layer3;
    for (int i = 0; i < complexity; ++i) {
        for (int j = 0; j < complexity; ++j) {
            // 六边形网格
            Path64 hex;
            int64_t centerX = 50 + i * 70;
            int64_t centerY = 300 + j * 70;
            for (int k = 0; k < 6; ++k) {
                double angle = 2 * 3.1415926 * k / 6;
                int64_t x = centerX + static_cast<int64_t>(25 * cos(angle));
                int64_t y = centerY + static_cast<int64_t>(25 * sin(angle));
                hex.push_back(Point64(x, y));
            }
            layer3.push_back(hex);
            
            // 三角形
            Path64 triangle;
            triangle.push_back(Point64(centerX + 100, centerY));
            triangle.push_back(Point64(centerX + 100 + 30, centerY + 50));
            triangle.push_back(Point64(centerX + 100 - 30, centerY + 50));
            layer3.push_back(triangle);
        }
    }
    geometryLayers.push_back(layer3);
    
    // 执行20个API操作链
    Paths64 accumulated;
    
    // 1-3. 每层独立并集
    for (const auto& layer : geometryLayers) {
        Paths64 unionLayer = Union(layer, FillRule::EvenOdd);
        accumulated.insert(accumulated.end(), unionLayer.begin(), unionLayer.end());
    }
    
    // 4. 所有层合并
    Paths64 allUnion = Union(accumulated, FillRule::EvenOdd);
    
    // 5-8. 分层处理
    Paths64 processed;
    for (size_t i = 0; i < geometryLayers.size(); ++i) {
        Paths64 layerUnion = Union(geometryLayers[i], FillRule::EvenOdd);
        
        // 与其他层交互
        for (size_t j = i + 1; j < geometryLayers.size(); ++j) {
            Paths64 otherUnion = Union(geometryLayers[j], FillRule::EvenOdd);
            
            // 5. 交集
            Paths64 intersect = Intersect(layerUnion, otherUnion, FillRule::EvenOdd);
            // 6. 差集
            Paths64 diff = Difference(layerUnion, otherUnion, FillRule::EvenOdd);
            // 7. 异或
            Paths64 xorbit = Xor(layerUnion, otherUnion, FillRule::EvenOdd);
            
            processed.insert(processed.end(), intersect.begin(), intersect.end());
            processed.insert(processed.end(), diff.begin(), diff.end());
            processed.insert(processed.end(), xorbit.begin(), xorbit.end());
        }
    }
    
    // 9. 处理结果合并
    Paths64 allProcessed = Union(processed, FillRule::NonZero);
    
    // 10-13. 偏移操作序列
    Paths64 offsetResults;
    double offsets[] = {2.0, -1.5, 3.0, -2.0};
    JoinType joinTypes[] = {JoinType::Round, JoinType::Miter, JoinType::Square, JoinType::Bevel};
    
    for (int i = 0; i < 4; ++i) {
        Paths64 offset = InflatePaths(allProcessed, offsets[i], joinTypes[i], EndType::Polygon);
        offsetResults.insert(offsetResults.end(), offset.begin(), offset.end());
    }
    
    // 14. 偏移结果合并
    Paths64 unionOffsets = Union(offsetResults, FillRule::EvenOdd);
    
    // 15. 与原始并集交互
    Paths64 finalIntersect = Intersect(unionOffsets, allUnion, FillRule::EvenOdd);
    Paths64 finalDiff = Difference(unionOffsets, allUnion, FillRule::EvenOdd);
    Paths64 finalXor = Xor(unionOffsets, allUnion, FillRule::EvenOdd);
    
    // 16. 合并所有最终结果
    Paths64 allFinal;
    allFinal.insert(allFinal.end(), finalIntersect.begin(), finalIntersect.end());
    allFinal.insert(allFinal.end(), finalDiff.begin(), finalDiff.end());
    allFinal.insert(allFinal.end(), finalXor.begin(), finalXor.end());
    
    // 17. 最终并集
    Paths64 finalUnion = Union(allFinal, FillRule::NonZero);
    
    // 18. 平移归一化
    Rect64 bounds = GetBounds(finalUnion);
    Paths64 normalized = TranslatePaths(finalUnion, -bounds.left, -bounds.top);
    
    // 19. 最终偏移
    Paths64 lastOffset = InflatePaths(normalized, 1.0, JoinType::Round, EndType::Polygon);
    
    // 20. 综合计算
    double totalArea = 0;
    double totalPerimeter = 0;
    int64_t totalVertices = 0;
    
    for (const auto& path : lastOffset) {
        double area = Area(path);
        totalArea += area;
        totalVertices += path.size();
        
        // 估算周长（简单方法）
        if (path.size() > 1) {
            for (size_t i = 0; i < path.size(); ++i) {
                size_t j = (i + 1) % path.size();
                double dx = path[j].x - path[i].x;
                double dy = path[j].y - path[i].y;
                totalPerimeter += sqrt(dx * dx + dy * dy);
            }
        }
    }
    
    double complexityScore = totalArea * 0.7 + totalPerimeter * 0.2 + totalVertices * 0.1;
    
    std::cout << complexityScore << std::endl;
    
    return 0;
}
