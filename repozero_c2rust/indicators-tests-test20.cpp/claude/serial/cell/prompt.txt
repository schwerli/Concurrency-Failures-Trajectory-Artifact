#include <iostream>
#include <string>
#include "../include/indicators.hpp"

int main(int argc, char* argv[]) {
    // 测试所有API的组合
    indicators::ProgressBar bar;
    bar.set_progress(45);
    bar.set_total(90);
    bar.set_width(40);
    std::cout << bar.str() << std::endl;
    std::cout << bar() << std::endl;
    
    indicators::Percentage pct1(0.45f, 1);
    indicators::Percentage pct2(0.67f, 2);
    std::cout << pct1.str() << std::endl;
    std::cout << pct2.str() << std::endl;
    std::cout << static_cast<std::string>(pct1) << std::endl;
    
    indicators::Counter cnt(45, 90);
    std::cout << cnt.str() << std::endl;
    std::cout << static_cast<std::string>(cnt) << std::endl;
    
    indicators::ProgressBuilder builder;
    std::cout << builder.set_progress(45).set_total(90).build() << std::endl;
    
    indicators::ProgressTracker tracker;
    tracker.add_stage("A").add_stage("B").add_stage("C");
    tracker.set_progress(1);
    tracker.set_total(3);
    std::cout << tracker.get_progress() << std::endl;
    std::cout << tracker.get_percentage() << std::endl;
    
    std::cout << indicators::Format::percentage(0.33f) << std::endl;
    std::cout << indicators::Format::progress(33, 100) << std::endl;
    std::cout << indicators::Format::bar(33, 100, 25) << std::endl;
    
    indicators::MultiProgress multi;
    indicators::ProgressBar b1, b2;
    b1.set_progress(30);
    b1.set_total(100);
    b2.set_progress(60);
    b2.set_total(100);
    multi.add_bar(b1);
    multi.add_bar(b2);
    std::cout << multi.str() << std::endl;
    
    indicators::ProgressDisplay display(100);
    display.update(50);
    std::cout << display.display() << std::endl;
    display.increment(25);
    std::cout << display.display() << std::endl;
    
    indicators::CyclicProgress cyclic(10);
    cyclic.next();
    cyclic.next();
    std::cout << cyclic.str() << std::endl;
    
    indicators::TaskProgress task("Test", 10);
    task.complete(3);
    std::cout << task.status() << std::endl;
    
    // 测试不同精度的百分比
    for (int i = 1; i <= 5; ++i) {
        std::cout << indicators::Format::percentage(0.12345f, i) << std::endl;
    }
    
    // 测试不同宽度的进度条
    for (int i = 10; i <= 50; i += 10) {
        std::cout << indicators::Format::bar(50, 100, i) << std::endl;
    }
    
    // 测试计数器数组
    indicators::Counter counters[3];
    for (int i = 0; i < 3; ++i) {
        counters[i].set_current((i+1)*25);
        counters[i].set_total(100);
        std::cout << counters[i].str() << std::endl;
    }
    
    // 测试百分比数组
    indicators::Percentage percentages[3] = {
        indicators::Percentage(0.25f, 1),
        indicators::Percentage(0.5f, 2),
        indicators::Percentage(0.75f, 3)
    };
    
    for (int i = 0; i < 3; ++i) {
        std::cout << percentages[i].str() << std::endl;
    }
    
    return 0;
}
